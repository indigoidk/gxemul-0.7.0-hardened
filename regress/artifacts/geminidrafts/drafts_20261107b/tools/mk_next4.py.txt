def span(n):
    old_hdr = re.compile(r"^##? .*\(#%s\)" % n)
    if not any(HDR.match(l) and l.startswith("# `#%s`" % n) for l in CL):
        first = next(i for i, l in enumerate(CL) if old_hdr.match(l))
        end = first + 1
        while end < len(CL) and not re.match(r"^##? ", CL[end]):
            end += 1
        return first + 1, end
    tag = "`#%s`" % n
    first = next(i for i, l in enumerate(CL) if HDR.match(l) and l.startswith("# " + tag))
    end = first + 1
    while end < len(CL) and not (HDR.match(CL[end]) and not CL[end].startswith("# " + tag)):
        end += 1
    body = CL[first:end]
    others = [l[:40] for l in body if re.match(r"^# `#(\d+)`", l) and not l.startswith("# " + tag)]
    assert not others, (n, others)
    return first + 1, end          # 1-based inclusive range
