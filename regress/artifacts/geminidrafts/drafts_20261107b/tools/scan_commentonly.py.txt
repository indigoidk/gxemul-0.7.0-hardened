"""scan_commentonly.py -- 04:4x 11-07: which BUILD_OK fix drafts change NO CODE?  The Gemini second-lens review found
`fbsignalcoverage`'s "fix" was one added comment line, scored BUILD_OK because it compiles.  For every BUILD_OK row: rebuild the
post-edit file in memory (HEAD + APPLIED.diff's .c/.h blocks, mk_fixreview.py's parser), strip C comments and whitespace from both
versions (string and char literals preserved), and compare.  Identical token streams = COMMENT_ONLY.  Read-only; prints a table and
writes commentonly.tsv."""
import io

sys.argv = ["mk_fixreview.py", "--list"]
src = io.open("C:/DocumentNoSnc/CC/GXEMUL/_scratchpad/gweb/jobs_20261106_kits/mk_fixreview.py", encoding="utf-8").read()
exec(src.split('if __name__ == "__main__":')[0])          # H, REPO, parse_udiff, apply, cands

TOK = re.compile(r'"(?:\\.|[^"\\\n])*"|\'(?:\\.|[^\'\\\n])*\'|/\*.*?\*/|//[^\n]*|\s+|.', re.S)


def code_only(text):
    out = []
    for m in TOK.finditer(text):
        t = m.group(0)
        if t.startswith("/*") or t.startswith("//") or t.isspace():
            out.append(" ")
        else:
            out.append(t)
    return re.sub(r" +", " ", "".join(out)).strip()


rows = []
for rid in cands:
    diff = io.open(H + "fix_%s/APPLIED.diff" % rid, encoding="latin-1").read()
    keep = [b for b in re.split(r"(?m)^(?=diff --git )", diff) if re.match(r"diff --git a/\S+\.[ch] ", b)
            and not re.match(r"diff --git a/src/cpus/tmp_\w+\.c ", b)]
    verdict, files = "CODE", []
    try:
        same = True
        for f, hunks in parse_udiff("".join(keep)).items():
            old = subprocess.run(["git", "-C", REPO, "show", "HEAD:" + f], capture_output=True).stdout.decode("latin-1").split("\n")
            new, _ = apply(old, hunks)
            if code_only("\n".join(old)) != code_only("\n".join(new)):
                same = False
            files.append(f)
        if not keep:
            verdict = "NO_SOURCE"
        elif same:
            verdict = "COMMENT_ONLY"
    except ValueError as e:
        verdict = "PARSE_ERR %s" % e
    rows.append((rid, verdict, " ".join(files)))
io.open(H + "commentonly.tsv", "w", encoding="utf-8", newline="\n").write("".join("%s\t%s\t%s\n" % r for r in rows))
from collections import Counter

        print("%-20s %-14s %s" % (rid, v, f))
