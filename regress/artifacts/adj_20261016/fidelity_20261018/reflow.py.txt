#!/usr/bin/env python3
"""reflow.py <block.md> -- reflow every paragraph and numbered item of a correction block to 120 columns: an item's first
line keeps its "N. " prefix and its continuation lines get three spaces; a code span (`...`) never breaks; a two-space
sentence break survives, including one that fell at a line end; headings and blank lines stay."""
import io, re, sys, textwrap
p = sys.argv[1]
t = io.open(p, encoding="utf-8", newline="").read()
SPAN = re.compile(r"`[^`\n]*`")
KEEP = "\x01"      # glued to a sentence's last word, it becomes the second space after wrapping
HARD = "\x00"      # a space inside a code span
END = re.compile(r"[.?!:][)*\"'`\]]*$")
START = re.compile(r"^[A-Z(*`\"']")

def join(lines):
    out = lines[0].strip()
    for l in lines[1:]:
        l = l.strip()
        out += (KEEP + " " if END.search(out) and START.match(l) else " ") + l
    return out

def wrap(first, rest, text, width=120):
    text = re.sub(r"(?<=\S)  +(?=\S)", KEEP + " ", text)
    text = SPAN.sub(lambda m: m.group(0).replace(" ", HARD), text)
    out = textwrap.wrap(text, width - 1, initial_indent=first, subsequent_indent=rest,
                        break_long_words=False, break_on_hyphens=False)
    return "\n".join(l.replace(HARD, " ").replace(KEEP, " ").rstrip() for l in out)

paras, cur = [], []
for line in t.split("\n"):
    if line.startswith("#") or line.strip() == "":
        if cur: paras.append(cur); cur = []
        paras.append([line])
    elif re.match(r"^\d+\. ", line):
        if cur: paras.append(cur)
        cur = [line]
    else:
        cur.append(line)
if cur: paras.append(cur)
res = []
for para in paras:
    if len(para) == 1 and (para[0].startswith("#") or para[0].strip() == ""):
        res.append(para[0]); continue
    text = join(para)
    m = re.match(r"^(\d+\. )(.*)$", text, re.S)
    res.append(wrap(m.group(1), "   ", m.group(2)) if m else wrap("", "", text))
io.open(p, "w", encoding="utf-8", newline="").write("\n".join(res))
print("reflowed %s, max width %d" % (p, max(len(l) for l in "\n".join(res).split("\n"))))
