#!/usr/bin/env python3
"""apply_fid.py <fid scratch dir> -- builds the corrected RULINGS_A_563.md and RULINGS_B_asic.md from HEAD's copies
(*_head.md) and the two correction blocks: each block goes after its file's roster paragraph, and every build-list line
an item changes gets an inline pointer to the item.  Nothing existing is deleted or reworded.  Writes *_commit.md."""
import io, os, sys
D = sys.argv[1]
def rd(n): return io.open(os.path.join(D, n), encoding="utf-8", newline="").read()
def wr(n, t): io.open(os.path.join(D, n), "w", encoding="utf-8", newline="").write(t)
def once(t, a, b, what):
    assert t.count(a) == 1, (what, t.count(a))
    return t.replace(a, b)
M = "**[2026-10-18, FIDELITY CORRECTIONS item %s.]**"

a = rd("RULINGS_A_563_head.md")
a = once(a, "\n## Rulings\n", "\n" + rd("block_A.md").rstrip("\n") + "\n\n## Rulings\n", "A block")
a = once(a, "## REPAIR4 as adjudicated (for the builder, Opus 5.5; PRETABLE first, its mtime before table 1)\n",
         "## REPAIR4 as adjudicated (for the builder, Opus 5.5; PRETABLE first, its mtime before table 1)\n\n"
         "**[2026-10-18: read FIDELITY CORRECTIONS items 1-6 above first.  Item 3 adds three PRETABLE measurements; items "
         "1, 2, 4 and 5 sharpen the N5, N4, N6 and preservation lines below.]**\n", "A build head")
a = once(a, "every existing row byte-identical):", "every existing row byte-identical) " + (M % "5: N4 and R14/R15 are the named exceptions") + ":", "A preserve")
a = once(a, "- **N4** VD1: a count read after the first 1-byte DATA read of the 1-byte walk, want 32767.",
         "- **N4** VD1: a count read after the first 1-byte DATA read of the 1-byte walk, want 32767.  " + (M % "2: VD1 is `consumed = len == 4 ? 4 : 2`"), "A N4")
a = once(a, "  oracle `rep_asserts == 0 && warned_null_xfer == 1` (FB-A).",
         "  oracle `rep_asserts == 0 && warned_null_xfer == 1` (FB-A).  " + (M % "1: the measured stores are 2, 4, 4, 4, then 2-byte stores to 514"), "A N5")
a = once(a, "  zeros).  A two-byte transfer would miss `n_E_cons_ge4` (#475 clamps it back).",
         "  zeros).  A two-byte transfer would miss `n_E_cons_ge4` (#475 clamps it back).  " + (M % "4: also measured against Astra's second-pop variants at both endian sites"), "A N6")
a = once(a, "- **G0/G2 rows** for each of Astra's three candidates that the PRETABLE measures as distinguishing.",
         "- **G0/G2 rows** for each of Astra's three candidates that the PRETABLE measures as distinguishing.  " + (M % "6: the flag-only member included, graded (e) with its delta"), "A G0")
wr("RULINGS_A_563_commit.md", a)

b = rd("RULINGS_B_asic_head.md")
b = once(b, "\n## Final rulings\n", "\n" + rd("block_B.md").rstrip("\n") + "\n\n## Final rulings\n", "B block")
b = once(b, "## Build list for the next asiceventaddr build (Opus 5.5; PRETABLE first)\n",
         "## Build list for the next asiceventaddr build (Opus 5.5; PRETABLE first)\n\n"
         "**[2026-10-18: read FIDELITY CORRECTIONS items 1-5 above first -- item 1 says which Record 3 text to leave alone, and "
         "items 2-5 are build actions this list implies but does not spell out.]**\n", "B build head")
wr("RULINGS_B_asic_commit.md", b)
print("built both")
