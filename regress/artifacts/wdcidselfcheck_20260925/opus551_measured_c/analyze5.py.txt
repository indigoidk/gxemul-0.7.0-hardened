#!/usr/bin/env python3
"""#551c: gate 2 emulated per (shape, device) -- the 73 wdc checks verbatim from gate_offline.sh, H (token
in SOURCE), G (text in LOG), the self-mutant lane (selfmutant.py's logic; arms = device, device+selfmut)."""
import io, os, re, subprocess
W = "/tmp/opus551c"
NAMED = "geometry words carry their high byte"
G_TEXT = "SELFCHECK the comparator can still fail"
io.open(W + "/gate_eval.sh", "w").write(r'''
WDCLOG="$1"; _f=0
check() { [ "$2" = "$3" ] || { echo "RED $1 got=$2 want=$3"; _f=$((_f+1)); }; }
check_min() { [ "$2" -ge "$3" ] 2>/dev/null || { echo "RED $1 got=$2 min=$3"; _f=$((_f+1)); }; }
note() { :; }
. ''' + W + '''/gatewdc.inc
''')


def rd(dev, h, ext, root=W):
    p = "%s/dev/%s/regress/%s.%s" % (root, dev, h, ext)
    return io.open(p, encoding="utf-8", errors="replace").read() if os.path.exists(p) else None


def summ(dev, h):
    t = rd(dev, h, "log")
    if rd(dev, h, "rc").strip() == "BUILD_FAIL":
        return "BUILD_FAIL"
    m = re.search(r"^(\d+) rows, (\d+) failures", t, re.M)
    tok = re.search(r"WDC_IDENTIFY_(PASS|FAIL)", t)
    return "%s/%s %s" % (m.group(1) if m else "?", m.group(2) if m else "?", tok.group(1) if tok else "NOTOKEN")


def gate(h, dev):
    if rd(dev, h, "rc").strip() == "BUILD_FAIL":
        return ["compile"]
    lane_dev = "selfmut" if dev == "pristine" else dev + "_sm"
    t = rd(dev, h, "log")
    out = subprocess.run(["bash", W + "/gate_eval.sh", "%s/dev/%s/regress/%s.log" % (W, dev, h)],
                         capture_output=True, text=True).stdout
    reds = [l[4:44] for l in out.split("\n") if l.startswith("RED ")]
    if "@@SELFCHECK@@" not in io.open("%s/h/%s.c" % (W, h), encoding="utf-8").read():
        reds.append("H")
    if G_TEXT not in t:
        reds.append("G")
    tm = rd(lane_dev, h, "log")
    named = [l for l in t.split("\n") if l.strip().startswith(("ok", "FAIL")) and NAMED in l and "@@SELFCHECK@@" not in l]
    lane = "SETUP"
    if tm is not None and re.search(r"\b[A-Z0-9_]+_PASS\b", t) and len(named) == 1 and not named[0].strip().startswith("FAIL"):
        killed = [l for l in tm.split("\n") if l.strip().startswith("FAIL") and NAMED in l and "@@SELFCHECK@@" not in l]
        lane = "OK" if (re.search(r"\b[A-Z0-9_]+_FAIL\b", tm) and killed) else "FAIL"
    if lane != "OK":
        reds.append("lane:" + lane)
    return reds


def cell(h, dev):
    r = gate(h, dev)
    return ("GREEN" if not r else "red") + "(" + summ(dev, h) + ")"


SHAPES = ["C", "C_reset1", "C_resetcmp", "C_reset2", "C_nocmpinc", "C_nocmpinc_reset1", "C_recon_ge_sub", "C_recon_sub",
          "C_recon_lt", "C_recon_del", "C_recon_uncond", "C_sub_before", "C_sub_after", "C_reset1_after", "C_scg_del",
          "Cfix", "Cfix_recon_ge_sub", "Cfix_sub_after", "Cfix_reset2"]
print("=== (1) the sweep: gate 2 on the real device and on three defects only COVERED rows see ===")
print("    %-19s %-22s %-22s %-22s %-22s" % ("shape", "pristine", "dropold (chk cmp)", "drqring (chk 0/1)", "w57hi (row_cmp)"))
for h in SHAPES:
    print("    %-19s %-22s %-22s %-22s %-22s" % (h, cell(h, "pristine"), cell(h, "dropold"), cell(h, "drqring"), cell(h, "w57hi")))
print("  (a covered row 'can no longer fail' where its defect's column is GREEN while the pristine column is GREEN)")

print("\n=== (3) healthy-run and defect-run output: MINE_c against HEAD's harness, MINE_b, and the closure ===")
for dev in ("pristine", "selfmut", "rev255", "cyl13", "srstbit", "drqring", "w57hi", "w58lo", "dropold"):
    c = rd(dev, "C", "log")
    cells = []
    o = rd(dev, "ORIG.O2", "log", "/tmp/opus551f2")
    if o is not None:
        cells.append("vs HEAD %s" % ("ID-exSC" if [l for l in o.split("\n") if "SELFCHECK" not in l] ==
                                     [l for l in c.split("\n") if "SELFCHECK" not in l] else "DIFFER"))
    b = rd(dev, "B", "log", "/tmp/opus551f2b")
    if b is not None:
        cells.append("vs MINE_b %s" % ("ID-exSC" if [l for l in b.split("\n") if "SELFCHECK" not in l] ==
                                       [l for l in c.split("\n") if "SELFCHECK" not in l] else "DIFFER"))
    f = rd(dev, "Cfix", "log")
    cells.append("Cfix vs C %s rc %s/%s" % ("WHOLE-ID" if f == c else "DIFFER", rd(dev, "C", "rc").strip(), rd(dev, "Cfix", "rc").strip()))
    print("    %-8s %s" % (dev, " | ".join(cells)))
print("  C's self-check lines:", " | ".join(re.sub(r"\s+", " ", l.strip()) for l in rd("pristine", "C", "log").split("\n")
                                         if "SELFCHECK" in l and "@@" not in l))
print("ANALYZE5_DONE")
