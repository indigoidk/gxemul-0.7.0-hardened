#!/usr/bin/env python3
"""#551c: the self-check also requires the EXACT change in cmp_fails (not only in failures), on top of Cfix."""
import io, sys
W = sys.argv[1]
def sub1(s, a, b):
    if s.count(a) != 1:
        raise SystemExit("anchor count %d: %r" % (s.count(a), a[:70]))
    return s.replace(a, b)
F = io.open(W + "/h/Cfix.c", encoding="utf-8", newline="").read()
P = sub1(F, "\t\tif (failures != sc_f + (cg[i] != cw[i]))\n",
         "\t\tif (failures != sc_f + (cg[i] != cw[i]) || cmp_fails != sc_cf + (cg[i] != cw[i]))\n")
P = sub1(P, "\t\tif (failures != sc_f + (r57[i] != rw[i] || r60[i] != rw[i]))\n",
         "\t\tif (failures != sc_f + (r57[i] != rw[i] || r60[i] != rw[i])\n"
         "\t\t    || cmp_fails != sc_cf + (r57[i] != rw[i] || r60[i] != rw[i]))\n")
CHK_CMP = "\t\tfailures ++;\n\t\tcmp_fails ++;\n\t\tprintf(\"  FAIL %-40s got %lld"
out = {"Cpin": P,
       "Cpin_nocmpinc_reset1": sub1(sub1(P, CHK_CMP, CHK_CMP.replace("\t\tcmp_fails ++;\n", "")),
                                    "\trow_ring_occupancy_sound();\n", "\trow_ring_occupancy_sound();\n\tfailures = 0;\n")}
for k, v in out.items():
    io.open("%s/h/%s.c" % (W, k), "w", encoding="utf-8", newline="").write(v)
print("wrote", " ".join(out))
