void debug(const char *fmt, ...) { (void) fmt; }
/*  Little-endian and SELF-CONSISTENT, which is all that is required: the device
 *  only ever reads back what this driver wrote.  Same bodies as diff_footbridge.c,
 *  which drives its own access handler the same way.  */
uint64_t memory_readmax64(struct cpu *cpu, unsigned char *buf, int len)
{
	uint64_t x = 0; int i;
	(void) cpu;
	for (i = 0; i < len; i++)
		x |= (uint64_t) buf[i] << (8*i);
	return x;
}

void memory_writemax64(struct cpu *cpu, unsigned char *buf, int len, uint64_t data)
{
	int i;
	(void) cpu;
	for (i = 0; i < len; i++)
		buf[i] = (unsigned char) (data >> (8*i));
}

/*  The SCSI layer, reached only by the packet-command path.
 *
 *  DEFAULT INERT, SELECTIVELY PRODUCTIVE.  Most rows here never issue a command -- they
 *  place the state machine one read from the arithmetic under test and then read.  For
 *  those, a scsicommand returning 1 would fabricate a transfer the fixture never asked
 *  for and would overwrite the atapi_st the row set up.
 *
 *  But `row_arm_overrun` MUST issue a real packet command, because the initial arm it
 *  tests is reachable no other way.  So the stub is controlled by a file-static byte
 *  count: zero (the default) keeps it inert for every other row, non-zero makes it
 *  fabricate exactly one DATA-IN transfer of that size.  */
static int stub_datain_len = 0;
static int stub_frees = 0;
static int stub_scsi_cmds = 0;	/*  #477: dispatches ATTEMPTED  */
static int stub_disk_writes = 0, stub_disk_bytes = 0;
static long stub_disk_offset = 0;
static int stub_disk_first = -1;


struct scsi_transfer *scsi_transfer_alloc(void)
{
	return (struct scsi_transfer *) calloc(1, sizeof(struct scsi_transfer));
}

void scsi_transfer_free(struct scsi_transfer *p)
{
	stub_frees ++;
	free(p);
}

int diskimage_scsicommand(struct cpu *cpu, int id, int type,
	struct scsi_transfer *xferp)
{
	size_t i;

	(void) cpu; (void) id; (void) type;

	/*  counted BEFORE the inert early-out, so it records the DISPATCH
	    regardless of whether this stub fabricates a transfer.  */
	stub_scsi_cmds ++;

	if (stub_datain_len <= 0)
		return 0;

	/*  One DATA-IN transfer, marked so the caller can tell WHICH bytes
	    survived: byte 0 alone carries 0x5a, everything else 0xc3.  A ring
	    overrun wraps block 514 onto block 0, so the first byte popped
	    becomes 0xc3 and the marker is destroyed.  */
	xferp->data_in = (unsigned char *) malloc(stub_datain_len);
	if (xferp->data_in == NULL)
		return 0;
	for (i = 0; i < (size_t) stub_datain_len; i++)
		xferp->data_in[i] = 0xc3;
	xferp->data_in[0] = 0x5a;
	xferp->data_in_len = stub_datain_len;
	return 1;
}

static int rows = 0, failures = 0;

/*  A 16-bit IDENTIFY word, assembled the way the transmit path presents it.  */
static uint32_t word(struct wdc_data *d, int w)
{
	return (d->identify_struct[2 * w + 0] << 8) | d->identify_struct[2 * w + 1];
}

static uint64_t sectors_from_block(struct wdc_data *d, int lo, int hi)
{
	return ((uint64_t)word(d, hi) << 16) | word(d, lo);
}

 *  diskimage_getsize, diskimage_is_a_cdrom and diskimage_getname calls). #405
 *  and #407 both passed NULL, and it "worked" only because -O2 inlines the stubs
 *  and deletes the load. MEASURED: the same driver SEGFAULTS at -O0 and at -O1

static struct machine stub_machine;
static struct cpu     stub_cpu;

/*  *** EVERY ROW IN THIS FILE TESTED ONE ENDIANNESS ARM UNTIL 2026-08-27.
 *
 *  `stub_cpu` is memset to zero and nothing ever set `byte_order`, while the
 *  device branches on `cpu->byte_order == EMUL_LITTLE_ENDIAN`.  So the BIG-endian
 *  arm ran for every row -- and for every kill table measured against them.
 *
 *  A seat proved it the only way that settles it: deleting `badlen_this = 1;`
 *  from the LITTLE-endian default alone produced a detector run BYTE-IDENTICAL
 *  to the unmutated one, same md5.  Not one row could tell them apart.
 *
 *  The sting is that the rung-3 witness machine for this defect family,
 *  `evbmips -e malta`, is LITTLE-endian.  The detector was testing the opposite
 *  arm from the machine the defect was witnessed on.
 *
 *  THE DEFAULT IS DELIBERATELY UNCHANGED.  Flipping it would silently re-point
 *  every existing row at the other arm, and any row that then failed would be
 *  indistinguishable from a real defect.  New rows opt IN by setting this.
 */
static int stub_byte_order = 0;	/*  0 = the arm this file always ran  */

static struct cpu *fake_cpu(void)
{
	static int done = 0;
	if (!done) {
		memset(&stub_machine, 0, sizeof(stub_machine));
		memset(&stub_cpu, 0, sizeof(stub_cpu));
		stub_cpu.machine = &stub_machine;
		done = 1;
	}
	stub_cpu.byte_order = stub_byte_order;
	return &stub_cpu;
}

static void build(struct wdc_data *d, uint64_t sectors, int c, int h, int s)
{
	memset(d, 0, sizeof(*d));
	d->cyls[0] = c;
	d->heads[0] = h;
	d->sectors_per_track[0] = s;
	stub_size = sectors * 512;
	stub_size_for_active = 0;			/*  single-drive rows  */
	wdc_initialize_identify_struct(fake_cpu(), d);
}

/*
 *  One row: an exact sector count, asserted through BOTH capacity word pairs
 *  (57-58 "current capacity" and 60-61 "total addressable"). `why` names the
 *  byte this row puts at or past the divergence point -- a row that cannot say
 *  that is a row that would pass unfixed.
 */
/*  #551: the comparison is split out of row() as a PURE comparator, so the self-check can feed it each
    disjunct alone, and a zero, without building a device.  row() and its eight callers are unchanged in
    what they print.  */
static int rowcmp_calls;
static int cmp_fails;	/*  failures the two comparators counted; only they touch it  */

static void row_cmp(const char *why, uint64_t sectors, uint64_t got57, uint64_t got60)
{
	rowcmp_calls ++;
	rows ++;

	if (got57 != sectors || got60 != sectors) {
		printf("  FAIL %-38s sectors=%llu -> 57/58=%llu 60/61=%llu\n",
		    why, (unsigned long long)sectors,
		    (unsigned long long)got57, (unsigned long long)got60);
		failures ++;
		cmp_fails ++;
	} else {
		printf("  ok   %-38s %llu\n", why, (unsigned long long)sectors);
	}
}

static void row(const char *why, uint64_t sectors)
{
	struct wdc_data d;
	uint64_t got57, got60;

	build(&d, sectors, 1, 1, 1);
	got57 = sectors_from_block(&d, 57, 58);
	got60 = sectors_from_block(&d, 60, 61);
	row_cmp(why, sectors, got57, got60);
}

/*
 *  The anchor. Word 53's bytes are 0x00,0x02 -- asymmetric, so a +0/+1 swap in
 *  the packing is visible here. Word 47 (0x8080) is a palindrome and would not
 *  be, which is exactly why it is not used.
 */
static void row_anchor(void)
{
	struct wdc_data d;

	build(&d, 1024, 1, 1, 1);
	rows ++;
	if (d.identify_struct[2 * 53 + 0] != 0x00 ||
	    d.identify_struct[2 * 53 + 1] != 0x02) {
		printf("  FAIL word-53 anchor: %02x %02x, want 00 02"
		    " (a +0/+1 swap in the packing)\n",
		    d.identify_struct[2 * 53 + 0], d.identify_struct[2 * 53 + 1]);
		failures ++;
	} else {
		printf("  ok   %-38s word 53 = 00 02\n", "packing anchor (non-palindromic)");
	}
}

/*
 *  #407: THE SLAVE ROW. Kills the smallest surviving mutant in the file -- ten
 *  characters, deleting `d->drive + ` from dev_wdc.c:179 -- which made drive 1
 *  report drive 0's capacity while still reporting its own geometry. It passed
 *  all ten of the original rows because every one of them left d->drive at 0.
 */
static void row_slave(void)
{
	struct wdc_data d;
	uint64_t master = 100ULL * 16 * 63;	/*  100 cylinders  */
	uint64_t slave  = 300ULL * 16 * 63;	/*  300 cylinders  */
	uint64_t got;
	int bad = 0;

	memset(&d, 0, sizeof(d));
	d.drive = 1;
	d.cyls[0] = 100; d.heads[0] = 16; d.sectors_per_track[0] = 63;
	d.cyls[1] = 300; d.heads[1] = 15; d.sectors_per_track[1] = 17;
	memset(stub_size_for, 0, sizeof(stub_size_for));
	stub_size_for[0] = master * 512;
	stub_size_for[1] = slave  * 512;
	stub_size = 0;				/*  poison: no fallback answer  */
	stub_size_for_active = 1;
	wdc_initialize_identify_struct(fake_cpu(), &d);
	stub_size_for_active = 0;

	rows ++;
	got = sectors_from_block(&d, 60, 61);
	if (got != slave) {
		printf("  FAIL %-38s drive 1 says %llu, want %llu\n",
		    "the SLAVE reports its own capacity",
		    (unsigned long long)got, (unsigned long long)slave);
		bad = 1;
	}

	/*
	 *  #408: THE GEOMETRY HALF, which #407 left open. It closed the mutant that
	 *  made the slave report the master's CAPACITY and never checked that the
	 *  slave reports its own CYLINDERS, HEADS and SECTORS. Measured: four
	 *  mutants -- cyls[d->drive]->cyls[0], heads[...]->heads[0],
	 *  sectors_per_track[...]->[0], on both bytes -- all SURVIVED the 12 rows.
	 *  Master and slave are given DELIBERATELY DIFFERENT geometries (16/63 vs
	 *  15/17) so that reading the wrong drive's field cannot coincide.
	 */
	if (word(&d, 1) != 300 || word(&d, 3) != 15 || word(&d, 6) != 17) {
		printf("  FAIL %-38s drive 1 reports c=%u h=%u s=%u, want 300/15/17\n",
		    "the SLAVE reports its own geometry",
		    word(&d, 1), word(&d, 3), word(&d, 6));
		bad = 1;
	}

	if (bad)
		failures ++;
	else
		printf("  ok   %-38s %llu sectors, c=300 h=15 s=17\n",
		    "the SLAVE reports its own capacity+geometry",
		    (unsigned long long)got);
}

/*
 *  #408: base_drive IS NOT HYPOTHETICAL -- dev_wdc_init() sets it to 2 for the
 *  SECONDARY controller (the 0x170 address arm), so every access on that
 *  controller adds it. #407's slave row never varied it, and MEASURED, dropping
 *  `+ d->base_drive` from all three call sites SURVIVED every row. Here drive 0
 *  of a base_drive=2 controller must read diskimage id 2, and ids 0/1 are poison
 *  so borrowing the primary controller's disk cannot look correct.
 */
static void row_base_drive(void)
{
	struct wdc_data d;
	uint64_t want = 500ULL * 16 * 63;
	uint64_t got;

	int bad = 0;

	memset(&d, 0, sizeof(d));
	d.drive = 0;
	d.base_drive = 2;
	d.cyls[0] = 500; d.heads[0] = 16; d.sectors_per_track[0] = 63;
	memset(stub_size_for, 0, sizeof(stub_size_for));
	stub_size_for[2] = want * 512;		/*  ids 0,1,3.. stay poison  */
	stub_size = 0;
	stub_size_for_active = 1;
	wdc_initialize_identify_struct(fake_cpu(), &d);
	stub_size_for_active = 0;

	rows ++;
	got = sectors_from_block(&d, 60, 61);
	if (got != want) {
		printf("  FAIL %-38s got %llu, want %llu (base_drive ignored?)\n",
		    "base_drive reaches the diskimage id",
		    (unsigned long long)got, (unsigned long long)want);
		bad = 1;
	}

	/*
	 *  #409: THE GEOMETRY HALF, and it kills an OUT-OF-BOUNDS READ.
	 *
	 *  base_drive is the DISKIMAGE id, not an index into the per-drive arrays,
	 *  which are `int cyls[2]`. A mutant indexing them as [d->drive +
	 *  d->base_drive] reads element 2 of a 2-element array -- and it SURVIVED
	 *  all five optimisation levels, because nothing asserted this drive's
	 *  geometry. Only ASan caught it. Asserting the geometry here kills it
	 *  without needing a sanitizer in the gate.
	 */
	if (word(&d, 1) != 500 || word(&d, 3) != 16 || word(&d, 6) != 63) {
		printf("  FAIL %-38s c=%u h=%u s=%u, want 500/16/63"
		    " (indexed by base_drive?)\n",
		    "base_drive is a disk id, not an array index",
		    word(&d, 1), word(&d, 3), word(&d, 6));
		bad = 1;
	}

	if (bad)
		failures ++;
	else
		printf("  ok   %-38s id 2 -> %llu sectors, c=500 h=16 s=63\n",
		    "base_drive reaches the diskimage id",
		    (unsigned long long)got);
}

/*
 *  #473 `wdcrearmzero`: the DATAIN re-arm reports ZERO remaining mid-transfer.
 *
 *  `if (atapi_len > 32768) atapi_len = 0;` at the re-arm, where the INITIAL arm clamps
 *  to 32768 in the identical condition.  With more than 32768 bytes remaining the guest
 *  is told zero remain, and there is no recovery: the next read does
 *  `atapi_len -= consumed` from 0, steps PAST zero, and `== 0` never fires again --
 *  precisely the #341 class this file's own comment describes, reintroduced by the
 *  re-arm.
 *
 *  MEASURED BAND [65537, 262655].  65536 completes (remaining is exactly 32768, so the
 *  `> 32768` test is false); 262656 also trips the ring overrun, so it is not purely
 *  this defect.
 *
 *  GUEST-VISIBLE at the first re-arm, read back through the handler: bytecount 0,
 *  STATUS 0x58 (DRDY|DSC|DRQ), ireason 0x02 -- "data-in phase, DRQ asserted, zero bytes
 *  to transfer".
 *
 *  THE IRQ SINK IS REQUIRED, and its absence is not a field omission: dev_wdc_tick() runs
 *  unconditionally after every access and calls INTERRUPT_ASSERT(d->irq).  DEVINIT wires
 *  that with INTERRUPT_CONNECT -- a macro, invisible to a scan for field assignments.
 *  Two earlier attempts died there and were removed rather than shipped as false reds.
 *
 *  THE STATE IS SYNTHESISED AND SAYS SO.  Under the SHIPPED code the machine breaks at
 *  the FIRST re-arm (received = 32768), so no real trajectory reaches these values; they
 *  are the arithmetic under test placed one read from firing.  131070 is on the FIXED
 *  code's trajectory (end of chunk 4), chosen so the row reads as a plausible state
 *  rather than an arbitrary one.
 */
static int rearm_asserts = 0;
static void rearm_sink(struct interrupt *a) { (void)a; rearm_asserts++; }

static void row_rearm_zero(void)
{
	struct wdc_data d;
	struct scsi_transfer st;
	unsigned char buf[4];
	unsigned char *payload;
	int occ;

	memset(&d, 0, sizeof(d));
	d.addr_mult  = 1;		/*  non-zero: :678 divides by it (SIGFPE otherwise)  */
	d.io_enabled = 1;		/*  or :680 goto ret skips the whole switch  */
	d.inbuf = (unsigned char *) calloc(1, WDC_INBUF_SIZE);
	d.irq.interrupt_assert = rearm_sink;	/*  what INTERRUPT_CONNECT would wire  */
	if (d.inbuf == NULL) { printf("  SETUP inbuf alloc failed\n"); return; }

	payload = (unsigned char *) calloc(1, 200000);
	if (payload == NULL) { free(d.inbuf); printf("  SETUP payload failed\n"); return; }

	/*  MARKERS, so the refill can be caught taking bytes from the WRONG place.
	    atapi_received advances BEFORE the re-arm (`received += consumed`), so
	    with 131070 received and 2 consumed the refill must start at 131072.
	    A refill that ignores atapi_received starts at 0 and yields 0xaa.  */
	payload[0]      = 0xaa;
	payload[131072] = 0x3c;

	memset(&st, 0, sizeof(st));
	st.data_in = payload;
	st.data_in_len = 200000;

	d.atapi_st = &st;
	d.atapi_phase = PHASE_DATAIN;
	d.atapi_cmd_in_progress = 1;
	d.atapi_received = 131070;	/*  on the FIXED code's trajectory  */
	d.atapi_len = 2;		/*  this read drains the chunk  */

	wdc_addtoinbuf(&d, 0x5a);
	wdc_addtoinbuf(&d, 0x5a);

	rearm_asserts = 0;
	memset(buf, 0, sizeof(buf));
	dev_wdc_access(fake_cpu(), NULL, wd_data, buf, 2, MEM_READ, &d);

	/*  68930 remain, so a correct re-arm reports min(remaining, 32768) = 32768.
	    The committed code reports 0 and the command never completes.  */
	rows ++;
	if (d.atapi_len != 32768) {
		printf("  FAIL %-38s atapi_len = %i, want 32768\n",
		    "re-arm: >32768 left is not zeroed", d.atapi_len);
		failures ++;
	} else {
		printf("  ok   %-38s %i\n",
		    "re-arm: >32768 left is not zeroed", d.atapi_len);
	}

	/*  FREE SECOND ORACLE: proves the re-arm BRANCH was taken, not merely that the
	    call survived.  A fixture that silently missed the branch would otherwise
	    look identical to one that took it and measured the wrong thing.  */
	rows ++;
	if (rearm_asserts != 1) {
		printf("  FAIL %-38s asserts = %i, want 1\n",
		    "re-arm: the branch was actually taken", rearm_asserts);
		failures ++;
	} else {
		printf("  ok   %-38s %i\n",
		    "re-arm: the branch was actually taken", rearm_asserts);
	}

	/*  *** THESE TWO ROWS CLOSE A MEASURED FLOOR CLAUSE (d) BREACH. ***  The length
	    check above passes under BOTH of these mutants, each of which keeps the
	    32768 guard and nulls its repair:

	      Md_norefill   delete the refill loop -- the guest is told 32768 bytes are
	                    ready and the ring is EMPTY
	      Md_wrongoff   refill from data_in[j] -- every chunk returns chunk 0 again,
	                    which is silent data corruption

	    The arm row already carried a corruption oracle and this one did not: the
	    sibling-parity check exists precisely to catch that asymmetry.  */
	occ = d.inbuf_head - d.inbuf_tail;
	while (occ < 0)
		occ += WDC_INBUF_SIZE;
	rows ++;
	if (occ != 32768) {
		printf("  FAIL %-38s occupancy = %i, want 32768\n",
		    "re-arm: the chunk is REFILLED", occ);
		failures ++;
	} else {
		printf("  ok   %-38s %i\n",
		    "re-arm: the chunk is REFILLED", occ);
	}

	rows ++;
	if (d.inbuf[d.inbuf_tail] != 0x3c) {
		printf("  FAIL %-38s first byte = 0x%02x, want 0x3c\n",
		    "re-arm: refill starts at received", d.inbuf[d.inbuf_tail]);
		failures ++;
	} else {
		printf("  ok   %-38s 0x%02x\n",
		    "re-arm: refill starts at received", d.inbuf[d.inbuf_tail]);
	}

	/*  *** THE REFILL IS NOT CONDITIONAL ON THE RING BEING EMPTY, AND A CLEAN
	    12-MUTANT TABLE DID NOT SHOW THAT. ***  Two panel seats independently
	    broke the shipped comment's claim that the ring IS empty here: the
	    data-port WRITE arm pushes guest bytes with no phase guard, and its
	    unexpected-write latch is suppressed precisely because
	    atapi_cmd_in_progress is set.  So a guest that writes mid-DATAIN grows
	    head without touching atapi_len.

	    One seat then turned that sentence into a mutant -- `if (inbuf_head ==
	    inbuf_tail)` around the refill -- and it SWEPT the detector clean, 23
	    rows 0 failures, because every other row leaves the ring empty here.
	    A guard built from a round's own safety argument is invisible to rows
	    written under that same argument.

	    This row is the only one that establishes a NON-EMPTY ring at the
	    re-arm.  It pins behaviour, not desirability: whether refilling onto a
	    dirty ring is RIGHT is a separate question, filed as
	    `wdcaddtoinbufbound`.  What must not happen silently is the refill
	    being skipped while the guest is still told 32768 bytes are ready.  */
	{
		struct wdc_data e;
		struct scsi_transfer es;
		unsigned char ebuf[4], *epay;
		int eocc, i;

		memset(&e, 0, sizeof(e));
		e.addr_mult = 1;  e.io_enabled = 1;
		e.inbuf = (unsigned char *) calloc(1, WDC_INBUF_SIZE);
		e.irq.interrupt_assert = rearm_sink;
		epay = (unsigned char *) calloc(1, 200000);
		if (e.inbuf == NULL || epay == NULL) {
			printf("  SETUP dirty-ring alloc failed\n");
			free(e.inbuf); free(epay);
		} else {
			memset(&es, 0, sizeof(es));
			es.data_in = epay;  es.data_in_len = 200000;
			e.atapi_st = &es;  e.atapi_phase = PHASE_DATAIN;
			e.atapi_cmd_in_progress = 1;
			e.atapi_received = 131070;
			e.atapi_len = 2;

			/*  the two chunk bytes the guest is about to drain...  */
			wdc_addtoinbuf(&e, 0x5a);
			wdc_addtoinbuf(&e, 0x5a);
			/*  ...and FIVE bytes behind them that atapi_len knows nothing
			    about, which is exactly what an interleaved data-port write
			    leaves in the ring.  */
			for (i = 0; i < 5; i++)
				wdc_addtoinbuf(&e, 0x77);

			memset(ebuf, 0, sizeof(ebuf));
			dev_wdc_access(fake_cpu(), NULL, wd_data, ebuf, 2, MEM_READ, &e);

			eocc = e.inbuf_head - e.inbuf_tail;
			while (eocc < 0)
				eocc += WDC_INBUF_SIZE;

			/*  5 leftover + 32768 refilled.  A refill skipped on a non-empty
			    ring leaves 5, and the guest is told 32768 are waiting.  */
			rows ++;
			if (eocc != 5 + 32768) {
				printf("  FAIL %-38s occupancy = %i, want %i\n",
				    "re-arm: refills a NON-EMPTY ring too", eocc, 5 + 32768);
				failures ++;
			} else {
				printf("  ok   %-38s %i\n",
				    "re-arm: refills a NON-EMPTY ring too", eocc);
			}
			free(epay);  free(e.inbuf);
		}
	}

	free(payload);
	free(d.inbuf);
}

static void arm_sink(struct interrupt *a) { (void)a; arm_asserts++; }

/*  *** THE FIFTH CLAUSE-(d) MUTANT, AND IT ESCAPED THE TABLE THAT HAD ALREADY

/*  PROPOSED ROW -- kills the fifth (d) mutant, and it is a SINGLE-RE-ARM
 *  trajectory, i.e. exactly the shape the shipped contract says IS in scope.

 *  LARGER than 32768, so `atapi_len` and the constant 32768 are the same

 *  chunk and the refill's bound becomes observable.

static int q2_asserts = 0;
static void q2_sink(struct interrupt *a) { (void)a; q2_asserts++; }

static void row_q2_short_last_chunk(void)
{
	struct wdc_data d;
	struct scsi_transfer st;
	unsigned char buf[4];
	unsigned char *payload;
	int occ;

	memset(&d, 0, sizeof(d));
	d.addr_mult  = 1;
	d.io_enabled = 1;
	d.inbuf = (unsigned char *) calloc(1, WDC_INBUF_SIZE);
	d.irq.interrupt_assert = q2_sink;
	if (d.inbuf == NULL) { printf("  SETUP q2 inbuf failed\n"); return; }

	/*  EXACT allocation, no slack: an over-read past data_in_len is then a real
	    heap-buffer-overflow rather than a read into padding.  */
	payload = (unsigned char *) malloc(40000);
	if (payload == NULL) { free(d.inbuf); printf("  SETUP q2 payload failed\n"); return; }
	memset(payload, 0xc3, 40000);
	payload[35002] = 0x7e;

	memset(&st, 0, sizeof(st));
	st.data_in = payload;
	st.data_in_len = 40000;

	d.atapi_st = &st;
	d.atapi_phase = PHASE_DATAIN;
	d.atapi_cmd_in_progress = 1;
	d.atapi_received = 35000;	/*  4998 will remain: a SHORT last chunk  */
	d.atapi_len = 2;

	wdc_addtoinbuf(&d, 0x5a);
	wdc_addtoinbuf(&d, 0x5a);

	q2_asserts = 0;
	memset(buf, 0, sizeof(buf));
	dev_wdc_access(fake_cpu(), NULL, wd_data, buf, 2, MEM_READ, &d);

	occ = d.inbuf_head - d.inbuf_tail;
	while (occ < 0)
		occ += WDC_INBUF_SIZE;

	rows ++;
	if (d.atapi_len != 4998) {
		printf("  FAIL %-38s atapi_len = %i, want 4998\n",
		    "last chunk: length is the REMAINDER", d.atapi_len);
		failures ++;
	} else {
		printf("  ok   %-38s %i\n",
		    "last chunk: length is the REMAINDER", d.atapi_len);
	}

	/*  THE DISCRIMINATOR.  A refill whose bound is the CONSTANT 32768 rather than
	    atapi_len pushes 32768 here and reads 27770 bytes past data_in.  */
	rows ++;
	if (occ != 4998) {
		printf("  FAIL %-38s occupancy = %i, want 4998 (refill bound is not atapi_len)\n",
		    "last chunk: refill is BOUNDED by it", occ);
		failures ++;
	} else {
		printf("  ok   %-38s %i\n",
		    "last chunk: refill is BOUNDED by it", occ);
	}

	free(payload);
	free(d.inbuf);
}


/*
 *  `wdcoddstep`: `atapi_len -= consumed` STEPS PAST ZERO on an odd-length chunk.
 *
 *  `consumed` is `len == 4 ? 4 : (len >= 2 ? 2 : 1)` -- what the reads actually POP, which
 *  is what #341 corrected.  But nothing clamps it to what REMAINS, so a four-byte read of
 *  a three-byte remainder drives atapi_len to -1 and atapi_received to 4 of 3.
 *
 *  TWO ROWS, KEPT APART ON PURPOSE.  The completion half and the truthfulness half are
 *  repaired by DIFFERENT candidate fixes: saturating atapi_len (or testing `<= 0`) fixes
 *  completion and leaves atapi_received over-counted; clamping `consumed` fixes both.  A
 *  single combined row could not distinguish them, and the fork's own precedents (#283
 *  ASC, #338 mb89352, #341 here) all hold that the count reported has to be the count that
 *  happened.
 *
 *  SMALL ODD TRANSFER, NO RE-ARM INVOLVED -- reachable by MODE SENSE / REQUEST SENSE and
 *  NOT a consequence of #473, which only added a second road via an odd FINAL chunk.
 */
static int odd_asserts = 0;
static void odd_sink(struct interrupt *a) { (void)a; odd_asserts++; }

static void row_odd_step(void)
{
	struct wdc_data d;
	struct scsi_transfer st;
	unsigned char buf[8];
	unsigned char *payload;

	memset(&d, 0, sizeof(d));
	d.addr_mult  = 1;
	d.io_enabled = 1;
	d.inbuf = (unsigned char *) calloc(1, WDC_INBUF_SIZE);
	d.irq.interrupt_assert = odd_sink;
	if (d.inbuf == NULL) { printf("  SETUP odd inbuf failed\n"); return; }

	payload = (unsigned char *) malloc(3);
	if (payload == NULL) { free(d.inbuf); printf("  SETUP odd payload failed\n"); return; }
	payload[0] = 0x11; payload[1] = 0x22; payload[2] = 0x33;

	memset(&st, 0, sizeof(st));
	st.data_in = payload;
	st.data_in_len = 3;

	d.atapi_st = &st;
	d.atapi_phase = PHASE_DATAIN;
	d.atapi_cmd_in_progress = 1;
	d.atapi_received = 0;
	d.atapi_len = 3;			/*  ODD  */

	/*  exactly three bytes queued: the fourth pop hits an empty ring, latches
	    warned_inbuf_overread, returns -1 and does NOT advance the tail.  */
	wdc_addtoinbuf(&d, 0x11);
	wdc_addtoinbuf(&d, 0x22);
	wdc_addtoinbuf(&d, 0x33);

	odd_asserts = 0;
	memset(buf, 0, sizeof(buf));
	dev_wdc_access(fake_cpu(), NULL, wd_data, buf, 4, MEM_READ, &d);

	/*  COMPLETION.  Today -1: `== 0` never fires again, so the command never
	    completes and the guest waits forever.  */
	rows ++;
	if (d.atapi_len != 0) {
		printf("  FAIL %-38s atapi_len = %i, want 0\n",
		    "odd chunk: the count stops at zero", d.atapi_len);
		failures ++;
	} else {
		printf("  ok   %-38s %i\n",
		    "odd chunk: the count stops at zero", d.atapi_len);
	}

	/*  TRUTHFULNESS.  Today 4 where data_in_len is 3.  Separate because the cheaper
	    fix options repair the row above and leave this one red.  */
	rows ++;
	if (d.atapi_received != st.data_in_len) {
		printf("  FAIL %-38s received = %i, want %i\n",
		    "odd chunk: received never exceeds len",
		    (int) d.atapi_received, (int) st.data_in_len);
		failures ++;
	} else {
		printf("  ok   %-38s %i\n",
		    "odd chunk: received never exceeds len", (int) d.atapi_received);
	}

	free(payload);
	free(d.inbuf);
}


/*  THE INTERMEDIATE-STATE ROW, and the two rows above do not imply it.
 *
 *  A panel seat named the escape: degrade the clamp from a MIN to a plain assignment,
 *  `consumed = d->atapi_len;`.  A plausible future simplification -- and both end-state
 *  rows still pass, because the transfer does end at 0 of 37.  What changes is WHEN: the
 *  FIRST four-byte read swallows the whole transfer and the command completes after four
 *  bytes were actually delivered.
 *
 *  So the detector needs a row that observes the walk MID-TRANSFER rather than at its end.
 *  One four-byte read of a 37-byte transfer must leave exactly 33.
 */
static void row_odd_midwalk(void)
{
	struct wdc_data d;
	struct scsi_transfer st;
	unsigned char buf[8];
	unsigned char *payload;
	int i;

	memset(&d, 0, sizeof(d));
	d.addr_mult  = 1;
	d.io_enabled = 1;
	d.inbuf = (unsigned char *) calloc(1, WDC_INBUF_SIZE);
	d.irq.interrupt_assert = odd_sink;
	if (d.inbuf == NULL) { printf("  SETUP midwalk inbuf failed\n"); return; }

	/*  37 -- the guest-chosen INQUIRY allocation length from the worked path.  */
	payload = (unsigned char *) malloc(37);
	if (payload == NULL) { free(d.inbuf); printf("  SETUP midwalk payload failed\n"); return; }
	for (i = 0; i < 37; i++)
		payload[i] = (unsigned char) i;

	memset(&st, 0, sizeof(st));
	st.data_in = payload;
	st.data_in_len = 37;

	d.atapi_st = &st;
	d.atapi_phase = PHASE_DATAIN;
	d.atapi_cmd_in_progress = 1;
	d.atapi_received = 0;
	d.atapi_len = 37;
	for (i = 0; i < 37; i++)
		wdc_addtoinbuf(&d, payload[i]);

	memset(buf, 0, sizeof(buf));
	dev_wdc_access(fake_cpu(), NULL, wd_data, buf, 4, MEM_READ, &d);

	/*  Correct: consumed is 4, so 33 remain.  Under `consumed = atapi_len` the whole
	    transfer is swallowed in one read and this reads 0 -- while both end-state rows
	    above stay green, because the transfer does finish at 0 of 37.  */
	rows ++;
	if (d.atapi_len != 33) {
		printf("  FAIL %-38s atapi_len = %i, want 33\n",
		    "mid-walk: one 4-byte read leaves 33", d.atapi_len);
		failures ++;
	} else {
		printf("  ok   %-38s %i\n",
		    "mid-walk: one 4-byte read leaves 33", d.atapi_len);
	}

	rows ++;
	if (d.atapi_received != 4) {
		printf("  FAIL %-38s received = %i, want 4\n",
		    "mid-walk: cursor advances by 4", (int) d.atapi_received);
		failures ++;
	} else {
		printf("  ok   %-38s %i\n",
		    "mid-walk: cursor advances by 4", (int) d.atapi_received);
	}

	free(payload);
	free(d.inbuf);
}


/*  THE WIDTH ROW.  Named by the codex seat on pass 1 and MEASURED to escape everything
 *  above: narrow the clamp to `len == 4 && consumed > atapi_len` and every other row in
 *  this file stays green, because every other row issues a FOUR-byte read.
 *
 *  This one issues a TWO-byte read with ONE byte remaining, and it sits on the same
 *  INQUIRY-37 trajectory as the mid-walk row: nine 4-byte reads leave exactly 1.
 *
 *  A detector whose rows all use one access width cannot see a fix narrowed to that
 *  width -- which is the padded-column trap's cousin: the row is present, runs, and is
 *  structurally unable to fail.
 */
static void row_odd_width(void)
{
	struct wdc_data d;
	struct scsi_transfer st;
	unsigned char buf[8];
	unsigned char *payload;
	int i;

	memset(&d, 0, sizeof(d));
	d.addr_mult  = 1;
	d.io_enabled = 1;
	d.inbuf = (unsigned char *) calloc(1, WDC_INBUF_SIZE);
	d.irq.interrupt_assert = odd_sink;
	if (d.inbuf == NULL) { printf("  SETUP width inbuf failed\n"); return; }

	payload = (unsigned char *) malloc(37);
	if (payload == NULL) { free(d.inbuf); printf("  SETUP width payload failed\n"); return; }
	for (i = 0; i < 37; i++)
		payload[i] = (unsigned char) i;

	memset(&st, 0, sizeof(st));
	st.data_in = payload;
	st.data_in_len = 37;

	d.atapi_st = &st;
	d.atapi_phase = PHASE_DATAIN;
	d.atapi_cmd_in_progress = 1;
	d.atapi_received = 36;		/*  nine 4-byte reads have happened  */
	d.atapi_len = 1;		/*  ...leaving exactly one  */
	wdc_addtoinbuf(&d, payload[36]);

	memset(buf, 0, sizeof(buf));
	dev_wdc_access(fake_cpu(), NULL, wd_data, buf, 2, MEM_READ, &d);

	/*  consumed is 2 for a 2-byte read; the clamp must cut it to 1.  A clamp that
	    only fires on len == 4 leaves atapi_len at -1 here.  */
	rows ++;
	if (d.atapi_len != 0) {
		printf("  FAIL %-38s atapi_len = %i, want 0\n",
		    "width: a 2-byte read of 1 remaining", d.atapi_len);
		failures ++;
	} else {
		printf("  ok   %-38s %i\n",
		    "width: a 2-byte read of 1 remaining", d.atapi_len);
	}

	rows ++;
	if (d.atapi_received != 37) {
		printf("  FAIL %-38s received = %i, want 37\n",
		    "width: received stops at data_in_len", (int) d.atapi_received);
		failures ++;
	} else {
		printf("  ok   %-38s %i\n",
		    "width: received stops at data_in_len", (int) d.atapi_received);
	}

	free(payload);
	free(d.inbuf);
}


/*  THE MULTI-CHUNK ROW.  This is a FLOOR fix, not strengthening.
 *
 *  The Opus seat built `Md5_xferbound` and MEASURED it past every row here: clamp to the
 *  TRANSFER remainder (`data_in_len - atapi_received`) instead of the CHUNK remainder
 *  (`atapi_len`).  It reads MORE correct, and on any single-chunk transfer it IS
 *  identical -- so it is, in that seat's words, the mutant a helpful future reviewer
 *  writes.  It diverges only at a chunk boundary, where it clamps nothing and the
 *  original defect returns.
 *
 *  *** AND IT EXPOSED A FRAMING ERROR IN THIS WHOLE ROUND. ***  "odd chunk" is the wrong
 *  name for the class.  The defect is the remainder not being a MULTIPLE OF THE ACCESS
 *  WIDTH -- so a perfectly EVEN 32768-byte chunk overshoots too, under mixed widths.
 *  Every row above tests one instance of the class; this one tests the other.
 *
 *  The state is placed at the boundary rather than walked to it: 32767 of 40000 received
 *  with 1 left in the chunk, then a 4-byte read.  Markers at data_in[32768] and [32769]
 *  make an off-by-one in the refill's source offset visible as well.
 */
static void row_odd_multichunk(void)
{
	struct wdc_data d;
	struct scsi_transfer st;
	unsigned char buf[8];
	unsigned char *payload;
	int occ;

	memset(&d, 0, sizeof(d));
	d.addr_mult  = 1;
	d.io_enabled = 1;
	d.inbuf = (unsigned char *) calloc(1, WDC_INBUF_SIZE);
	d.irq.interrupt_assert = odd_sink;
	if (d.inbuf == NULL) { printf("  SETUP multichunk inbuf failed\n"); return; }

	payload = (unsigned char *) calloc(1, 40000);
	if (payload == NULL) { free(d.inbuf); printf("  SETUP multichunk payload failed\n"); return; }
	payload[32768] = 0xa1;		/*  where chunk 2 MUST start  */
	payload[32769] = 0xb2;		/*  one past -- catches an off-by-one  */

	memset(&st, 0, sizeof(st));
	st.data_in = payload;
	st.data_in_len = 40000;

	d.atapi_st = &st;
	d.atapi_phase = PHASE_DATAIN;
	d.atapi_cmd_in_progress = 1;
	d.atapi_received = 32767;	/*  one byte left in chunk 1...  */
	d.atapi_len = 1;		/*  ...of a 40000-byte transfer  */
	wdc_addtoinbuf(&d, 0x5a);

	memset(buf, 0, sizeof(buf));
	dev_wdc_access(fake_cpu(), NULL, wd_data, buf, 4, MEM_READ, &d);

	/*  Correct: consumed clamps to 1, atapi_len reaches 0, the re-arm fires and
	    refills min(40000 - 32768, 32768) = 7232.  Under Md5_xferbound the clamp
	    compares against 7233 remaining in the TRANSFER, does not fire, and
	    atapi_len lands on -3 -- so `== 0` never fires and no re-arm happens.  */
	rows ++;
	if (d.atapi_len != 7232) {
		printf("  FAIL %-38s atapi_len = %i, want 7232\n",
		    "chunk edge: the re-arm still fires", d.atapi_len);
		failures ++;
	} else {
		printf("  ok   %-38s %i\n",
		    "chunk edge: the re-arm still fires", d.atapi_len);
	}

	/*  The ANCHORED half: chunk 2 must begin at data_in[32768], not [32769].  */
	occ = d.inbuf_head - d.inbuf_tail;
	while (occ < 0)
		occ += WDC_INBUF_SIZE;
	rows ++;
	if (occ != 7232 || d.inbuf[d.inbuf_tail] != 0xa1) {
		printf("  FAIL %-38s occ = %i, first = 0x%02x\n",
		    "chunk edge: chunk 2 starts at 32768", occ, d.inbuf[d.inbuf_tail]);
		failures ++;
	} else {
		printf("  ok   %-38s 0x%02x\n",
		    "chunk edge: chunk 2 starts at 32768", d.inbuf[d.inbuf_tail]);
	}

	free(payload);
	free(d.inbuf);
}


/*  THE GUEST-VISIBLE ROW, and it is the only one in this file.
 *
 *  Every other row here reads DEVICE STATE.  A guest cannot see `atapi_len`; what it CAN
 *  see is the ATAPI byte-count-limit pair, served straight from it:
 *
 *        int x = d->atapi_len;
 *        if (x > 32768) x = 32768;
 *        odata = x & 255;            (CYL_LO)  /  (x >> 8) & 255  (CYL_HI)
 *
 *  So a negative count reaches the guest as 0xffff.  The Opus seat measured that this is
 *  the row separating a real fix from one that only repairs completion: an option that
 *  saturates the TEST but leaves atapi_len negative lets the command finish while the
 *  guest is still told 65535 bytes remain.
 *
 *  cur_command stays 0 so the COMMAND_RESET arm short-circuits before
 *  diskimage_is_a_cdrom(), which this harness does not stub.
 */
static void row_odd_bytecount(void)
{
	struct wdc_data d;
	struct scsi_transfer st;
	unsigned char buf[8];
	unsigned char *payload;
	uint64_t lo, hi;

	memset(&d, 0, sizeof(d));
	d.addr_mult  = 1;
	d.io_enabled = 1;
	d.inbuf = (unsigned char *) calloc(1, WDC_INBUF_SIZE);
	d.irq.interrupt_assert = odd_sink;
	if (d.inbuf == NULL) { printf("  SETUP bytecount inbuf failed\n"); return; }
	payload = (unsigned char *) malloc(3);
	if (payload == NULL) { free(d.inbuf); printf("  SETUP bytecount payload failed\n"); return; }
	payload[0] = 0x11; payload[1] = 0x22; payload[2] = 0x33;

	memset(&st, 0, sizeof(st));
	st.data_in = payload;
	st.data_in_len = 3;
	d.atapi_st = &st;
	d.atapi_phase = PHASE_DATAIN;
	d.atapi_cmd_in_progress = 1;
	d.atapi_received = 0;
	d.atapi_len = 3;
	wdc_addtoinbuf(&d, 0x11);
	wdc_addtoinbuf(&d, 0x22);
	wdc_addtoinbuf(&d, 0x33);

	memset(buf, 0, sizeof(buf));
	dev_wdc_access(fake_cpu(), NULL, wd_data, buf, 4, MEM_READ, &d);

	memset(buf, 0, sizeof(buf));
	dev_wdc_access(fake_cpu(), NULL, wd_cyl_lo, buf, 1, MEM_READ, &d);
	lo = memory_readmax64(fake_cpu(), buf, 1);
	memset(buf, 0, sizeof(buf));
	dev_wdc_access(fake_cpu(), NULL, wd_cyl_hi, buf, 1, MEM_READ, &d);
	hi = memory_readmax64(fake_cpu(), buf, 1);

	rows ++;
	if (lo != 0 || hi != 0) {
		printf("  FAIL %-38s guest reads 0x%02x%02x, want 0\n",
		    "guest byte count is not 0xffff", (int) hi, (int) lo);
		failures ++;
	} else {
		printf("  ok   %-38s 0x%02x%02x\n",
		    "guest byte count is not 0xffff", (int) hi, (int) lo);
	}

	free(payload);
	free(d.inbuf);
}

/*
 *  *** `#475` SHIPPED A GUEST-REACHABLE HOST KILL, AND THIS ROW IS THE BISECT. ***
 *
 *  `#475`'s whole code change is `if (consumed > d->atapi_len) consumed = d->atapi_len;`.
 *  Before it, a data-port READ with `atapi_len == 0` drove the counter NEGATIVE and the
 *  `== 0` test never fired again.  After it, `consumed` clamps to 0, `atapi_len` STAYS 0,
 *  the test FIRES -- and the next line is
 *
 *        if (d->atapi_received < d->atapi_st->data_in_len)
 *
 *  with `atapi_st == NULL`.  The only NULL check on this pointer anywhere in the file is
 *  in the dispatch, hundreds of lines away.
 *
 *  TWO REGISTER ACCESSES FROM A FRESH DEVICE: write ATAPI_PKT_CMD to COMMAND, then read
 *  DATA.  A panel seat measured `SEGV on unknown address 0x48` -- which is
 *  `offsetof(struct scsi_transfer, data_in_len)` from NULL -- and confirmed the pre-`#475`
 *  blob survives the same input.
 *
 *  *A fix that stops a counter going negative made a previously-unreachable branch
 *  reachable.  The clamp is right; the branch it enabled was never NULL-safe.*
 *
 *  THE ROW FORKS, because the defect is a SIGNAL: an in-process oracle would take the
 *  whole differential down with it and report nothing.
 */
static void row_null_atapi_st(void)
{
	pid_t pid;
	int st = 0;

	fflush(stdout);
	pid = fork();
	if (pid == 0) {
		struct wdc_data d;
		unsigned char buf[4];
		memset(&d, 0, sizeof(d));
		d.addr_mult  = 1;
		d.io_enabled = 1;
		d.inbuf = (unsigned char *) calloc(1, WDC_INBUF_SIZE);
		d.irq.interrupt_assert = arm_sink;
		/*  exactly what the two guest accesses leave behind: a packet command
		    announced, no transfer yet allocated.  */
		d.atapi_cmd_in_progress = 1;
		d.atapi_phase = PHASE_DATAIN;
		d.atapi_st = NULL;
		d.atapi_len = 0;
		d.atapi_received = 0;
		memset(buf, 0, sizeof(buf));
		dev_wdc_access(fake_cpu(), NULL, wd_data, buf, 2, MEM_READ, &d);
		_exit(0);
	}
	if (pid < 0) { printf("  SETUP fork failed\n"); return; }
	waitpid(pid, &st, 0);

	rows ++;
	if (WIFSIGNALED(st)) {
		printf("  FAIL %-38s host killed, signal %i\n",
		    "a DATA read with no transfer survives", WTERMSIG(st));
		failures ++;
	} else {
		printf("  ok   %-38s exit %i\n",
		    "a DATA read with no transfer survives", WEXITSTATUS(st));
	}
}


/*
 *  `wdccdbmidxfer`: a mid-DATAIN data-port WRITE can FREE THE IN-FLIGHT TRANSFER.
 *
 *  The packet dispatch fires on ring occupancy == 12, gated only by
 *  `atapi_cmd_in_progress` -- which stays 1 for the whole of DATAIN.  The data-port WRITE
 *  arm pushes guest bytes with NO PHASE CHECK, so a guest that writes twelve bytes while a
 *  DATAIN transfer is live drives occupancy to exactly 12 and re-enters the dispatch:
 *
 *        if (d->atapi_st != NULL)
 *                scsi_transfer_free(d->atapi_st);     <- the LIVE transfer
 *        d->atapi_st = scsi_transfer_alloc();
 *
 *  *** AN EARLIER VERSION OF THIS COMMENT CALLED THAT A USE-AFTER-FREE.  A PANEL SEAT
 *  READ IT FALSE AND IT IS RIGHT. ***  After the free the code allocates a replacement
 *  and then RESETS `atapi_len = 0` and `atapi_received = 0` on both exits -- the reject
 *  path and the normal one.  The cursor is re-based onto the NEW transfer inside the
 *  same access, so no later read dangles through `data_in[atapi_received]`.
 *
 *  *A hazard inferred from a free site is a hypothesis about what happens NEXT, and what
 *  happens next has to be read separately.  The reset is nine lines below the free.*
 *
 *  WHAT THE DEFECT ACTUALLY IS, and it is still worth fixing:
 *    (1) the in-flight transfer is DESTROYED -- the guest's read stream is silently
 *        aborted mid-delivery, with no error reported to it;
 *    (2) a SPURIOUS SCSI COMMAND is executed against the disk image, and its opcode is
 *        `scsi_cmd[0]` -- a leftover DATAIN PAYLOAD byte.  What command runs is
 *        influenced by data the guest chose to have read back to it.
 *
 *  That is an integrity and spurious-command defect, NOT a host memory-safety
 *  primitive.  Ranked accordingly.
 *
 *  THE ORACLE IS A FREE COUNT, not a sanitiser, so the row works in an ordinary build:
 *  a mid-DATAIN write must not free the transfer the guest is still reading.
 *
 *  NOT A #475 DEFECT and not introduced by any round here -- the dispatch has always been
 *  phase-blind.  Filed from reading, then measured.
 */
static void row_cdb_midxfer(void)
{
	struct wdc_data d;
	unsigned char buf[4];
	int i, freed_during;

	memset(&d, 0, sizeof(d));
	d.addr_mult  = 1;
	d.io_enabled = 1;
	d.inbuf = (unsigned char *) calloc(1, WDC_INBUF_SIZE);
	d.irq.interrupt_assert = arm_sink;
	if (d.inbuf == NULL) { printf("  SETUP midxfer inbuf failed\n"); return; }

	/*  arm a real DATAIN transfer through the packet path, as row_arm_overrun does  */
	/*  #477 pass 2: the real device sets cur_command with the flag and the
	    phase (`cur_command = idata`, then the ATAPI_PKT_CMD arm).  Without
	    it a mutant conjoining `cur_command == ATAPI_PKT_CMD` kills this row
	    by removing LEGAL dispatch -- a false-path kill.  */
	d.cur_command = ATAPI_PKT_CMD;
	d.atapi_cmd_in_progress = 1;
	/*  SETUP RE-PIN, and it is INERT on the committed build: the real
	    device sets BOTH flags together when the packet command
	    arrives, so a fixture setting only atapi_cmd_in_progress
	    builds a state the device never produces.  A candidate fix
	    that keys on the PHASE would then fail here for the fixture's
	    reason rather than its own.  */
	d.atapi_phase = PHASE_CMDOUT;
	for (i = 0; i < 10; i++)
		wdc_addtoinbuf(&d, 0x00);
	stub_datain_len = 10;
	memset(buf, 0, sizeof(buf));
	dev_wdc_access(fake_cpu(), NULL, wd_data, buf, 2, MEM_WRITE, &d);
	stub_datain_len = 0;

	/*  the guest has drained nothing; the transfer is LIVE and data_in is in use  */
	stub_frees = 0;

	/*  *** THE ARITHMETIC THAT MAKES THIS REACHABLE, and my first fixture had
	    it wrong. ***  During DATAIN the ring holds exactly `atapi_len` bytes,
	    so occupancy == 12 means TEN UNDRAINED plus a two-byte write.  The
	    first attempt armed 4096 bytes, so occupancy went 4096 -> 4108 and the
	    `== 12` equality could never fire: the row was GREEN and the defect was
	    simply not reached.

	    A ten-byte transfer is the smallest live state that reaches it, and it
	    is an ordinary one -- any DATAIN with ten bytes still queued.  */
	memset(buf, 0, sizeof(buf));
	dev_wdc_access(fake_cpu(), NULL, wd_data, buf, 2, MEM_WRITE, &d);
	freed_during = stub_frees;

	rows ++;
	if (freed_during != 0) {
		printf("  FAIL %-38s freed %i live transfer(s)\n",
		    "a mid-DATAIN write frees nothing", freed_during);
		failures ++;
	} else {
		printf("  ok   %-38s %i\n",
		    "a mid-DATAIN write frees nothing", freed_during);
	}

	if (d.atapi_st != NULL) {
		free(d.atapi_st->data_in);
		free(d.atapi_st->cmd);
		free(d.atapi_st);
	}
	free(d.inbuf);
}


/*  OCCUPANCY ALREADY 12 -- the row that would have caught the fix I preferred.
 *
 *  A panel seat argued (REASONED) that guarding only the PUSH is not enough, because the
 *  dispatch runs whether or not a push happened.  Arm a DATAIN whose queued occupancy is
 *  ALREADY 12 -- a 12-byte INQUIRY response, or a larger chunk read down to twelve -- and
 *  any data-port write reaches the dispatch with occupancy still 12.
 *
 *  The row above tests TEN QUEUED PLUS A TWO-BYTE WRITE, so a push-only guard would have
 *  scored it GREEN while this case still freed the live transfer.  This row exists to make

	/*  arm a TWELVE-byte transfer: occupancy is then exactly 12 with nothing drained  */

	    it a mutant conjoining `cur_command == ATAPI_PKT_CMD` kills this row

/*  THE THREE ROWS THE BRIEF DID NOT HAVE, each named by a measuring seat and each the
 *  SOLE carrier of a one-token escape past a DATAIN-only detector.
 *
 *    no-push    a len-3 write pushes ZERO bytes -- the width switch's `default:` arm only
 *               latches -- and control falls straight through to the dispatch.  This is
 *               the row that proves the defect is NOT "writes push foreign bytes"; the
 *               dispatch runs on every data-port write regardless of what the write did.
 *    DATAOUT    twelve DATA-OUT payload bytes are read as a CDB.
 *    COMPLETED  the flag is never cleared at completion, so twelve bytes AFTER a finished
 *               command still free the completed transfer and issue a garbage command.
 *
 *  A guard written `atapi_phase != PHASE_DATAIN` passes every DATAIN row and leaves all
 *  three of these open.  A seat measured exactly that mutant sweeping the earlier table.
 */
static void midxfer_case(const char *name, int phase, int prefill,
	int alen, int wlen)
{
	struct wdc_data d;
	struct scsi_transfer *st;
	unsigned char buf[8], *payload;
	int i;

	/*  *** THE TRANSFER MUST BE HEAP-ALLOCATED, and the first version of this
	    fixture put it on the STACK. ***  On the unfixed build the dispatch calls
	    scsi_transfer_free() on it -- so a stack object gave
	    `double free or corruption (out)` and the run ABORTED partway, losing
	    every row after it.  A measuring seat hit the identical flaw in its own
	    first fixture.

	    An abort is not a FAIL: the rows that never ran print nothing at all, and
	    the verdict line is missing rather than red.  */

	memset(&d, 0, sizeof(d));
	d.addr_mult  = 1;
	d.io_enabled = 1;
	d.inbuf = (unsigned char *) calloc(1, WDC_INBUF_SIZE);
	d.irq.interrupt_assert = arm_sink;
	payload = (unsigned char *) calloc(1, 64);
	if (d.inbuf == NULL || payload == NULL) {
		printf("  SETUP %s alloc failed\n", name);
		free(d.inbuf); free(payload); return;
	}
	st = (struct scsi_transfer *) calloc(1, sizeof(*st));
	if (st == NULL) { free(d.inbuf); free(payload);
		printf("  SETUP %s xfer alloc failed@", name); return; }
	st->data_in = payload;  st->data_in_len = 64;

	/*  a live transfer, in the phase under test, with `prefill` bytes queued  */
	d.atapi_st = st;
	/*  #477 pass 2, codex: THE REAL DEVICE SETS ALL THREE TOGETHER.
	    `cur_command = idata` then the ATAPI_PKT_CMD arm sets the flag AND
	    the phase, so a fixture setting only two builds a state the device
	    never produces -- and a mutant adding `cur_command == ATAPI_PKT_CMD`
	    then dies because LEGAL DISPATCH vanishes, not because of the defect.
	    That is a FALSE-PATH KILL: the table scores it caught and it proves
	    nothing.  (Inert on the committed build: the only other reads compare
	    against COMMAND_RESET = 0x100, which 0xa0 is not.)  */
	d.cur_command = ATAPI_PKT_CMD;
	d.atapi_cmd_in_progress = 1;
	d.atapi_phase = phase;
	/*  `alen` is DECOUPLED from `prefill`: prefill is what the RING holds,
	    alen is the transfer's remaining count.  They coincide in DATAIN and
	    must NOT in COMPLETED -- see the caller.  */
	d.atapi_len = alen;
	d.atapi_received = 0;
	for (i = 0; i < prefill; i++)
		wdc_addtoinbuf(&d, 0x5a);

	stub_frees = 0;
	memset(buf, 0, sizeof(buf));
	dev_wdc_access(fake_cpu(), NULL, wd_data, buf, wlen, MEM_WRITE, &d);

	rows ++;
	if (stub_frees != 0) {
		printf("  FAIL %-38s freed %i\n", name, stub_frees);
		failures ++;
	} else {
		printf("  ok   %-38s %i\n", name, stub_frees);
	}
	/*  the dispatch may have freed and replaced it; free whatever is there now  */
	if (d.atapi_st != NULL && d.atapi_st != st) {
		free(d.atapi_st->data_in);
		free(d.atapi_st->cmd);
		free(d.atapi_st);
	} else if (d.atapi_st == st) {
		free(st);
	}
	free(payload);
	free(d.inbuf);
}

static void row_midxfer_phases(void)
{
	/*  occupancy 12 already, then a len-3 write that pushes NOTHING  */
	midxfer_case("no-push write does not dispatch", PHASE_DATAIN, 12, 12, 3);
	/*  ten queued plus a two-byte write, in the two phases DATAIN rows miss  */
	midxfer_case("a DATAOUT write does not dispatch", PHASE_DATAOUT, 10, 10, 2);
	/*  *** THE COMPLETED FIXTURE USED TO SET atapi_len = 10, WHICH THE DEVICE
	    CANNOT PRODUCE.  Found by the codex seat on #477 pass 2. ***

	    EVERY real transition to PHASE_COMPLETED leaves atapi_len == 0: the
	    reject path assigns zero explicitly, the ordinary path assigns zero
	    before selecting its phase, and terminal DATAIN only completes once
	    the count has reached zero.  So a guard widened by `|| atapi_len == 0`
	    passed this row for the FIXTURE's reason and swept the table, while a
	    real completed command plus twelve guest DATA writes still re-enters
	    dispatch and frees the completed transfer.

	    Ten bytes still sit in the RING (so the 2-byte write reaches occupancy
	    12 and the dispatch is genuinely attempted); only the COUNT is zero.
	    That is the state a real device is in, and it is the row codex named
	    `row_completed_zero_len`.  */
	midxfer_case("a COMPLETED write does not dispatch", PHASE_COMPLETED, 10, 0, 2);
}


/*  #477 pass 2 -- THE ACCEPT SIDE, WHICH THE SHIPPED TABLE DID NOT HAVE.
 *
 *  Two seats found this independently: NO row proved a legal CDB actually
 *  DISPATCHES.  `row_cdb_already12` asserts nothing is FREED, which is a reject
 *  assertion -- so a fix that suppressed every dispatch would have passed the
 *  whole table.  The de-escalation clause calls these rows part of the FIX'S
 *  COST, not future strengthening.
 *
 *  The stale state is the point.  ATAPI_PKT_CMD assigns only the flag and the
 *  phase, so after a PRIOR COMPLETED command a new packet begins with
 *  atapi_len == 0, atapi_received == data_in_len and atapi_st STALE NON-NULL.
 *  Every fixture in this file is memset-fresh and none crosses a command
 *  boundary, so that entire dimension was invisible.
 */
static void pkt_cdb_case(const char *name, int stray_read)
{
	struct wdc_data d;
	struct scsi_transfer *st;
	unsigned char buf[4], *payload;
	int i;

	memset(&d, 0, sizeof(d));
	d.addr_mult  = 1;
	d.io_enabled = 1;
	d.inbuf = (unsigned char *) calloc(1, WDC_INBUF_SIZE);
	d.irq.interrupt_assert = arm_sink;
	payload = (unsigned char *) calloc(1, 64);
	st = (struct scsi_transfer *) calloc(1, sizeof(*st));
	if (d.inbuf == NULL || payload == NULL || st == NULL) {
		printf("  SETUP %s alloc failed\n", name);
		free(d.inbuf); free(payload); free(st); return;
	}
	st->data_in = payload;  st->data_in_len = 64;

	/*  a PRIOR COMPLETED command: fully received, count exhausted, transfer
	    still hanging off the device (it is freed at the NEXT dispatch and
	    never NULLed).  */
	d.atapi_st = st;
	d.atapi_received = 64;
	d.atapi_len = 0;

	/*  then A0h arrives -- it assigns EXACTLY these three and resets nothing.  */
	d.cur_command = ATAPI_PKT_CMD;
	d.atapi_cmd_in_progress = 1;
	d.atapi_phase = PHASE_CMDOUT;

	if (stray_read) {
		/*  ONE out-of-protocol data-port read in the CDB window.  Without the
		    #477 read-arm guard this rewrites the phase to PHASE_COMPLETED.  */
		memset(buf, 0, sizeof(buf));
		dev_wdc_access(fake_cpu(), NULL, wd_data, buf, 2, MEM_READ, &d);
	}

	stub_scsi_cmds = 0;
	/*  a legal twelve-byte CDB: six 16-bit writes from an empty ring.  */
	for (i = 0; i < 6; i++) {
		memset(buf, 0, sizeof(buf));
		dev_wdc_access(fake_cpu(), NULL, wd_data, buf, 2, MEM_WRITE, &d);
	}

	rows ++;
	if (stub_scsi_cmds != 1) {
		printf("  FAIL %-38s dispatched %i, want 1\n",
			name, stub_scsi_cmds);
		failures ++;
	} else {
		printf("  ok   %-38s %i\n", name, stub_scsi_cmds);
	}

	if (d.atapi_st != NULL && d.atapi_st != st) {
		free(d.atapi_st->data_in);
		free(d.atapi_st->cmd);
		free(d.atapi_st);
	} else if (d.atapi_st == st) {
		free(st);
	}
	free(payload);
	free(d.inbuf);
}

static void row_pkt_cdb_accept(void)
{
	/*  CONTROL first: the same fixture with no stray read MUST dispatch, so a
	    red below is attributable to the read and not to the fixture.  */
	pkt_cdb_case("completed, PKT then CDB dispatches", 0);
	/*  THE SOLE CARRIER of the #477 read-arm guard.  MEASURED by the Fable seat:
	    HEAD services this sequence; the phase guard alone wedges it FOREVER,
	    because the 12-byte residue means occupancy can never EQUAL 12 again.  */
	pkt_cdb_case("a stray read does not wedge the CDB", 1);
}

/*  #477 pass 2 -- THE ROW THAT PINS THE `malloc(12)` BOUND.
 *
 *  The kill table's `M7_ge12` mutant (`inbuf_len == 12` relaxed to `>= 12`) swept
 *  44 rows clean, and the table's first draft classified it as an IMPROVEMENT --
 *  dispatching on an overshoot beating never dispatching.  The Opus seat measured
 *  it under ASAN instead and got a heap-buffer-overflow WRITE, because nine lines
 *  below the guard:
 *
 *      CHECK_ALLOCATION(scsi_cmd = (unsigned char *) malloc(12));
 *      while (inbuf_len > 0) { scsi_cmd[x++] = wdc_get_inbuf(d); inbuf_len--; }
 *
 *  `x` is unbounded.  The equality is not only selecting the dispatch -- it is the
 *  ONLY thing bounding that copy.  Four readers reasoned about it as a dispatch
 *  predicate and none opened the tail it protects.
 *
 *  This row reds on `>= 12` WITHOUT needing a sanitiser: the oracle is the dispatch
 *  count, so an ordinary build catches it.  Reaching occupancy 13 without passing
 *  THROUGH 12 is the whole trick -- preload an odd count, then write one byte.
 */
static void row_cdb_overshoot(void)
{
	struct wdc_data d;
	unsigned char buf[4];
	int i;

	memset(&d, 0, sizeof(d));
	d.addr_mult  = 1;
	d.io_enabled = 1;
	d.inbuf = (unsigned char *) calloc(1, WDC_INBUF_SIZE);
	d.irq.interrupt_assert = arm_sink;
	if (d.inbuf == NULL) { printf("  SETUP overshoot inbuf failed\n"); return; }

	d.cur_command = ATAPI_PKT_CMD;
	d.atapi_cmd_in_progress = 1;
	d.atapi_phase = PHASE_CMDOUT;

	/*  THIRTEEN queued, so the one-byte write below lands on 14 and the ring
	    never passes through 12.  A fixture that wrote its way up two bytes at
	    a time would dispatch at 12 and never reach the overshoot at all.  */
	for (i = 0; i < 13; i++)
		wdc_addtoinbuf(&d, 0x5a);

	stub_scsi_cmds = 0;
	memset(buf, 0, sizeof(buf));
	dev_wdc_access(fake_cpu(), NULL, wd_data, buf, 1, MEM_WRITE, &d);

	rows ++;
	if (stub_scsi_cmds != 0) {
		printf("  FAIL %-38s dispatched %i at occupancy 14\n",
			"occupancy above 12 does not dispatch", stub_scsi_cmds);
		failures ++;
	} else {
		printf("  ok   %-38s %i\n",
			"occupancy above 12 does not dispatch", stub_scsi_cmds);
	}

	if (d.atapi_st != NULL) {
		free(d.atapi_st->data_in);
		free(d.atapi_st->cmd);
		free(d.atapi_st);
	}
	free(d.inbuf);
}

/*  wdcdataoutdrain -- REPRODUCTION, not yet a shipped row.
 *
 *  `inbuf_len == 0` satisfies `0 % 512 == 0`, so the write-back block fires with an
 *  EMPTY ring and hands diskimage_access 512 bytes of whatever the ring happens to
 *  hold, at the guest's chosen LBA.
 *
 *  The entrance is #337's own de-fatalised arm: a 3- or 8-byte data-port write pushes
 *  NOTHING (the width switch has no case for it) and `break` leaves the SWITCH, not the
 *  handler -- so execution falls straight through to the drain with occupancy 0.
 *
 *  The ring is deliberately POISONED with 0xDD first, so the oracle can say the bytes
 *  written are ones the guest never supplied rather than merely counting a write.
 */
static void row_dataout_drain(void)
{
	struct wdc_data d;
	unsigned char buf[8];
	int i;

	memset(&d, 0, sizeof(d));
	d.addr_mult  = 1;
	d.io_enabled = 1;
	d.inbuf = (unsigned char *) calloc(1, WDC_INBUF_SIZE);
	d.irq.interrupt_assert = arm_sink;
	if (d.inbuf == NULL) { printf("  SETUP drain inbuf failed\n"); return; }

	/*  poison: any byte reaching the image is one the guest never wrote  */
	for (i = 0; i < WDC_INBUF_SIZE; i++)
		d.inbuf[i] = 0xdd;

	/*  a WRITE command is in progress and the ring is EMPTY (head == tail == 0)  */
	d.write_in_progress = WDCC_WRITE;
	d.write_count  = 1;
	d.write_offset = 0x8000;

	stub_disk_writes = 0; stub_disk_bytes = 0; stub_disk_offset = 0;
	stub_disk_first = -1;

	/*  ONE 3-byte data-port write.  Pushes nothing; falls through to the drain.  */
	memset(buf, 0, sizeof(buf));
	dev_wdc_access(fake_cpu(), NULL, wd_data, buf, 3, MEM_WRITE, &d);

	rows ++;
	if (stub_disk_writes != 0) {
		printf("  FAIL %-38s %i write(s), %i bytes to 0x%lx, first=0x%02x\n",
			"an empty ring writes nothing to disk",
			stub_disk_writes, stub_disk_bytes, stub_disk_offset,
			stub_disk_first);
		failures ++;
	} else {
		printf("  ok   %-38s %i\n",
			"an empty ring writes nothing to disk", stub_disk_writes);
	}

	/*  the SIBLING, and it is the same condition: an 8-byte access.  A 64-bit guest
	    writing a sector with `sd` stores would fire this 64 times at consecutive
	    offsets.  Separate device so the first row's drain cannot mask it.  */
	{
		struct wdc_data e;
		memset(&e, 0, sizeof(e));
		e.addr_mult  = 1;
		e.io_enabled = 1;
		e.inbuf = (unsigned char *) calloc(1, WDC_INBUF_SIZE);
		e.irq.interrupt_assert = arm_sink;
		if (e.inbuf == NULL) { free(d.inbuf); return; }
		for (i = 0; i < WDC_INBUF_SIZE; i++)
			e.inbuf[i] = 0xdd;
		e.write_in_progress = WDCC_WRITE;
		e.write_count  = 1;
		e.write_offset = 0x8000;
		stub_disk_writes = 0; stub_disk_bytes = 0;
		memset(buf, 0, sizeof(buf));
		dev_wdc_access(fake_cpu(), NULL, wd_data, buf, 8, MEM_WRITE, &e);
		rows ++;
		if (stub_disk_writes != 0) {
			printf("  FAIL %-38s %i write(s), %i bytes\n",
				"an 8-byte write drains nothing",
				stub_disk_writes, stub_disk_bytes);
			failures ++;
		} else {
			printf("  ok   %-38s %i\n",
				"an 8-byte write drains nothing", stub_disk_writes);
		}
		free(e.inbuf);
	}

	/*  *** THE LEFTOVER ENTRANCE, named by the grok seat and NOT closed by
	    `inbuf_len > 0`.  Leave 512 bytes UNREAD in the ring (an unread IDENTIFY or
	    DATA-IN response), start a WRITE, then do the same 8-byte store.  Occupancy
	    is 512, not 0 -- so `512 %% 512 == 0` AND `512 > 0` both pass, and the
	    guest's own returned payload is written to the disk image at its chosen
	    LBA.  Poisoned 0xBE so the byte identifies WHICH entrance fired.  ***  */
	{
		struct wdc_data f;
		memset(&f, 0, sizeof(f));
		f.addr_mult  = 1;
		f.io_enabled = 1;
		f.inbuf = (unsigned char *) calloc(1, WDC_INBUF_SIZE);
		f.irq.interrupt_assert = arm_sink;
		if (f.inbuf == NULL) { free(d.inbuf); return; }
		for (i = 0; i < 512; i++)
			wdc_addtoinbuf(&f, 0xbe);
		f.write_in_progress = WDCC_WRITE;
		f.write_count  = 1;
		f.write_offset = 0x9000;
		stub_disk_writes = 0; stub_disk_bytes = 0; stub_disk_first = -1;
		memset(buf, 0, sizeof(buf));
		dev_wdc_access(fake_cpu(), NULL, wd_data, buf, 8, MEM_WRITE, &f);
		rows ++;
		if (stub_disk_writes != 0) {
			printf("  FAIL %-38s %i write(s), first=0x%02x\n",
				"leftover payload not written to disk",
				stub_disk_writes, stub_disk_first);
			failures ++;
		} else {
			printf("  ok   %-38s %i\n",
				"leftover payload not written to disk", stub_disk_writes);
		}
		free(f.inbuf);
	}

	free(d.inbuf);
}

/*  THE WRITE+PKT DOOR, measured independently by the Fable seat.
 *
 *  No bad-len access is involved at all.  A WRITE is in progress AND a packet
 *  command is being assembled; the 12-byte CDB dispatches, and the dispatch loop
 *  consumes the LOCAL `inbuf_len` down to 0 while pushing fresh DATA-IN bytes into
 *  the ring.  The drain below therefore reads a STALE ZERO and fires, putting the
 *  guest's own returned DATA-IN payload on the disk image.
 *
 *  This door is closed by `inbuf_len > 0` (the stale local is 0) and is NOT closed
 *  by declining the bad-len width, because no bad-len access occurs.  The leftover
 *  door is the exact mirror.  That is why neither edit alone is sufficient.
 */
static void row_pkt_drain(void)
{
	struct wdc_data g;
	unsigned char buf[8];
	int i;

	memset(&g, 0, sizeof(g));
	g.addr_mult  = 1;
	g.io_enabled = 1;
	g.inbuf = (unsigned char *) calloc(1, WDC_INBUF_SIZE);
	g.irq.interrupt_assert = arm_sink;
	if (g.inbuf == NULL) {
		printf("  SETUP pkt_drain inbuf failed\n");
		return;
	}

	/*  the real device sets all three together  */
	g.cur_command = ATAPI_PKT_CMD;
	g.atapi_cmd_in_progress = 1;
	g.atapi_phase = PHASE_CMDOUT;

	/*  ...and a disk WRITE is live at the same time  */
	g.write_in_progress = WDCC_WRITE;
	g.write_count  = 1;

	for (i = 0; i < 10; i++)
		wdc_addtoinbuf(&g, 0x5a);

	stub_disk_first = -1;

		failures ++;
	} else {

		    "a CDB dispatch does not drain to disk", stub_disk_writes);

}


/*  THE ACCEPT ROW FOR THE BAD-LEN DECLINE.
 *
 *  A fix that declines the drain after an unimplemented width MUST NOT latch that
 *  decision for the life of the device.  If the flag is a struct field, one 8-byte
 *  store silences every subsequent legal write -- turning a corruption defect into
 *  a silent data-loss defect, which is worse.
 *
 *  Sequence: one bad-len (8-byte) store, then a full LEGAL 512-byte write via
 *  256 two-byte stores.  The legal write MUST reach the disk image.
 *
 *  Expected: RED under a persistent struct-field flag; GREEN under a per-access
 *  local.  That is the whole point of the row -- it distinguishes two fixes that
 *  the four corruption rows score identically.
 */
static void row_badlen_then_legal(void)
{
	struct wdc_data h;
	unsigned char buf[8];
	int i;

	memset(&h, 0, sizeof(h));
	h.addr_mult  = 1;
	h.io_enabled = 1;
	h.inbuf = (unsigned char *) calloc(1, WDC_INBUF_SIZE);
	h.irq.interrupt_assert = arm_sink;
	if (h.inbuf == NULL) {
		printf("  SETUP badlen_then_legal inbuf failed\n");
		return;
	}

	h.write_in_progress = WDCC_WRITE;
	h.write_count  = 1;
	h.write_offset = 0xb000;

	/*  (1) the bad-len store -- pushes nothing, must drain nothing  */
	memset(buf, 0, sizeof(buf));
	dev_wdc_access(fake_cpu(), NULL, wd_data, buf, 8, MEM_WRITE, &h);

	/*  (2) a full legal sector, 256 two-byte stores carrying 0x77  */
	stub_disk_writes = 0;
	stub_disk_bytes = 0;
	stub_disk_first = -1;
	for (i = 0; i < 256; i++) {
		buf[0] = 0x77; buf[1] = 0x77;
		buf[2] = 0; buf[3] = 0;
		dev_wdc_access(fake_cpu(), NULL, wd_data, buf, 2, MEM_WRITE, &h);
	}

	rows ++;
	if (stub_disk_writes != 1 || stub_disk_bytes != 512) {
		printf("  FAIL %-38s %i write(s), %i bytes\n",
		    "a legal write after a bad-len lands",
		    stub_disk_writes, stub_disk_bytes);
		failures ++;
	} else {
		printf("  ok   %-38s %i write(s), %i bytes\n",
		    "a legal write after a bad-len lands",
		    stub_disk_writes, stub_disk_bytes);
	}

	free(h.inbuf);
}


/*  THE PARTIAL-SECTOR ROW.  Named by the codex seat after `M10_modinv` -- the
 *  modulus inverted from `== 0` to `!= 0` -- swept a 50-row table clean.
 *
 *  That mutant does not open the empty-ring door (the `inbuf_len > 0` half still
 *  holds), so every corruption row stays green.  What it does is authorise a
 *  512-byte disk write on ONE buffered byte, while exactly 512 bytes fail to
 *  drain.  The accept row cannot see it either: it observes one write of 512
 *  bytes, which is the right SHAPE at the wrong MOMENT.
 *
 *  So the sufficiency condition needs pinning from below as well as at the
 *  boundary: occupancy 1 and needed-1 must write NOTHING.
 */
static void row_partial_sector(void)
{
	struct wdc_data j;
	unsigned char buf[8];
	int i;

	memset(&j, 0, sizeof(j));
	j.addr_mult  = 1;
	j.io_enabled = 1;
	j.inbuf = (unsigned char *) calloc(1, WDC_INBUF_SIZE);
	j.irq.interrupt_assert = arm_sink;
	if (j.inbuf == NULL) {
		printf("  SETUP partial_sector inbuf failed\n");
		return;
	}

	j.write_in_progress = WDCC_WRITE;
	j.write_count  = 1;
	j.write_offset = 0xc000;

	stub_disk_writes = 0;
	stub_disk_bytes = 0;

	/*  255 two-byte stores: occupancy climbs 2..510, one short of a sector.
	    NOT ONE of them may reach the disk.  */
	for (i = 0; i < 255; i++) {
		buf[0] = 0x33; buf[1] = 0x33;
		buf[2] = 0; buf[3] = 0;
		dev_wdc_access(fake_cpu(), NULL, wd_data, buf, 2, MEM_WRITE, &j);
	}

	rows ++;
	if (stub_disk_writes != 0) {
		printf("  FAIL %-38s %i write(s) at occupancy < 512\n",
		    "a partial sector does not drain", stub_disk_writes);
		failures ++;
	} else {
		printf("  ok   %-38s %i\n",
		    "a partial sector does not drain", stub_disk_writes);
	}

	/*  the 256th store completes the sector: exactly one write, exactly 512
	    bytes, at exactly the offset the command named.  */
	buf[0] = 0x33; buf[1] = 0x33;
	buf[2] = 0; buf[3] = 0;
	dev_wdc_access(fake_cpu(), NULL, wd_data, buf, 2, MEM_WRITE, &j);

	rows ++;
	if (stub_disk_writes != 1 || stub_disk_bytes != 512 ||
	    stub_disk_offset != 0xc000) {
		printf("  FAIL %-38s %i w, %i b, off 0x%lx\n",
		    "completing store drains one sector",
		    stub_disk_writes, stub_disk_bytes, stub_disk_offset);
		failures ++;
	} else {
		printf("  ok   %-38s %i w, %i b\n",
		    "completing store drains one sector",
		    stub_disk_writes, stub_disk_bytes);
	}

	free(j.inbuf);
}


/*  THE ROW `#477` OWED.  Prescribed by the codex seat on `#477`'s pass 2, after it
 *  reasoned that deleting the `atapi_cmd_in_progress` conjunct from the dispatch
 *  guard would sweep the table.  MEASURED: it swept all 52 rows clean.
 *
 *  THE STALE PHASE IS THE POINT.  `wdc_command(ATAPI_PKT_CMD)` sets PHASE_CMDOUT.
 *  A following `wdc_command(WDCC_WRITE)` clears the flag at the top of the function
 *  and arms the write -- but NOTHING RESETS THE PHASE.  So the device sits with
 *  `phase == PHASE_CMDOUT`, `atapi_cmd_in_progress == 0`, and a disk write armed:
 *  exactly the state a phase-only guard cannot tell from a real CDB collection.
 *
 *  Under that mutant, the guest's own sector payload reaches occupancy 12 and the
 *  first twelve bytes are issued as a SCSI command AND removed from the pending
 *  sector.  Both conjuncts are load-bearing; `#477`'s "key on the phase, not on
 *  the flag" was the wrong headline and is corrected in place.
 *
 *  (The stale phase is itself a defect and is filed as part of `wdcwipclear` /
 *  `wdcringnoflush` -- a new command does not reset what the old one left.)
 */
static void row_stale_phase_write(void)
{
	struct wdc_data d;
	unsigned char buf[4];
	int i;

	memset(&d, 0, sizeof(d));
	d.addr_mult  = 1;
	d.io_enabled = 1;
	d.inbuf = (unsigned char *) calloc(1, WDC_INBUF_SIZE);

		printf("  SETUP stale_phase inbuf failed\n");
		return;
	}

	stub_disk_exists = 1;
	d.seccnt = 1;
	d.sector = 1;

	/*  a packet command, which is what leaves PHASE_CMDOUT behind  */
	wdc_command(fake_cpu(), &d, ATAPI_PKT_CMD);

	/*  then an ordinary ATA write.  This clears the flag and arms the write; the
	    phase is NOT reset, which is the state under test.  */
	wdc_command(fake_cpu(), &d, WDCC_WRITE);

	/*  *** ORACLE CORRECTION, 2026-08-27.  THIS ROW USED TO PIN THE DEFECT.
	    The old assertion was `LIVENESS: stale CMDOUT with a write`, wanting
	    phase == PHASE_CMDOUT after a WRITE command -- and it was CORRECT WHEN
	    IT SHIPPED.  A phase-only dispatch guard has to be shown a state where
	    the phase is stale, or the row proves nothing. ***

	    `#479` RESETS the phase at the top of `wdc_command`, so the assertion
	    inverts: a new command must not leave the previous one's phase behind.
	    A seat asked whether the fix would break this row BEFORE any code was
	    written, which is the only reason this is an oracle correction rather
	    than a laundering.

	    PROVENANCE, quoted rather than summarised.  `wdc_command()` opens:

	        d->cur_command = idata;
	        d->atapi_cmd_in_progress = 0;
	        d->error = 0;

	    It clears the flag and the error and NOT the phase.  That is the
	    defect, and this row is its oracle.

	    `== PHASE_CMDOUT` as the FAILING condition, not a pin on a specific
	    replacement value: what the round establishes is that the phase does
	    not SURVIVE a new command.  Pinning which value it takes would
	    over-specify a choice the fix has not made.

	    THE OTHER TWO ASSERTIONS IN THIS ROW ARE UNCHANGED -- they concern the
	    dispatch guard, not the phase, and must still red under the phase-only

	rows ++;

	    d.write_in_progress == 0) {

		    "a new command does not leave CMDOUT",
		    d.atapi_phase, d.atapi_cmd_in_progress,
		    d.write_in_progress);
		failures ++;
	} else {
		printf("  ok   %-38s phase 0x%x\n",
		    "a new command does not leave CMDOUT", d.atapi_phase);
	}

	/*  the guest streams one legal sector, 256 sixteen-bit stores  */
	stub_scsi_cmds = 0;
	stub_disk_writes = 0;
	stub_disk_bytes = 0;
	for (i = 0; i < 256; i++) {
		buf[0] = 0x6b; buf[1] = 0x6b;
		buf[2] = 0; buf[3] = 0;
		dev_wdc_access(fake_cpu(), NULL, wd_data, buf, 2, MEM_WRITE, &d);
	}

	rows ++;

		    "disk payload is not issued as a CDB", stub_scsi_cmds);

	} else {
		printf("  ok   %-38s %i\n",
		    "disk payload is not issued as a CDB", stub_scsi_cmds);
	}

	/*  ...and the sector must arrive INTACT, which is the half that catches a
	    mutant that merely eats the bytes without dispatching.  */
	rows ++;
	if (stub_disk_writes != 1 || stub_disk_bytes != 512) {
		printf("  FAIL %-38s %i w, %i b\n",
		    "the sector survives a stale phase",
		    stub_disk_writes, stub_disk_bytes);
		failures ++;
	} else {
		printf("  ok   %-38s %i w, %i b\n",
		    "the sector survives a stale phase",
		    stub_disk_writes, stub_disk_bytes);

		free(d.atapi_st->cmd);
		free(d.atapi_st);
	}
	free(d.inbuf);
}

/*  #478's DECLINE-AND-ANNOUNCE, measured by EMISSION COUNT rather than by the latch.
 *
 *  The latch field is a proxy and a weak one: a fix that sets the flag without
 *  printing, or prints without setting it, satisfies a latch check and fails the
 *  property.  `fatal()` is counted instead, so both halves of "N=1 PER DEVICE
 *  INSTANCE" are observable -- the ONE, and the PER INSTANCE.
 *
 *  The third row is the one that matters for the idiom.  `#269` records that a
 *  shared counter across instances is the wrong shape; a `static` latch would pass
 *  rows one and two and fail row three.

static void row_drain_announce(void)
{
	struct wdc_data d, e;
	unsigned char buf[8];

	memset(&d, 0, sizeof(d));
	d.addr_mult  = 1;

		return;
	}
	d.write_in_progress = WDCC_WRITE;
	d.write_count  = 1;
	d.write_offset = 0xd000;

	/*  the input where a spurious sector USED to reach the disk image  */
	stub_fatals = 0; stub_drain_msgs = 0;
	memset(buf, 0, sizeof(buf));
	dev_wdc_access(fake_cpu(), NULL, wd_data, buf, 8, MEM_WRITE, &d);
	first = stub_drain_msgs;

	/*  the SAME device again: the latch must hold  */
	stub_fatals = 0; stub_drain_msgs = 0;

	memset(&e, 0, sizeof(e));

	e.write_in_progress = WDCC_WRITE;
	e.write_count  = 1;
	e.write_offset = 0xd000;
	stub_fatals = 0; stub_drain_msgs = 0;
	memset(buf, 0, sizeof(buf));
	dev_wdc_access(fake_cpu(), NULL, wd_data, buf, 8, MEM_WRITE, &e);
	other = stub_drain_msgs;

	/*  the bad-len arm ALSO latches once, so the first access emits two
	    messages: #337's width notice and #478's dropped-write notice.  What
	    this row pins is that the count does not GROW on repetition.  */
	rows ++;

		printf("  FAIL %-38s emitted %i on repeat, want 0\n",
		    "the announce is latched per device", second);
		failures ++;
	} else {
		printf("  ok   %-38s %i\n",
		    "the announce is latched per device", second);
	}

	rows ++;
	if (other < 1) {
		printf("  FAIL %-38s emitted %i, want >= 1\n",
		    "a second device is still told", other);
		failures ++;
	} else {
		printf("  ok   %-38s %i\n",
		    "a second device is still told", other);
	}

	free(e.inbuf);
	free(d.inbuf);
}


/*  THE DEVICE-CONTROL ACCEPT ROW.
 *
 *  A pending sector write must SURVIVE an ordinary Device Control write.
 *
 *  *** THE WARRANT FIRST WRITTEN HERE WAS WRONG, and it was wrong in the
 *  reassuring direction.  It said this row is "the only instrument in the tree
 *  that can see the regression it guards against: MEASURED, no gate arms a CDROM,
 *  so the ALT-CTRL/COMMAND_RESET path is never exercised by any rig."  Corrected
 *  2026-08-27 after a seat refuted it. ***
 *
 *  The ALT-CTRL arm NEEDS NO CDROM.  `if (idata & WDCTL_4BIT) d->cur_command =
 *  COMMAND_RESET;` is unconditional -- only the SIGNATURE BYTES read back at
 *  wd_cyl_lo/wd_cyl_hi are CDROM-gated.  So "no CDROM" never implied "not
 *  exercised", and the row's own justification did not follow.
 *
 *  The defensible statement is narrower and still supports the row: no RIG writes
 *  Device Control, and on the fork's only SH4 wdc machine it CANNOT -- the driver
 *  writes addr+0x2c while GXemul registers alt-status at addr+0x206, so twelve
 *  probe writes land on unmapped memory (filed separately as `landiskaux`).  A
 *  PROBE can write it, and a seat built one at rung 3.
 *
 *  *** THAT MAKES THIS ROW'S WARRANT REST ON A SECOND DEFECT. ***  Fix the
 *  mapping and the ALT-CTRL arm goes live on that machine too.  A row justified by
 *  another bug's existence is worth keeping and worth SAYING SO, because the
 *  justification expires the day that bug is fixed.
 *
 *  0x08 is not an exotic value.  From cited driver source: Linux libata holds
 *  ATA_DEVCTL_OBS (0x08) as `ap->ctl`, the STEADY STATE between commands, and older
 *  OpenBSD resets with 0x06 (bit 3 clear) and then writes 0x08 separately.  A guest
 *  may write it at any moment, including mid-transfer.
 *
 *  So a future round that hangs a ring-flush or a writer-cancel off the ALT-CTRL arm
 *  -- while it still triggered on WDCTL_4BIT (0x08) rather than WDCTL_RST (0x04) --
 *  would cancel a legal transfer on a routine register write.  This row reds on that
 *  edit.  It is the de-escalation clause's accept side.
 *
 *  *** THAT FIX IS `#481`, AND THE PREDICTION HELD -- BUT ONLY BECAUSE THE ROUND DID
 *  THE PREREQUISITE FIRST.  `#481` hung exactly this writer-cancel off the ALT-CTRL
 *  arm, and this row is GREEN, because the same change corrected the bit: a routine
 *  0x08 never reaches the cancel.  The row DOES red under `Md_bitswap` -- which is
 *  that same edit WITHOUT the prerequisite.  So this row measured the ordering
 *  argument rather than merely asserting it, which is what an accept row is for. ***
 */
static void row_devctl_preserves_write(void)
{
	struct wdc_data d;
	unsigned char buf[8];
	int i;

	memset(&d, 0, sizeof(d));
	d.addr_mult  = 1;
	d.io_enabled = 1;
	d.inbuf = (unsigned char *) calloc(1, WDC_INBUF_SIZE);
	d.irq.interrupt_assert = arm_sink;
	if (d.inbuf == NULL) {
		printf("  SETUP devctl inbuf failed\n");
		return;
	}

	stub_disk_exists = 1;
	d.seccnt = 1;
	d.sector = 1;
	wdc_command(fake_cpu(), &d, WDCC_WRITE);

	/*  half the sector, then a perfectly ordinary Device Control write  */
	stub_disk_writes = 0;
	stub_disk_bytes = 0;
	stub_disk_first = -1;
	for (i = 0; i < 128; i++) {
		buf[0] = 0x6d; buf[1] = 0x6d;
		buf[2] = 0; buf[3] = 0;
		dev_wdc_access(fake_cpu(), NULL, wd_data, buf, 2, MEM_WRITE, &d);
	}

	memset(buf, 0, sizeof(buf));
	buf[0] = WDCTL_4BIT;		/*  0x08 -- libata's steady state  */
	dev_wdc_altstatus_access(fake_cpu(), NULL, 0, buf, 1, MEM_WRITE, &d);

	/*  the transfer must still be armed and the queued bytes still queued  */
	rows ++;
	if (d.write_in_progress == 0) {
		printf("  FAIL %-38s writer cancelled by 0x08\n",
		    "a devctl write keeps the writer");
		failures ++;
	} else {
		printf("  ok   %-38s 0x%x\n",
		    "a devctl write keeps the writer", d.write_in_progress);
	}

	/*  finish the sector: it must land, and it must be the GUEST'S bytes  */
	for (i = 0; i < 128; i++) {
		buf[0] = 0x6d; buf[1] = 0x6d;
		buf[2] = 0; buf[3] = 0;
		dev_wdc_access(fake_cpu(), NULL, wd_data, buf, 2, MEM_WRITE, &d);
	}

	rows ++;
	if (stub_disk_writes != 1 || stub_disk_bytes != 512 ||
	    stub_disk_first != 0x6d) {
		printf("  FAIL %-38s %i w, %i b, first=0x%02x\n",
		    "a devctl write does not lose bytes",
		    stub_disk_writes, stub_disk_bytes, stub_disk_first);
		failures ++;
	} else {
		printf("  ok   %-38s %i w, %i b\n",
		    "a devctl write does not lose bytes",
		    stub_disk_writes, stub_disk_bytes);
	}

	stub_disk_exists = 0;
	if (d.atapi_st != NULL) {
		free(d.atapi_st->data_in);
		free(d.atapi_st->cmd);
		free(d.atapi_st);
	}
	free(d.inbuf);
}


/*  THE THREE ROWS `#479`'s OWN MUTANTS REQUIRE.  Each was named by a seat that
 *  built the mutant and found the existing suite blind to it.
 *
 *  `#479` flushes the ring at the TOP of wdc_command() -- and IDENTIFY and READ
 *  both load their answer INTO that ring further down the same function.  Move
 *  only the RING FLUSH to the end and all three pinned reproductions stay GREEN
 *  while the guest's data is destroyed.  That mutant reads as the more principled
 *  placement, which is what makes it worth a row.
 *
 *  READ is deliberately here as well as IDENTIFY: a boot issues WDCC_READ far
 *  more often than WDCC_IDENTIFY, so a round that rows only IDENTIFY has pinned
 *  the rarer path.
 */
static void row_command_fills_ring(void)
{
	struct wdc_data d;
	int occ;

	memset(&d, 0, sizeof(d));
	d.addr_mult  = 1;
	d.io_enabled = 1;
	d.inbuf = (unsigned char *) calloc(1, WDC_INBUF_SIZE);
	d.irq.interrupt_assert = arm_sink;
	if (d.inbuf == NULL) {
		printf("  SETUP fills_ring inbuf failed\n");
		return;
	}

	stub_disk_exists = 1;

	/*  IDENTIFY must leave its response queued for the guest to read back.  */
	wdc_command(fake_cpu(), &d, WDCC_IDENTIFY);
	occ = d.inbuf_head - d.inbuf_tail;
	while (occ < 0)
		occ += WDC_INBUF_SIZE;

	rows ++;
	if (occ != 512) {
		printf("  FAIL %-38s %i queued, want 512\n",
		    "IDENTIFY leaves its answer queued", occ);
		failures ++;
	} else {
		printf("  ok   %-38s %i\n",
		    "IDENTIFY leaves its answer queued", occ);
	}

	/*  and the HOT path: an ordinary READ fills the ring the same way.  */
	d.inbuf_head = d.inbuf_tail = 0;
	d.seccnt = 1;
	d.sector = 1;
	wdc_command(fake_cpu(), &d, WDCC_READ);
	occ = d.inbuf_head - d.inbuf_tail;
	while (occ < 0)
		occ += WDC_INBUF_SIZE;

	rows ++;
	if (occ != 512) {
		printf("  FAIL %-38s %i queued, want 512\n",
		    "READ leaves a sector queued", occ);
		failures ++;
	} else {
		printf("  ok   %-38s %i\n",
		    "READ leaves a sector queued", occ);
	}

	stub_disk_exists = 0;
	free(d.inbuf);
}


/*  A LATCH MUST SURVIVE THE NEXT COMMAND.
 *
 *  `#479`'s helper clears three things a new command must not inherit.  Adding the
 *  `warned_*` flags to that list reads as COMPLETING it -- and undoes `#337` and
 *  `#464`, because those latches exist to stop a guest-drivable log flood.
 *
 *  It is strictly worse here than the equivalent mistake would be in dev_asc.c,
 *  which states the rule outright at its own latch declarations: the latches are
 *  "Deliberately NOT cleared by dev_asc_reset() or dev_asc_fifo_flush(): a guest
 *  could otherwise re-arm the flood."  `wdc_command()` runs far more often than a
 *  chip reset, so a guest alternating two opcodes would flood forever.
 *
 *  The oracle counts EMISSIONS, not the flag: a mutant that clears the latch while
 *  something else happens to suppress the message would pass a flag check.
 */
static void row_latch_survives_command(void)
{
	struct wdc_data d;
	unsigned char buf[8];
	int first, second;

	memset(&d, 0, sizeof(d));
	d.addr_mult  = 1;
	d.io_enabled = 1;
	d.inbuf = (unsigned char *) calloc(1, WDC_INBUF_SIZE);
	d.irq.interrupt_assert = arm_sink;
	if (d.inbuf == NULL) {
		printf("  SETUP latch inbuf failed\n");
		return;
	}
	d.write_in_progress = WDCC_WRITE;
	d.write_count  = 1;
	d.write_offset = 0xe000;

	/*  fire the drain announce once  */
	stub_fatals = 0; stub_drain_msgs = 0;
	memset(buf, 0, sizeof(buf));
	dev_wdc_access(fake_cpu(), NULL, wd_data, buf, 8, MEM_WRITE, &d);
	first = stub_drain_msgs;

	/*  a new command arrives, then the same provoking access again.  The latch
	    must NOT have been re-armed by the command.  */
	stub_disk_exists = 1;
	d.seccnt = 1;
	d.sector = 1;
	wdc_command(fake_cpu(), &d, WDCC_WRITE);
	stub_disk_exists = 0;

	stub_fatals = 0; stub_drain_msgs = 0;
	memset(buf, 0, sizeof(buf));
	dev_wdc_access(fake_cpu(), NULL, wd_data, buf, 8, MEM_WRITE, &d);
	second = stub_drain_msgs;

	rows ++;
	if (first < 1) {
		printf("  FAIL %-38s never armed (%i)\n",
		    "LIVENESS: the announce fired once", first);
		failures ++;
	} else {
		printf("  ok   %-38s %i\n",
		    "LIVENESS: the announce fired once", first);
	}

	rows ++;
	if (second != 0) {
		printf("  FAIL %-38s re-armed by a command (%i)\n",
		    "a latch survives the next command", second);
		failures ++;
	} else {
		printf("  ok   %-38s %i\n",
		    "a latch survives the next command", second);
	}

	free(d.inbuf);
}


/*  wdcwipclear -- REPRODUCTION.  `wdc_command()` clears `atapi_cmd_in_progress`
 *  and `error`, and does NOT clear `write_in_progress`.
 *
 *  So a guest can issue WRITE SECTORS and then a packet command, and the device
 *  holds BOTH flags live at once: the one shared input ring is then serving the
 *  disk-write path and the CDB path simultaneously.  That is the root of the
 *  alias -- #478 guards the sink, this is the source.
 *
 *  LIVENESS CONTROL FIRST.  The row asserts the flag IS set after the WRITE
 *  command before asserting it is cleared by the next one, because otherwise a
 *  fixture where nothing happened at all would pass and read as a fix.
 */
static void row_wip_not_cleared(void)
{
	struct wdc_data d;

	memset(&d, 0, sizeof(d));
	d.addr_mult  = 1;
	d.io_enabled = 1;
	d.inbuf = (unsigned char *) calloc(1, WDC_INBUF_SIZE);
	d.irq.interrupt_assert = arm_sink;
	if (d.inbuf == NULL) {
		printf("  SETUP wip inbuf failed\n");
		return;
	}

	/*  the command path aborts on a missing image, so the write would never be
	    set up at all and the row would pass for the wrong reason.  */
	stub_disk_exists = 1;
	d.seccnt = 1;
	d.sector = 1;

	wdc_command(fake_cpu(), &d, WDCC_WRITE);

	rows ++;
	if (d.write_in_progress == 0) {
		printf("  FAIL %-38s flag never set -- row is vacuous\n",
		    "LIVENESS: a WRITE arms write_in_prog");
		failures ++;
	} else {
		printf("  ok   %-38s 0x%x\n",
		    "LIVENESS: a WRITE arms write_in_prog", d.write_in_progress);
	}

	/*  now a packet command supersedes it.  wdc_command clears the ATAPI flag
	    here; it must clear this one too, or both are live at once.  */
	wdc_command(fake_cpu(), &d, ATAPI_PKT_CMD);

	rows ++;
	if (d.write_in_progress != 0) {
		printf("  FAIL %-38s still 0x%x with ATAPI live\n",
		    "a new command clears write_in_prog",
		    d.write_in_progress);
		failures ++;
	} else {
		printf("  ok   %-38s %i\n",
		    "a new command clears write_in_prog", d.write_in_progress);
	}

	stub_disk_exists = 0;
	if (d.atapi_st != NULL) {
		free(d.atapi_st->data_in);
		free(d.atapi_st->cmd);
		free(d.atapi_st);
	}
	free(d.inbuf);
}

/*  wdcringnoflush -- REPRODUCTION.  A new command does not flush the input ring.
 *
 *  MEASURED by a seat: under-drain a DATA-IN transfer and the residue is never
 *  cleared.  ATAPI_PKT_CMD does not flush it, ATAPI_SOFT_RESET does not flush it,
 *  and the ALT-CTRL device reset does not flush it -- so occupancy only ever
 *  climbs, and once it can no longer EQUAL 12 the CDB dispatch is dead for the
 *  life of the device.  There is no sequence a guest can issue to recover.
 *
 *  Same function and same semantic class as `wdcwipclear` -- "a new command must
 *  reset what the old one left" -- and observable through the same oracle, which
 *  is why both are built against one detector.
 *
 *  The in-tree idiom for the fix already exists: dev_asc.c's dev_asc_fifo_flush(),
 *  called from NCRCMD_FLUSH and NCRCMD_RSTCHIP.
 */
static void row_ring_not_flushed(void)
{
	struct wdc_data d;
	int occ;

	memset(&d, 0, sizeof(d));
	d.addr_mult  = 1;
	d.io_enabled = 1;
	d.inbuf = (unsigned char *) calloc(1, WDC_INBUF_SIZE);
	d.irq.interrupt_assert = arm_sink;
	if (d.inbuf == NULL) {
		printf("  SETUP flush inbuf failed\n");
		return;
	}

	stub_disk_exists = 1;

	/*  an under-drained transfer: 20 bytes the guest never read back  */
	{
		int i;
		for (i = 0; i < 20; i++)
			wdc_addtoinbuf(&d, 0x5a);
	}

	occ = d.inbuf_head - d.inbuf_tail;
	while (occ < 0)
		occ += WDC_INBUF_SIZE;
	rows ++;
	if (occ != 20) {
		printf("  FAIL %-38s occupancy %i, want 20\n",
		    "LIVENESS: the residue is really there", occ);
		failures ++;
	} else {
		printf("  ok   %-38s %i\n",
		    "LIVENESS: the residue is really there", occ);
	}

	/*  a fresh packet command.  A device that resets its command state must not
	    carry the previous transfer's residue into the new CDB.  */
	wdc_command(fake_cpu(), &d, ATAPI_PKT_CMD);

	occ = d.inbuf_head - d.inbuf_tail;
	while (occ < 0)
		occ += WDC_INBUF_SIZE;
	rows ++;
	if (occ != 0) {
		printf("  FAIL %-38s %i bytes survive the command\n",
		    "a packet command flushes the ring", occ);
		failures ++;
	} else {
		printf("  ok   %-38s %i\n",
		    "a packet command flushes the ring", occ);
	}

	/*  and a SOFT RESET is the guest's documented recovery.  If neither the
	    command nor the reset clears it, nothing does.  */
	wdc_command(fake_cpu(), &d, ATAPI_SOFT_RESET);

	occ = d.inbuf_head - d.inbuf_tail;
	while (occ < 0)
		occ += WDC_INBUF_SIZE;
	rows ++;
	if (occ != 0) {
		printf("  FAIL %-38s %i bytes survive a soft reset\n",
		    "a soft reset flushes the ring", occ);
		failures ++;
	} else {
		printf("  ok   %-38s %i\n",
		    "a soft reset flushes the ring", occ);
	}

	stub_disk_exists = 0;
	if (d.atapi_st != NULL) {
		free(d.atapi_st->data_in);
		free(d.atapi_st->cmd);
		free(d.atapi_st);
	}
	free(d.inbuf);
}

/*  THE RESIDUE ROW, RE-CUT.  L1 construction: this never shipped.
 *
 *  Its scratchpad form pinned the DEFECT'S SHAPE -- that a residue-contaminated
 *  drain fires, leaving occupancy 0 with `first=0xaa`.  That form earned its keep:
 *  it refuted this round's own "the residue is unrecoverable" premise and turned a
 *  denial-of-service claim into a data-integrity one.
 *
 *  But it goes red under every candidate fix, for the FIXTURE's reason rather than
 *  the code's, so it is replaced rather than kept alongside -- two rows disagreeing
 *  about one behaviour is how a suite stops meaning anything.
 *
 *  The new pin is the round's actual claim: after a new command the residue is gone,
 *  and the guest's next sector reaches the disk as the guest wrote it.
 */
static void row_residue_recovery(void)
{
	struct wdc_data d;
	unsigned char buf[8];
	int i, occ;

	memset(&d, 0, sizeof(d));
	d.addr_mult  = 1;
	d.io_enabled = 1;
	d.inbuf = (unsigned char *) calloc(1, WDC_INBUF_SIZE);
	d.irq.interrupt_assert = arm_sink;
	if (d.inbuf == NULL) {
		printf("  SETUP residue inbuf failed\n");
		return;
	}

	stub_disk_exists = 1;
	for (i = 0; i < 20; i++)
		wdc_addtoinbuf(&d, 0xaa);

	d.seccnt = 1;
	d.sector = 1;
	wdc_command(fake_cpu(), &d, WDCC_WRITE);

	occ = d.inbuf_head - d.inbuf_tail;
	while (occ < 0)
		occ += WDC_INBUF_SIZE;

	rows ++;
	if (occ != 0) {
		printf("  FAIL %-38s %i bytes survive the command\n",
		    "a new command leaves an empty ring", occ);
		failures ++;
	} else {
		printf("  ok   %-38s %i\n",
		    "a new command leaves an empty ring", occ);
	}

	/*  the guest-visible consequence: one sector, its OWN bytes.  A tree that
	    keeps the residue drains 20 bytes early and puts 0xaa at the head.  */
	stub_disk_writes = 0;
	stub_disk_bytes = 0;
	stub_disk_first = -1;
	for (i = 0; i < 256; i++) {
		buf[0] = 0x55; buf[1] = 0x55;
		buf[2] = 0; buf[3] = 0;
		dev_wdc_access(fake_cpu(), NULL, wd_data, buf, 2, MEM_WRITE, &d);
	}

	rows ++;
	if (stub_disk_writes != 1 || stub_disk_bytes != 512 ||
	    stub_disk_first != 0x55) {
		printf("  FAIL %-38s %i w, %i b, first=0x%02x\n",
		    "the sector after a command is guest's",
		    stub_disk_writes, stub_disk_bytes, stub_disk_first);
		failures ++;
	} else {
		printf("  ok   %-38s %i w, %i b\n",
		    "the sector after a command is guest's",
		    stub_disk_writes, stub_disk_bytes);
	}

	stub_disk_exists = 0;
	if (d.atapi_st != NULL) {
		free(d.atapi_st->data_in);
		free(d.atapi_st->cmd);
		free(d.atapi_st);
	}
	free(d.inbuf);
}

/*  A REJECTED ACCESS MUST HAVE NO SIDE EFFECT -- including the CDB dispatch.
 *
 *  Named by the codex seat on `#478`'s pass 2: `badlen_this` guards the sector
 *  drain and NOT the CDB dispatch, so an access of an unimplemented width -- which
 *  pushes nothing and is reported "(ignored)" -- can still issue a 12-byte SCSI
 *  command when occupancy was already 12.
 *
 *  The existing `no-push write does not dispatch` row does not cover this: it sits
 *  in PHASE_DATAIN, so `#477`'s phase guard blocks the dispatch for an unrelated
 *  reason.  Here the phase is CMDOUT -- where a real CDB is collected -- which is
 *  the state that makes the question live.
 *
 *  If this row is RED, `#478`'s claim that `#337`'s "(ignored)" became TRUE is a
 *  wrong record in product code, and a rejected access still has a side effect.
 */
static void row_badlen_no_dispatch(void)
{
	struct wdc_data d;
	unsigned char buf[8];
	int i;

	memset(&d, 0, sizeof(d));
	d.addr_mult  = 1;
	d.io_enabled = 1;
	d.inbuf = (unsigned char *) calloc(1, WDC_INBUF_SIZE);
	d.irq.interrupt_assert = arm_sink;
	if (d.inbuf == NULL) {
		printf("  SETUP badlen_cdb inbuf failed\n");
		return;
	}

	/*  the real device sets all three together  */
	d.cur_command = ATAPI_PKT_CMD;
	d.atapi_cmd_in_progress = 1;
	d.atapi_phase = PHASE_CMDOUT;

	/*  a CDB's worth already queued -- occupancy is exactly 12 BEFORE the
	    rejected access, so the access itself pushes nothing and changes it
	    not at all.  */
	for (i = 0; i < 12; i++)
		wdc_addtoinbuf(&d, 0x12);

	stub_scsi_cmds = 0;
	stub_fatals = 0;

	/*  an 8-byte store: the width switch rejects it and says so.  */
	memset(buf, 0, sizeof(buf));
	dev_wdc_access(fake_cpu(), NULL, wd_data, buf, 8, MEM_WRITE, &d);

	rows ++;
	if (stub_scsi_cmds != 0) {
		printf("  FAIL %-38s dispatched %i\n",
		    "a rejected width issues no command", stub_scsi_cmds);
		failures ++;
	} else {
		printf("  ok   %-38s %i\n",
		    "a rejected width issues no command", stub_scsi_cmds);
	}

	/*  and it must SAY something -- a silently rejected access with a side
	    effect is the worst of both.  */
	rows ++;
	if (stub_fatals < 1) {
		printf("  FAIL %-38s emitted %i\n",
		    "a rejected width is announced", stub_fatals);
		failures ++;
	} else {
		printf("  ok   %-38s %i\n",
		    "a rejected width is announced", stub_fatals);
	}

	if (d.atapi_st != NULL) {
		free(d.atapi_st->data_in);
		free(d.atapi_st->cmd);
		free(d.atapi_st);
	}
	free(d.inbuf);
}


/*  WRITEMULTI -- the arm that had NO ROW AT ALL.
 *
 *  Every drain row in this file used WDCC_WRITE with write_count = 1, so the
 *  multi-sector half of the guard AND of the announce arm `#478` added was
 *  untested.  A seat measured FOUR mutants sweeping all 60 rows on that one gap,
 *  including one that reads as pure de-duplication -- folding the two arms, since
 *  they differ only in a modulus -- and writes 131,072 bytes.
 *
 *  THE MOMENT IS PINNED, NOT JUST THE SHAPE.  A first attempt asserted "one write,
 *  1024 bytes" and a mutant passed it by draining at occupancy 512 with count=2:
 *  the right shape at the wrong moment.  511 stores must drain NOTHING.
 */
static void row_writemulti_moment(void)
{
	struct wdc_data d;
	unsigned char buf[8];
	int i;

	memset(&d, 0, sizeof(d));
	d.addr_mult  = 1;
	d.io_enabled = 1;
	d.inbuf = (unsigned char *) calloc(1, WDC_INBUF_SIZE);
	d.irq.interrupt_assert = arm_sink;
	if (d.inbuf == NULL) {
		printf("  SETUP writemulti inbuf failed\n");
		return;
	}

	d.write_in_progress = WDCC_WRITEMULTI;
	d.write_count  = 2;
	d.write_offset = 0xf000;

	stub_disk_writes = 0;
	stub_disk_bytes = 0;
	stub_disk_first = -1;

	/*  511 two-byte stores = 1022 bytes: two bytes short of two sectors.  */
	for (i = 0; i < 511; i++) {
		buf[0] = 0x7c; buf[1] = 0x7c;
		buf[2] = 0; buf[3] = 0;
		dev_wdc_access(fake_cpu(), NULL, wd_data, buf, 2, MEM_WRITE, &d);
	}

	rows ++;
	if (stub_disk_writes != 0) {
		printf("  FAIL %-38s %i w at occupancy 1022\n",
		    "WRITEMULTI partial does not drain", stub_disk_writes);
		failures ++;
	} else {
		printf("  ok   %-38s %i\n",
		    "WRITEMULTI partial does not drain", stub_disk_writes);
	}

	/*  the completing store: exactly one write of BOTH sectors, guest bytes.  */
	buf[0] = 0x7c; buf[1] = 0x7c;
	buf[2] = 0; buf[3] = 0;
	dev_wdc_access(fake_cpu(), NULL, wd_data, buf, 2, MEM_WRITE, &d);

	rows ++;
	if (stub_disk_writes != 1 || stub_disk_bytes != 1024 ||
	    stub_disk_first != 0x7c) {
		printf("  FAIL %-38s %i w, %i b, first=0x%02x\n",
		    "WRITEMULTI drains both sectors once",
		    stub_disk_writes, stub_disk_bytes, stub_disk_first);
		failures ++;
	} else {
		printf("  ok   %-38s %i w, %i b\n",
		    "WRITEMULTI drains both sectors once",
		    stub_disk_writes, stub_disk_bytes);
	}

	free(d.inbuf);
}


/*  THE LEFTOVER DOOR MUST STAY SHUT ON THE SECOND BAD-LEN STORE TOO.
 *
 *  Named by a seat as the highest-plausibility slip: move `badlen_this = 1;`
 *  inside the `if (!d->warned_badlen)` block -- one line, and it looks like
 *  tidying the latch.  Because `warned_badlen` latches for the device's life,
 *  EVERY bad-length store after the first then drains 512 bytes of unread payload
 *  to the guest's LBA.
 *
 *  The existing leftover row uses ONE bad-len store and cannot see it.
 */
static void row_leftover_second_badlen(void)
{
	struct wdc_data d;
	unsigned char buf[8];
	int i;

	memset(&d, 0, sizeof(d));
	d.addr_mult  = 1;
	d.io_enabled = 1;
	d.inbuf = (unsigned char *) calloc(1, WDC_INBUF_SIZE);
	d.irq.interrupt_assert = arm_sink;
	if (d.inbuf == NULL) {
		printf("  SETUP second_badlen inbuf failed\n");
		return;
	}

	/*  an unread payload, marked so the oracle names WHICH bytes escaped  */
	for (i = 0; i < 512; i++)
		wdc_addtoinbuf(&d, 0xbe);

	d.write_in_progress = WDCC_WRITE;
	d.write_count  = 1;
	d.write_offset = 0x11000;

	/*  the FIRST bad-len store also arms #337's latch  */
	memset(buf, 0, sizeof(buf));
	dev_wdc_access(fake_cpu(), NULL, wd_data, buf, 8, MEM_WRITE, &d);

	/*  the SECOND is the one under test: the latch is now set, so a fix that
	    ties badlen_this to it stops declining here.  */
	stub_disk_writes = 0;
	stub_disk_first = -1;
	memset(buf, 0, sizeof(buf));
	dev_wdc_access(fake_cpu(), NULL, wd_data, buf, 8, MEM_WRITE, &d);

	rows ++;
	if (stub_disk_writes != 0) {
		printf("  FAIL %-38s %i write(s), first=0x%02x\n",
		    "leftover safe on the 2nd bad-len too",
		    stub_disk_writes, stub_disk_first);
		failures ++;
	} else {
		printf("  ok   %-38s %i\n",
		    "leftover safe on the 2nd bad-len too", stub_disk_writes);
	}

	free(d.inbuf);
}


/*  THE LITTLE-ENDIAN ARM, which had ZERO coverage until now.
 *
 *  The width switch is duplicated once per endianness branch, and only the
 *  big-endian copy was ever reached.  A one-line deletion in the little-endian
 *  copy swept all 60 rows -- byte-identical output, same md5.
 *
 *  This runs the decline path through the OTHER arm.  It is not a duplicate of
 *  the big-endian rows: it is the same assertion against a second copy of the
 *  code, which is exactly what a duplicated switch requires.
 *
 *  `#337`'s comment records that `warned_badlen` is shared across the two arms
 *  BECAUSE they are mutually exclusive at runtime -- true, and it is also why a
 *  latch-based reading of the flag cannot distinguish them.  The rows must.
 */
static void row_little_endian_arm(void)
{
	struct wdc_data d;
	unsigned char buf[8];
	int i;

	memset(&d, 0, sizeof(d));
	d.addr_mult  = 1;
	d.io_enabled = 1;
	d.inbuf = (unsigned char *) calloc(1, WDC_INBUF_SIZE);
	d.irq.interrupt_assert = arm_sink;
	if (d.inbuf == NULL) {
		printf("  SETUP le_arm inbuf failed\n");
		return;
	}

	stub_byte_order = EMUL_LITTLE_ENDIAN;

	/*  an unread payload, so a failed decline writes something identifiable  */
	for (i = 0; i < 512; i++)
		wdc_addtoinbuf(&d, 0xbe);

	d.write_in_progress = WDCC_WRITE;
	d.write_count  = 1;
	d.write_offset = 0x12000;

	stub_disk_writes = 0;
	stub_disk_first = -1;
	memset(buf, 0, sizeof(buf));
	dev_wdc_access(fake_cpu(), NULL, wd_data, buf, 8, MEM_WRITE, &d);

	rows ++;
	if (stub_disk_writes != 0) {
		printf("  FAIL %-38s %i w, first=0x%02x\n",
		    "LE arm: leftover is not drained",
		    stub_disk_writes, stub_disk_first);
		failures ++;
	} else {
		printf("  ok   %-38s %i\n",
		    "LE arm: leftover is not drained", stub_disk_writes);
	}

	/*  and a LIVENESS control -- prove the LE arm was actually taken, or the
	    row above passes for the same reason it always did.  A 2-byte store
	    on this arm pushes low byte first; on the other, high byte first.  */
	d.inbuf_head = d.inbuf_tail = 0;
	buf[0] = 0x34; buf[1] = 0x12;
	buf[2] = 0; buf[3] = 0;
	dev_wdc_access(fake_cpu(), NULL, wd_data, buf, 2, MEM_WRITE, &d);

	rows ++;
	if (d.inbuf[0] != 0x34) {
		printf("  FAIL %-38s first pushed 0x%02x, want 0x34\n",
		    "LIVENESS: the LE arm really ran", d.inbuf[0]);
		failures ++;
	} else {
		printf("  ok   %-38s 0x%02x\n",
		    "LIVENESS: the LE arm really ran", d.inbuf[0]);
	}

	stub_byte_order = 0;
	free(d.inbuf);
}


/*  #480: THREE ROWS THAT WATCH THE HELPER'S CALL, NOT ITS BODY.
 *
 *  #479's thirteen mutants all edit what wdc_new_command_reset() DOES.  Every one
 *  dies.  Three edits to WHETHER IT IS CALLED escape with byte-identical output,
 *  because every #479 row calls wdc_command() directly with two DIFFERENT opcodes
 *  and a disk that exists -- so no row is in a position to notice a condition
 *  appearing in front of the call.
 *
 *  Each row below is aimed at one of them, and each asserts a value MEASURED on
 *  the committed build by a register-level witness probe.
 */
static void row_same_opcode_resets(void)
{
	struct wdc_data d;
	unsigned char buf[8];
	int i, occ;

	memset(&d, 0, sizeof(d));
	d.addr_mult  = 1;
	d.io_enabled = 1;
	d.inbuf = (unsigned char *) calloc(1, WDC_INBUF_SIZE);
	d.irq.interrupt_assert = arm_sink;
	if (d.inbuf == NULL) {
		printf("  SETUP same_opcode inbuf failed\n");
		return;
	}
	stub_disk_exists = 1;

	/*  The guest ISSUES a WRITE -- as a real command, so the device sets
	    cur_command and arms the transfer itself.  This is what makes the
	    reissue below a genuine same-opcode reissue instead of a fixture
	    that asserts one: a row that pre-sets cur_command makes M11's
	    `cur_command == idata` true BY CONSTRUCTION, which is the artifact
	    the kimi-k3:cloud seat named.  Here the device produces it.  */
	d.seccnt = 1;
	wdc_command(fake_cpu(), &d, WDCC_WRITE);

	/*  ...streams 100 bytes, then gives up.  */
	for (i = 0; i < 50; i++) {
		buf[0] = 0xaa; buf[1] = 0xaa;
		dev_wdc_access(fake_cpu(), NULL, wd_data, buf, 2, MEM_WRITE, &d);
	}

	/*  It re-issues THE SAME OPCODE.  A retry is a new command: the abandoned
	    100 bytes are not part of it.  M11 reads this as "a repeated opcode is
	    a retry, not a new command" and skips the reset.  */
	/*  SETUP LIVENESS.  Without this the row asserts an outcome without
	    proving its premise: if the staging loop stopped working, the ring
	    would be empty and a clean 0xbb sector would land for a reason that
	    has nothing to do with the reset.  Named by the codex seat.  */
	{
		int staged = d.inbuf_head - d.inbuf_tail;
		while (staged < 0)
			staged += WDC_INBUF_SIZE;
		rows ++;
		if (staged != 100) {
			printf("  FAIL %-38s staged %i\n",
			    "setup: 100 bytes are abandoned", staged);
			failures ++;
		} else {
			printf("  ok   %-38s %i\n",
			    "setup: 100 bytes are abandoned", staged);
		}
	}

	stub_disk_writes = 0;
	stub_disk_bytes = 0;
	stub_disk_first = -1;
	/*  NOT pre-set: wdc_command() assigns cur_command itself, and pre-setting
	    it makes M11's `cur_command == idata` true BY CONSTRUCTION.  Here the
	    same-opcode condition must come from the SCENARIO -- the previous
	    WDCC_WRITE left it set -- not from the fixture.  */
	wdc_command(fake_cpu(), &d, WDCC_WRITE);

	/*  Now a full, legal sector of its own.  */
	for (i = 0; i < 256; i++) {
		buf[0] = 0xbb; buf[1] = 0xbb;
		dev_wdc_access(fake_cpu(), NULL, wd_data, buf, 2, MEM_WRITE, &d);
	}

	occ = d.inbuf_head - d.inbuf_tail;
	while (occ < 0)
		occ += WDC_INBUF_SIZE;

	rows ++;
	if (stub_disk_first != 0xbb || occ != 0) {
		printf("  FAIL %-38s first 0x%02x residue %i\n",
		    "a same-opcode reissue resets the ring",
		    stub_disk_first, occ);
		failures ++;
	} else {
		printf("  ok   %-38s first 0x%02x residue %i\n",
		    "a same-opcode reissue resets the ring",
		    stub_disk_first, occ);
	}

	stub_disk_exists = 0;
	free(d.inbuf);
}


static void row_rejected_cmd_resets(void)
{
	struct wdc_data d;
	unsigned char buf[8];
	int i, before;

	memset(&d, 0, sizeof(d));
	d.addr_mult  = 1;
	d.io_enabled = 1;
	d.inbuf = (unsigned char *) calloc(1, WDC_INBUF_SIZE);
	d.irq.interrupt_assert = arm_sink;
	if (d.inbuf == NULL) {
		printf("  SETUP rejected_cmd inbuf failed\n");
		return;
	}
	stub_disk_exists = 1;
	stub_cdrom_poison = 1;		/*  an ATAPI-only controller  */

	/*  A packet command collects EIGHT of its twelve CDB bytes...  */
	wdc_command(fake_cpu(), &d, ATAPI_PKT_CMD);
	for (i = 0; i < 4; i++) {
		buf[0] = 0x28; buf[1] = 0x00;
		dev_wdc_access(fake_cpu(), NULL, wd_data, buf, 2, MEM_WRITE, &d);
	}

	/*  ...and the guest then issues IDENTIFY, which an ATAPI drive ABORTS.
	    That is the single most routine probe any ATA driver makes, and ABRT
	    is the correct answer -- but the command still HAPPENED, so the
	    abandoned CDB must not survive it.  M4 moves the reset below the
	    early-return guards, so a rejected command clears nothing.  */
	before = stub_scsi_cmds;
	/*  NOT pre-set.  A real guest reaching here has cur_command ==
	    ATAPI_PKT_CMD, so IDENTIFY is a DIFFERENT opcode -- which is the
	    scenario.  Pre-setting it made M11 skip the reset here too and
	    contributed a red that was an artifact of the fixture rather than
	    of the defect.  Named by the kimi-k3:cloud seat.  */
	wdc_command(fake_cpu(), &d, WDCC_IDENTIFY);

	/*  Four more stray bytes.  Under M4 these COMPLETE the abandoned CDB and
	    the device issues a SCSI command assembled from it -- on a command it
	    has just told the guest was aborted.  */
	for (i = 0; i < 2; i++) {
		buf[0] = 0xff; buf[1] = 0xff;
		dev_wdc_access(fake_cpu(), NULL, wd_data, buf, 2, MEM_WRITE, &d);
	}

	rows ++;
	/*  The WDCE_ABRT conjunct is NOT decoration.  MEASURED: with
	    stub_cdrom_poison forced to 0 so the guard CANNOT fire, the other
	    two conditions stay green -- a plain IDENTIFY on a non-CD-ROM runs
	    the reset too and issues no SCSI command.  So without this the row
	    passes on a command that was never rejected, which is not what its
	    name claims.  It still killed M4 either way; a row that is green for
	    the wrong reason is a row whose next edit is unattributable.  */
	if (stub_scsi_cmds != before || d.atapi_cmd_in_progress != 0 ||
	    !(d.error & WDCE_ABRT)) {
		printf("  FAIL %-38s %i cmd(s) flag %i abrt %i\n",
		    "a rejected command still resets",
		    stub_scsi_cmds - before, d.atapi_cmd_in_progress,
		    !!(d.error & WDCE_ABRT));
		failures ++;
	} else {
		printf("  ok   %-38s %i cmd(s) flag %i abrt %i\n",
		    "a rejected command still resets",
		    stub_scsi_cmds - before, d.atapi_cmd_in_progress,
		    !!(d.error & WDCE_ABRT));
	}

	stub_cdrom_poison = 0;
	stub_disk_exists = 0;
	if (d.atapi_st != NULL) {
		free(d.atapi_st->data_in);
		free(d.atapi_st->cmd);
		free(d.atapi_st);
	}
	free(d.inbuf);
}


static void row_refill_cmd_resets(void)
{
	struct wdc_data d;
	unsigned char buf[8];
	int occ;

	memset(&d, 0, sizeof(d));
	d.addr_mult  = 1;
	d.io_enabled = 1;
	d.inbuf = (unsigned char *) calloc(1, WDC_INBUF_SIZE);
	d.irq.interrupt_assert = arm_sink;
	if (d.inbuf == NULL) {
		printf("  SETUP refill_cmd inbuf failed\n");
		return;
	}
	stub_disk_exists = 1;

	/*  IDENTIFY fills the ring with its 512-byte answer...  */
	wdc_command(fake_cpu(), &d, WDCC_IDENTIFY);

	/*  ...the guest reads one word and abandons it...  */
	memset(buf, 0, sizeof(buf));
	dev_wdc_access(fake_cpu(), NULL, wd_data, buf, 2, MEM_READ, &d);

	/*  ...and re-issues IDENTIFY.  The answer must be a FRESH block, not the
	    old tail with a new block queued behind it.  M23 exempts exactly these
	    refilling commands from the reset -- "they load their answer INTO the
	    ring, so do not flush for them" -- which is a direct misreading of the
	    round's own comment about why the helper runs at the TOP.  */
	/*  SETUP LIVENESS.  510 bytes must actually be sitting there, or the
	    final occupancy-512 assertion is satisfied by a ring that was empty
	    all along and proves nothing about the flush.  Named by codex.  */
	{
		int left = d.inbuf_head - d.inbuf_tail;
		while (left < 0)
			left += WDC_INBUF_SIZE;
		rows ++;
		if (left != 510) {
			printf("  FAIL %-38s left %i\n",
			    "setup: 510 bytes remain unread", left);
			failures ++;
		} else {
			printf("  ok   %-38s %i\n",
			    "setup: 510 bytes remain unread", left);
		}
	}

	wdc_command(fake_cpu(), &d, WDCC_IDENTIFY);

	occ = d.inbuf_head - d.inbuf_tail;
	while (occ < 0)
		occ += WDC_INBUF_SIZE;

	rows ++;
	if (occ != 512) {
		printf("  FAIL %-38s occupancy %i\n",
		    "a refill command starts from empty", occ);
		failures ++;
	} else {
		printf("  ok   %-38s occupancy %i\n",
		    "a refill command starts from empty", occ);
	}

	stub_disk_exists = 0;
	free(d.inbuf);
}


/*  #480: THE CALL CAN STOP RUNNING ON A PATH WITHOUT BEING "CONDITIONED".
 *
 *  wdc_command() has TWO early returns.  This round's first three rows covered the
 *  second (the CD-ROM/IDENTIFY abort) and left the first -- the no-disk guard --
 *  reachable by no row at all: every one of the seventeen wdc_command() calls in
 *  the file runs with stub_disk_exists = 1.
 *
 *  So moving the helper call below the FIRST guard only, which is strictly smaller
 *  than the M4 mutant this round was built for, escaped every row.  Measured worse
 *  than pre-#479: a driver probing an absent drive -- which every ATA bus scan does
 *  -- leaves an abandoned CDB live, and stray bytes then complete it into a SCSI
 *  command on a command the device has just aborted.
 */
static void row_absent_drive_resets(void)
{
	struct wdc_data d;
	unsigned char buf[8];
	int i, before;

	memset(&d, 0, sizeof(d));
	d.addr_mult  = 1;
	d.io_enabled = 1;
	d.inbuf = (unsigned char *) calloc(1, WDC_INBUF_SIZE);
	d.irq.interrupt_assert = arm_sink;
	if (d.inbuf == NULL) {
		printf("  SETUP absent_drive inbuf failed\n");
		return;
	}

	/*  A packet command is in flight and has collected eight of twelve bytes.  */
	stub_disk_exists = 1;
	stub_cdrom_poison = 1;
	wdc_command(fake_cpu(), &d, ATAPI_PKT_CMD);
	for (i = 0; i < 4; i++) {
		buf[0] = 0x28; buf[1] = 0x00;
		dev_wdc_access(fake_cpu(), NULL, wd_data, buf, 2, MEM_WRITE, &d);
	}

	/*  The guest now addresses a drive that is not there.  The command is
	    aborted at the FIRST guard -- but it still happened, so the abandoned
	    CDB must not survive it.  */
	before = stub_scsi_cmds;
	stub_disk_exists = 0;
	wdc_command(fake_cpu(), &d, WDCC_IDENTIFY);

	/*  Stray bytes that would complete the abandoned CDB.  */

		buf[0] = 0xff; buf[1] = 0xff;

	}

	rows ++;
	if (stub_scsi_cmds != before || d.atapi_cmd_in_progress != 0 ||
	    !(d.error & WDCE_ABRT)) {
		printf("  FAIL %-38s %i cmd(s) flag %i abrt %i\n",
		    "an absent-drive command still resets",
		    stub_scsi_cmds - before, d.atapi_cmd_in_progress,
		    !!(d.error & WDCE_ABRT));
		failures ++;
	} else {
		printf("  ok   %-38s %i cmd(s) flag %i abrt %i\n",
		    "an absent-drive command still resets",
		    stub_scsi_cmds - before, d.atapi_cmd_in_progress,
		    !!(d.error & WDCE_ABRT));
	}

	stub_cdrom_poison = 0;
	if (d.atapi_st != NULL) {
		free(d.atapi_st->data_in);
		free(d.atapi_st->cmd);
		free(d.atapi_st);
	}
	free(d.inbuf);
}


/*  #480: ONE ROW OVER EVERY DISPATCHED OPCODE, because a four-token exemption is
 *  killed by a row that only exercises one of its tokens.
 *
 *  M23_exemptfill exempts IDENTIFY, READ, READMULTI and ATAPI_IDENTIFY_DEVICE and
 *  is scored KILLED -- yet THREE OF ITS FOUR CONJUNCTS ESCAPE ON THEIR OWN.  The
 *  sharpest is ATAPI_IDENTIFY_DEVICE, which shares a case label with WDCC_IDENTIFY
 *  and fills the ring identically: it is the standard ATAPI probe, so it is the
 *  MORE reachable half, and `row_refill_cmd_resets` uses WDCC_IDENTIFY only.
 *
 *  A row per opcode would be twenty rows.  The oracle that collapses them to one is
 *  `occ % 512 == 0`: a command that legitimately FILLS the ring leaves exactly 512,
 *  a command that does not leaves 0, and an unflushed 100-byte residue makes both
 *  impossible whatever the opcode does afterwards.
 *
 *  The enumeration is the function's own switch arms, read off the source -- not the
 *  opcodes some row happens to use.  That is what keeps it closed.
 */
static void row_every_opcode_resets(void)
{
	static const int op[] = {
		WDCC_READ, WDCC_READMULTI, WDCC_WRITE, WDCC_WRITEMULTI,
		WDCC_IDP, SET_FEATURES, WDCC_RECAL, WDCC_IDENTIFY,
		ATAPI_IDENTIFY_DEVICE, WDCC_IDLE_IMMED, WDCC_SETMULTI,
		ATAPI_SOFT_RESET, ATAPI_PKT_CMD
	};
	int k, bad = 0, firstbad = -1, badocc = -1;

	for (k = 0; k < (int) (sizeof(op) / sizeof(op[0])); k++) {
		struct wdc_data d;
		unsigned char buf[8];
		int i, occ;

		memset(&d, 0, sizeof(d));
		d.addr_mult  = 1;
		d.io_enabled = 1;
		d.inbuf = (unsigned char *) calloc(1, WDC_INBUF_SIZE);
		d.irq.interrupt_assert = arm_sink;
		if (d.inbuf == NULL)
			continue;
		stub_disk_exists = 1;

		/*  An abandoned WRITE with a 100-byte residue -- the exact state
		    #479 exists to clear.  */
		d.seccnt = 1;
		wdc_command(fake_cpu(), &d, WDCC_WRITE);
		for (i = 0; i < 50; i++) {
			buf[0] = 0xaa; buf[1] = 0xaa;
			dev_wdc_access(fake_cpu(), NULL, wd_data, buf, 2,
			    MEM_WRITE, &d);
		}

		/*  ...and then THIS opcode arrives.  */
		d.seccnt = 1;
		wdc_command(fake_cpu(), &d, op[k]);

		occ = d.inbuf_head - d.inbuf_tail;
		while (occ < 0)
			occ += WDC_INBUF_SIZE;

		if ((occ % 512) != 0) {
			bad ++;
			if (firstbad < 0) {
				firstbad = op[k];
				badocc = occ;
			}
		}

		if (d.atapi_st != NULL) {
			free(d.atapi_st->data_in);
			free(d.atapi_st->cmd);
			free(d.atapi_st);
		}
		free(d.inbuf);
		stub_disk_exists = 0;
	}

	rows ++;
	if (bad != 0) {
		printf("  FAIL %-38s %i of 13, first 0x%02x occ %i\n",
		    "every opcode clears the residue", bad, firstbad, badocc);
		failures ++;
	} else {
		printf("  ok   %-38s %i of 13\n",
		    "every opcode clears the residue", bad);
	}
}


/*  THE ALT-CTRL HANDLER: FOUR LINES, AND THE PANEL MEASURED THREE DEFECTS AND A
 *  MISSING FEATURE BEHIND THEM.
 *
 *      if (idata & WDCTL_4BIT)
 *              d->cur_command = COMMAND_RESET;
 *
 *  WDCTL_4BIT is 0x08, "use four head bits (wd1003)".  The reset bit is WDCTL_RST,
 *  0x04.  And COMMAND_RESET is not a reset ACTION at all -- it is read at exactly
 *  two sites and both make the cylinder registers return the ATAPI signature
 *  0xEB14 for a CD-ROM.  IT IS THE SIGNATURE LATCH.
 *
 *  Because it triggers on bit 3, which Linux libata holds as its steady-state
 *  ap->ctl, the signature is on almost always and the defect has never shown.
 *
 *  EACH ROW BUILDS ITS OWN DEVICE.  An earlier draft ran three rows off one
 *  struct and row 2's control write arrived carrying row 1's state -- its
 *  independence compromised by its neighbour.  Named by a panel seat.
 */
static void altctl_fresh(struct wdc_data *d)
{
	memset(d, 0, sizeof(*d));
	d->addr_mult  = 1;
	d->io_enabled = 1;
	d->inbuf = (unsigned char *) calloc(1, WDC_INBUF_SIZE);
	d->irq.interrupt_assert = arm_sink;
}


static void altctl_free(struct wdc_data *d)
{
	if (d->atapi_st != NULL) {
		free(d->atapi_st->data_in);
		free(d->atapi_st->cmd);
		free(d->atapi_st);
	}
	free(d->inbuf);
}


static void row_srst_is_a_reset(void)
{
	struct wdc_data d;
	unsigned char buf[8];

	altctl_fresh(&d);
	if (d.inbuf == NULL) {
		printf("  SETUP srst_is_a_reset\n");
		return;
	}

	/*  0x06 = WDCTL_RST | WDCTL_IDS -- what older OpenBSD writes to assert a
	    soft reset.  BIT 3 IS CLEAR, so today's handler does not see it.  */
	memset(buf, 0, sizeof(buf));
	buf[0] = WDCTL_RST | WDCTL_IDS;
	dev_wdc_altstatus_access(fake_cpu(), NULL, 0, buf, 1, MEM_WRITE, &d);

	rows ++;
	if (d.cur_command != COMMAND_RESET) {
		printf("  FAIL %-38s cur_command 0x%x\n",
		    "a real SRST (0x06) is a reset", d.cur_command);
		failures ++;
	} else {
		printf("  ok   %-38s 0x%x\n",
		    "a real SRST (0x06) is a reset", d.cur_command);
	}
	altctl_free(&d);
}


/*  *** THIS ROW IS THE ORACLE CORRECTION, and the reason is worth the space.
 *
 *  It first read `cur_command != COMMAND_RESET` -- "a routine 0x08 must not leave
 *  the device in the reset state".  A seat MEASURED that going RED on CORRECT code
 *  the moment a power-on signature is modelled, because then the device legitimately
 *  IS in the reset state before any command runs.  The old oracle conflated "this
 *  write did not latch" with "the device is not latched".
 *
 *  The delta form asks the question that actually matters and is robust to A4: DID
 *  THIS WRITE CHANGE cur_command?  And the harm it states is the sharper one the
 *  brief had understated -- `cur_command` is also the value assigned into
 *  write_in_progress, so a routine Device Control write DESTROYS THE RECORD OF WHICH
 *  COMMAND IS EXECUTING.  That is safe today only because wdc__write() has exactly
 *  one caller, immediately after the assignment: an accident, not an invariant.  ***
 */
static void row_routine_ctl_preserves_cmd(void)
{
	struct wdc_data d;
	unsigned char buf[8];
	int before;

	altctl_fresh(&d);
	if (d.inbuf == NULL) {
		printf("  SETUP routine_ctl\n");
		return;
	}

	/*  A command is executing.  0xec is IDENTIFY -- an ordinary one.  */
	d.cur_command = WDCC_IDENTIFY;
	before = d.cur_command;

	/*  ...and the driver writes its steady-state Device Control byte.  libata
	    does this constantly; it is not a reset and must change nothing.  */
	memset(buf, 0, sizeof(buf));
	buf[0] = WDCTL_4BIT;
	dev_wdc_altstatus_access(fake_cpu(), NULL, 0, buf, 1, MEM_WRITE, &d);

	rows ++;
	if (d.cur_command != before) {
		printf("  FAIL %-38s 0x%x -> 0x%x\n",
		    "a routine 0x08 preserves the command",
		    before, d.cur_command);
		failures ++;
	} else {
		printf("  ok   %-38s 0x%x\n",
		    "a routine 0x08 preserves the command", d.cur_command);
	}
	altctl_free(&d);
}


/*  The `m1` escape, built as a row before the fix exists.  A mask widened to
 *  (WDCTL_RST | WDCTL_IDS) escaped every row a seat could throw at it -- because the
 *  rows' entire control vocabulary was {0x06, 0x08, 0x0c} and nothing wrote an
 *  IDS-without-RST value.  0x0a is libata's polled-mode staple and 0x02 is a bare
 *  nIEN write, and NEITHER carries bit 2 (WDCTL_RST), so neither may latch.
 *
 *  *** ONLY 0x0a LATCHED ON THE COMMITTED DEVICE, NOT BOTH.  An earlier version of
 *  this comment said "both carry bit 3 and so latch TODAY".  0x02 & 0x08 == 0, and
 *  the round's own pre-fix log says so: `0x0a->0x100 0x02->0xec`.  The row is
 *  unaffected -- it asserts NEITHER may latch, which is true before and after -- but
 *  the stated reason was wrong and a wrong reason gets reused.  Named by the grok
 *  seat reading the log against the comment. ***
 */
static void row_nien_is_not_a_reset(void)
{
	struct wdc_data d;
	unsigned char buf[8];
	int a, b;

	altctl_fresh(&d);
	if (d.inbuf == NULL) {
		printf("  SETUP nien\n");
		return;
	}

	d.cur_command = WDCC_IDENTIFY;
	memset(buf, 0, sizeof(buf));
	buf[0] = WDCTL_IDS | WDCTL_4BIT;		/*  0x0a  */
	dev_wdc_altstatus_access(fake_cpu(), NULL, 0, buf, 1, MEM_WRITE, &d);
	a = d.cur_command;

	altctl_free(&d);
	altctl_fresh(&d);
	if (d.inbuf == NULL)
		return;
	d.cur_command = WDCC_IDENTIFY;
	memset(buf, 0, sizeof(buf));
	buf[0] = WDCTL_IDS;				/*  0x02  */
	dev_wdc_altstatus_access(fake_cpu(), NULL, 0, buf, 1, MEM_WRITE, &d);
	b = d.cur_command;

	rows ++;
	/*  (B) A DELTA, NOT AN ABSOLUTE.  This asked only whether the value BECAME
	    COMMAND_RESET, where its sibling row asks whether the value CHANGED.
	    A mutant that clobbers cur_command to 0 on an nIEN write -- which is
	    this round's own defect #2, reached through a different bit -- walked
	    straight through that gap and the row still reported ok.  Measured by
	    the opus seat (`Mo_nienzero`, `Mo_nien4bit`).  The re-pin's own
	    rationale applied to both rows; only one of them got it.  */
	if (a != WDCC_IDENTIFY || b != WDCC_IDENTIFY) {
		printf("  FAIL %-38s 0x0a->0x%x 0x02->0x%x\n",
		    "an nIEN write is not a reset", a, b);
		failures ++;
	} else {
		printf("  ok   %-38s 0x%x 0x%x\n",
		    "an nIEN write is not a reset", a, b);
	}
	altctl_free(&d);
}


static void row_srst_aborts(void)
{
	struct wdc_data d;
	unsigned char buf[8];
	int i, occ;

	altctl_fresh(&d);
	if (d.inbuf == NULL) {
		printf("  SETUP srst_aborts\n");
		return;
	}

	/*  A write is armed, a packet is mid-CDB, and the ring holds 40 bytes.  */
	d.write_in_progress = WDCC_WRITE;
	d.write_count  = 1;
	d.atapi_cmd_in_progress = 1;
	d.atapi_phase = PHASE_CMDOUT;
	for (i = 0; i < 40; i++)
		wdc_addtoinbuf(&d, 0x5a);

	memset(buf, 0, sizeof(buf));
	buf[0] = WDCTL_RST | WDCTL_4BIT;		/*  0x0c -- assert  */
	dev_wdc_altstatus_access(fake_cpu(), NULL, 0, buf, 1, MEM_WRITE, &d);

	occ = d.inbuf_head - d.inbuf_tail;
	while (occ < 0)
		occ += WDC_INBUF_SIZE;

	rows ++;
	if (d.write_in_progress != 0 || d.atapi_cmd_in_progress != 0 || occ != 0) {
		printf("  FAIL %-38s wip 0x%x flag %i occ %i\n",
		    "an SRST aborts what was in flight",
		    d.write_in_progress, d.atapi_cmd_in_progress, occ);
		failures ++;
	} else {
		printf("  ok   %-38s %i\n",
		    "an SRST aborts what was in flight", occ);
	}
	altctl_free(&d);
}


/*  THE SRST SIGNATURE ACCEPT ROWS.  THE POWER-ON CASE IS FILED AS `wdcponsig` AND
 *  NO ROW HERE COVERS IT -- do not read these rows as power-on coverage.
 *
 *  ATA/ATAPI-4 has an ATAPI device post 0x14/0xEB after POWER-ON OR RESET.  This
 *  model implements only the reset half, and it implemented that on the wrong bit.
 *  Correcting the bit changes WHEN the signature appears, so these rows are the
 *  de-escalation clause's accept side: a real reset must still produce it.
 *
 *  *** MEASURED, and it corrects TWO earlier claims in this very comment. ***  With a
 *  CD-ROM attached, pre-fix vs post-fix:
 *
 *      ctl 0x08 alone (libata steady)     0xeb14 -> 0x0000   REMOVED
 *      ctl 0x0a alone (polled mode)       0xeb14 -> 0x0000   REMOVED
 *      LBA48 HOB traffic alone            0xeb14 -> 0x0000   REMOVED
 *      ctl 0x06 (OpenBSD SRST)            0x0000 -> 0xeb14   GAINED
 *      ctl 0x06 -> 0x02                   0x0000 -> 0xeb14   GAINED
 *      power-on, no control write at all  0x0000 -> 0x0000   UNCHANGED
 *
 *  This header first said the rows were "folded in rather than filed", and that a
 *  round fixing the bit and not the power-on case "ships a regression".  BOTH ARE
 *  FALSE.  Power-on is dark on both sides, so nothing regresses; and the accident
 *  was never power-on -- it was the signature being available after ANY
 *  4BIT-carrying control write.  Removing that IS the repair, two real reset paths
 *  GAINED a signature they never had, and `wdcponsig` stands on its own merits
 *  rather than as this round's debt.  Named by the opus seat, which built the
 *  matrix above at rung 3 on unmodified `evbmips -e malta`.
 *
 *  *** THE COMMENTS PIN THE MODEL'S OWN CONSTANTS, NOT A SPECIFICATION.  There is no
 *  ATA document in this tree -- fifteen PDFs and none of them ATA -- and this project
 *  shipped manual page numbers from panel output three times in #357/#362/#365 and
 *  got away with it on luck.  The claim here is change-detection, not conformance. ***
 */
static void row_srst_signature(void)
{
	struct wdc_data d;
	unsigned char buf[8];
	int lo, hi;

	altctl_fresh(&d);
	if (d.inbuf == NULL) {
		printf("  SETUP srst_sig\n");
		return;
	}
	stub_disk_exists = 1;
	stub_cdrom_poison = 1;

	/*  ACCEPT ROW.  A REAL soft reset must still produce the signature after
	    the bit is corrected -- this is the de-escalation clause's accept side,
	    and it is the row wdcatapisig was filed for.  BOTH assert forms are read
	    here: 0x06 (OpenBSD) and 0x0c (libata).
	
	    *** THE 0x0c READ IS NEW, AND IT REPAIRS A FALSE CLAIM BY MAKING IT TRUE.
	    This comment used to say the 0x0c form was "covered by the abort row
	    above".  That row reads ZERO cylinder registers and cannot cover it --
	    the comment was writing a cheque the row did not cash.  Named by the
	    grok seat, which also proposed the repair: one more read is cheaper
	    than deleting the claim, and it kills a mutant no other row kills. ***  */
	memset(buf, 0, sizeof(buf));
	buf[0] = WDCTL_RST | WDCTL_IDS;
	dev_wdc_altstatus_access(fake_cpu(), NULL, 0, buf, 1, MEM_WRITE, &d);
	memset(buf, 0, sizeof(buf));
	dev_wdc_access(fake_cpu(), NULL, wd_cyl_lo, buf, 1, MEM_READ, &d);
	lo = buf[0];
	memset(buf, 0, sizeof(buf));
	dev_wdc_access(fake_cpu(), NULL, wd_cyl_hi, buf, 1, MEM_READ, &d);
	hi = buf[0];

	rows ++;
	if (lo != 0x14 || hi != 0xeb) {
		printf("  FAIL %-38s 0x%02x%02x want 0xeb14\n",
		    "a real SRST still signs the CD-ROM", hi, lo);
		failures ++;
	} else {
		printf("  ok   %-38s 0x%02x%02x\n",
		    "a real SRST still signs the CD-ROM", hi, lo);
	}

	/*  CONTROL: the same registers on a NON-CD-ROM must not sign.  Without this
	    a row demanding 0xeb14 is satisfied by a device that signs
	    unconditionally -- a different defect wearing this one's clothes.  */
	stub_cdrom_poison = 0;
	memset(buf, 0, sizeof(buf));
	dev_wdc_access(fake_cpu(), NULL, wd_cyl_lo, buf, 1, MEM_READ, &d);
	lo = buf[0];
	memset(buf, 0, sizeof(buf));
	dev_wdc_access(fake_cpu(), NULL, wd_cyl_hi, buf, 1, MEM_READ, &d);
	hi = buf[0];

	rows ++;
	if (lo == 0x14 && hi == 0xeb) {
		printf("  FAIL %-38s an ATA disk signed 0xeb14\n",
		    "control: a non-CD-ROM does not sign");
		failures ++;
	} else {
		printf("  ok   %-38s 0x%02x%02x\n",
		    "control: a non-CD-ROM does not sign", hi, lo);
	}

	/*  CONTROL: the signature the device DOES produce still works, so a failure
	    above reads as "the power-on half is missing" and not "the signature is
	    broken".  Two different repairs; the row has to say which.  */
	stub_cdrom_poison = 1;
	d.cur_command = COMMAND_RESET;
	memset(buf, 0, sizeof(buf));
	dev_wdc_access(fake_cpu(), NULL, wd_cyl_lo, buf, 1, MEM_READ, &d);
	lo = buf[0];
	memset(buf, 0, sizeof(buf));
	dev_wdc_access(fake_cpu(), NULL, wd_cyl_hi, buf, 1, MEM_READ, &d);
	hi = buf[0];

	rows ++;
	if (lo != 0x14 || hi != 0xeb) {
		printf("  FAIL %-38s 0x%02x%02x want 0xeb14\n",
		    "control: the reset signature works", hi, lo);
		failures ++;
	} else {
		printf("  ok   %-38s 0x%02x%02x\n",
		    "control: the reset signature works", hi, lo);
	}

	/*  (C) THE RESET CONJUNCT, pinned.  The "non-CD-ROM does not sign" control
	    above tests the OTHER half of the read-site guard -- `is_a_cdrom` -- so
	    dropping the `cur_command == COMMAND_RESET` conjunct entirely escaped
	    every signature check this row had.  An UNRESET CD-ROM must read back
	    the bytes the GUEST WROTE, not a signature.  Named and measured by the
	    opus seat (`Mo_sigalways`).  */
	altctl_free(&d);
	altctl_fresh(&d);
	if (d.inbuf == NULL) {
		stub_cdrom_poison = 0;
		stub_disk_exists = 0;
		return;
	}
	stub_cdrom_poison = 1;

	memset(buf, 0, sizeof(buf));
	buf[0] = 0x55;
	dev_wdc_access(fake_cpu(), NULL, wd_cyl_lo, buf, 1, MEM_WRITE, &d);
	memset(buf, 0, sizeof(buf));
	buf[0] = 0x66;
	dev_wdc_access(fake_cpu(), NULL, wd_cyl_hi, buf, 1, MEM_WRITE, &d);
	memset(buf, 0, sizeof(buf));
	dev_wdc_access(fake_cpu(), NULL, wd_cyl_lo, buf, 1, MEM_READ, &d);
	lo = buf[0];
	memset(buf, 0, sizeof(buf));
	dev_wdc_access(fake_cpu(), NULL, wd_cyl_hi, buf, 1, MEM_READ, &d);
	hi = buf[0];

	rows ++;
	if (lo != 0x55 || hi != 0x66) {
		printf("  FAIL %-38s 0x%02x%02x want 0x6655\n",
		    "an unreset CD-ROM does not sign", hi, lo);
		failures ++;
	} else {
		printf("  ok   %-38s 0x%02x%02x\n",
		    "an unreset CD-ROM does not sign", hi, lo);
	}

	/*  (A) THE OTHER ASSERT FORM.  0x0c is what libata writes to assert SRST and
	    differs from OpenBSD's 0x06 in BIT 3 -- the exact bit this round moved away
	    from.  A predicate accepting 0x06 but not 0x0c would pass every other row
	    in this file.  Proposed by the grok seat, which found this row's comment
	    already CLAIMING 0x0c coverage that no row provided.  Fresh device, so it
	    cannot inherit the latch set above.  */
	altctl_free(&d);
	altctl_fresh(&d);
	if (d.inbuf == NULL) {
		stub_cdrom_poison = 0;
		stub_disk_exists = 0;
		return;
	}
	stub_cdrom_poison = 1;

	memset(buf, 0, sizeof(buf));
	buf[0] = WDCTL_RST | WDCTL_4BIT;
	dev_wdc_altstatus_access(fake_cpu(), NULL, 0, buf, 1, MEM_WRITE, &d);
	memset(buf, 0, sizeof(buf));
	dev_wdc_access(fake_cpu(), NULL, wd_cyl_lo, buf, 1, MEM_READ, &d);
	lo = buf[0];
	memset(buf, 0, sizeof(buf));
	dev_wdc_access(fake_cpu(), NULL, wd_cyl_hi, buf, 1, MEM_READ, &d);
	hi = buf[0];

	rows ++;
	if (lo != 0x14 || hi != 0xeb) {
		printf("  FAIL %-38s 0x%02x%02x want 0xeb14\n",
		    "the libata SRST form signs too", hi, lo);
		failures ++;
	} else {
		printf("  ok   %-38s 0x%02x%02x\n",
		    "the libata SRST form signs too", hi, lo);
	}

	stub_cdrom_poison = 0;
	stub_disk_exists = 0;
	altctl_free(&d);
}


/*  #482: ONE TENANT EACH, because a row that arms them all cannot say which one
 *  the guard reads.
 *
 *  #481's abort row sets write_in_progress, atapi_cmd_in_progress, the phase and
 *  the ring together and asserts all four are cleared.  Both flagship seats
 *  measured the same hole independently: putting ANY ONE of those fields in front
 *  of the helper call still passes it.  The worst is the first -- a plain ATA
 *  write is the majority reset path, and `if (d->atapi_cmd_in_progress)` leaves
 *  its writer and 16 staged bytes alive across the reset.
 *
 *  These rows are ACCEPT-SIDE: green on the committed build, which aborts
 *  correctly, and red under the qualifying mutants.
 *
 *  *** THE WITNESS KIND IS `harness:`, NOT A REPRODUCTION.  An earlier version of
 *  this comment said "the reproduction is the kill table".  There was never a live
 *  defect to reproduce -- the defect was in the APPARATUS, and the carrier's ladder
 *  has a term for exactly that: `harness:` is valid when the round's subject is the
 *  test apparatus, proven normally by a MUTATION.  Both flagship seats named this
 *  independently.  "Reproduction" invites a reader to think a device misbehaviour
 *  was observed, and none was. ***
 *
 *  The proof has the same shape test-first always has, applied one level up: the
 *  mutants ESCAPE the 96-row form and DIE on the 100-row form, while the product
 *  stays green through both.  The rows change the MUTANT's colour, not the
 *  PRODUCT's -- which is the discriminator separating this from fitting a test to a
 *  green binary.
 */
static void row_srst_aborts_ata_only(void)
{
	struct wdc_data d;
	unsigned char buf[8];
	int i, occ;

	altctl_fresh(&d);
	if (d.inbuf == NULL) {
		printf("  SETUP ata_only\n");
		return;
	}

	/*  ONLY an ATA write is in flight.  No packet, no phase, and the ring
	    holds a partial sector -- the commonest thing a guest resets out of.  */
	d.write_in_progress = WDCC_WRITE;
	d.write_count = 1;
	for (i = 0; i < 16; i++)
		wdc_addtoinbuf(&d, 0x5a);

	memset(buf, 0, sizeof(buf));
	buf[0] = WDCTL_RST | WDCTL_4BIT;
	dev_wdc_altstatus_access(fake_cpu(), NULL, 0, buf, 1, MEM_WRITE, &d);

	occ = d.inbuf_head - d.inbuf_tail;
	while (occ < 0)
		occ += WDC_INBUF_SIZE;

	rows ++;
	if (d.write_in_progress != 0 || occ != 0) {
		printf("  FAIL %-38s wip 0x%x occ %i\n",
		    "an SRST aborts an ATA write alone",
		    d.write_in_progress, occ);
		failures ++;
	} else {
		printf("  ok   %-38s %i\n",
		    "an SRST aborts an ATA write alone", occ);
	}
	altctl_free(&d);
}

/*  #482: THE THIRD TENANT IS THE RING ITSELF, and the first draft of this round

/*  #482: THE MASK'S FREE BITS.  The predicate must contain bit 2 and exclude bits
 *  1 and 3; bits 0, 4, 5, 6 and 7 are unconstrained by every row in this file,
 *  because the whole control vocabulary here is {0x02, 0x06, 0x08, 0x0a, 0x0c}.
 *
 *  0x80 is the one with a real driver behind it, and a seat MEASURED `0x88`
 *  latching a reset and destroying cur_command while escaping all 96 rows.
 *
 *  *** THIS ROW PINS THE MODEL, NOT A SPECIFICATION.  The identification of 0x80
 *  as an HOB-style control bit is READ from general ATA knowledge and THERE IS NO
 *  ATA DOCUMENT IN THIS TREE -- fifteen PDFs and not one of them covers it.  What
 *  is measured is that a control byte carrying bit 7 must not be a reset, which
 *  follows from the bit the round chose regardless of what 0x80 means. ***
 */
static void row_hob_is_not_a_reset(void)
{
	struct wdc_data d;
	unsigned char buf[8];
	int i, occ, before;

	altctl_fresh(&d);
	if (d.inbuf == NULL) {
		printf("  SETUP hob\n");
		return;
	}

	d.cur_command = WDCC_IDENTIFY;
	before = d.cur_command;
	d.write_in_progress = WDCC_WRITE;
	d.write_count = 1;
	for (i = 0; i < 40; i++)
		wdc_addtoinbuf(&d, 0x5a);

	memset(buf, 0, sizeof(buf));
	buf[0] = 0x80 | WDCTL_4BIT;			/*  0x88  */
	dev_wdc_altstatus_access(fake_cpu(), NULL, 0, buf, 1, MEM_WRITE, &d);

	occ = d.inbuf_head - d.inbuf_tail;
	while (occ < 0)
		occ += WDC_INBUF_SIZE;

	rows ++;
	if (d.cur_command != before || d.write_in_progress == 0 || occ != 40) {
		printf("  FAIL %-38s 0x%x->0x%x wip 0x%x occ %i\n",
		    "a 0x88 control write is not a reset",
		    before, d.cur_command, d.write_in_progress, occ);
		failures ++;
	} else {
		printf("  ok   %-38s 0x%x occ %i\n",
		    "a 0x88 control write is not a reset", d.cur_command, occ);
	}
	altctl_free(&d);
}


/*  #483: THE ORPHAN RING -- the state no row reached, and the one that kills the
 *  edit a maintainer is most likely to make.
 *
 *  `if (d->write_in_progress || d->atapi_cmd_in_progress)` in front of the helper
 *  call reads as an obvious tidy-up: only abort if something is actually in
 *  flight.  BOTH flagship seats built it independently against #482, and it
 *  defeats all four of that round's rows, because every one of them arms one of
 *  those two flags.
 *
 *  *** THE STATE IS GUEST-REACHABLE, and that is what makes this a row rather
 *  than a curiosity.  The #464 unexpected-write test gates ONLY the one-shot
 *  diagnostic; the wdc_addtoinbuf() calls sit OUTSIDE it and push
 *  unconditionally.  So a guest can stage bytes with NEITHER tenant armed and
 *  then assert SRST, and a disjunctively-guarded abort would leave them. ***
 */
static void row_srst_purges_orphan_ring(void)
{
	struct wdc_data d;
	unsigned char buf[8];
	int i, live, occ, benign, small_live, small;

	/*  LIVENESS FIRST: the same traffic with NO reset must LEAVE the bytes.
	    Without this the row is satisfied by a ring that was empty all along --
	    the vacuity shape this file has been bitten by before.  */
	altctl_fresh(&d);
	if (d.inbuf == NULL) {
		printf("  SETUP orphan live\n");
		return;
	}
	for (i = 0; i < 8; i++) {
		buf[0] = 0x5a; buf[1] = 0x5a;
		dev_wdc_access(fake_cpu(), NULL, wd_data, buf, 2, MEM_WRITE, &d);
	}
	live = d.inbuf_head - d.inbuf_tail;
	while (live < 0)
		live += WDC_INBUF_SIZE;
	altctl_free(&d);

	/*  Now the same stores, with NEITHER tenant armed, followed by SRST.  */
	altctl_fresh(&d);
	if (d.inbuf == NULL) {
		printf("  SETUP orphan\n");
		return;
	}
	for (i = 0; i < 8; i++) {
		buf[0] = 0x5a; buf[1] = 0x5a;
		dev_wdc_access(fake_cpu(), NULL, wd_data, buf, 2, MEM_WRITE, &d);
	}
	memset(buf, 0, sizeof(buf));
	buf[0] = WDCTL_RST | WDCTL_4BIT;
	dev_wdc_altstatus_access(fake_cpu(), NULL, 0, buf, 1, MEM_WRITE, &d);

	occ = d.inbuf_head - d.inbuf_tail;
	while (occ < 0)
		occ += WDC_INBUF_SIZE;

	/*  THIRD LEG: A CONTROL WRITE WITH BIT 2 CLEAR MUST *NOT* PURGE.
	    The liveness leg above proves the bytes were staged.  It does NOT prove
	    the purge is specific to SRST -- it performs no altstatus access at all,
	    so a mutant purging on ANY control write leaves it untouched and passes
	    the reset leg as well.  This leg is the difference between "SRST purges"
	    and "any control write purges", which is what the row claims.
	    Named by the kimi-k3:cloud seat.  */
	altctl_fresh(&d);
	if (d.inbuf == NULL) {
		printf("  SETUP orphan benign\n");
		return;
	}
	for (i = 0; i < 8; i++) {
		buf[0] = 0x5a; buf[1] = 0x5a;
		dev_wdc_access(fake_cpu(), NULL, wd_data, buf, 2, MEM_WRITE, &d);
	}
	memset(buf, 0, sizeof(buf));
	buf[0] = WDCTL_4BIT;			/*  bit 2 CLEAR -- not a reset  */
	dev_wdc_altstatus_access(fake_cpu(), NULL, 0, buf, 1, MEM_WRITE, &d);
	benign = d.inbuf_head - d.inbuf_tail;
	while (benign < 0)
		benign += WDC_INBUF_SIZE;
	altctl_free(&d);

	/*  FOURTH LEG: A SMALLER RING, so a THRESHOLD guard cannot hide.  The row
	    staged exactly 16 bytes, so `occ > 8` in front of the abort passed it
	    while leaving every smaller orphan ring stale across a reset.  Two
	    bytes is the least a single data-port store can produce, so any
	    threshold at all is caught.  The escape space is multi-dimensional;
	    pin a SPACE, not a point -- the same move the bit-2 sweep makes.  */
	altctl_fresh(&d);
	if (d.inbuf == NULL) {
		printf("  SETUP orphan small\n");
		return;
	}
	buf[0] = 0x5a; buf[1] = 0x5a;
	dev_wdc_access(fake_cpu(), NULL, wd_data, buf, 2, MEM_WRITE, &d);
	small_live = d.inbuf_head - d.inbuf_tail;
	while (small_live < 0)
		small_live += WDC_INBUF_SIZE;
	memset(buf, 0, sizeof(buf));
	buf[0] = WDCTL_RST | WDCTL_4BIT;
	dev_wdc_altstatus_access(fake_cpu(), NULL, 0, buf, 1, MEM_WRITE, &d);
	small = d.inbuf_head - d.inbuf_tail;
	while (small < 0)
		small += WDC_INBUF_SIZE;

	rows ++;
	if (live != 16 || occ != 0 || benign != 16 ||
	    small_live != 2 || small != 0) {
		printf("  FAIL %-38s live %i occ %i benign %i sm %i/%i\n",
		    "an SRST purges an orphan ring",
		    live, occ, benign, small_live, small);
		failures ++;
	} else {
		printf("  ok   %-38s %i/%i/%i/%i/%i\n",
		    "an SRST purges an orphan ring",
		    live, occ, benign, small_live, small);
	}
	altctl_free(&d);
}

/*  #483: THE WHOLE INPUT SPACE, and it retires an appeal this tree cannot support.
 *
 *  #482 pinned ONE high bit (0x88) and its comment had to disclose that reading
 *  0x80 as an HOB-style control bit comes from general ATA knowledge -- there is
 *  no ATA document anywhere in this tree.  A seat then measured bit 6 escaping,
 *  which is exactly what a single-point pin cannot see.
 *
 *  This asserts the model's own decode across all 256 values: THE RESET LATCHES
 *  IF AND ONLY IF BIT 2 IS SET.  No specification is cited or needed.
 *
 *  *** AND AN IFF IS STRICTLY STRONGER THAN A SET OF POINT PINS.  It fails on a
 *  mask that is too WIDE (a value without bit 2 latches) and on one that is too
 *  NARROW (a value with bit 2 does not), where each point pin sees one direction
 *  only. ***
 */
static void row_reset_bit_iff(void)
{
	int v, ran = 0, wide = -1, narrow = -1, half = -1;

	for (v = 0; v < 256; v++) {
		struct wdc_data d;
		unsigned char buf[8];
		int i, occ, want, got;

		altctl_fresh(&d);
		if (d.inbuf == NULL)
			continue;
		ran ++;
		d.cur_command = WDCC_IDENTIFY;

		/*  STATE TO PURGE.  Without it this row could only ever observe
		    the LATCH, and a guard qualifying the abort alone escaped all
		    103 rows -- built independently by both flagship seats.  */
		d.write_in_progress = WDCC_WRITE;
		d.write_count = 1;
		for (i = 0; i < 8; i++)
			wdc_addtoinbuf(&d, 0x5a);

		memset(buf, 0, sizeof(buf));
		buf[0] = v;
		dev_wdc_altstatus_access(fake_cpu(), NULL, 0, buf, 1,
		    MEM_WRITE, &d);

		occ = d.inbuf_head - d.inbuf_tail;
		while (occ < 0)
			occ += WDC_INBUF_SIZE;

		/*  BOTH EFFECTS, IFF BIT 2.  got == 1 is a HALF RESET and is
		    wrong whichever way bit 2 went: a latch without a purge, or
		    a purge without a latch.  */
		want = (v & WDCTL_RST) ? 2 : 0;
		got  = ((d.cur_command == COMMAND_RESET) ? 1 : 0)
		     + ((d.write_in_progress == 0 && occ == 0) ? 1 : 0);

		if (got == 1 && half < 0)
			half = v;
		else if (got == 2 && want == 0 && wide < 0)
			wide = v;
		else if (got == 0 && want == 2 && narrow < 0)
			narrow = v;
		altctl_free(&d);
	}

	/*  *** THE DENOMINATOR IS COUNTED, NOT WRITTEN.  An earlier version printed
	    a LITERAL "256 of 256" after an unconditional rows++, with the NULL case
	    doing `continue` rather than `return` -- so a sweep in which every
	    allocation failed would have reported a full pass, and the row-count
	    tripwire could not have caught it either.  The comment above warns about
	    exactly that shape and the row reproduced it.  A hardcoded denominator is
	    a denominator that cannot fail.  Named by the kimi-k3:cloud seat. ***

	    The sentinels print as %i, not %02x: they are -1 when unset, and %02x
	    rendered that as 0xffffffff in every one-sided failure line.  */
	rows ++;
	if (ran != 256 || wide >= 0 || narrow >= 0 || half >= 0) {
		printf("  FAIL %-38s %i of 256 w %i n %i h %i\n",
		    "reset ACTS IFF bit 2 is set", ran, wide, narrow, half);
		failures ++;
	} else {
		printf("  ok   %-38s %i of 256\n",
		    "reset ACTS IFF bit 2 is set", ran);
	}
}


/*
 *  #484: the ring-occupancy rows.  MEASURED before they were ported --
 *  7/16 on the committed build, 16/16 with the fix, and every mutant in
 *  `artifacts/wdcringambig/killtable_20260827/` scored against them.
 *
 *  *** THE ANCHORED ROWS ARE R2 AND R4, AND THEY ARE NOT OPTIONAL. ***  Two
 *  mutants that keep occupancy sound while destroying data -- the `return`
 *  moved inside the latch, and drop-oldest -- are GREEN under every
 *  occupancy-shaped and DRQ-shaped row here.  Only the marker byte at the
 *  tail cell and the real consumer's FIFO order kill them.  That is clause
 *  (f) one level down: an oracle computed the same way as the code under
 *  test cannot see an edit that preserves the computation.
 *
 *  R7 and R8 cover the BULK arm, which bypasses wdc_addtoinbuf() entirely.
 *  Three reading seats judged it unreachable AND THEY WERE RIGHT ABOUT THE
 *  PRODUCT -- wdc__read()'s one caller resets the ring first, so this state is
 *  a FUNCTION state, not a guest route.  What was MEASURED is the function
 *  past tail -- occupancy 262156 -> 1548 with unread guest data overwritten.
 *  Deleting the bulk bound scores 13/16, so those rows carry it.
 */
static int gx_occ(struct wdc_data *d)
{
	int o = d->inbuf_head - d->inbuf_tail;
	while (o < 0)
		o += WDC_INBUF_SIZE;
	return o;
}

static void gx_push(struct wdc_data *d, int n, int w, unsigned char v)
{
	unsigned char b[8];
	int i, k;
	for (k = 0; k < n; k += w) {
		for (i = 0; i < w; i++)
			b[i] = v;
		dev_wdc_access(fake_cpu(), NULL, wd_data, b, w, MEM_WRITE, d);
	}
}

/*  Each assertion is a ROW in this file's sense: it bumps `rows`, and a
    mismatch bumps `failures`.  The seat's private counters are deliberately
    NOT carried over -- a row this file does not count is a row the identity
    pin cannot protect.  */
static int chk_calls;	/*  #551: counted, so the LAST self-check can print the coverage  */

static void chk(const char *name, long long got, long long want)
{
	chk_calls ++;
	rows ++;
	if (got != want) {
		failures ++;
		cmp_fails ++;
		printf("  FAIL %-40s got %lld want %lld\n", name, got, want);
	} else
		printf("  ok   %-40s %lld\n", name, got);
}

static void row_ring_occupancy_sound(void)
{
	struct wdc_data d;
	unsigned char buf[8];
	int i;

	/*  R6 LIVENESS -- one short of full must ALL land.  Without this every
	    row below is satisfied by a ring that refuses everything.  */
	altctl_fresh(&d);
	for (i = 0; i < WDC_INBUF_SIZE - 1; i++) wdc_addtoinbuf(&d, 0x11);
	chk("R6 liveness: SIZE-1 all land", gx_occ(&d), WDC_INBUF_SIZE - 1);
	chk("R6 liveness: DRQ asserted",
	    (status_byte(&d, fake_cpu()) & WDCS_DRQ) ? 1 : 0, 1);
	free(d.inbuf);

	/*  R1 -- THE WRAP ROW IN THE EXISTING (head-tail) IDIOM.  */
	altctl_fresh(&d);
	for (i = 0; i < WDC_INBUF_SIZE; i++) wdc_addtoinbuf(&d, 0x11);
	chk("R1 full ring: pointer occupancy", gx_occ(&d), WDC_INBUF_SIZE - 1);
	chk("R3 full ring: DRQ still set",
	    (status_byte(&d, fake_cpu()) & WDCS_DRQ) ? 1 : 0, 1);
	free(d.inbuf);

	/*  R2 -- THE ANCHORED ROW.  A marker byte the ring did not compute.  */
	altctl_fresh(&d);
	wdc_addtoinbuf(&d, 0xA5);			/*  the anchor  */
	for (i = 1; i < WDC_INBUF_SIZE + 64; i++) wdc_addtoinbuf(&d, 0x11);
	chk("R2 anchor: inbuf[tail] survives", d.inbuf[d.inbuf_tail], 0xA5);
	chk("R2 anchor: cell 0 untouched", d.inbuf[0], 0xA5);
	free(d.inbuf);

	/*  R4 -- FIFO ORDER through the real consumer.  */
	altctl_fresh(&d);
	wdc_addtoinbuf(&d, 0xA5);
	for (i = 1; i < WDC_INBUF_SIZE + 64; i++) wdc_addtoinbuf(&d, 0x11);
	chk("R4 fifo: first pop is first push",
	    (long)(wdc_get_inbuf(&d) & 0xff), 0xA5);
	free(d.inbuf);

	/*  R5 -- THE DISK-IMAGE SINK (the rider's fifth consumer).  Wrap the ring
	    while an ATA WRITE is armed, then land on a multiple of 512.
	    THE ACCEPT LEG COMES FIRST: without it every refusing fix passes by
	    writing nothing at all, which is the vacuity the de-escalation clause
	    exists to refuse.  */
	altctl_fresh(&d);
	stub_disk_exists = 1;
	stub_disk_writes = 0; stub_disk_bytes = 0; stub_disk_first = -1;
	d.seccnt = 1; d.sector = 1;
	buf[0] = WDCC_WRITE;
	dev_wdc_access(fake_cpu(), NULL, wd_command, buf, 1, MEM_WRITE, &d);
	gx_push(&d, 2, 2, 0xA5);
	gx_push(&d, 510, 2, 0x11);
	chk("R5a ACCEPT: an ordinary sector still lands", stub_disk_writes, 1);
	chk("R5a ACCEPT: with the guest's own first byte", stub_disk_first, 0xA5);
	stub_disk_exists = 0; free(d.inbuf);

	altctl_fresh(&d);
	stub_disk_exists = 1;
	stub_disk_writes = 0; stub_disk_bytes = 0; stub_disk_first = -1;
	d.seccnt = 1; d.sector = 1;
	buf[0] = WDCC_WRITE;
	dev_wdc_access(fake_cpu(), NULL, wd_command, buf, 1, MEM_WRITE, &d);
	gx_push(&d, 2, 2, 0xA5);			/*  the guest's FIRST bytes  */
	{ int n = 0;
	  while (n < WDC_INBUF_SIZE + 4096 && stub_disk_writes == 0) {
		gx_push(&d, 4, 4, 0x55); n += 4; } }
	gx_push(&d, 2, 2, 0x66);
	{ int g = 0;
	  while ((gx_occ(&d) % 512) != 0 && g < 2000 && stub_disk_writes == 0) {
		gx_push(&d, 4, 4, 0x77); g++; } }
	/*  Either the drain was refused, or it committed the guest's OWN first
	    byte.  What it must never do is commit 0x55 -- a byte the guest wrote
	    262000 stores later, at the LBA it chose for its first sector.  */
	chk("R5b sink: no MIS-SEQUENCED sector reaches the image",
	    (stub_disk_first < 0 || stub_disk_first == 0xA5) ? 1 : 0, 1);
	stub_disk_exists = 0; free(d.inbuf);

	/*  R7 -- THE BULK memcpy ARM (P4), AND THE ONE PLACE AN OCCUPANCY
	    ORACLE IS GENUINELY BLIND.  Put head BELOW tail with a marker byte
	    sitting at tail, then run a bulk refill whose length steps head PAST
	    tail while still satisfying `head + 512*to_read <= WDC_INBUF_SIZE`.
	    Occupancy goes 262156 -> 1548, a perfectly plausible number; only the
	    marker says 1548 bytes of unread data were destroyed.  */
	altctl_fresh(&d);
	stub_disk_exists = 1; stub_size = 512L*1024*1024; stub_read_fill = 0xEE;
	for (i = 0; i < 1500; i++) wdc_addtoinbuf(&d, 0x11);
	for (i = 0; i < 1500; i++) (void) wdc_get_inbuf(&d);	/*  h = t = 1500  */
	wdc_addtoinbuf(&d, 0xA5);				/*  THE ANCHOR at [1500]  */
	for (i = 1; i < 262156; i++) wdc_addtoinbuf(&d, 0x11);
	chk("R7 setup: head is below tail",
	    (d.inbuf_head < d.inbuf_tail) ? 1 : 0, 1);
	chk("R7 setup: the anchor is at tail", d.inbuf[d.inbuf_tail], 0xA5);
	{
		int occ_before = gx_occ(&d);
		d.seccnt = 4; d.sector = 1; d.head = 0; d.cyl_lo = 0; d.cyl_hi = 0;
		wdc__read(fake_cpu(), &d);
		chk("R7 memcpy arm: the anchor survives", d.inbuf[1500], 0xA5);
		chk("R7 memcpy arm: occupancy did not COLLAPSE",
		    (gx_occ(&d) >= occ_before) ? 1 : 0, 1);
	}
	stub_disk_exists = 0; stub_read_fill = 0; free(d.inbuf);

	/*  R8 -- THE BULK ARM MUST HONOUR THE SAME CAPACITY AS THE PRODUCER.
	    A bulk refill of exactly `room + 1` bytes is the smallest input that
	    tells a `> room` bound from a `> room - 1` one.  Without this row the
	    `- 1` in the bulk bound is unpinned and its loss is invisible.  */
	altctl_fresh(&d);
	stub_disk_exists = 1; stub_size = 512L*1024*1024; stub_read_fill = 0xEE;
	wdc_addtoinbuf(&d, 0xA5);				/*  ANCHOR at [0]  */
	for (i = 1; i < WDC_INBUF_SIZE - 512; i++) wdc_addtoinbuf(&d, 0x11);
	{
		d.seccnt = 1; d.sector = 1; d.head = 0; d.cyl_lo = 0; d.cyl_hi = 0;
		wdc__read(fake_cpu(), &d);
		chk("R8 bulk edge: the anchor survives a 512 refill",
		    d.inbuf[0], 0xA5);
		chk("R8 bulk edge: head never reaches tail",
		    (d.inbuf_head == d.inbuf_tail) ? 0 : 1, 1);
	}
	stub_disk_exists = 0; stub_read_fill = 0; free(d.inbuf);

	/*  R9 -- THE ACCEPT SIDE OF THE SAME BOUND, and THREE SEATS NAMED THE
	    ESCAPE INDEPENDENTLY on pass 2 (codex, glm, deepseek): `>` -> `>=`.

	    R8 above pins only the REFUSAL side -- room 511, ask 512, refused.  A
	    bound tightened by one character still refuses everything R8 asks, so
	    R8 cannot see it.  This row asks for EXACTLY the room available, which
	    the shipped bound accepts (512 > 512 is false) and the tightened one
	    refuses, silently shortening the transfer.

	    *** A GUARD IS HALF-TESTED BY SHOWING IT REFUSES WHAT IT SHOULD.  The
	    other half is showing it ACCEPTS WHAT IT MUST, and that is the half a
	    bound-tightening edit escapes. ***  Same reasoning as the de-escalation
	    clause's accept rows, reached from the opposite direction: there the
	    lost tripwire was a crash, here it is a refusal, and in both cases only
	    the accept side can tell a correct bound from an over-restrictive one.

	    Arithmetic: occupancy SIZE-513 leaves room SIZE-1-(SIZE-513) = 512; one
	    sector is 512 bytes; acceptance takes occupancy to SIZE-1 exactly.  */
	altctl_fresh(&d);
	stub_disk_exists = 1; stub_size = 512L*1024*1024; stub_read_fill = 0xEE;
	wdc_addtoinbuf(&d, 0xA5);				/*  ANCHOR at [0]  */
	for (i = 1; i < WDC_INBUF_SIZE - 513; i++) wdc_addtoinbuf(&d, 0x11);
	{
		chk("R9 setup: room is exactly one sector",
		    (d.inbuf_tail - d.inbuf_head - 1 + WDC_INBUF_SIZE)
		    % WDC_INBUF_SIZE, 512);
		d.seccnt = 1; d.sector = 1; d.head = 0; d.cyl_lo = 0; d.cyl_hi = 0;
		wdc__read(fake_cpu(), &d);
		chk("R9 exact fit: the refill is ACCEPTED", gx_occ(&d),
		    WDC_INBUF_SIZE - 1);
		chk("R9 exact fit: the anchor is still intact", d.inbuf[0], 0xA5);
	}
	stub_disk_exists = 0; stub_read_fill = 0; free(d.inbuf);

	/*  R10 -- THE MULTIPLIER.  Every earlier bulk row asks for ONE sector, so
	    `512 * to_read` is never distinguished from a bare `512`.  Deleting those
	    two tokens scored 0 reds of 122 and caused 275 integrity violations.

	    Same free space as R9 (512) but TWO sectors requested: the shipped bound
	    refuses (1024 > 512) and the mutant admits (512 > 512 is false), then
	    memcpys 1024 bytes over a ring with room for 512 and destroys 511 unread
	    ones.  *** THAT IS CLAUSE (d) -- A BOUND RELAXED ON THE PREDICATE THIS
	    DIFF ADDED -- so it was fixed in-round rather than filed. ***  */
	altctl_fresh(&d);
	stub_disk_exists = 1; stub_size = 512L*1024*1024; stub_read_fill = 0xEE;
	wdc_addtoinbuf(&d, 0xA5);				/*  ANCHOR at [0]  */
	for (i = 1; i < WDC_INBUF_SIZE - 513; i++) wdc_addtoinbuf(&d, 0x11);
	{
		int occ_before = gx_occ(&d);
		d.seccnt = 2; d.sector = 1; d.head = 0; d.cyl_lo = 0; d.cyl_hi = 0;
		wdc__read(fake_cpu(), &d);
		chk("R10 multiplier: two sectors into one sector of room is REFUSED",
		    gx_occ(&d), occ_before);
		chk("R10 multiplier: the anchor survives the refusal", d.inbuf[0], 0xA5);
	}
	stub_disk_exists = 0; stub_read_fill = 0; free(d.inbuf);

	/*  R11 -- THE SECOND CALL.  No row called wdc__read() twice, so the bulk
	    arm's LATCH INTERACTION and its COMMON TAIL were both unpinned.

	    Moving the bulk `break` inside the warning latch scored 0 reds of 122:
	    the first refusal announces and breaks, and the SECOND falls straight
	    through, landing head ON tail -- the exact ambiguity this round removed.
	    It is the identical mutant to the producer's Md_inside, which IS pinned;
	    the bulk twin was not.  Turning that `break` into a `return` also scored
	    0 reds and drops the completion interrupt, because `int_assert` is set in
	    the common tail AFTER the loop.

	    So this row asserts BOTH: the ring stays unambiguous across two refused
	    reads, and the interrupt still reaches the guest.  */
	/*  *** THE SETUP IS THE ROW.  A first cut filled the ring to free == 0,
	    and the mutant SURVIVED at 126/126 -- because head + 512*to_read then
	    exceeds the buffer, so the fall-through takes the BYTE-WISE arm, whose
	    per-byte producer guard refuses every push and saves it.  The mutant
	    needs the MEMCPY arm, and that needs head low and tail just above it.
	    A mutant that will not die is a RESULT: here it was the row mis-cut,
	    not the target dead. ***

	    push 512, pop 512, push SIZE-512  ->  head = 0, tail = 512, free = 511.
	    One sector asks 512 > 511, so the guard fires; the fall-through memcpy
	    then lands head exactly ON tail and occupancy reads 0.  */
	altctl_fresh(&d);
	stub_disk_exists = 1; stub_size = 512L*1024*1024; stub_read_fill = 0xEE;
	for (i = 0; i < 512; i++) wdc_addtoinbuf(&d, 0x11);
	for (i = 0; i < 512; i++) (void) wdc_get_inbuf(&d);	/*  h = t = 512  */
	for (i = 0; i < WDC_INBUF_SIZE - 512; i++) wdc_addtoinbuf(&d, 0x11);
	{
		chk("R11 setup: head is 0 and tail is 512",
		    (d.inbuf_head == 0 && d.inbuf_tail == 512) ? 1 : 0, 1);
		d.seccnt = 1; d.sector = 1; d.head = 0; d.cyl_lo = 0; d.cyl_hi = 0;
		d.int_assert = 0;
		wdc__read(fake_cpu(), &d);
		wdc__read(fake_cpu(), &d);		/*  the SECOND one is the test  */
		chk("R11 second refusal: head has NOT landed on tail",
		    (d.inbuf_head == d.inbuf_tail) ? 0 : 1, 1);
		chk("R11 second refusal: the completion irq still reaches the tail",
		    d.int_assert, 1);
	}
	stub_disk_exists = 0; stub_read_fill = 0; free(d.inbuf);
}

/*  #485: a command that TERMINATES must raise its completion interrupt.
 *
 *  THE INVARIANT, derived over all 256 opcodes: every command wdc_command()
 *  FINISHES asserts.  The only silent opcodes are the THREE that hand control
 *  back for a data phase -- WDCC_WRITE, WDCC_WRITEMULTI, ATAPI_PKT_CMD -- each
 *  of which asserts when its data phase drains.
 *
 *  *** EVERY LEG BELOW EXISTS BECAUSE A MEASURED MUTANT ESCAPED THE
 *  TWO-OBSERVABLE SWEEP THIS ROUND FIRST DESIGNED. ***  The reproduction called
 *  wdc_command() directly and ran the tick by hand; that cannot see a fix whose
 *  repair depends on dev_wdc_access()'s tail calling the tick.  So leg 1 drives
 *  the REAL REGISTER PATH.
 */
static void row_completion_irq(void)
{
	struct wdc_data d;
	unsigned char buf[8];
	int v, ran = 0, abrt_noirq = 0, first_bad = -1;
	int silent[8], nsilent = 0, i;
	int second = -1, second_sec = -1;
	int retract_de = -1, retract_field = -1, retract_re = -1;
	int sf_irq = -1, sf_abrt = -1;
	int tf_sec = -1, tf_def = -1, held = -1, abrt_sec = -1, abrt_def = -1;

	/*  LEG 1 -- THE SWEEP, THROUGH THE REGISTER PATH.  A command STORE, not a
	    direct call, so dev_wdc_access's tail tick is inside the measurement.  */
	for (v = 0; v < 256; v++) {
		altctl_fresh(&d);
		if (d.inbuf == NULL)
			continue;
		ran ++;
		stub_disk_exists = 1;
		d.irq.interrupt_assert = arm_sink;
		d.irq.interrupt_deassert = cmpl_desink;
		arm_asserts = 0;
		buf[0] = v;
		dev_wdc_access(fake_cpu(), NULL, wd_command, buf, 1,
		    MEM_WRITE, &d);
		if (arm_asserts == 0) {
			if (nsilent < 8)
				silent[nsilent] = v;
			nsilent ++;
			if ((d.error & WDCE_ABRT) && abrt_noirq ++ == 0)
				first_bad = v;
		}
		altctl_free(&d);
	}

	/*  LEG 2 -- A SECOND UNIMPLEMENTED COMMAND ON *ONE* DEVICE.  Kills the
	    edit that moves the assert inside the warned_unimpl_cmd latch, which a
	    fresh-device-per-opcode sweep cannot see at any oracle width.  */
	altctl_fresh(&d);
	if (d.inbuf != NULL) {
		stub_disk_exists = 1;
		d.irq.interrupt_assert = arm_sink;
		d.irq.interrupt_deassert = cmpl_desink;
		buf[0] = 0xaa;
		dev_wdc_access(fake_cpu(), NULL, wd_command, buf, 1, MEM_WRITE, &d);
		buf[0] = wd_command;
		dev_wdc_access(fake_cpu(), NULL, wd_command, buf, 1, MEM_READ, &d);
		arm_asserts = 0;
		buf[0] = 0xab;			/*  a DIFFERENT unknown opcode  */
		dev_wdc_access(fake_cpu(), NULL, wd_command, buf, 1, MEM_WRITE, &d);
		second = arm_asserts;

		/*  *** AND THE SECOND COMMAND MUST ALSO BE TRIED AT THE **OTHER**
		    ARM.  A pass-2 seat that cannot read a single file named this
		    gap, and the mutant it implied ESCAPED all four legs. ***

		    M10_seclock: give the WDCC_SEC_* arm a latch of its own and put
		    its assert inside it.  Leg 1 cannot see that -- one fresh device
		    per opcode means the latch is 0 on every leg -- and the sub-leg
		    above cannot either, because it replays only `default:` opcodes.
		    Measured: 128 rows, 0 failures.  A CLEAN ESCAPE.

		    So the SECOND command here is 0xF5, a WDCC_SEC_* opcode, on a
		    device whose unimplemented latch is ALREADY SET by the 0xaa
		    above.  Any qualification of the SEC arm by latch state now
		    reds.  Same clause-(d) shape as the default arm's M5_inlatch,
		    which the round DID catch -- at the other arm, which it did
		    not.  */
		/*  *** CLEAR THE PENDING REQUEST FIRST.  int_assert is a LEVEL, and
		    a first cut of this sub-leg left it SET from the 0xab above -- so
		    the next access's tail tick asserted from the STALE request and
		    the sub-leg read 1 no matter what the SEC arm did.  M10 escaped a
		    row written specifically to catch it.  A mutant that will not die
		    is a RESULT: that one was the row mis-cut, not the target dead,
		    and it is the same level-versus-edge subtlety leg 3 exists for. */
		buf[0] = wd_command;
		dev_wdc_access(fake_cpu(), NULL, wd_command, buf, 1, MEM_READ, &d);
		arm_asserts = 0;
		buf[0] = WDCC_SEC_FREEZE_LOCK;
		dev_wdc_access(fake_cpu(), NULL, wd_command, buf, 1, MEM_WRITE, &d);
		second_sec = arm_asserts;
		altctl_free(&d);
	}

	/*  LEG 3 -- RETRACTION.  int_assert is a LEVEL: dev_wdc_tick re-raises it
	    every tick until a STATUS read clears it, so a set-and-never-cleared
	    request is asserted forever.  Only this leg sees that.  */
	altctl_fresh(&d);
	if (d.inbuf != NULL) {
		stub_disk_exists = 1;
		d.irq.interrupt_assert = arm_sink;
		d.irq.interrupt_deassert = cmpl_desink;
		buf[0] = 0xaa;
		dev_wdc_access(fake_cpu(), NULL, wd_command, buf, 1, MEM_WRITE, &d);
		cmpl_de = 0;
		dev_wdc_access(fake_cpu(), NULL, wd_command, buf, 1, MEM_READ, &d);
		retract_de = cmpl_de;
		retract_field = d.int_assert;
		arm_asserts = 0;
		dev_wdc_tick(fake_cpu(), &d);
		retract_re = arm_asserts;
		altctl_free(&d);
	}

	/*  LEG 4 -- SET_FEATURES' *SUCCESS* SUB-CASE.  The sweep varies the command
	    byte and never `precomp`, so this path never ran: input coverage in one
	    dimension is not input coverage.  It must assert AND not abort.  */
	altctl_fresh(&d);
	if (d.inbuf != NULL) {
		stub_disk_exists = 1;
		d.irq.interrupt_assert = arm_sink;
		d.irq.interrupt_deassert = cmpl_desink;
		d.precomp = WDSF_SET_MODE;
		arm_asserts = 0;
		buf[0] = SET_FEATURES;
		dev_wdc_access(fake_cpu(), NULL, wd_command, buf, 1, MEM_WRITE, &d);
		sf_irq = arm_asserts;
		sf_abrt = (d.error & WDCE_ABRT) ? 1 : 0;
		altctl_free(&d);
	}

	/*  ROW F -- A TASK FILE THAT IS NOT ALL ZERO.  *** THE ROOT CAUSE OF
	    ELEVEN MEASURED ESCAPES: altctl_fresh() memsets the device and every
	    leg above writes ONLY the command register, so the whole task file is
	    ZERO on every leg. ***  A guard added on ANY of those fields is then
	    true by default and the fix looks intact.

	    These are INSIDE clause (d)'s bound and are not invented predicates:
	    the `default:` arm's guarded body reads drive, head, cyl_hi, cyl_lo,
	    sector and seccnt -- they are literally the fatal() arguments -- so
	    the GUARD-FIELD-vs-BODY-FIELD check enumerates them mechanically.

	    Measured escapes closed here: the `default:` arm guarded on seccnt,
	    head, cyl, sector or drive, and BOTH arms guarded on drive, seccnt,
	    precomp or lba.  Nine variants, one leg.

	    The round had already learned this for `precomp` -- 'input coverage
	    in one dimension is not input coverage' -- and applied it only to
	    SET_FEATURES, not to the two arms the diff edits.  */
	altctl_fresh(&d);
	if (d.inbuf != NULL) {
		stub_disk_exists = 1;
		d.irq.interrupt_assert = arm_sink;
		d.irq.interrupt_deassert = cmpl_desink;
		d.precomp = 0x5a; d.seccnt = 0x11; d.sector = 0x22;
		d.cyl_lo = 0x33; d.cyl_hi = 0x44;
		d.drive = 0; d.head = 0x0e; d.lba = 1;
		arm_asserts = 0;
		buf[0] = WDCC_SEC_FREEZE_LOCK;
		dev_wdc_access(fake_cpu(), NULL, wd_command, buf, 1, MEM_WRITE, &d);
		tf_sec = arm_asserts;
		buf[0] = wd_command;
		dev_wdc_access(fake_cpu(), NULL, wd_command, buf, 1, MEM_READ, &d);
		arm_asserts = 0;
		buf[0] = 0xaa;
		dev_wdc_access(fake_cpu(), NULL, wd_command, buf, 1, MEM_WRITE, &d);
		tf_def = arm_asserts;
		altctl_free(&d);
	}

	/*  ROW D -- THE LEVEL IS HELD, WHICH IS THE DUAL OF LEG 3.  Leg 3 pins
	    RETRACTION: a STATUS read lowers the line.  *** NOTHING PINNED THAT
	    THE REQUEST SURVIVES UNTIL THAT READ, so an EDGE assert satisfies leg
	    3 exactly. ***  Two measured escapes lived in that gap: asserting the
	    macro directly instead of setting the field, and a tick that clears
	    int_assert after asserting.  Both make the interrupt an edge.

	    int_assert is a LEVEL: the tick re-raises it every tick until the
	    STATUS read.  So: one command, then TWO ticks with NO read between,
	    and the sink must count both.  */
	altctl_fresh(&d);
	if (d.inbuf != NULL) {
		stub_disk_exists = 1;
		d.irq.interrupt_assert = arm_sink;
		d.irq.interrupt_deassert = cmpl_desink;
		arm_asserts = 0;
		buf[0] = 0xaa;
		dev_wdc_access(fake_cpu(), NULL, wd_command, buf, 1, MEM_WRITE, &d);
		dev_wdc_tick(fake_cpu(), &d);
		held = arm_asserts;		/*  the store's own tick, plus one more  */
		altctl_free(&d);
	}

	/*  ROW E -- THE ABRT BIT, ON THE FIRST COMMAND, AT BOTH ARMS.  *** LEG 1
	    COUNTS ABRT ONLY AMONG OPCODES THAT STAYED SILENT, so the moment the
	    fix makes them assert, the ABRT half of the very same two-statement
	    pair goes UNPINNED. ***  Deleting `d->error |= WDCE_ABRT;` from
	    either arm escaped every leg, measured.  The fix added a statement
	    beside an existing one and the detector stopped watching the one it
	    did not add.  */
	altctl_fresh(&d);
	if (d.inbuf != NULL) {
		stub_disk_exists = 1;
		d.irq.interrupt_assert = arm_sink;
		d.irq.interrupt_deassert = cmpl_desink;
		buf[0] = WDCC_SEC_FREEZE_LOCK;
		dev_wdc_access(fake_cpu(), NULL, wd_command, buf, 1, MEM_WRITE, &d);
		abrt_sec = (d.error & WDCE_ABRT) ? 1 : 0;
		altctl_free(&d);
	}
	altctl_fresh(&d);
	if (d.inbuf != NULL) {
		stub_disk_exists = 1;
		d.irq.interrupt_assert = arm_sink;
		d.irq.interrupt_deassert = cmpl_desink;
		buf[0] = 0xaa;
		dev_wdc_access(fake_cpu(), NULL, wd_command, buf, 1, MEM_WRITE, &d);
		abrt_def = (d.error & WDCE_ABRT) ? 1 : 0;
		altctl_free(&d);
	}
	stub_disk_exists = 0;

	rows ++;
	printf("  IRQ swept %i | ABRT+NO-irq %i first 0x%02x | silent %i:",
	    ran, abrt_noirq, first_bad, nsilent);
	for (i = 0; i < nsilent && i < 8; i++)
		printf(" 0x%02x", silent[i]);
	printf(" | 2nd %i secarm %i | retract de=%i f=%i re=%i | sf irq=%i abrt=%i"
	    " | taskfile %i/%i held %i abrt %i/%i\n",
	    second, second_sec, retract_de, retract_field, retract_re,
	    sf_irq, sf_abrt, tf_sec, tf_def, held, abrt_sec, abrt_def);

	/*  THE SILENT SET IS PINNED AS AN IDENTITY, NOT A COUNT.  A single
	    `int_assert = 1` hoisted after the switch reads ABRT+NO-irq == 0 and
	    passes a count-only oracle, while starting to assert at DISPATCH for
	    the three deferred opcodes -- a spurious completion interrupt on every
	    disk write a real guest issues.  */
	if (ran != 256 || abrt_noirq != 0 || nsilent != 3 ||
	    silent[0] != WDCC_WRITE || silent[1] != ATAPI_PKT_CMD ||
	    silent[2] != WDCC_WRITEMULTI ||
	    second != 1 || second_sec != 1 ||
	    retract_de != 1 || retract_field != 0 ||
	    retract_re != 0 || sf_irq != 1 || sf_abrt != 0 ||
	    tf_sec != 1 || tf_def != 1 || held != 2 ||
	    abrt_sec != 1 || abrt_def != 1) {
		printf("  FAIL %-38s see the IRQ line above\n",
		    "a terminated command raises its irq");
		failures ++;
	} else {
		printf("  ok   %-38s 256 swept, 3 deferred\n",
		    "a terminated command raises its irq");
	}
}


static void row_arm_overrun(void)
{
	struct wdc_data d;
	unsigned char buf[4];
	int i, occ, first;

	memset(&d, 0, sizeof(d));
	d.addr_mult  = 1;
	d.io_enabled = 1;
	d.inbuf = (unsigned char *) calloc(1, WDC_INBUF_SIZE);
	d.irq.interrupt_assert = arm_sink;
	if (d.inbuf == NULL) { printf("  SETUP inbuf alloc failed\n"); return; }

	/*  The packet fires when inbuf holds 12 bytes AFTER this access adds its
	    own -- the test is `inbuf_len == 12` and the write pushes first.  So
	    ten are queued here and the 2-byte write below completes the packet.
	    Queueing twelve gives 14 and the command silently never fires, which
	    is what the first version of this row measured.  */
	/*  #477 pass 2: cur_command too -- the arm below issues a REAL packet
	    command, so without it a `cur_command` conjunct blocks the arm and
	    reds these rows for the fixture's reason.  */
	d.cur_command = ATAPI_PKT_CMD;
	d.atapi_cmd_in_progress = 1;
	/*  SETUP RE-PIN, and it is INERT on the committed build: the real
	    device sets BOTH flags together when the packet command
	    arrives, so a fixture setting only atapi_cmd_in_progress
	    builds a state the device never produces.  A candidate fix
	    that keys on the PHASE would then fail here for the fixture's
	    reason rather than its own.  */
	d.atapi_phase = PHASE_CMDOUT;
	for (i = 0; i < 10; i++)
		wdc_addtoinbuf(&d, 0x00);

	/*  514 blocks: one more than the ring holds.  */
	stub_datain_len = 514 * 512;
	arm_asserts = 0;
	memset(buf, 0, sizeof(buf));
	dev_wdc_access(fake_cpu(), NULL, wd_data, buf, 2, MEM_WRITE, &d);
	stub_datain_len = 0;		/*  inert again for every later row  */

	occ = d.inbuf_head - d.inbuf_tail;
	while (occ < 0)
		occ += WDC_INBUF_SIZE;

	/*  FIXED: one DRQ chunk, 32768.  BROKEN: 263168 pushed into a 262656-byte
	    ring reports 512 -- the ring cannot even tell you how much it lost.  */
	rows ++;
	if (occ != 32768) {
		printf("  FAIL %-38s occupancy = %i, want 32768\n",
		    "arm: pushes one chunk, not the whole", occ);
		failures ++;
	} else {
		printf("  ok   %-38s %i\n",
		    "arm: pushes one chunk, not the whole", occ);
	}

	/*  THE CORRUPTION ORACLE.  Block 0's first byte carries 0x5a; an overrun wraps
	    block 514 onto it and leaves 0xc3.  Occupancy alone says the count is wrong;
	    this says the DATA is gone, which is the guest-visible harm.  */
	first = d.inbuf[d.inbuf_tail];
	rows ++;
	if (first != 0x5a) {
		printf("  FAIL %-38s first byte = 0x%02x, want 0x5a\n",
		    "arm: block 0 survives the push", first);
		failures ++;
	} else {
		printf("  ok   %-38s 0x%02x\n",
		    "arm: block 0 survives the push", first);
	}

	if (d.atapi_st != NULL) {
		free(d.atapi_st->data_in);
		free(d.atapi_st->cmd);
		free(d.atapi_st);
	}
	free(d.inbuf);
}


static void row_identity(void)
{
	struct wdc_data d;
	uint64_t want = 700ULL * 16 * 63;
	char model[41];
	int bad = 0;

	memset(&d, 0, sizeof(d));
	d.drive = 1;
	d.base_drive = 2;			/*  so the wanted id is 3  */
	d.cyls[1] = 700; d.heads[1] = 16; d.sectors_per_track[1] = 63;
	memset(stub_size_for, 0, sizeof(stub_size_for));
	stub_size_for[3] = want * 512;
	stub_size = 0;
	stub_size_for_active = 1;
	stub_noncdrom_id = 3;		/*  ONLY id 3 is a plain disk  */
	stub_cdrom_poison = 1;		/*  every other id is a CD-ROM  */
	wdc_initialize_identify_struct(fake_cpu(), &d);
	stub_cdrom_poison = 0; stub_noncdrom_id = -1;
	stub_size_for_active = 0;

	rows ++;
	if (word(&d, 0) != 0x0040) {
		printf("  FAIL %-38s word 0 = %04x, want 0040"
		    " (0x8580 = it asked is_a_cdrom about the wrong id)\n",
		    "this drive is not the CD-ROM next to it", word(&d, 0));
		bad = 1;
	}

	memcpy(model, &d.identify_struct[2 * 27], 40);
	model[40] = '\0';
	if (strstr(model, "id3") == NULL) {
		printf("  FAIL %-38s model \"%s\" lacks id3\n",
		    "the model number names THIS drive", model);
		bad = 1;
	}

	if (bad)
		failures ++;
	else
		printf("  ok   %-38s word 0 = 0040, model names id3\n",
		    "is_a_cdrom and getname get the right id");
}

/*
 *  #410: THE POSITIVE HALF OF THE CD-ROM DECISION.
 *
 *  #409's inverted poison made the drive under test the ONLY non-CD-ROM, so
 *  row_identity can observe nothing but the NEGATIVE answer -- and MEASURED,
 *  four mutants therefore survived all fifteen rows: deleting the
 *  `if (cdrom) flags = 0x8580` branch outright, forcing cdrom to 0, changing
 *  0x8580 to 0x8500, and dropping word 0's high byte. *** The entire ATAPI
 *  announcement was deletable with the whole table green. ***
 *
 *  A fix that only ever exercises one branch of a two-branch decision is half a
 *  test. This row takes the other branch: the drive under test IS the CD-ROM,
 *  and word 0 must read 0x8580 (ATAPI, CDROM, removable) rather than the plain
 *  0x0040 fixed-disk flag.
 *
 *  Reachability, so this is not a synthetic configuration: wdc_command() rejects
 *  WDCC_IDENTIFY for a CD-ROM before the switch, but ATAPI_IDENTIFY_DEVICE falls
 *  into the SAME case and calls this same initializer with cdrom set -- so a
 *  guest issuing ATAPI IDENTIFY reaches exactly this state.
 */
static void row_atapi_flags(void)
{
	struct wdc_data d;

	memset(&d, 0, sizeof(d));
	d.drive = 0;
	d.cyls[0] = 50; d.heads[0] = 16; d.sectors_per_track[0] = 63;
	memset(stub_size_for, 0, sizeof(stub_size_for));
	stub_size_for[0] = 50ULL * 16 * 63 * 512;
	stub_size = 0;
	stub_size_for_active = 1;
	stub_noncdrom_id = -1;		/*  nothing is exempt: THIS drive is a CD-ROM  */
	stub_cdrom_poison = 1;
	wdc_initialize_identify_struct(fake_cpu(), &d);
	stub_cdrom_poison = 0;
	stub_size_for_active = 0;

	rows ++;
	if (word(&d, 0) != 0x8580) {
		printf("  FAIL %-38s word 0 = %04x, want 8580"
		    " (ATAPI/CDROM/removable)\n",
		    "a CD-ROM announces itself as one", word(&d, 0));
		failures ++;
	} else {
		printf("  ok   %-38s word 0 = 8580\n",
		    "a CD-ROM announces itself as one");
	}
}

/*
 *  #409: EXERCISE THE HIGH BYTE OF THE GEOMETRY WORDS.
 *
 *  Every other fixture keeps heads and sectors-per-track below 256, so the
 *  `>> 8` half of words 3 and 6 is never non-zero -- MEASURED: forcing either
 *  high byte to a literal 0 SURVIVES the whole table. Cylinders were already
 *  covered (300 > 255).
 *
 *  #410 CORRECTION, and it was a FALSE PASS in this very row. #409 shipped it
 *  with s = 17, so word 6's high byte was STILL permanently zero while this
 *  comment, the row's name, the gate and the commit all claimed words 3 AND 6.
 *  Measured: forcing word 6's high byte to 0 survived at all five optimisation
 *  levels, exactly as before the row existed. Word 3 was genuinely closed; word
 *  6 was not. The three values now carry DISTINCT high bytes -- 4096 = 0x1000,
 *  300 = 0x012c, 770 = 0x0302 -- so a mutant that SWAPS two high bytes is caught
 *  as well. A fixture whose fields share a byte value proves less than it looks:
 *  a first attempt at the multi-drive fixture used heads 400 and 271, both high
 *  byte 0x01, and killed nothing. These values are not reachable from the command line
 *  today only because the geometry override is broken (a separate round), and
 *  wdc copies whatever geometry the diskimage layer hands it.
 */
static void row_wide_geometry(void)
{
	struct wdc_data d;

	build(&d, 4096ULL * 300 * 770, 4096, 300, 770);
	rows ++;
	if (word(&d, 1) != 4096 || word(&d, 3) != 300 || word(&d, 6) != 770) {
		printf("  FAIL %-38s c=%u h=%u s=%u, want 4096/300/770\n",
		    "geometry words carry their high byte",
		    word(&d, 1), word(&d, 3), word(&d, 6));
		failures ++;
	} else {
		printf("  ok   %-38s c/h/s high bytes 10/01/03\n",
		    "geometry words carry their high byte");
	}
}

/*
 *  #407: WORD 49 MUST STAY ZERO, and this row is worth one character of mutant.
 *  Setting the LOW byte of word 49 (`identify_struct[2 * 49 + 1]`, dev_wdc.c:236 at
 *  085766a) to 2 gives word 0x0002 -- a reserved bit, NOT LBA (LBA is 0x0200, the
 *  high byte); that mutant passed all ten original rows, and this row pins both
 *  bytes so either mutant reds.  (This comment said "advertises LBA" until the
 *  2026-09-05 retrospective review.)
 *  It is not cosmetic: the LBA offset arms in wdc__read/wdc__write are `#if 0`
 *  (:315-321, :361-367), the live offsets at :311-313/:358-360 are pure CHS, and
 *  d->lba is parsed at :970 and never reaches an offset computation. A guest
 *  told LBA is supported would have its LBA addresses read as C/H/S.
 *
 *  This asserts the CURRENT, DELIBERATE state. If LBA is ever implemented, this
 *  row must be re-authored in the SAME change -- that is the point of pinning it.
 */
static void row_no_lba_claim(void)
{
	struct wdc_data d;

	build(&d, 1024, 1, 1, 1);
	rows ++;
	if (d.identify_struct[2 * 49 + 0] != 0 ||
	    d.identify_struct[2 * 49 + 1] != 0) {
		printf("  FAIL word 49 claims a capability: %02x %02x"
		    " (0x200 = LBA, 0x100 = DMA -- neither is implemented)\n",
		    d.identify_struct[2 * 49 + 0], d.identify_struct[2 * 49 + 1]);
		failures ++;
	} else {
		printf("  ok   %-38s word 49 = 00 00\n",
		    "no unimplemented capability claimed");
	}
}

/*
 *  THE SPEC-FREE ORACLE. Images are rounded to whole cylinders, so the sector
 *  count must equal cyls * heads * sectors_per_track -- both readable from the
 *  same block. No ATA document is needed to see the block contradict itself.
 *
 *  #407: H WAS 15 AND THE BOUND WAS 2000. Both were wrong for their purpose.
 *  diskimage.c:255-256 pins H=16/S=63 for every non-floppy, so H=15 measured a
 *  geometry no image has. And an 11-bit cylinder truncation, `(cyls >> 8) & 7`,
 *  SURVIVED all ten rows because its first self-contradiction is at c=2048 --
 *  the loop stopped 48 short of catching it. 65535 is the last count word 1 can
 *  represent, so it is the right bound; c=65536 cannot be represented at all and
 *  is a separate defect (the missing clamp) -- filed as wdccylwrap on 2026-09-05
 *  (this line promised "its own round" with no row behind it).
 */
static void row_selfconsistent(void)
{
	struct wdc_data d;
	int c, bad = 0, first_bad = 0;
	const int H = 16, S = 63;

	for (c = 1; c <= 65535; c++) {
		uint64_t sectors = (uint64_t)c * H * S;
		uint64_t claimed, geom;

		build(&d, sectors, c, H, S);
		claimed = sectors_from_block(&d, 60, 61);
		geom    = (uint64_t)word(&d, 1) * word(&d, 3) * word(&d, 6);

		if (claimed != geom) {
			if (!bad)
				first_bad = c;
			bad ++;
		}
	}
	rows ++;
	if (bad) {
		printf("  FAIL self-consistency: %d of 65535 cylinder counts give a\n"
		       "       block that contradicts its own geometry, first at %d\n",
		    bad, first_bad);
		failures ++;
	} else {
		printf("  ok   %-38s 0 of 65535 contradict their geometry\n",
		    "IDENTIFY agrees with itself");
	}
}

/*
 *  @@SELFCHECK@@ -- #551.  CAN THIS FILE'S COMPARATORS STILL FAIL?  Feeds chk() and row_cmp() known answers --
 *  each operand zero and past 32 bits, each of row_cmp()'s two disjuncts alone and on either side of its
 *  want, and one pair that must NOT
 *  fail -- and requires the EXACT change in `failures' for each, restoring `rows' and `failures' after every
 *  probe.  Run FIRST (before any row) and LAST (after every row, before the identity row), so a comparator
 *  that goes inert partway through is caught at one end or the other.  Only the success arm prints gate 2's G
 *  text; the failure arm does not, and forces one failure.
 *
 *  WHAT IT COVERS: the rows that go through chk() and row_cmp() -- the last run prints how many calls
 *  each comparator took.  Of chk()'s calls, the ones in row_ring_occupancy_sound() that pass a
 *  `cond ? 1 : 0' (or `? 0 : 1') hand it a VERDICT computed at the call, not a comparison, and every
 *  other row in this file compares inline: neither is covered by this block.  "Covered" means the
 *  COMPARATOR is live, not that a row is.  NOT CAUGHT, DISCLOSED: the self-check proves only that the
 *  comparators answer right on ITS OWN calls, so an edit that keeps those answers and changes the rest
 *  keeps every gate-2 row green -- a test keyed on a value, a name, a position or the self-check's own
 *  phase that the probes do not share; an edit of the code that FEEDS the comparators (row() handing
 *  row_cmp() its own want); an edit of this block itself.  Measured on #551's passes; reviewed in the
 *  differential, not chased.  A reset of `failures' ALONE after the comparators is not among them: they
 *  also count cmp_fails, which only they touch, and the trailer reports the larger.  A reset of BOTH
 *  tallies is, and is disclosed.  No line this block prints carries the self-mutant row's
 *  text or "high byte" (gate 2 counts both).  Gate 2's G text is printed ONCE, by the last run, only
 *  when both runs passed -- and the run's own PASS needs that line (sc_g), so a self-check that failed
 *  fails the run even if its forced failure is gone.  The rest of the file is filed as
 *  `wdcidcomparators' (#551; owner ruling 2026-09-25).
 */
static int sc_first_bad = -1;	/*  -1 until the FIRST run has happened  */
static int sc_g;	/*  set ONLY where gate 2's G line prints: the run's PASS needs it  */

static void selfcheck(const char *where)
{
	static const long long cg[] = { 1, 0x100000000LL, 0, 0, 7 };
	static const long long cw[] = { 2, 0, 1, 0x100000000LL, 7 };
	static const uint64_t rw[] = { 2, 2, 1, 3, 0x100000000ULL, 0x100000000ULL, 0, 0 };
	static const uint64_t r57[] = { 1, 2, 0, 3, 0, 0x100000000ULL, 1, 0 };
	static const uint64_t r60[] = { 2, 1, 0, 3, 0x100000000ULL, 0, 0, 1 };
	int sc_f = failures, sc_r = rows, sc_c = chk_calls, sc_rc = rowcmp_calls, sc_cf = cmp_fails, sc_bad = 0;
	int i, n = 0;

	for (i = 0; i < 5; i++, n++) {
		chk("@@SELFCHECK@@/chk", cg[i], cw[i]);
		if (failures != sc_f + (cg[i] != cw[i]))
			sc_bad++;
		failures = sc_f; rows = sc_r; cmp_fails = sc_cf;
	}
	for (i = 0; i < 8; i++, n++) {
		row_cmp("@@SELFCHECK@@/row", rw[i], r57[i], r60[i]);
		if (failures != sc_f + (r57[i] != rw[i] || r60[i] != rw[i]))
			sc_bad++;
		failures = sc_f; rows = sc_r; cmp_fails = sc_cf;
	}
	chk_calls = sc_c; rowcmp_calls = sc_rc;
	if (sc_bad) {
		printf("  FAIL  SELFCHECK(%s): %d of %d probes answered wrongly\n", where, sc_bad, n);
		failures = sc_f + 1; rows = sc_r; cmp_fails = sc_cf;
	}
	if (where[0] == 'f') {
		sc_first_bad = sc_bad;
		if (!sc_bad)
			printf("  ok    SELFCHECK(first): all %d probes answered right\n", n);
		return;
	}
	if (!sc_bad && sc_first_bad == 0) {
		sc_g = 1;
		printf("  ok    SELFCHECK the comparator can still fail (first and last)\n");
	} else if (!sc_bad)
		printf("  FAIL  SELFCHECK(last): the first run did not run, or did not pass\n");
	printf("  note  SELFCHECK comparator calls this run: chk() %d, row_cmp() %d\n",
	    chk_calls, rowcmp_calls);
}
