#!/usr/bin/env python3
"""#551 wdcidselfcheck, revision b (on #550 v3's goff_550v3.sh): gate 2 (regress/gate_offline.sh) -- wdc_identify moves from SC_EXEMPT to SC_COVERED, with its
coverage stated as the Fable adjudication ruled (fable5_f2_550_551.md, #551 rulings 4 and 7) and the owner ruled
("24/128 now, file the rest", 2026-09-25).  Applies on top of #550's gate_offline.sh.  Exactly-one-match edits.
usage: edit551_goff.py <in gate_offline.sh> <out>"""
import io
import sys


def sub1(s, old, new):
    n = s.count(old)
    if n != 1:
        sys.exit("REFUSED: %d matches for %r" % (n, old[:80]))
    return s.replace(old, new)

    s = sub1(s, 'diskimage_io diskimage_geom sh4_tmu rtc_range fbpending_bound"\n#  *** COVERED IS NOT UNIFORM',
             'diskimage_io diskimage_geom sh4_tmu rtc_range fbpending_bound wdc_identify"\n#  *** COVERED IS NOT UNIFORM')

                "#  stem was added; a proportion that has to be re-derived on every addition is a comment that\n"
                "#  will be wrong more often than right, so it is stated as an exception list instead.  The\n"
                "#  three that do NOT route through shared helpers:\n",
             "TREATING A GREEN ROW AS \"the file is vouched for\". ***  Most COVERED stems route every row\n"
             "#  through one shared helper, so the sentinel covers the file.  No count is kept here: it used to say\n"
             "#  \"seven of the ten\", then \"all but three\" -- stale the first time a stem was added, and wrong for\n"
             "#  geom and sync, which each have two helpers.  The stems whose sentinel does NOT cover every row, or\n"
             "#  needs more than one probe:\n")
    s = sub1(s, "#    * sync      -- both checks() and checkn(), for the same reason.\n",
             "#    * sync      -- both checks() and checkn(), for the same reason.\n"
             "#    * wdc_identify -- 24 of 128 rows compare inside a comparator the sentinel feeds: 16 of\n"
             "#                   chk()'s 24 calls and all 8 of row_cmp()'s.  chk()'s other 8 are handed a 0/1\n"
             "#                   verdict computed at the call (derive: grep -cE '\\? [01] : [01], 1\\);'\n"
             "#                   regress/diff_wdc_identify.c), and the other\n"
             "#                   96 compare inline -- NOT covered.  \"Covered\" means the COMPARATOR is live,\n"
             "#                   not that a row is: the self-check proves only its own calls, so an edit that\n"

             "#                   PASS needs the G line.  #551, by owner ruling 2026-09-25 (\"24/128 now, file\n"
             "#                   the rest\"); the rest is `wdcidcomparators'.\n")
    s = sub1(s, "#  Named here rather than left to be re-derived, because the three-file exception list above\n",
             "#  Named here rather than left to be re-derived, because the exception list above\n")
    s = sub1(s, "#  over the Q4 one I proposed for the selfmutant list: 2 uncovered differentials, 1311 and 766\n"
                "#  lines, whose every row could be inert with this gate green.\n",
             "#  over the Q4 one I proposed for the selfmutant list: 2 uncovered differentials, 1311 and 766\n"
             "#  lines, whose every row could be inert with this gate green.  (True when written; #551 moved\n"

    old_para_head = "#    wdc_identify  NOTHING TO FEED A MISMATCH TO.  Read every row function: row_anchor,\n"

    j = s.index(old_para_tail, i) + len(old_para_tail)

def main():
    src, dst = sys.argv[1], sys.argv[2]
    raw = io.open(src, encoding="utf-8", newline="").read()
    eol = "\r\n" if "\r\n" in raw else "\n"
    s = goff(raw.replace("\r\n", "\n"))
    io.open(dst, "w", encoding="utf-8", newline="").write(s.replace("\n", eol))
    print("edit551_goff ->", dst)


if __name__ == "__main__":
    main()
