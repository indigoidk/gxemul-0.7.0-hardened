"""#551 PASS 2: the WIDE row_cmp probes the ruling did not ask for -- one per disjunct, differing
only above bit 32 -- added to the FINAL file, plus the two edits they should now red."""
import io, sys
W = sys.argv[1]
F = io.open(W + "/final.c", encoding="utf-8", newline="").read()
def sub1(s, a, b):
    if s.count(a) != 1:
        raise SystemExit("anchor count %d: %r" % (s.count(a), a[:70]))
    return s.replace(a, b)
old = ("\tstatic const uint64_t rw[] = { 2, 2, 1, 3 };\n"
       "\tstatic const uint64_t r57[] = { 1, 2, 0, 3 };\n"
       "\tstatic const uint64_t r60[] = { 2, 1, 0, 3 };\n")
new = ("\tstatic const uint64_t rw[] = { 2, 2, 1, 3, 0x100000000ULL, 0x100000000ULL };\n"
       "\tstatic const uint64_t r57[] = { 1, 2, 0, 3, 0, 0x100000000ULL };\n"
       "\tstatic const uint64_t r60[] = { 2, 1, 0, 3, 0x100000000ULL, 0 };\n")
Fw = sub1(F, old, new)
Fw = sub1(Fw, "\tfor (i = 0; i < 4; i++, n++) {\n\t\trow_cmp(", "\tfor (i = 0; i < 6; i++, n++) {\n\t\trow_cmp(")
out = {"F_wide": Fw,
       "F_wide_u16": sub1(Fw, "static void row_cmp(const char *why, uint64_t sectors, uint64_t got57, uint64_t got60)",
                          "static void row_cmp(const char *why, uint16_t sectors, uint16_t got57, uint16_t got60)"),
       "F_wide_sect4": sub1(Fw, "\tif (got57 != sectors || got60 != sectors) {",
                            "\tif ((got57 != sectors || got60 != sectors) && sectors<4) {")}
for k, v in out.items():
    io.open("%s/h/%s.c" % (W, k), "w", encoding="utf-8", newline="").write(v)
print("wrote", " ".join(out))
