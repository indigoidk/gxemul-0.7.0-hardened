#!/usr/bin/env python3
"""#551d option: the LAST self-check also requires the comparators' call counts the file is built with
(chk() 24, row_cmp() 8), so covered calls moved past it (with the identity pin re-counted) are red.
Measured as an option only -- it pins two counts the way the identity row pins 128."""
import io, sys
W = sys.argv[1]
def sub1(s, a, b):
    if s.count(a) != 1:
        raise SystemExit("anchor count %d: %r" % (s.count(a), a[:70]))
    return s.replace(a, b)
D = io.open(W + "/final.c", encoding="utf-8", newline="").read()
P = sub1(D, "\tif (!sc_bad && sc_first_bad == 0) {\n\t\tsc_g = 1;",
         "\tif (!sc_bad && (chk_calls != 24 || rowcmp_calls != 8))\n\t\tsc_bad = 1;\n"
         "\tif (!sc_bad && sc_first_bad == 0) {\n\t\tsc_g = 1;")
AFTER = "\trow_ring_occupancy_sound();\n"
TOKEN_END = "\telse\n\t\tprintf(\"WDC_IDENTIFY_FAIL" + chr(92) + "n\");\n"
M = sub1(sub1(P, AFTER, ""), TOKEN_END, TOKEN_END + AFTER)
M = sub1(M, "\tif (rows != 128) {\n", "\tif (rows != 104) {\n")
io.open(W + "/h/Dpin.c", "w", encoding="utf-8", newline="").write(P)
io.open(W + "/h/Dpin_move_repin.c", "w", encoding="utf-8", newline="").write(M)
print("wrote Dpin Dpin_move_repin")
