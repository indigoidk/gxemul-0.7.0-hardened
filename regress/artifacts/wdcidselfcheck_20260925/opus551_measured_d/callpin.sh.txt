#!/bin/bash
set -u
D6=/mnt/c/Users/User00/AppData/Local/Temp/claude/C--DocumentNoSnc-cc-GXEMUL/8e474f41-4d47-4c99-bf11-3406a84c4fcd/scratchpad/opus551p1/d

FLAGS="-std=c99 -I$W/sec/src/include -I$W/sec/src/include/thirdparty -ffunction-sections -fdata-sections -Wl,--gc-sections"
run() { local h=$1 dev=$2 d=$W/dev/$2/regress; cp "$W/h/$h.c" "$d/"
  if cc -O2 $FLAGS -o "$d/$h.bin" "$d/$h.c" > "$d/$h.cc" 2>&1 && [ "$d/$h.bin" -nt "$d/$h.c" ]; then
    timeout 300 "$d/$h.bin" > "$d/$h.log" 2>&1; echo $? > "$d/$h.rc"; else echo BUILD_FAIL > "$d/$h.rc"; fi; }
for dev in pristine selfmut drqring w57hi dropold; do run Dpin $dev; done
for dev in pristine drqring; do run Dpin_move_repin $dev; done
for dev in pristine selfmut drqring w57hi dropold; do
  printf "  Dpin vs D %-8s %s rc %s/%s\n" $dev "$(cmp -s $W/dev/$dev/regress/Dpin.log $W/dev/$dev/regress/D.log && echo WHOLE-LOG-IDENTICAL || echo DIFFER)" \
    "$(cat $W/dev/$dev/regress/D.rc)" "$(cat $W/dev/$dev/regress/Dpin.rc)"; done
for dev in pristine drqring; do
  printf "  Dpin_move_repin %-8s %s | %s\n" $dev "$(grep -E '^[0-9]+ rows|WDC_IDENTIFY' $W/dev/$dev/regress/Dpin_move_repin.log | tr '\n' ' ')" \
    "$(grep -E 'SELFCHECK\((first|last)\)|comparator can still' $W/dev/$dev/regress/Dpin_move_repin.log | tr -s ' ' | tr '\n' ' ')"; done
