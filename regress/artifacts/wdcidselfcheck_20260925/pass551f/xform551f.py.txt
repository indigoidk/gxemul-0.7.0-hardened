BS = chr(92)
NL = BS + "n"
ART = "regress/artifacts/wdcidselfcheck_20260925/measured/run551f.out"


def sub1(s, old, new):
    n = s.count(old)
    if n != 1:
        raise SystemExit("REFUSED: %d matches for %r" % (n, old[:70]))
    return s.replace(old, new)


def emit(var):
    return ("\t\tif (%s[0] != '@')\t/*  #551f: contract (B) -- a covered failure reports itself  */\n"
            "\t\t\tprintf(\"  FAIL  covered comparator failure -- %%d failures so far" + NL + "\", cmp_fails);\n") % var


CHK_ARM = "\t\tcmp_fails ++;\n\t\tprintf(\"  FAIL %-40s got %lld want %lld" + NL + "\", name, got, want);\n"
ROW_ARM = "\t\tfailures ++;\n\t\tcmp_fails ++;\n\t} else {\n"

BLOCK_OLD_HEAD = " *  NOT CAUGHT, DISCLOSED.  This block pins the two COMPARATORS and nothing else: it proves they answer\n"

    " *  that the probes do not share; a predicate rewritten to agree with the original on every probe and differ\n"
    " *  elsewhere (the sum of the two gots against twice the want, say); and the code that FEEDS a comparator\n"

def make(orig):
    s = E.make(orig)
    for old, new in EDITS:
        s = sub1(s, old, new)
    i = s.index(BLOCK_OLD_HEAD)
    j = s.index(BLOCK_OLD_TAIL, i) + len(BLOCK_OLD_TAIL)
    if s.count(BLOCK_OLD_HEAD) != 1 or s.count(BLOCK_OLD_TAIL) != 1:
        raise SystemExit("REFUSED: the contract's anchors are not unique")
    return s[:i] + BLOCK_NEW + s[j:]


def main():
    orig = io.open(sys.argv[1], encoding="utf-8", newline="").read()
    assert "\r" not in orig
    s = make(orig)
    io.open(sys.argv[2], "w", encoding="utf-8", newline="").write(s)
    print("xform551f ->", sys.argv[2])


if __name__ == "__main__":
    main()
