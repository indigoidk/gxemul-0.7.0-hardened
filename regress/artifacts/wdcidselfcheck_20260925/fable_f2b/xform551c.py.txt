  * two more row_cmp() probes, (sectors 0, got57 1, got60 0) and (0, 0, 1): each disjunct wrong ALONE with got ABOVE
    its want, and sectors 0 -- `<' for `!=' (minimax) and `&& sectors' (the Opus seat; grok, kimi-k3) both answered
    every earlier probe right; this also makes "each operand zero" true of every row_cmp() operand;
  * cmp_fails: the comparators count their own failures, which nothing but they touch (the self-check restores it with
    `failures'), and the trailer reports the larger of the two -- so `failures = 0;' after the covered rows (the Opus
    seat, grok, agy) cannot hide a comparator's failure; an untampered run is unchanged, since `failures' always
    counts at least what the comparators counted;
  * a self-check that did not pass counts as a failure in the trailer (grok: the double edit printed "0 failures" and
    WDC_IDENTIFY_FAIL);
  * the disclosure in its general form (glm): the self-check proves only its own calls.

usage: xform551c.py <orig diff_wdc_identify.c> <out.c> [variant]
  variants: xform551b.py's, plus lt andsect reset phase"""

    n = s.count(old)
    if n != 1:
        raise SystemExit("REFUSED: %d matches for %r" % (n, old[:70]))
    return s.replace(old, new)

def make(orig):
    s = B.make(orig)
    for old, new in EDITS:
        s = sub1(s, old, new)
    return s


def main():
    orig = io.open(sys.argv[1], encoding="utf-8", newline="").read()
    assert "\r" not in orig
    s = make(orig)
    v = sys.argv[3] if len(sys.argv) > 3 else None
    if v == "nosc":
        s = s.replace('\tselfcheck("first");\n', "").replace('\tselfcheck("last");\n', "")
        i = s.index("/*\n *  @@SELFCHECK@@")
        j = s.index("int main(void)\n{\n", i)
        s = s[:i] + s[j:]
        s = sub1(s, B.NEW_VERDICT, B.VERDICT)
        s = sub1(s, "\tif (!sc_g && failures == 0)\n\t\tfailures = 1;\n", "")
    elif v in COMBOS:
        for part in COMBOS[v]:
            s = sub1(s, *VARIANTS[part])
    elif v:
        s = sub1(s, *VARIANTS[v])
    io.open(sys.argv[2], "w", encoding="utf-8", newline="").write(s)
    print("xform551c ->", sys.argv[2], v or "")


if __name__ == "__main__":
    main()
