#!/usr/bin/env python3
"""#551f: score every (shape, device) TWICE and compare with the coordinator's run551f.out cell by cell.

  scorer M (mine, as analyze7): gate 2's wdc block with a MOCK check()/check_min() of lib.sh's semantics.
  scorer R (independent of the mock): the SAME extracted block run under gate 2's REAL lib.sh (sourced; gate_begin,
           then the block, then its own _fails counter), plus H and G as the literal greps gate_offline.sh runs.
  Both add the self-mutant lane (selfmutant.py's pristine/mutant logic; arms = device and device+self-mutant).
A shape ESCAPES on a device when the unedited #551f file reds it, the shape keeps gate 2 green there, and the
pristine device is green too."""
import io, os, re, subprocess
W = "/tmp/opus551f"
ART = "/mnt/c/DocumentNoSnc/cc/GXEMUL/GXEMUL-SEC/regress/artifacts/wdcidselfcheck_20260925"
NAMED = "geometry words carry their high byte"
G_TEXT = "SELFCHECK the comparator can still fail"
EMIT = "covered comparator failure"
os.makedirs(W + "/logdir", exist_ok=True)
io.open(W + "/gate_mock.sh", "w").write(
    'WDCLOG="$1"; _f=0\n'
    'check() { [ "$2" = "$3" ] || { echo "RED $1 got=$2 want=$3"; _f=$((_f+1)); }; }\n'
    'check_min() { [ "$2" -ge "$3" ] 2>/dev/null || { echo "RED $1 got=$2 min=$3"; _f=$((_f+1)); }; }\n'
    'note() { :; }\n. ' + W + '/gatewdc.inc\n')
io.open(W + "/gate_real.sh", "w").write(
    'export LOGDIR=' + W + '/logdir\nWDCLOG="$1"\n. ' + W + '/lib.sh > /dev/null 2>&1\n'
    'gate_begin "wdc-only" > /dev/null 2>&1\n. ' + W + '/gatewdc.inc > /dev/null 2>&1\necho "REALFAILS=$_fails CHECKS=$_checks"\n')


def rd(dev, h, ext):
    p = "%s/dev/%s/regress/%s.%s" % (W, dev, h, ext)
    return io.open(p, encoding="utf-8", errors="replace").read() if os.path.exists(p) else None


def summ(dev, h):
    if (rd(dev, h, "rc") or "").strip() == "BUILD_FAIL":
        return "BUILD_FAIL"
    t = rd(dev, h, "log") or ""
    m = re.search(r"^(\d+) rows, (\d+) failures", t, re.M)
    e = len([l for l in t.split("\n") if EMIT in l])
    return "%s/%s %s%s" % (m.group(1) if m else "?", m.group(2) if m else "?",
                           "PASS" if "WDC_IDENTIFY_PASS" in t else ("FAIL" if "WDC_IDENTIFY_FAIL" in t else "NOTOKEN"),
                           " r%d" % e if e else "")


def lane(h, dev):
    lane_dev = "selfmut" if dev == "pristine" else dev + "_sm"
    t, tm = rd(dev, h, "log"), rd(lane_dev, h, "log")
    named = [l for l in t.split("\n") if l.strip().startswith(("ok", "FAIL")) and NAMED in l and "@@SELFCHECK@@" not in l]
    if tm is not None and re.search(r"\b[A-Z0-9_]+_PASS\b", t) and len(named) == 1 and not named[0].strip().startswith("FAIL"):
        killed = [l for l in tm.split("\n") if l.strip().startswith("FAIL") and NAMED in l and "@@SELFCHECK@@" not in l]
        return "OK" if (re.search(r"\b[A-Z0-9_]+_FAIL\b", tm) and killed) else "FAIL"
    return "SETUP"


def gate(h, dev, real):
    if (rd(dev, h, "rc") or "").strip() == "BUILD_FAIL":
        return ["compile"]
    logp = "%s/dev/%s/regress/%s.log" % (W, dev, h)
    if real:
        out = subprocess.run(["bash", W + "/gate_real.sh", logp], capture_output=True, text=True).stdout
        m = re.search(r"REALFAILS=(\d+)", out)
        reds = ["real-lib fails=%s" % (m.group(1) if m else "?")] if (not m or m.group(1) != "0") else []
        src = "%s/h/%s.c" % (W, h)
        if subprocess.run(["grep", "-q", "@@SELFCHECK@@", src]).returncode != 0:
            reds.append("H")
        if subprocess.run(["grep", "-q", G_TEXT, logp]).returncode != 0:
            reds.append("G")
    else:
        out = subprocess.run(["bash", W + "/gate_mock.sh", logp], capture_output=True, text=True).stdout
        reds = [l[4:44] for l in out.split("\n") if l.startswith("RED ")]
        if "@@SELFCHECK@@" not in io.open("%s/h/%s.c" % (W, h), encoding="utf-8").read():
            reds.append("H")
        if G_TEXT not in rd(dev, h, "log"):
            reds.append("G")
    ln = lane(h, dev)
    if ln != "OK":
        reds.append("lane:" + ln)
    return reds


def cell(h, dev, real):
    return ("GREEN" if not gate(h, dev, real) else "red") + "(" + summ(dev, h) + ")"


#  the coordinator's table, parsed
theirs = {}
for line in io.open(ART + "/measured/run551f.out", encoding="utf-8").read().split("\n"):
    m = re.match(r"\s+(F\w*)\s+((?:GREEN|red)\([^)]*\))\s+((?:GREEN|red)\([^)]*\))\s+((?:GREEN|red)\([^)]*\))\s+((?:GREEN|red)\([^)]*\))", line)
    if m:
        theirs[m.group(1)] = [m.group(i) for i in (2, 3, 4, 5)]
DEVS = ("pristine", "dropold", "drqring", "w57hi")
shapes = sorted(f[:-2] for f in os.listdir(W + "/h") if f.endswith(".c") and f[:-2] not in ("F", "E", "ORIG"))
print("=== (1) %d shapes: my scorer (M), gate 2's real lib.sh (R), and the coordinator's cell (C), each device ===" % len(shapes))
mism, mr = [], []
for h in ["F"] + shapes:
    M = [cell(h, d, False) for d in DEVS]
    R = [cell(h, d, True) for d in DEVS]
    C = theirs.get(h)
    esc = [d for d in DEVS[1:] if not gate(h, "pristine", True) and not gate(h, d, True) and gate("F", d, True)] if h != "F" else []
    flag = ""
    if M != R:
        mr.append(h); flag += " M!=R"
    if C is not None and C != M:
        mism.append(h); flag += " C!=M"
    print("    %-24s %-26s %-26s %-26s %-26s esc=%-18s %s" % (h, M[0], M[1], M[2], M[3], ",".join(esc) or "-",
                                                             ("theirs: " + ("same" if C == M else "|".join(C))) if C else "(not in run551f)") + flag)
print("  cells where my mock and gate 2's real lib.sh disagree: %s" % (" ".join(mr) or "NONE"))
print("  shapes whose cells differ from run551f.out: %s" % (" ".join(mism) or "NONE"))
print("  shapes in run551f.out not rebuilt here: %s" % (" ".join(sorted(set(theirs) - set(["F"] + shapes))) or "NONE"))

print("\n=== (2) healthy and defect runs: #551f against #551e (whole log) ===")
for dev in sorted(os.listdir(W + "/dev")):
    f, e = rd(dev, "F", "log"), rd(dev, "E", "log")
    extra = [l for l in f.split("\n") if l not in e.split("\n")]
    print("    %-11s %s | rc F=%s E=%s" % (dev, "WHOLE-LOG IDENTICAL" if f == e else "+%d lines, all report lines: %s" % (
        len(extra), all(EMIT in l for l in extra)), rd(dev, "F", "rc").strip(), rd(dev, "E", "rc").strip()))

print("\n=== (3) the report line against every gate-2 pattern (would a report line satisfy or break a named check?) ===")
pats = re.findall(r"grep -c(F?) '([^']+)' \"\$WDCLOG\"", io.open(W + "/gatewdc.inc", encoding="utf-8").read())
rep = "  FAIL  covered comparator failure -- 3 failures so far"
hits = []
for fixed, p in pats:
    r = subprocess.run(["grep", "-c" + ("F" if fixed else ""), p], input=rep + "\n", capture_output=True, text=True).stdout.strip()
    if r != "0":
        hits.append(p)
print("    %d named-row/text patterns checked; patterns the report line matches: %s" % (len(pats), hits or "NONE"))
print("    probe names starting with '@': chk(\"@@SELFCHECK@@/chk\") and row_cmp(\"@@SELFCHECK@@/row\") only -- %s" % (
    subprocess.run(["bash", "-c", "grep -oE '\\b(chk|row|row_cmp)\\(\"@[^\"]*\"' " + W + "/h/F.c | sort | uniq -c | tr '\\n' ' '"],
                   capture_output=True, text=True).stdout.strip()))
print("ANALYZE9_DONE")
