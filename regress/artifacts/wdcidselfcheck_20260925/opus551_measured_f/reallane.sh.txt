#!/bin/bash
#  The REAL regress/selfmutant.py (copied, unmodified) run on a mirror tree per cell: src = the device variant's
#  dev_wdc.c over HEAD's headers, regress/diff_wdc_identify.c = the shape.  SERIAL; two builds per cell.
set -u
REPO=/mnt/c/DocumentNoSnc/cc/GXEMUL/GXEMUL-SEC
W=/tmp/opus551f
L=/tmp/opus551f_lane; rm -rf $L; mkdir -p $L
for cellspec in "F pristine" "F_macro drqring" "F_macro w57hi" "F_guard1_chk dropold" "F_skipforge drqring"; do
  set -- $cellspec; h=$1; dev=$2; T=$L/${h}_$dev
  mkdir -p $T/regress/selfmutants
  cp -a $W/sec/src $T/src
  cp $W/dev/$dev/src/devices/dev_wdc.c $T/src/devices/dev_wdc.c
  cp $W/sec/config.h $T/config.h
  cp $REPO/regress/selfmutant.py $T/regress/selfmutant.py
  cp $W/h/$h.c $T/regress/diff_wdc_identify.c
  cp $W/sec/regress/selfmutants/wdc_identify.old $W/sec/regress/selfmutants/wdc_identify.new $T/regress/selfmutants/
  out=$(cd $T/regress && python3 selfmutant.py diff_wdc_identify.c src/devices/dev_wdc.c \
        $T/regress/selfmutants/wdc_identify.old $T/regress/selfmutants/wdc_identify.new \
        "geometry words carry their high byte" 2>&1)
  echo "  real lane $h on $dev: $(printf '%s\n' "$out" | grep -oE 'SELFMUTANT_(OK|FAIL|SETUP)') | $(printf '%s\n' "$out" | grep -E '^ +(ok|FAIL|SETUP) ' | tr -s ' ' | tr '\n' ' ' | cut -c1-200)"
done
echo REALLANE_DONE
