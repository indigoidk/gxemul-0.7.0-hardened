#!/bin/bash
#  #551i: the comment-stripped preprocessor output of #551f/g/h, raw and whitespace-normalized; #551g's record gives
#  c10d453c for f and g.  Also the gate script with every comment-only line removed, g vs i.
cd /mnt/c/Users/User00/AppData/Local/Temp/claude/C--DocumentNoSnc-cc-GXEMUL/8e474f41-4d47-4c99-bf11-3406a84c4fcd/scratchpad/wdc551 || exit 9
for f in MINE_g.c MINE_h.c MINE_i.c; do
    a=$(gcc -fpreprocessed -dD -E -P "$f" 2>/dev/null | md5sum | cut -c1-8)
    b=$(gcc -fpreprocessed -dD -E -P "$f" 2>/dev/null | tr -s ' \t\n' ' ' | md5sum | cut -c1-8)
    echo "$f raw=$a normalized=$b"
done
for f in goff_551g.sh goff_551i.sh; do
    echo "$f non-comment lines md5=$(grep -vE '^[[:space:]]*#' "$f" | md5sum | cut -c1-8) ($(grep -vcE '^[[:space:]]*#' "$f") lines)"
done
echo STRIP551I_DONE
