#!/usr/bin/env python3
"""mk_551h.py -- #551's comment corrections from GPT-6 Astra's confirmation (final551g/codex_astra551.txt), applied to
the pinned #551g sources as NEW files (MINE_h.c, goff_551h.sh); the _g versions stay as history.  COMMENT TEXT ONLY --
the script refuses if any changed line is not inside a comment.  Exact-match edits; a find-text that is not unique is a
hard failure."""
import hashlib, io, os, sys

HERE = os.path.dirname(os.path.abspath(__file__))


def load(n):
    return io.open(os.path.join(HERE, n), encoding="utf-8", newline="").read()


def rep(s, old, new):
    n = s.count(old)
    if n != 1:
        sys.exit("TEXT NOT UNIQUE (%d): %r" % (n, old[:70]))
    return s.replace(old, new)


c = load("MINE_g.c")
if hashlib.md5(c.encode("utf-8")).hexdigest()[:8] != "db79dad9":
    sys.exit("MINE_g.c is not db79dad9")
c = rep(c,
        "/*  #551: the comparison is split out of row() as a PURE comparator, so the self-check can feed it each\n"
        "    disjunct alone, and a zero, without building a device.  row() and its eight callers are unchanged in\n"
        "    what they print.  */\n",
        "/*  #551: the comparison is split out of row() as a DEVICE-INDEPENDENT comparator (it counts and prints, so it\n"
        "    is not pure), so the self-check can feed it each disjunct alone, and a zero, without building a device.\n"
        "    row() built the device and compared the extracted got57/got60 against `sectors'; nothing could inject an\n"
        "    observed value independently.  row()'s eight call sites keep their original output and add a report line\n"
        "    for each COVERED failure (w57hi adds four, w58lo five -- GPT-6 Astra, confirming #551).  */\n")
c = rep(c,
        " *  `rows' and chk_calls is one insertion and green (measured); an exit handler that seeks a file-redirected\n"
        " *  stdout to 0 and writes a healthy log over the real one is one insertion and green (measured)\n"
        " *  (M19, #551f's pass).  The identity row's 128 and the LAST run's call counts raise the cost of a skip;\n",
        " *  `rows' and chk_calls is one insertion and green (measured); an exit handler that seeks a file-redirected\n"
        " *  stdout to 0 and writes a healthy log over the real one is one insertion and passes the file-redirected\n"
        " *  MAIN checks (M19, #551f's pass) -- the existing piped self-mutant lane rejects it on all three tested\n"
        " *  defects (dropold, drqring, w57hi), so it does not keep the complete gate green (GPT-6 Astra, confirming\n"
        " *  #551).  The identity row's 128 and the LAST run's call counts raise the cost of a skip;\n")

g = load("goff_551g.sh")
if hashlib.md5(g.encode("utf-8")).hexdigest()[:8] != "813a4023":
    sys.exit("goff_551g.sh is not 813a4023")
g = rep(g,
        "#  needs more than one probe:\n"
        "#    * sh4_tmu   -- 12 of 18 rows.  row1() is the only comparator in the file that takes an\n"
        "#                   expected value; row_three_timers/row_interrupt/row_interrupt_unie_clear/\n"
        "#                   row_stopped/row_no_freeze each hardcode their own, and cannot be probed.\n",
        "#  has multiple comparators:\n"
        "#    * sh4_tmu   -- 12 of 19 rows.  row1() is the only comparator in the file that takes an\n"
        "#                   expected value; row_three_timers/row_rtcsr_falls_through/row_interrupt/\n"
        "#                   row_interrupt_unie_clear/row_stopped/row_no_freeze each hardcode their own,\n"
        "#                   and cannot be probed; the 19th is the identity row.  (This said \"12 of 18\" and\n"
        "#                   \"needs more than one probe\" -- GPT-6 Astra, confirming #551.)\n")
g = rep(g,
        "#                   it its own want, or its failure report kept from this gate's count check while\n"
        "#                   the tallies are hidden -- one edit or two, in the comparator's text or through\n"
        "#                   a file-scope macro); an edit of the self-check itself; and the observables\n"
        "#                   forged (covered rows skipped with their labels and counts printed, or the log\n"
        "#                   rewritten by the process after the rows ran).  Anything else -- a trailer\n",
        "#                   it its own want, or its failure report kept from this gate's count check while\n"
        "#                   the tallies are hidden -- one edit or two, in the comparator's text or through\n"
        "#                   a file-scope macro); an edit of the self-check itself; and covered rows skipped\n"
        "#                   with their labels and counts printed.  (A log REWRITTEN by the process after\n"
        "#                   the rows ran passes the file-redirected MAIN checks, but the existing piped\n"
        "#                   self-mutant lane rejects it on all three tested defects, so it does NOT keep\n"
        "#                   this gate green -- GPT-6 Astra, confirming #551.)  Anything else -- a trailer\n")

# comment-only proof: every line that differs must be a comment line in its file's syntax
for name, old, new, ok in (("MINE_h.c", load("MINE_g.c"), c, lambda l: l.lstrip().startswith(("/*", "*", "*/")) or l.strip() == "" or l.startswith("    ")),
                           ("goff_551h.sh", load("goff_551g.sh"), g, lambda l: l.lstrip().startswith("#") or l.strip() == "")):
    import difflib
    changed = [l[1:] for l in difflib.unified_diff(old.split("\n"), new.split("\n"), lineterm="", n=0)
               if l[:1] in "+-" and not l.startswith(("+++", "---"))]
    bad = [l for l in changed if not ok(l)]
    if bad:
        sys.exit("%s: NON-COMMENT LINE CHANGED: %r" % (name, bad[0]))
    io.open(os.path.join(HERE, name), "w", encoding="utf-8", newline="").write(new)
    print("wrote %s %s (%d changed lines, all comment text)" % (name, hashlib.md5(new.encode("utf-8")).hexdigest()[:8], len(changed)))
