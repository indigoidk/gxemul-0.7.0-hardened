#!/bin/bash
#  #551f2 opus seat: build M19 (grok's heal handler) and M18 (four resets), run each with BOTH the gate's
#  file redirect AND the proposed pipe, score with gate 2's REAL lib.sh + H + G.  SERIAL.
set -u
F2=/mnt/c/Users/User00/AppData/Local/Temp/claude/C--DocumentNoSnc-cc-GXEMUL/8e474f41-4d47-4c99-bf11-3406a84c4fcd/scratchpad/opus551p1/f2
OLD=/tmp/opus551f
W=/tmp/opus551f2
rm -rf "$W"; mkdir -p "$W/h" "$W/logdir"
cp -a "$OLD/sec" "$W/"
cp "$OLD/h/F.c" "$W/h/F.c"
cp "$OLD/gatewdc.inc" "$OLD/lib.sh" "$W/"
DEVS="pristine dropold drqring w57hi"
SM="selfmut dropold_sm drqring_sm w57hi_sm"
for d in $DEVS $SM; do mkdir -p "$W/dev/$d/regress"; cp "$OLD/dev/$d/src/devices/dev_wdc.c" "$W/dev/$d/dev_wdc.c" 2>/dev/null; cp -a "$OLD/dev/$d/src" "$W/dev/$d/src"; done
echo "F md5 $(md5sum < "$W/h/F.c" | cut -c1-8) (want 7fd15d1d)"
python3 "$F2/xf2.py" "$W/h" "$OLD/dev/pristine/regress/F.log" || exit 1
echo "variants: $(ls "$W/h")"
FLAGS="-std=c99 -I$W/sec/src/include -I$W/sec/src/include/thirdparty -ffunction-sections -fdata-sections -Wl,--gc-sections"
HARN="F M19_heal M18_f_before_recon M18_f_after_printf M18_cf_before_recon M18_cf_after_printf"
t0=$(date +%s)
for h in $HARN; do
  for dev in $DEVS $SM; do
    d="$W/dev/$dev/regress"; cp "$W/h/$h.c" "$d/$h.c"
    if cc -O2 $FLAGS -o "$d/$h.bin" "$d/$h.c" > "$d/$h.cc" 2>&1 && [ "$d/$h.bin" -nt "$d/$h.c" ]; then
      timeout 300 "$d/$h.bin" > "$d/$h.file.log" 2>&1; echo "$? $(wc -c < "$d/$h.file.log")" > "$d/$h.file.rc"
      ( timeout 300 "$d/$h.bin" 2>&1 | cat > "$d/$h.pipe.log" ); echo "${PIPESTATUS[0]} $(wc -c < "$d/$h.pipe.log")" > "$d/$h.pipe.rc"
    else
      echo BUILD_FAIL > "$d/$h.file.rc"; echo BUILD_FAIL > "$d/$h.pipe.rc"; head -3 "$d/$h.cc"
    fi
  done
done
echo "built+ran serially in $(( $(date +%s) - t0 )) s"
echo "build failures: $(grep -l BUILD_FAIL "$W"/dev/*/regress/*.file.rc 2>/dev/null | sed "s|$W/dev/||" | tr '\n' ' ')"
echo RUN_F2_DONE
