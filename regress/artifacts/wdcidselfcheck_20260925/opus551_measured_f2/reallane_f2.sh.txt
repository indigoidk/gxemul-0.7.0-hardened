#!/bin/bash
#  The REAL regress/selfmutant.py (copied, unmodified) on a mirror tree per cell.  selfmutant.py runs the
#  binary through subprocess.Popen(stdout=PIPE) -- an UNSEEKABLE pipe -- so this is also the natural test of
#  whether M19's fseek-rewrite survives a pipe.  The gate's real lane runs on the PRISTINE device; the
#  drqring/w57hi/dropold cells are the covered-row-failing devices, shown for contrast.  SERIAL.
set -u
REPO=/mnt/c/DocumentNoSnc/cc/GXEMUL/GXEMUL-SEC
W=/tmp/opus551f2
OLD=/tmp/opus551f
L=/tmp/opus551f2_lane; rm -rf "$L"; mkdir -p "$L"
for cellspec in "F pristine" "M19_heal pristine" "M19_heal drqring" "M19_heal w57hi" "M19_heal dropold" \
                "M18_f_before_recon drqring" "M18_cf_after_printf drqring"; do
  set -- $cellspec; h=$1; dev=$2; T=$L/${h}_$dev
  mkdir -p "$T/regress/selfmutants"
  cp -a "$W/sec/src" "$T/src"
  cp "$W/dev/$dev/src/devices/dev_wdc.c" "$T/src/devices/dev_wdc.c"
  cp "$W/sec/config.h" "$T/config.h"
  cp "$REPO/regress/selfmutant.py" "$T/regress/selfmutant.py"
  cp "$W/h/$h.c" "$T/regress/diff_wdc_identify.c"
  cp "$OLD/sec/regress/selfmutants/wdc_identify.old" "$OLD/sec/regress/selfmutants/wdc_identify.new" "$T/regress/selfmutants/"
  out=$(cd "$T/regress" && python3 selfmutant.py diff_wdc_identify.c src/devices/dev_wdc.c \
        "$T/regress/selfmutants/wdc_identify.old" "$T/regress/selfmutants/wdc_identify.new" \
        "geometry words carry their high byte" 2>&1)
  echo "  lane $h on $dev: $(printf '%s\n' "$out" | grep -oE 'SELFMUTANT_(OK|FAIL|SETUP)') | $(printf '%s\n' "$out" | grep -E '^ +(ok|FAIL|SETUP) ' | tr -s ' ' | tr '\n' '/' | cut -c1-160)"
done
echo REALLANE_F2_DONE
