def move_repin(s):
    """the Opus seat's D_move_repin: covered chk() rows moved past the verdict, the identity row re-pinned to 104"""
    s = sub1(s, MAIN_RING, "")
    s = sub1(s, VERDICT_TAIL, VERDICT_TAIL + MAIN_RING)
    s = sub1(s, ID_PIN, "\tif (rows != 104) {\n")
    s = sub1(s, ID_MSG, "rows = %i, want 104" + NL)
    return s


def nopin(s):
    i = s.index(PIN)
    return s[:i] + s[i + len(PIN):]


RECON = "\tif (cmp_fails > failures)\n\t\tfailures = cmp_fails;\n"
VARIANTS = dict(D.VARIANTS)


def make(orig):
    s = D.make(orig)
    for old, new in EDITS:
        s = sub1(s, old, new)
    return s


def main():
    orig = io.open(sys.argv[1], encoding="utf-8", newline="").read()
    assert "\r" not in orig
    s = make(orig)
    v = sys.argv[3] if len(sys.argv) > 3 else None
    if v == "nosc":
        s = s.replace('\tselfcheck("first");\n', "").replace('\tselfcheck("last");\n', "")
        i = s.index("/*\n *  @@SELFCHECK@@")
        j = s.index("int main(void)\n{\n", i)
        s = s[:i] + s[j:]
        s = sub1(s, "\tif (failures == 0 && cmp_fails == 0 && sc_g)\n", "\tif (failures == 0 && cmp_fails == 0)\n")

        s = sub1(s, "\tif (!sc_g && failures == 0)\n\t\tfailures = 1;\n", "")
    elif v == "incdel_reset":
        s = sub1(s, *D.INCDEL)
        s = sub1(s, *VARIANTS["reset"])
    elif v == "move_repin":
        s = move_repin(s)
    elif v == "nopin":
        s = nopin(s)
    elif v == "nopin_move_repin":
        s = move_repin(nopin(s))
    elif v == "condreset":
        #  the disclosed class (3), conditional form: both tallies reset when they are equal, at the trailer
        s = sub1(s, RECON, "\tif (failures == cmp_fails)\n\t\tfailures = cmp_fails = 0;\n" + RECON)
    elif v in D.COMBOS:
        for part in D.COMBOS[v]:
            s = sub1(s, *VARIANTS[part])
    elif v:
        s = sub1(s, *VARIANTS[v])
    io.open(sys.argv[2], "w", encoding="utf-8", newline="").write(s)
    print("xform551e ->", sys.argv[2], v or "")


if __name__ == "__main__":
    main()
