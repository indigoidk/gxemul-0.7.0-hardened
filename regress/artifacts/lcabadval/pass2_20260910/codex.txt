def seq(consts, phases, settle=1.6, spin=2.2):
    """consts: list of 64-bit values -> SCRATCH slots 0,8,16,...  phases: list of lists of
    (op, addr_slot, val_slot) with op in {'st', 'ld'}; each phase ends with `print t3'.
    Returns (txt, alive, [read-backs in phase order as strings])."""
    setup = ["put d 0x%x, 0x%x" % (SCRATCH + 8 * i, v) for i, v in enumerate(consts)]
    prog = load_const(T0, SCRATCH)
    lens = [len(prog)]
    for ph in phases:
        ops = []
        for op, a, v in ph:
            ops.append(LDQ(T1, 8 * a, T0))
            if op == "st":
                ops.append(LDQ(T2, 8 * v, T0))
                ops.append(STQ(T2, 0, T1))
            else:
                ops.append(LDQ(T3, 0, T1))
        prog += ops
        lens.append(len(ops))
    cmds = setup + ["put w 0x%x, 0x%08x" % (k0(CODE) + 4 * i, op) for i, op in enumerate(prog)]
    cmds += ["pc=0x%x" % k0(CODE), "step %d" % lens[0]]
    for n in lens[1:]:

            re.findall(r"print\s+t3\s*\n\s*(0x[0-9a-f]+)", txt)]

def mem(op, ra, rb, disp):
    assert -32768 <= disp <= 32767, "displacement %d out of range" % disp
    return (op << 26) | (ra << 21) | (rb << 16) | (disp & 0xffff)

def LDA (ra, disp, rb=ZERO): return mem(0x08, ra, rb, disp)
def LDAH(ra, disp, rb=ZERO): return mem(0x09, ra, rb, disp)
def LDQ (ra, disp, rb):      return mem(0x29, ra, rb, disp)
def STQ (ra, disp, rb):      return mem(0x2d, ra, rb, disp)
#  QUADWORD ONLY, and that is architectural rather than a preference: HRM S6.4 says the IOC
#  registers are accessed with quadword operations, so LDL/STL are excluded by the manual.

def load_const(reg, val):
    #  32-bit only, and deliberately so: this builds the SCRATCH pointer, never a 64-bit operand.
    #  Every operand this probe drives is 64-bit-capable and arrives via `put d' + LDQ.
    lo = val & 0xffff
    hi = (val >> 16) & 0xffff
    if lo >= 0x8000:
        hi = (hi + 1) & 0xffff
    ops = [LDAH(reg, hi if hi < 0x8000 else hi - 0x10000)]
    if lo:
        ops.append(LDA(reg, lo if lo < 0x8000 else lo - 0x10000, reg))
    return ops

# ---------------------------------------------------------------- session
def session(cmds, settle=1.6, spin=2.2):
    pid, fd = pty.fork()
    if pid == 0:
        os.execv(BIN, [BIN, "-V", "-x", "-E", "alpha", "-e", MACH, "-M", "64",
                       "0x20000000:" + DUMMY])
        os._exit(127)
    buf = ""
    def rd(t):
        nonlocal buf
        end = time.time() + t
        while time.time() < end:
            r, _, _ = select.select([fd], [], [], 0.1)
            if not r:
                continue
            try:
                d = os.read(fd, 65536)
                if not d:
                    return False
                buf += d.decode("latin-1", "replace")
            except OSError:
                return False
        return True
    alive = rd(settle)
    for i, c in enumerate(cmds):
        try:
            os.write(fd, (c + "\n").encode("latin-1"))
        except OSError:
            alive = False; break
        if not rd(spin if i == len(cmds) - 1 else 0.35):
            alive = False; break
    try:
        os.write(fd, b"quit\n"); rd(0.4)
    except OSError:
        pass
    try:
        os.close(fd)
    except OSError:
        pass
    return buf.replace("\r", ""), alive

def printed(t, reg):
    m = re.search(r"print\s+%s\s*\n\s*(0x[0-9a-f]+)" % reg, t)
    return "absent" if not m else "0x%x" % int(m.group(1), 16)

# ---------------------------------------------------------------- the per-arm driver
#  Register discipline, copied from lca_confbus_probe.py: T0 holds the SCRATCH base and is read
#  for the LAST time by the third LDQ, so every operand is in a register before anything else
#  happens.  The four accesses are labelled [A]..[D] here and in the report.
def arm(addr, accepted, legal, settle=1.6, spin=2.2):
    setup = ["put d 0x%x, 0x%x" % (SCRATCH + 0,  k0(addr)),
             "put d 0x%x, 0x%x" % (SCRATCH + 8,  accepted),
             "put d 0x%x, 0x%x" % (SCRATCH + 16, legal)]
    prologue = load_const(T0, SCRATCH) + [
        LDQ(T1, 0,  T0),    # t1 = k0(register)
        LDQ(T4, 8,  T0),    # t4 = the ACCEPTED value
        LDQ(T5, 16, T0),    # t5 = the manual-LEGAL value  (T0 is dead after this)
        LDQ(T2, 0,  T1),    # [A] PRE-read       -- must be 0 (the model memsets its state)
        STQ(T4, 0,  T1),    # [B] ACCEPTED store -- must survive
        LDQ(T3, 0,  T1)]    # [C] READ-BACK      -- must echo the accepted value
    armed = [STQ(T5, 0, T1)]  # [D] the manual-LEGAL store -- ends the host on HEAD
    cmds = setup + ["put w 0x%x, 0x%08x" % (k0(CODE) + 4 * i, op)
                    for i, op in enumerate(prologue + armed)]
    cmds += ["pc=0x%x" % k0(CODE),
             "step %d" % len(prologue),   # up to and including the read-back
             "print t2",                  # [A], while the host is certainly still alive
             "print t3",                  # [C], likewise
             "step %d" % len(armed)]      # [D]
    txt, alive = session(cmds, settle, spin)
    return txt, alive, printed(txt, "t2"), printed(txt, "t3")

#  each phase reloads T1/T2 from a `put d' slot, so eight registers are never a constraint.  A
#  phase is stepped as a block and its read-back printed while the host is certainly alive.

def seq(consts, phases, settle=1.6, spin=2.2):
    """consts: list of 64-bit values -> SCRATCH slots 0,8,16,...  phases: list of lists of
    (op, addr_slot, val_slot) with op in {'st', 'ld'}; each phase ends with `print t3'.
    Returns (txt, alive, [read-backs in phase order as strings])."""
    setup = ["put d 0x%x, 0x%x" % (SCRATCH + 8 * i, v) for i, v in enumerate(consts)]
    prog = load_const(T0, SCRATCH)
    lens = [len(prog)]
    for ph in phases:
        ops = []
        for op, a, v in ph:
            ops.append(LDQ(T1, 8 * a, T0))
            if op == "st":
                ops.append(LDQ(T2, 8 * v, T0))
                ops.append(STQ(T2, 0, T1))
            else:
                ops.append(LDQ(T3, 0, T1))
        prog += ops
        lens.append(len(ops))
    cmds = setup + ["put w 0x%x, 0x%08x" % (k0(CODE) + 4 * i, op) for i, op in enumerate(prog)]
    cmds += ["pc=0x%x" % k0(CODE), "step %d" % lens[0]]
    for n in lens[1:]:
        cmds += ["step %d" % n, "print t3"]
    txt, alive = session(cmds, settle, spin)
    vals = ["0x%x" % int(m, 16) for m in
            re.findall(r"print\s+t3\s*\n\s*(0x[0-9a-f]+)", txt)]
    return txt, alive, vals

 109: 	    round, not this one.
 110: 
 111: 	    So the disposition is DECLINE plus a bounded announce, and what it buys is
 112: 	    stated narrowly on purpose: THE HOST STOPS DYING.  Interrupts do not start
 113: 	    working; they were never delivered on this architecture and still are not.
 114: 
 115: 	    The latch is PER-INSTANCE (`interrupt->extra'), not function-static: a
 116: 	    function-static bit would let one guest access to one device instance
 117: 	    silence the diagnostic for every other instance, which is a class already
 118: 	    filed against this tree.  */
 119: 	if (d != NULL && !(d->warned & (1u << LCA_W_IRQ_ASSERT))) {	/*  #516  */
 120: 		d->warned |= 1u << LCA_W_IRQ_ASSERT;
 121: 		fatal("[ lca: ISA interrupt assert not implemented (Alpha has no "
 122: 		    "interrupt delivery); ignored  (once) ]\n");
 123: 	}
 124: }

 128:  *  lca_interrupt_deassert():
 129:  *
 130:  *  Line 0 = ISA interrupt.
 131:  */
 132: void lca_interrupt_deassert(struct interrupt *interrupt)
 133: {
 134: 	struct lca_data *d = (struct lca_data *) interrupt->extra;	/*  #516  */
 135: 
 136: 	/*  #516: the sibling of lca_interrupt_assert() above, same reasoning.  A
 137: 	    SEPARATE latch bit, so the two are distinguishable: a driver normally
 138: 	    reaches both within one interrupt cycle, and one shared bit would report
 139: 	    whichever arrived first and hide the other entirely.  */
 140: 	if (d != NULL && !(d->warned & (1u << LCA_W_IRQ_DEASSERT))) {	/*  #516  */
 141: 		d->warned |= 1u << LCA_W_IRQ_DEASSERT;
 142: 		fatal("[ lca: ISA interrupt deassert not implemented (Alpha has no "
 143: 		    "interrupt delivery); ignored  (once) ]\n");
 144: 	}
 145: }

 148: DEVICE_ACCESS(lca_pci_conf)
 149: {
 150: 	uint64_t idata = 0, odata = 0;
 151: 	int tag, type1, bus, dev, func, reg;	/*  #515  */
 152: 	struct lca_data *d = (struct lca_data *) extra;
 153: 
 154: 	if (writeflag == MEM_WRITE)
 155: 		idata = memory_readmax64(cpu, data, len);
 156: 
 157: 	/*
 158: 	 *  1. Decompose the address into a tag.
 159: 	 *
 160: 	 *  According to NetBSD's lca_pci.c, the address is composed like this:
 161: 	 *
 162: 	 *	addr = tag << 5 | (regoffset & ~0x03) << 5 | 0x3 << 3
 163: 	 */
 164: 	reg = (relative_addr >> 5) & 0xfc;
 165: 	tag = (relative_addr >> 5) & ~0xff;
 166: 
 167: 	/*
 168: 	 *  2. Decompose the tag into bus, dev, and func.
 169: 	 *
 170: 	 *  The tag can be constructed in one of two ways. On the primary
 171: 	 *  bus (nr 0):
 172: 	 *
 173: 	 *	tag = (1 << (device + 11)) | (function << 8);
 174: 	 *
 175: 	 *  and on other busses, the tag is a normal:
 176: 	 *
 177: 	 *	tag = (bus << 16) | (device << 11) | (function << 8)
 178: 	 */
 179: 	/*  printf("tag = 0x%x\n", (int)tag);  */
 180: 
 181: 	/*  #515: `bus' used to be BOTH the configuration CYCLE TYPE and the
 182: 	    DESTINATION BUS, and conflating them is the whole defect.  IOC_CONF bit 0
 183: 	    selects the cycle type only -- 21066/21066A HRM S6.2.9: "PCI address bits
 184: 	    ad<1:0> come directly from configuration cycle type register (IOC_CONF,
 185: 	    Section 6.4.2) bits <1:0>" -- and it does NOT carry a bus number.  The
 186: 	    destination bus of a type-1 cycle lives in the ADDRESS, per the PCI Local
 187: 	    Bus specification's type-1 layout (bus 23..16, dev 15..11, fn 10..8,
 188: 	    reg 7..2), which S6.2.9 defers to explicitly.
 189: 
 190: 	    Before this, a guest that merely SET bit 0 -- a value the CONF guard
 191: 	    ACCEPTS -- made any subsequent access in this window reach
 192: 	    `fatal("TODO. Non-zero bus."); exit(1)'.  MEASURED: two ordinary guest
 193: 	    accesses, nothing refused anywhere on the path
 194: 	    (regress/lca_confbus_probe.py, 3/4 pre-fix; artifacts/lcaconfbus1kill/).  */
 195: 	type1 = d->ioc_conf & 1;	/*  #515  */
 196: 	bus = 0;			/*  #515  */
 197: 
 198: 	if (!type1) {			/*  #515  */
 199: 		/*  #511: was `0x800 << dev'.  `0x800' is a plain int (2^11) and `dev'
 200: 		    reaches 20, so the last iteration evaluates 2^31 -- a SIGNED
 201: 		    OVERFLOW, undefined by C99 6.5.7p4, not merely
 202: 		    implementation-defined.  MEASURED under UBSAN before the change:
 203: 		    "dev_lca.c:125:21: runtime error: left shift of 2048 by 20 places
 204: 		    cannot be represented in type 'int'", reached by an ordinary guest
 205: 		    ldq into the PCI config window (regress/lca_confbus_probe.py against
 206: 		    build-asan; artifacts/lcashiftub/ubsan_HEAD_20260907.raw).
 207: 
 208: 		    `0x800U' makes the operand unsigned, so 2048U << 20 is 0x80000000U --
 209: 		    inside unsigned int -- and NOTHING ELSE MOVES.
 210: 
 211: 		    *** THE REASON, CORRECTED: AN EARLIER DRAFT SAID `tag' IS "ALREADY A
 212: 		    WIDER UNSIGNED QUANTITY".  THAT IS FALSE -- `tag' is a plain signed
 213: 		    `int' (:92) -- AND FOUR PASS-2 SEATS CAUGHT IT.  Worse, the claim was
 214: 		    backwards: being WIDER is what would CREATE a difference, not remove
 215: 		    one.  Had `tag' been `uint64_t', the old `int' 0x80000000 is NEGATIVE
 216: 		    and would SIGN-EXTEND to 0xffffffff80000000 for the `&', while
 217: 		    `0x800U << 20' zero-extends to 0x0000000080000000 -- two different
 218: 		    masks.  The seats were right about the mechanism and it does not fire
 219: 		    here only because of the type.
 220: 
 221: 		    WHAT ACTUALLY HOLDS: both operands are 32-bit, so the `&' is performed
 222: 		    at 32 bits either way -- `int & int' before, and after, `tag' converted
 223: 		    to `unsigned int' by the usual arithmetic conversions.  The RESULT'S
 224: 		    BIT PATTERN IS IDENTICAL; only its type changes, and it is consumed
 225: 		    solely as a truth test in `if (...)'.  So no guest-visible value moves,
 226: 		    and the repair is owed purely by the LANGUAGE. ***
 227: 
 228: 		    *** THE LOOP THIS SITS IN IS THIS FILE'S OWN GRACEFUL-DEGRADATION
 229: 		    PRECEDENT -- the `dev >= 21' arm below has a commented-out
 230: 		    fatal()/exit(1) and substitutes `dev = 0' instead, and `lcabadval'
 231: 		    cites it as a transferable warrant.  The safe path contained the UB. ***  */
 232: 		for (dev=0; dev<21; dev++)
 233: 			if (tag & (0x800U << dev))	/*  #511  */
 234: 				break;
 235: 		if (dev >= 21) {
 236: 			/*  fatal("[ LCA: No bus 0 device? TODO ]\n");
 237: 			exit(1);  */
 238: 			dev = 0;
 239: 		}
 240: 	} else {
 241: 		/*  #515: a type-1 cycle carries its destination in the address.  */
 242: 		bus = (tag >> 16) & 0xff;	/*  #515  */
 243: 		dev = (tag >> 11) & 0x1f;	/*  #515  */
 244: 	}
 245: 
 246: 	func = (tag >> 8) & 7;
 247: 	/*  printf("bus=%i dev=%i func=%i reg=%i\n", bus,dev,func,reg);  */
 248: 
 249: 	/*  #515: a type-1 cycle whose bus field is zero selects NOTHING.  Type-1 is
 250: 	    how a bridge is asked to reach a bus BEHIND it; the primary bus is reached
 251: 	    with type-0 cycles, whose IDSEL encoding is decoded above.  Forwarding this
 252: 	    case would set cur_bus = 0 and hand the access to whichever primary-bus
 253: 	    device matched -- `i82378zb' is registered at 0:7:0 by DEVINIT below -- so
 254: 	    simply decoding `bus' and stopping there would replace a host kill with a
 255: 	    NEW MISROUTING.  Refuse it instead, matching `bus_pci_data_access''s own
 256: 	    empty-slot contract (bus_pci.c, the `dev == NULL' arm: all-ones at register
 257: 	    0, zero elsewhere) so the guest sees the ordinary "nothing there" answer.
 258: 	    Writes are discarded, which leaves backend state untouched.  */
 259: 	if (type1 && bus == 0) {		/*  #515  */
 260: 		odata = reg == 0? (uint64_t) -1 : 0;
 261: 	} else {
 262: 		/*  Pass PCI accesses onto bus_pci:  */
 263: 		bus_pci_setaddr(cpu, d->pci_data, bus, dev, func, reg);
 264: 		bus_pci_data_access(cpu, d->pci_data, writeflag == MEM_READ?
 265: 		    &odata : &idata, len, writeflag);
 266: 	}
 267: 
 268: 	if (writeflag == MEM_READ)
 269: 		memory_writemax64(cpu, data, len, odata);
 270: 
 271: 	return 1;
 272: }

 275: DEVICE_ACCESS(lca_isa)
 276: {
 277: 	unsigned int ofs, i;
 278: 	uint8_t byte;
 279: 
 280: 	relative_addr >>= 5;
 281: 
 282: 	ofs = relative_addr & 3;
 283: 	if (ofs > len) {
 284: 		fatal("[ ofs=%i len=%i in lca_isa access function. "
 285: 		    "aborting ]\n", ofs, len);
 286: 		exit(1);
 287: 	}
 288: 
 289: 	if (writeflag == MEM_WRITE) {
 290: 		byte = data[ofs % len];
 291: 		return cpu->memory_rw(cpu, cpu->mem, LCA_ISA_BASE +
 292: 		    relative_addr, &byte, 1, writeflag, CACHE_NONE);
 293: 	}
 294: 
 295: 	cpu->memory_rw(cpu, cpu->mem, LCA_ISA_BASE + relative_addr,
 296: 	    &byte, 1, MEM_READ, CACHE_NONE);
 297: 
 298: 	for (i=0; i<len; i++)
 299: 		data[i] = i == ofs? byte : 0x00;
 300: 
 301: 	return 1;
 302: }

 304: 
 305: /*
 306:  *  #518 (lcabadval): one bounded announce per (device instance, reason).  Six
 307:  *  IOC write arms used to end the host with fatal()+exit(1) on any value outside
 308:  *  a tiny accepted set -- and the sets were WRONG, not merely harsh: the 21066
 309:  *  HRM (Table 6-6) lists THIRTEEN legal W_MASK encodings where the model took
 310:  *  one, and S6.4.14 REQUIRES SROM to write ZERO to W_BASE1, which was fatal.
 311:  *  The arms below store the ARCHITECTED FIELDS, ignore reserved bits (type X),
 312:  *  validate W_MASK by its field rule, and decline what the model cannot honour
 313:  *  with this announce -- fatal() prints and returns (it cannot be silenced by
 314:  *  -q, which is why a latch and not a verbosity flag).
 315:  */
 316: static void lca_ioc_announce(struct lca_data *d, int bit, const char *what,
 317: 	uint64_t v)
 318: {
 319: 	if (d->warned & (1u << bit))
 320: 		return;
 321: 	d->warned |= 1u << bit;
 322: 	fatal("[ lca_ioc: %s: 0x%" PRIx64 "  (once) ]\n", what, v);
 323: }

 325: 
 326: DEVICE_ACCESS(lca_ioc)
 327: {
 328: 	uint64_t idata = 0, odata = 0;
 329: 	struct lca_data *d = (struct lca_data *) extra;
 330: 
 331: 	if (writeflag == MEM_WRITE)
 332: 		idata = memory_readmax64(cpu, data, len);
 333: 
 334: 	switch (relative_addr + LCA_IOC_BASE) {
 335: 
 336: 	case LCA_IOC_BASE:
 337: 		/*  Ignore? Linux reads from the base at startup.  */
 338: 		break;
 339: 
 340: 	case LCA_IOC_CONF:
 341: 		if (writeflag == MEM_READ) {
 342: 			odata = d->ioc_conf;
 343: 		} else {
 344: 			/*  #518: was `if (idata & ~1) { fatal(); exit(1); }' with the
 345: 			    store BEFORE the check (#506 had widened its diagnostic).
 346: 			    HRM S6.2.9: IOC_CONF<1:0> select the configuration cycle
 347: 			    TYPE; everything above is reserved and is IGNORED, not
 348: 			    fatal.  Bit 0 is what lca_pci_conf reads; bit 1 (type-1
 349: 			    cycles) is stored and announced once, not modelled.  */
 350: 			d->ioc_conf = idata & 0x3ULL;
 351: 			if (idata & 0x2ULL)
 352: 				lca_ioc_announce(d, LCA_W_IOC_CONF_CFG1, "IOC_CONF<1> set"
 353: 				    " (type-1 configuration cycles): stored, not modelled", idata);
 354: 		}
 355: 		break;
 356: 
 357: 	case LCA_IOC_TBIA:
 358: 		/*  TLB Invalidate All.  */
 359: 		/*  TODO: For now, let's just ignore it.  */
 360: 		break;
 361: 
 362: 	case LCA_IOC_TB_ENA:
 363: 		if (writeflag == MEM_READ) {
 364: 			odata = d->tlb_enable;
 365: 		} else {
 366: 			/*  #518: TEN is the only architected bit (HRM Table 6-13);
 367: 			    the rest are type X -- IGNORED on write, never fatal.  The
 368: 			    old arm stored the whole value and then died on it.  */
 369: 			d->tlb_enable = idata & IOC_TB_ENA_TEN;
 370: 		}
 371: 		break;
 372: 
 373: 	case LCA_IOC_W_BASE0:
 374: 		if (writeflag == MEM_READ) {
 375: 			odata = d->window_base_0;
 376: 		} else {
 377: 			/*  #518: was a two-value whitelist + exit(1).  The HRM defines
 378: 			    this register by FIELDS -- W_BASE<31:20>, SG<32>, WEN<33> --
 379: 			    not by a list of numbers, and S6.4.14 REQUIRES SROM to write
 380: 			    zero here.  Store the fields (reserved bits dropped); the
 381: 			    window is not translated by this model, which is announced
 382: 			    once when any window is ENABLED.  */
 383: 			d->window_base_0 = idata & (IOC_W_BASE_W_BASE | IOC_W_BASE_SG | IOC_W_BASE_WEN);
 384: 			if (idata & IOC_W_BASE_WEN)
 385: 				lca_ioc_announce(d, LCA_W_IOC_WEN, "a DMA window was"
 386: 				    " enabled; window translation is not modelled", idata);
 387: 		}
 388: 		break;
 389: 
 390: 	case LCA_IOC_W_MASK0:
 391: 		if (writeflag == MEM_READ) {
 392: 			odata = d->window_mask_0;
 393: 		} else {
 394: 			/*  #518: was `!= 0x700000ULL' + exit(1): ONE accepted value where
 395: 			    HRM Table 6-6 lists THIRTEEN (W_MASK<31:20> = 0, 1, 3, 7, ...
 396: 			    0xfff: a contiguous low-ones field), and the old arm STORED
 397: 			    before it checked.  Validate the field, store only when it is
 398: 			    legal, decline once otherwise with the previous value kept.  */
 399: 			{
 400: 				uint64_t v = idata & 0xfff00000ULL, f = v >> 20;
 401: 				if ((f & (f + 1)) == 0)
 402: 					d->window_mask_0 = v;
 403: 				else
 404: 					lca_ioc_announce(d, LCA_W_IOC_WMASK0, "unsupported"
 405: 					    " IOC_W_MASK0 encoding declined (HRM Table 6-6);"
 406: 					    " previous value kept", idata);
 407: 			}
 408: 		}
 409: 		break;
 410: 
 411: 	case LCA_IOC_W_T_BASE0:
 412: 		if (writeflag == MEM_READ) {
 413: 			odata = d->window_t_base_0;
 414: 		} else {
 415: 			d->window_t_base_0 = idata;
 416: 			/*  TODO: Actually implement this.  */
 417: 		}
 418: 		break;
 419: 
 420: 	case LCA_IOC_W_BASE1:
 421: 		if (writeflag == MEM_READ) {
 422: 			odata = d->window_base_1;
 423: 		} else {
 424: 			/*  #518: was a ONE-value whitelist + exit(1) -- and the one value
 425: 			    was not zero, which S6.4.14 item 3 requires SROM to write.
 426: 			    Same disposition as W_BASE0 above.  */
 427: 			d->window_base_1 = idata & (IOC_W_BASE_W_BASE | IOC_W_BASE_SG | IOC_W_BASE_WEN);
 428: 			if (idata & IOC_W_BASE_WEN)
 429: 				lca_ioc_announce(d, LCA_W_IOC_WEN, "a DMA window was"
 430: 				    " enabled; window translation is not modelled", idata);
 431: 		}
 432: 		break;
 433: 
 434: 	case LCA_IOC_W_MASK1:
 435: 		if (writeflag == MEM_READ) {
 436: 			odata = d->window_mask_1;
 437: 		} else {
 438: 			/*  #518: was `!= 0x3ff00000ULL' + exit(1): ONE accepted value where
 439: 			    HRM Table 6-6 lists THIRTEEN (W_MASK<31:20> = 0, 1, 3, 7, ...
 440: 			    0xfff: a contiguous low-ones field), and the old arm STORED
 441: 			    before it checked.  Validate the field, store only when it is
 442: 			    legal, decline once otherwise with the previous value kept.  */
 443: 			{
 444: 				uint64_t v = idata & 0xfff00000ULL, f = v >> 20;
 445: 				if ((f & (f + 1)) == 0)
 446: 					d->window_mask_1 = v;
 447: 				else
 448: 					lca_ioc_announce(d, LCA_W_IOC_WMASK1, "unsupported"
 449: 					    " IOC_W_MASK1 encoding declined (HRM Table 6-6);"
 450: 					    " previous value kept", idata);
 451: 			}
 452: 		}
 453: 		break;
 454: 
 455: 	case LCA_IOC_W_T_BASE1:
 456: 		if (writeflag == MEM_READ) {
 457: 			odata = d->window_t_base_1;
 458: 		} else {
 459: 			d->window_t_base_1 = idata;
 460: 			/*  TODO: Actually implement this.  */
 461: 		}
 462: 		break;
 463: 
 464: 	default:
 465: 		/*  #518: was fatal()+exit(1) for ANY unlisted offset.  Decline
 466: 		    the write (no state changes), answer a read with zero (odata
 467: 		    is already 0), announce each direction once.  The value keeps
 468: 		    #506's full width.  */
 469: 		{
 470: 			int bit = writeflag == MEM_WRITE ?
 471: 			    LCA_W_IOC_DEF_W : LCA_W_IOC_DEF_R;
 472: 			if (!(d->warned & (1u << bit))) {
 473: 				d->warned |= 1u << bit;
 474: 				fatal("[ lca_ioc: unimplemented %s to offset 0x%x",
 475: 				    writeflag == MEM_WRITE? "write" : "read", (int)
 476: 				    relative_addr);
 477: 				if (writeflag == MEM_WRITE)
 478: 					fatal(": 0x%" PRIx64 " -- declined",
 479: 					    (uint64_t)idata);	/*  #506  */
 480: 				else
 481: 					fatal(" -- answered with zero");
 482: 				fatal("  (once per direction) ]\n");
 483: 			}
 484: 		}
 485: 		break;
 486: 	}
 487: 
 488: 	if (writeflag == MEM_READ)
 489: 		memory_writemax64(cpu, data, len, odata);
 490: 
 491: 	return 1;
 492: }
