  CHANGELOG.md               HEAD's with the #555 block prepended
and write spec555.json and msg555.txt for commit_unit.py.  Every number in the three texts is derived here from the

TEXT_EXT = (".c", ".h", ".py", ".sh", ".md", ".json", ".txt", ".tsv")
first = lambda k: int(re.match(r"\d+", k).group(0))


def git(*a, binary=False):
    r = subprocess.run(["git", "-C", SEC] + list(a), capture_output=True)
    if r.returncode:
        sys.exit("REFUSED: git %s: %s" % (" ".join(a), r.stderr.decode("utf-8", "replace")))
    return r.stdout if binary else r.stdout.decode("utf-8")


def md5(b):
    return hashlib.md5(b).hexdigest()[:8]


def put(rel, data):
    p = os.path.join(ST, *rel.split("/"))
    os.makedirs(os.path.dirname(p), exist_ok=True)
    open(p, "wb").write(data)
    return p


def rd(p):
    return io.open(p, encoding="utf-8", errors="replace").read().replace("\r\n", "\n")


def tsv(name):
    return [l.split("\t") for l in rd(os.path.join(D, name)).split("\n")[1:] if l.strip()]


def verdict(*rel):
    v = re.findall(r"VERDICT: ([^\n]*)", rd(rel[0] if os.path.isabs(rel[0]) else os.path.join(D, *rel)))
    if not v:
        sys.exit("REFUSED: no VERDICT line in %s" % "/".join(rel))
    return v[0].replace("`", "").strip().rstrip(".")


def carried(rows, ledger, queue):
    """passage -> [(id, note)]: what the checks ruled (IDLESS_CARRIED.tsv), then the FILED dispositions no check
    touched (IDLESS_FILED_CHECK.tsv).  Every id must exist; its ledger state is given where it is not `open`."""
    new = set(r["id"] for r in rows)
    state = dict((x["id"], x.get("state") or "?") for x in ledger["rows"])
    spans = set(re.findall(r"`([A-Za-z][A-Za-z0-9_-]{2,})`", queue))
    car = [c for c in tsv("IDLESS_CARRIED.tsv") if c[1] != "(UNSETTLED)"]
    uns = [c for c in tsv("IDLESS_CARRIED.tsv") if c[1] == "(UNSETTLED)"]
    ruled = set(c[0] for c in car)
    for f in tsv("IDLESS_FILED_CHECK.tsv"):
        if f[1] in ruled:
            continue
        ids = f[3].split() if f[3] != "?" else re.findall(r"DISPOSITION: FILED ([A-Za-z][\w-]*)", f[6])
        if not ids:
            sys.exit("REFUSED: the FILED entry at %s names no id this script can read" % f[1])
        for i in ids:
            car.append([f[1], i, "", ""])
    by, only_shut = {}, []
    for q, rid, scope, _src in car:
        q = re.match(r"\d+(?:-\d+)?(?:; ?\d+(?:-\d+)?)*", q).group(0)      # "2614-2615 scoped to PPC" is the passage 2614-2615
        if rid in new:
            tag = "a `#555` row above"
        elif rid in state:
            tag = "" if state[rid] == "open" else "ledger: %s" % state[rid]
        elif rid in spans or re.search(r"\| %s \|" % re.escape(rid), queue):
            tag = "named in this file only: no ledger row of its own"
        else:
            sys.exit("REFUSED: %s is carried by `%s`, which exists nowhere" % (q, rid))
        note = "; ".join(x for x in (tag, scope) if x)
        if rid not in [i for i, _ in by.setdefault(q, [])]:
            by[q].append((rid, note))
    for q in by:
        if all(state.get(i) in ("closed", "dropped") for i, _ in by[q]):
            only_shut.append(q)
    return by, uns, only_shut


def agy_jobs():
    """the agy seat's audits of the table: (jobs queued, answered, walled at launch, never launched)"""
    jobs = [l.split("\t")[1] for l in rd(os.path.join(D, "agy_idless_queue.tsv.txt")).split("\n") if l.strip() and not l.startswith("#")]
    log = rd(os.path.join(D, "agy_results.tsv"))
    ans = [j for j in jobs if re.search(r"\t%s\trc=0\t.*seat answered" % re.escape(j), log)]
    wall = [j for j in jobs if j not in ans and re.search(r"\t%s\trc=\d+\t.*SEAT WALLED" % re.escape(j), log)]
    return jobs, ans, wall, [j for j in jobs if j not in ans and j not in wall]


def main():
    dry = "--dry" in sys.argv
    args = [a for a in sys.argv[1:] if a != "--dry"]
    if len(args) != 1:
        sys.exit(__doc__)
    closing = args[0]
    head = git("rev-parse", "--short=8", "HEAD").strip()
    if not git("log", "-1", "--format=%s", "HEAD").startswith("#554 "):
        sys.exit("REFUSED: HEAD %s is not #554's commit" % head)
    if os.path.isdir(ST):
        for root, dirs, files in os.walk(ST, topdown=False):
            for f in files:
                os.remove(os.path.join(root, f))
            for d in dirs:
                os.rmdir(os.path.join(root, d))
    os.makedirs(ST, exist_ok=True)
    man = []
    r = subprocess.run([sys.executable, os.path.join(REC, "records555.py"), "gen"], capture_output=True, text=True)
    if r.returncode:
        sys.exit("REFUSED: records555 gen: %s%s" % (r.stdout, r.stderr))
    man.append("records555 gen: " + r.stdout.strip())
    rows = R.table()
    g = dict((s, sum(1 for x in rows if x["sev"] == s)) for s in R.ORDER)
    queue = git("show", "HEAD:OUTSTANDING_BUGS.md")
    ledger = json.loads(git("show", "HEAD:tools/pipeline/ledger.json"))
    by, uns, only_shut = carried(rows, ledger, queue)
    sents = apply555.rows()
    nsettled = sum(1 for s in sents if "UNSETTLED" in s[3])
    # the sentences that name a LIVE row for what stays open.  Keyed on what makes an id a live carrier -- a #555 row, a
    # ledger row that is not closed or dropped, or a row this file lists with no ledger row of its own -- and not on
    # the words around it: the first form counted the phrase "filed as" and missed "remains `armalignexc`" (GPT-6
    # Astra's ifile_05, E1).
    state = dict((x["id"], x.get("state")) for x in ledger["rows"])
    qrows = set(re.findall(r"(?m)^(?:\* |#### )`([a-z][a-zA-Z0-9_]{3,})` ", queue))
    newids = set(x["id"] for x in rows)
    live = lambda s: [i for i in dict.fromkeys(re.findall(r"`([a-z][a-zA-Z0-9_]{3,})`", s))
                      if i in newids or state.get(i) in ("open", "held", "in work") or (i not in state and i in qrows)]
    rem = [(s[0], live(s[2])) for s in sents if live(s[2])]
    nrem = len(rem)
    c5 = [c[1] for c in tsv("IDLESS_CARRIED.tsv") if c[3].startswith("grok/grok_irule_c.txt 5 ")]
    if c5 != ["sh4sqexpevt"]:
        sys.exit("REFUSED: the changelog's sentence on `qhshsqat` is written for one carrier, `sh4sqexpevt`; the table has %s" % c5)
    disp = rd(os.path.join(D, "IDLESS_DISPOSITIONS.md"))
    ndisp = int(re.search(r"Totals: (\d+) dispositions", disp).group(1))
    nclaimed = int(re.search(r"(\d+) claimed entries\)", disp).group(1))
    nmissed = int(re.search(r"the\s+(\d+) it found the scan had missed", disp).group(1))
    notowed = tsv("IDLESS_NOTOWED.tsv")
    keep = [x for x in notowed if x[1] == "KEEP-AS-IS"]
    kmoved = [x for x in keep if x[4] in by]
    retired = tsv("IDLESS_RETIRED.tsv")
    fix = tsv("IDLESS_TITLEFIX.tsv")
    if len(fix) != 1:
        sys.exit("REFUSED: IDLESS_TITLEFIX.tsv has %d rows; the texts below are written for one" % len(fix))
    fid, fold, fnew, _why = fix[0]
    cls = R.classrows()
    cov = [x["id"] for x in rows if x["src"].startswith("grok/grok_grok_irule_d.txt 4 ")]
    if len(cls) != 1 or len(cov) != 1:
        sys.exit("REFUSED: the texts below are written for one class row and one coverage row from ruling D; found %d and %d" % (len(cls), len(cov)))
    cid = list(cls)[0]
    mem, evidence, _where = cls[cid]
    unread = R.sweep_unread(mem, evidence)
    jobs, ans, wall, never = agy_jobs()

    # 1. the artifact files
    files, skipped, empty, nul = [], 0, 0, []
    complete = re.compile(r"seat answered|SEAT WALLED|REFUSED|The answer is LOST|PANEL_COMPLETE|rc=")
    tracked = set(git("ls-files", "regress/artifacts").split("\n"))
    now = time.time()
    for d in DIRS:
        for root, _, fs in os.walk(os.path.join(ARTS, d)):
            for f in sorted(fs):
                p = os.path.join(root, f)
                rel = os.path.relpath(p, SEC).replace(os.sep, "/")
                if rel in tracked:
                    continue
                if f.endswith(".err"):
                    skipped += 1
                    continue
                if os.path.getsize(p) == 0:
                    empty += 1
                    continue
                if now - os.path.getmtime(p) < 120 and not dry:
                    sys.exit("REFUSED: %s changed in the last two minutes -- a job in flight?" % p)
                if f.endswith(".launch.log") and not complete.search(rd(p)):
                    if not dry:
                        sys.exit("REFUSED: %s has no completion line -- a job in flight?" % p)
                    man.append("(dry) IN FLIGHT, would refuse: %s has no completion line" % rel)
                if f.endswith(TEXT_EXT) and b"\0" in open(p, "rb").read():
                    nul.append(rel)
                files.append(rel)
    if nul:
        sys.exit("REFUSED: %d artifact text file(s) carry a NUL byte: %s" % (len(nul), "; ".join(nul[:6])))
    man.append("artifacts: %d files to add under %s; %d *.err and %d zero-byte files left on disk" % (len(files), ", ".join(DIRS), skipped, empty))

    fmt = dict(nnew=len(rows), nh=g["HIGH"], nm=g["MED"], nl=g["LOW"], nind=g["HIGH"] + g["MED"], nledger=g["HIGH"] + g["MED"] + 1,
               nsent=len(sents), nclose=len(sents) - nsettled, nsettled=nsettled, nrem=nrem, ncar=len(by), nshut=len(only_shut), nuns=len(uns),
               remids=", ".join("`%s`" % i for _q, ids in rem for i in ids),
               ndisp=ndisp, nclaimed=nclaimed, nmissed=nmissed, nkeep=len(keep), nkmoved=len(kmoved), nkstay=len(keep) - len(kmoved),
               nret=len(retired), nmerged=sum(1 for x in retired if not x[1].startswith("(none")), fid=fid,
               v1=verdict("ifile_01", "codex_ifile_01.txt"), v2=verdict("ifile_02", "codex_ifile_02.txt"),
               v3=verdict("ifile_03", "codex_ifile_03.txt"), ga=verdict("grok", "grok_irule_a.txt"),
               gb=verdict("grok", "grok_irule_b.txt"), gc=verdict("grok", "grok_irule_c.txt"), closing=closing,
               gr=verdict(os.path.join(ARTS, "idlessread_20261002", "grok", "grok_grok_iunsettled.txt")),
               v4=verdict("ifile_04", "codex_ifile_04.txt"), gd=verdict("grok", "grok_grok_irule_d.txt"),
               v5=verdict("ifile_05", "codex_ifile_05.txt"),
               cid=cid, covid=cov[0], nmem=len(mem), nprose=len(rows) - 1, nunread=len(unread),
               unread=", ".join("`%s`" % u for u in unread),
               nfiles=len(files), nagy=len(jobs), nagyans=len(ans), nagywall=len(wall), nagynever=len(never),
               nagynot=len(wall) + len(never), agynot=", ".join("`%s`" % j for j in wall + never))

    # 2. OUTSTANDING_BUGS.md
    ob, rep = apply555.apply(queue)
    if ob.count(fold) != 1:
        sys.exit("REFUSED: the title #554 filed for %s was found %d times in the queue file" % (fid, ob.count(fold)))
    ob = ob.replace(fold, fnew + "  **[TITLE CORRECTED by `#555`, 2026-10-02 -- until then it stated one half of the entry: \"" + fold +
                    "\"; the row's ledger note has the two seats' readings]**")
    body = io.open(os.path.join(REC, "bugs555.body.md"), encoding="utf-8", newline="").read()
    tail = ["", "### The passages an id carries -- %d (lines as of `db8b1a33` -> the id or ids; a note says where the id is not an open ledger row)" % len(by), ""]
    for q in sorted(by, key=first):
        tail.append("* %s -> %s" % (q, ", ".join("`%s`%s" % (i, " (%s)" % n if n else "") for i, n in by[q])))
    tail += ["", "### The entries that stay UNSETTLED -- %d (lines as of `db8b1a33`; what is unsettled, in the ruling seat's words)" % len(uns), ""]
    for u in sorted(uns, key=lambda u: first(u[0])):
        tail.append("* %s -- %s  [%s]" % (u[0], u[2].rstrip(". ") + ".", u[3]))
    nread = sum(1 for u in uns if "READ SINCE" in u[2])
    nmanual = sum(1 for s in sents if "local manuals" in s[3])
    if (nread, nmanual) != (2, 1):
        sys.exit("REFUSED: the texts below say grok's reading of the local manuals settled 1 entry and left 2; the tables say %d and %d" % (nmanual, nread))
    sec = SECTION.format(**fmt) + body + "\n".join(tail) + "\n"
    put("OUTSTANDING_BUGS.md", (ob.rstrip("\n") + "\n" + sec).encode("utf-8"))
    man.append("bugs: %d sentences added in place; the heading of %s corrected; a section of %d lines appended" % (len(rep), fid, sec.count("\n")))

    # 3. ledger and matrix
    lp = put("tools/pipeline/ledger.json", git("show", "HEAD:tools/pipeline/ledger.json", binary=True))
    for verb in ("ledger", "titlefix", "members"):

            sys.exit("REFUSED: records555 %s: %s%s" % (verb, r.stdout, r.stderr))
        man.append(r.stdout.strip())
    mp = os.path.join(ST, "tools", "pipeline", "matrix.html")

                       env=dict(os.environ, GXLEDGER=lp), capture_output=True, text=True)

    # 4. CHANGELOG, the message, the spec
    blk = CHANGELOG.format(**fmt)
    put("CHANGELOG.md", (blk.rstrip("\n") + "\n\n" + git("show", "HEAD:CHANGELOG.md")).encode("utf-8"))
    msg = MESSAGE.format(**fmt)
    io.open(os.path.join(REC, "msg555.txt"), "w", encoding="utf-8", newline="\n").write(msg)
    for t in (SECTION.format(**fmt), blk, msg):
        if ("<OWED" in t and not dry) or re.search(r"\{[a-z_0-9]+\}", t):
            sys.exit("REFUSED: a placeholder survives in a generated text")
    spec = {"unit": "#555", "head_subject_re": "^#554 ", "stage_dir": ST,
            "files": ["CHANGELOG.md", "OUTSTANDING_BUGS.md", "tools/pipeline/ledger.json", "tools/pipeline/matrix.html"],
            "neutral": ["tools/pipeline/check_probe_wiring.py"], "add_dirs": [], "add_files": files, "exclude_dirs": [],
            "message": os.path.join(REC, "msg555.txt"), "sidecars": [], "pre": [],
            "post": [["python", os.path.join(REC, "post555.py")]],
            "no_merge": ["tools/pipeline/ledger.json", "tools/pipeline/matrix.html"]}
    io.open(os.path.join(REC, "spec555.json"), "w", encoding="utf-8", newline="\n").write(json.dumps(spec, indent=1) + "\n")

    for root, _, fs in os.walk(ST):
        for f in sorted(fs):
            p = os.path.join(root, f)
            rel = os.path.relpath(p, ST).replace(os.sep, "/")
            if rel == "MANIFEST.txt":
                continue
            new = open(p, "rb").read()
            r0 = subprocess.run(["git", "-C", SEC, "show", "HEAD:" + rel], capture_output=True)
            old = r0.stdout if r0.returncode == 0 else b""
            d2 = list(difflib.unified_diff(old.decode("utf-8", "replace").splitlines(), new.decode("utf-8", "replace").splitlines(), lineterm="", n=0))
            man.append("%s  %-34s +%d -%d  (HEAD %s)" % (md5(new), rel, sum(1 for l in d2 if l.startswith("+") and not l.startswith("+++")),
                                                        sum(1 for l in d2 if l.startswith("-") and not l.startswith("---")), md5(old)))
    man.insert(0, "# #555 staged committed versions -- HEAD %s%s" % (head, "  (DRY)" if dry else ""))
    man.append("derived: " + ", ".join("%s=%s" % (k, fmt[k]) for k in sorted(fmt) if k.startswith("n")))
    io.open(os.path.join(ST, "MANIFEST.txt"), "w", encoding="utf-8", newline="\n").write("\n".join(man) + "\n")
    print("\n".join(man))
