#!/usr/bin/env python3
"""records555.py gen | ledger | setcommits <row> <sha>  -- the records of #555 (the id-less entries of OUTSTANDING_BUGS.md).

Reads the adjudicated table (regress/artifacts/idless_20261002/IDLESS_TABLE2.tsv, IDLESS_CARRIED.tsv; made by
mk_itable2.py.txt beside them) and writes:
  gen      bugs555.body.md (the rows of the queue section), filed555.json (the ledger rows), counts555.txt
  ledger   appends filed555.json to $LEDGER_PATH (default: the working ledger); refuses on an id that exists
The form is #554's: every HIGH and MED row has its own heading and ledger row; the LOW rows are written out by id
under ONE class row, `idlesslow`, and a LOW row gets its own ledger row when a round takes it up."""
import io
import json
import os
import sys

REC = os.path.dirname(os.path.abspath(__file__))
SEC = r"C:\DocumentNoSnc\cc\GXEMUL\GXEMUL-SEC"
D = os.environ.get("IDLESS_DIR") or os.path.join(SEC, "regress", "artifacts", "idless_20261002")
P = os.environ.get("LEDGER_PATH") or os.path.join(SEC, "tools", "pipeline", "ledger.json")
DATE = "2026-10-02"
EV = "regress/artifacts/idless_20261002/IDLESS_DISPOSITIONS.md"
CLASSID = "idlesslow"
ORDER = ["HIGH", "MED", "LOW"]


def tsv(name):
    return [l.split("\t") for l in io.open(os.path.join(D, name), encoding="utf-8").read().split("\n")[1:] if l.strip()]


def table():
    rows = [dict(id=r[0], sev=r[1], lines=r[2], title=r[3], src=r[4]) for r in tsv("IDLESS_TABLE2.tsv")]
    rows.sort(key=lambda r: ORDER.index(r["sev"]))
    return rows


def classrows():
    """IDLESS_CLASSROW.tsv: id -> (members, the ruling seat's evidence, where it is ruled).  A class row is NOT an id
    for a prose entry: it is a records defect of the LEDGER that the review found while checking which rows carry this
    file's prose (GPT-6 Astra's ifile_04 part D; grok's ruling D)."""
    p = os.path.join(D, "IDLESS_CLASSROW.tsv")
    return dict((r[0], (r[1].split(), r[2], r[3])) for r in tsv("IDLESS_CLASSROW.tsv")) if os.path.isfile(p) else {}


def sweep_unread(mem, evidence):
    """rows the 2026-08-27 sweep closed with no commit that the ruling seat did NOT read (derived from the ledger at
    HEAD and from the seat's own "Checked and left out" clause; they belong to the class row's audit)"""
    import re
    import subprocess
    L = json.loads(subprocess.run(["git", "-C", SEC, "show", "HEAD:tools/pipeline/ledger.json"], capture_output=True).stdout.decode("utf-8"))
    m = re.search(r"Checked and left out: (.*?) One carrier", evidence)
    if not m:
        sys.exit("REFUSED: the ruling's evidence has no 'Checked and left out' clause")
    checked = set(mem) | set(re.findall(r"`([a-z][a-zA-Z0-9_]{3,})`", m.group(1)))
    return [x["id"] for x in L["rows"] if "EXECUTING A RULING THAT ALREADY EXISTED" in (x.get("state_note") or "")
            and x.get("state") == "closed" and not (x.get("commits") or "").strip() and x["id"] not in checked]


MEMBER_NOTE = ("[%s] A MEMBER OF `%s` (#555).  This row was closed on 2026-08-27 by a sweep that executed its `CLOSE (assess)` note as a "
               "row closure; that note signs off the ASSESS STAGE.  The grok seat found the work this row names NOT done and carried by "
               "no other row (regress/artifacts/idless_20261002/%s, item 5).  The row is NOT re-opened here: whether to re-open "
               "it, re-file it or close it with a commit is `%s`'s work.  ")

    cls = classrows()
    out, filed = [], []
    for sev in ("HIGH", "MED"):
        sr = [r for r in rows if r["sev"] == sev]
        out += ["", "### %s -- %d rows" % (sev, len(sr)), ""]
        for r in sr:
            if r["id"] in cls:
                mem, evidence, where = cls[r["id"]]
                unread = sweep_unread(mem, evidence)
                out.append("#### `%s` (%s, a CLASS row: a records defect of the LEDGER, found while checking which rows carry this file's prose) -- %s"
                           % (r["id"], r["sev"], r["title"]))
                out += ["**What it is.**  On 2026-08-27 one sweep closed ledger rows that carried the adjudicator's note `CLOSE (assess)` -- \"on "
                        "the adjudicator's recorded authority\", with \"no independent re-verification of the underlying defect\" (each row's "
                        "own `state_note`).  That note signs off a row's ASSESS STAGE: seven rows carry it while held or in work.  For the "
                        "%d members below the grok seat found the work the row names NOT done and carried by no other row.  **Nothing is "
                        "re-opened by this row or by `#555`**: whether each member is re-opened, re-filed or closed with a commit is this "
                        "row's work, one decision per member.  Each member's ledger row carries a note saying so." % len(mem),
                        "",
                        "**Members (%d):** %s." % (len(mem), ", ".join("`%s`" % m for m in mem)),
                        "",
                        "**Read by no seat (%d):** %s -- closed by the same sweep with no commit; they belong to this row's audit."
                        % (len(unread), ", ".join("`%s`" % m for m in unread)),
                        "",
                        "**The ruling seat's evidence, in its words** (`regress/artifacts/idless_20261002/%s`; the `ledger.json` line "
                        "numbers are those of commit `1202d342`): %s" % (where.split(" ")[0], evidence),
                        "",
                        "**The passages of this file that point here:** %s (line numbers as of commit `db8b1a33`).  [%s]" % (r["lines"], r["src"]),
                        ""]
                filed.append({"id": r["id"], "title": r["title"], "sev": r["sev"], "state": "open", "opened": DATE, "commits": "",
                              "witness": "", "hold_question": "",
                              "state_note": "A CLASS ROW (a records defect of the ledger).  MEMBERS (%d): %s.  READ BY NO SEAT (%d), closed by "
                                            "the same sweep with no commit: %s.  Nothing is re-opened by this row: the audit of each member "
                                            "-- re-open, re-file, or close with a commit -- is its work.  The ruling seat's evidence (%s): %s"
                                            % (len(mem), ", ".join(mem), len(unread), ", ".join(unread), where, evidence),
                              "entries": [{"phase": "assess", "seat": "fable5", "date": DATE, "grade": "", "grader": "",
                                           "ev": "regress/artifacts/idless_20261002/%s" % where.split(" ")[0],
                                           "note": "[NOT A SEAT ANSWER] Filed by GXEMUL [087a90] (main loop) by #555.  GPT-6 Astra's closing "
                                                   "check of the id-less table (ifile_04, part D) found open work in four queue passages that "
                                                   "were pointed only at closed rows; the main loop read the closures and proposed the class; the "
                                                   "grok seat ruled it first-hand and cut the members to the ones it found undone.  NOT FIXED "
                                                   "by #555."}]})
                continue
            out.append("#### `%s` (%s, an id for an entry this file already carries in prose) -- %s" % (r["id"], r["sev"], r["title"]))
            out.append("the prose: %s (line numbers of this file as of commit `db8b1a33`) -- `%s`  [%s]" % (r["lines"], EV, r["src"]))
            out.append("")
            filed.append({"id": r["id"], "title": r["title"], "sev": r["sev"], "state": "open", "opened": DATE, "commits": "",
                          "witness": "", "hold_question": "",
                          "state_note": "AN ID FOR AN EXISTING PROSE ENTRY of OUTSTANDING_BUGS.md (lines %s at db8b1a33) -- no second entry "
                                        "is opened.  Where the text comes from: %s" % (r["lines"], r["src"]),
                          "entries": [{"phase": "assess", "seat": "fable5", "date": DATE, "grade": "", "grader": "", "ev": EV,
                                       "note": "[NOT A SEAT ANSWER] Filed by GXEMUL [087a90] (main loop) by #555 from the id-less review "
                                               "of OUTSTANDING_BUGS.md: GPT-6 Astra's scan, the grok seat's first-hand ruling, Astra's "
                                               "confirmation, Astra's check of the table and grok's ruling of that check (%s and the files "
                                               "beside it).  NOT FIXED by #555." % EV}]})
    low = [r for r in rows if r["sev"] == "LOW"]
    out += ["", "### LOW -- %d rows, under one class row" % len(low), "",
            "#### `%s` (LOW, a CLASS row) -- the %d LOW rows of the id-less review, one line each; a row is promoted to its own "

    out.append("")
    filed.append({"id": CLASSID, "sev": "LOW", "state": "open", "opened": DATE, "commits": "", "witness": "", "hold_question": "",
                  "title": "CLASS ROW: the %d LOW rows the id-less review of OUTSTANDING_BUGS.md filed -- prose entries of the queue that "
                           "now have ids; each is listed with its id in OUTSTANDING_BUGS.md (2026-10-02, #555) and promoted to its own "
                           "row when a round takes it up" % len(low),
                  "state_note": "MEMBERS (%d), id -- title: " % len(low) + " || ".join("%s -- %s" % (r["id"], r["title"]) for r in low),

    summ = ["rows %d" % len(rows)] + ["%s %d" % (s, sum(1 for r in rows if r["sev"] == s)) for s in ORDER] + ["ledger rows %d" % len(filed)]
    io.open(os.path.join(REC, "counts555.txt"), "w", encoding="utf-8", newline="\n").write("\n".join(summ) + "\n")
    print("; ".join(summ))


def _load():
    raw = open(P, "rb").read()
    L = json.loads(raw.decode("utf-8"))
    dump = lambda x: json.dumps(x, indent=1, ensure_ascii=False).replace("\n", "\r\n").encode("utf-8")
    if dump(L) != raw:
        sys.exit("REFUSED: the ledger does not round-trip byte-identical; its format changed")
    return L, dump


def ledger():
    L, dump = _load()
    filed = json.load(io.open(os.path.join(REC, "filed555.json"), encoding="utf-8"))
    ids = {x["id"] for x in L["rows"]}
    clash = [r["id"] for r in filed if r["id"] in ids]
    if clash:
        sys.exit("REFUSED: %d ids already exist: %s" % (len(clash), " ".join(clash[:8])))
    L["rows"].extend(filed)
    open(P, "wb").write(dump(L))
    print("ledger: +%d rows filed by #555 (%d rows now)" % (len(filed), len(L["rows"])))


def titlefix():
    """IDLESS_TITLEFIX.tsv: a row #554 filed whose TITLE stated half of the entry it names.  The title is corrected;
    what it read is kept in the row's note."""
    L, dump = _load()
    done = []
    for rid, old, new, why in tsv("IDLESS_TITLEFIX.tsv"):
        r = [x for x in L["rows"] if x["id"] == rid]
        if len(r) != 1 or r[0]["title"].count(old) != 1:
            sys.exit("REFUSED: the ledger row %s does not carry the title #554 filed" % rid)
        r[0]["title"] = r[0]["title"].replace(old, new)
        r[0]["state_note"] = ("[%s] TITLE CORRECTED by #555; it read: \"%s\".  %s (regress/artifacts/idless_20261002/"
                              "IDLESS_TITLEFIX.tsv).  " % (DATE, old, why)) + (r[0].get("state_note") or "")
        done.append(rid)
    open(P, "wb").write(dump(L))
    print("ledger: %d title corrected by #555 (%s)" % (len(done), " ".join(done)))


def members():
    """each member row of a class row gets a note saying it is one (its state is NOT changed)"""
    L, dump = _load()
    done = []
    for cid, (mem, _evidence, where) in sorted(classrows().items()):
        for rid in mem:
            r = [x for x in L["rows"] if x["id"] == rid]
            if len(r) != 1 or r[0].get("state") != "closed":
                sys.exit("REFUSED: the member %s is not one closed ledger row" % rid)
            note = MEMBER_NOTE % (DATE, cid, where.split(" ")[0], cid)
            if note in (r[0].get("state_note") or ""):
                sys.exit("REFUSED: the member %s already carries the note" % rid)
            r[0]["state_note"] = note + (r[0].get("state_note") or "")
            done.append(rid)
    open(P, "wb").write(dump(L))
    print("ledger: %d member rows noted by #555 (%s)" % (len(done), " ".join(done)))


def setcommits():
    L, dump = _load()
    a = sys.argv[2:]
    for rid, sha in zip(a[0::2], a[1::2]):
        r = [x for x in L["rows"] if x["id"] == rid]
        if len(r) != 1:
            sys.exit("REFUSED: row %s found %d times" % (rid, len(r)))
        r[0]["commits"] = sha
    open(P, "wb").write(dump(L))
    print("ledger: commits recorded -- " + ", ".join("%s %s" % p for p in zip(a[0::2], a[1::2])))


if __name__ == "__main__":
    {"gen": gen, "ledger": ledger, "titlefix": titlefix, "members": members, "setcommits": setcommits}[sys.argv[1]]()
