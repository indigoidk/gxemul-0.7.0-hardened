#!/usr/bin/env python3
"""mk_itable2.py [--show]  -- the id-less unit's table AFTER adjudication.

IDLESS_TABLE.tsv (mk_itable.py) was checked by GPT-6 Astra (ifile_01, ifile_02: 68 findings), every finding was ruled
first-hand by the grok seat (grok/grok_irule_a.txt, grok_irule_b.txt; grok_irule_c.txt for eight flags of the agy
seat's audits), and what grok HELD went back to Astra (ifile_03).  Each grok line ends in a RESULT field that says, in
grok's words, what the table should hold.  This script applies those RESULT fields and nothing else:
    `retire `x``, "`x` is retired"                    the row x is removed; the row named beside it keeps the work
    `NEW <grade> `id`, queue <q>. Title: <t>`           the row is set (added, renamed, re-titled, re-scoped)
    `NEW <grade> `id` "<t>"`                            a row for a remainder; its queue passage is the item's own
    `FILED ... `id``, `CROSS-REFERENCE `id``            the passage is carried by an existing id: listed, no new row
    a closing sentence (S items), `SETTLED: ...`        the sentence that stands at the passage
A title is cut out of grok's line; it is never retyped.  Retired ids are replaced by their keepers wherever they are
named.  Every id a passage is pointed at must exist -- as a row of this table, in the ledger at HEAD, or in the queue
file -- or the script says so.  Writes IDLESS_TABLE2.tsv (the NEW rows), IDLESS_SENTENCES.tsv (the sentences to add in
place), IDLESS_CARRIED.tsv (passages carried by an id that already exists) beside the dispositions."""
import io
import json
import os
import re
import subprocess
import sys

SEC = "C:/DocumentNoSnc/CC/GXEMUL/GXEMUL-SEC"
D = os.environ.get("IDLESS_DIR") or os.path.join(SEC, "regress", "artifacts", "idless_20261002")
ID = r"`([a-z][a-zA-Z0-9_]{3,})`"


def rd(p):
    return io.open(p, encoding="utf-8", errors="replace").read().replace("\r\n", "\n")


def tsv(name):
    return [l.split("\t") for l in rd(os.path.join(D, name)).split("\n")[1:] if l.strip()]


def ruling(name, keys):
    """item -> (key text, word, why, RESULT, cites) for one grok ruling file"""
    out = {}
    for l in rd(os.path.join(D, "grok", name)).split("\n"):
        m = re.match(r"^(" + keys + r"\d+) \| (.*?) \| (CONCEDE IN PART|CONCEDE|HOLD)\b(.*?) \| RESULT: (.*?) \| ([^|]*)$", l)
        if m:
            out[m.group(1)] = (m.group(2).strip(), m.group(3), m.group(4).strip(": "), m.group(5).strip(), m.group(6).strip())
    return out


table = tsv("IDLESS_TABLE.tsv")
new = {}        # id -> [grade, slice, queue text, title, source]
order = []
for r in table:
    if r[0] == "NEW":
        new[r[1]] = [r[2], r[3], r[4], r[5], r[6]]
        order.append(r[1])
sent = {}       # queue lines -> [slice, sentence, source]
for r in table:
    if r[0] == "CLOSE-RECORD":
        sent[r[4]] = [r[3], r[5], r[6]]
unsettled = {r[4]: [r[3], r[5], r[6]] for r in table if r[0] == "UNSETTLED"}
retired, carried, log, held = {}, [], [], []


def setrow(rid, grade, queue, title, src, item):
    title = title.strip().rstrip(".").strip()
    if rid in new:
        old = new[rid]
        if old[0] != grade:
            log.append("%s: %s grade %s -> %s" % (item, rid, old[0], grade))
        new[rid] = [grade, old[1], queue or old[2], title or old[3],
                    src if title else (old[4] + "; its queue lines from " + src) if queue and queue != old[2] else old[4]]
    else:
        new[rid] = [grade, "", queue, title, src]
        order.append(rid)
        log.append("%s: ADDED %s %s" % (item, grade, rid))


# backticked words in the seats' RESULT fields and sentences that are NOT ids (each with why); any other unknown word
# is reported, so a mistyped id cannot hide behind this list
NOT_IDS = {"wait_from": "K14: a function of the probe harness", "baremips": "a machine name, in the sentence at 1875-1885",
           "calibrated": "a budget class, in the sentence at 5411-5414", "zeroed": "a budget class, in the same sentence"}

A = ruling("grok_irule_a.txt", "[A-C]")
B = ruling("grok_irule_b.txt", "[SFKU]")
if len(A) != 25 or len(B) != 43:
    sys.exit("REFUSED: parsed %d of 25 lines of grok_irule_a.txt and %d of 43 of grok_irule_b.txt" % (len(A), len(B)))

# ---- list A/B/C of ifile_01
for item in sorted(A, key=lambda k: (k[0], int(k[1:]))):
    key, word, why, res, cites = A[item]
    src = "grok/grok_irule_a.txt %s (%s Astra's ifile_01 finding)" % (item, {"CONCEDE": "conceding", "CONCEDE IN PART": "conceding in part", "HOLD": "holding against"}[word])
    for x in re.findall(r"retire " + ID, res) + re.findall(ID + r" is retired", res):
        keep = [i for i in re.findall(r"(?:NEW (?:HIGH|MED|LOW)|FILED(?: (?:HIGH|MED|LOW))?(?: under)?) " + ID, res) if i != x]
        retired[x] = keep[0] if keep else retired.get(x, "")
    f = re.search(r"FILED (?:(HIGH|MED|LOW) )?(?:under )?" + ID + r"([^.]*)", res)
    n = re.search(r"NEW (HIGH|MED|LOW) " + ID + r", queue (.*?)(?:\. Title: (.*?))?(?:\. `|\.$|$)", res)
    if n:
        t = n.group(4) or ""
        t = re.split(r"\. (?=`[a-z])", t)[0]
        ids = key.split(" | ")
        if ids[0] in new and ids[0] != n.group(2) and ids[0] not in retired and item.startswith("A"):
            retired[ids[0]] = n.group(2)                    # a rename: the old id goes, the row is the new one
            new[n.group(2)] = list(new[ids[0]])
            del new[ids[0]]
            order[order.index(ids[0])] = n.group(2)
            log.append("%s: RENAMED %s -> %s" % (item, ids[0], n.group(2)))
        # "queue 4839-4848 and 4953-4954, with the A4 title": the words after the lines are not lines (Astra's
        # ifile_04, C8: they had reached the queue-range field)
        setrow(n.group(2), n.group(1), re.split(r",? with the ", n.group(3))[0].strip(), t, src, item)
    if f and not (n and res.index("FILED") > res.index("NEW ")):
        # The passage a FILED id carries is the one its OWN clause names ("for the 5262-5266 transport sweep only");

        # ifile_04, C11 and C12: the range of the NEW row beside it had been given to the FILED id.)
        tail = f.group(3).strip(" ,")
        rng = re.search(r"(?<![\w.:`-])(\d{3,5}(?:-\d{3,5})?)(?![\w.:`-])", tail)
        qall = re.search(r"queue ([0-9][^.]*?)(?:,|\.| with)", res)
        passage = rng.group(1) if rng else re.match(r"\d+(?:-\d+)?(?: and \d+(?:-\d+)?)*", qall.group(1)).group(0) if qall else key
        what = re.sub(r"^in whole,?\s*", "", tail)
        what = re.sub(r"^queue [0-9]+(?:-[0-9]+)?,?\s*", "", what)
        what = re.sub(r"^with the A\d+ title$", "", what)
        what = re.sub(r"^for ", "", what).strip()
        scope = ("in whole" if ("in whole" in tail or not n) else "in part") + (": " + what if what else "")
        carried.append([passage, f.group(2), scope, src])
for x, k in retired.items():
    if x in new:
        del new[x]
        order.remove(x)
        log.append("retired %s (the work stays on %s)" % (x, k or "?"))

# ---- ifile_02: sentences, FILED entries, KEEP-AS-IS entries, UNSETTLED
for item in sorted(B, key=lambda k: ("SFKU".index(k[0]), int(k[1:]))):
    q, word, why, res, cites = B[item]
    src = "grok/grok_irule_b.txt %s" % item
    if word == "HOLD":
        held.append([item, q, why, res])
    for g, rid, t in re.findall(r"NEW (HIGH|MED|LOW) " + ID + r" \"([^\"]+)\"", res):

    elif item[0] == "U":

    elif item[0] in "FK" and res.strip(". ").lower() not in ("nothing owed", "yes"):
