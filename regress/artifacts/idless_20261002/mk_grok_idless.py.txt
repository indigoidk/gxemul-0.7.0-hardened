#!/usr/bin/env python3
"""mk_grok_idless.py  -- grok RULING briefs for the id-less scan of OUTSTANDING_BUGS.md (codexpack_20260928/idless_NN:
279 claimed id-less open entries; second-seat verified by agy, agypack_20261001/idlessverify_NN).  Owner 2026-10-02: the
working seats are GPT-6 Astra, agy and grok -- the scan is Astra's family's, so the grok seat RULES each entry
first-hand and Astra then confirms the dispositions; agy's agreement carries no weight, its flags do.
Writes regress/artifacts/idless_20261002/briefs/grok_ruleidless_<tag>.md (ten: slice 02 is split in two) and
<scratchpad>/astraretro/idless_queue.tsv (tag<TAB>brief).  Refuses to overwrite a brief."""
import io
import os
import re
import sys

SEC = r"C:\DocumentNoSnc\cc\GXEMUL\GXEMUL-SEC"
ARTS = os.path.join(SEC, "regress", "artifacts")
CP = os.path.join(ARTS, "codexpack_20260928")
AP = os.path.join(ARTS, "agypack_20261001")
OUT = os.path.join(ARTS, "idless_20261002")
HERE = os.path.dirname(os.path.abspath(__file__))
REV = "db8b1a33"
# four row shapes across the nine answers: `821-828 | ...`, `| 821-828 | ...`, `- 2614-2615 | ...`,
# `OUTSTANDING_BUGS.md:20873-20878 | ...`, each optionally `; repeated at ...` before the first bar
ROW = re.compile(r"^[|*\s-]*(?:OUTSTANDING_BUGS\.md:)?(\d{1,5})(?:\s*[\u2013\u2014-]\s*\d{1,5})?(?:;[^|]*)? \|")

COMMON = """
**The tree.**  `C:\\DocumentNoSnc\\cc\\GXEMUL\\GXEMUL-SEC` (git): a hardened fork of the GXemul 0.7.0 full-system machine
emulator, C99.  Read COMMITTED files at `{rev}` only (`git -C C:\\DocumentNoSnc\\cc\\GXEMUL\\GXEMUL-SEC show {rev}:<path>`,
numbered with `awk '{{print NR": "$0}}'` or `grep -n`): the line numbers below are that revision's, HEAD may have moved
past it, and the working tree carries other people's uncommitted work.  Use `git show`, never `git status` or
`git diff` (another process may be committing).  Never read `C:\\DocumentNoSnc\\cc\\GXEMUL\\src\\`, `est/`, `build/`,
`build-asan/` or `_archive/` (other copies; a copy reports line numbers off by hundreds).  Read nothing under
`regress/artifacts/` except the files this brief names.  Hardware manuals: `C:\\DocumentNoSnc\\cc\\GXEMUL\\_scratchpad\\*.txt`.
**Write nothing; run nothing that changes a file; build nothing.**  Return text only.  Cite `path:line` at `{rev}` for
every claim; say "not checked" rather than guess.
"""

BODY = """ID-LESS QUEUE ENTRIES -- RULING by the grok seat, slice {tag}: `OUTSTANDING_BUGS.md` lines {lo}-{hi} at `{rev}`{part}

FIRST line of your answer, exactly in this form:
`VERDICT: <n> NEW, <f> FILED, <c> CLOSE-RECORD, <k> KEEP-AS-IS, <u> UNSETTLED, of {nrows} claimed entries (+<m> missed entries ruled)`
SECOND line: the token `GROK_IDLESS_{TAG}_ALIVE_1002`.  THIRD line: quote verbatim line {lo} of the file as
`git -C C:\\DocumentNoSnc\\cc\\GXEMUL\\GXEMUL-SEC show {rev}:OUTSTANDING_BUGS.md` prints it (an empty line is quoted as `(empty)`).
{common}
**Context.**  The queue file `OUTSTANDING_BUGS.md` names most rows by a short backticked id (`wdcnien`, `sh4sqexpevt`,
...).  Some entries record a defect, an owed measurement or an owed correction in PROSE only, with no id -- and prose
without an id gets filed twice or lost.  A reviewing model (GPT-6 Astra) scanned lines {lo}-{hi} and claims
**{claimed} id-less OPEN entries**, each with a line range, the entry's own words, a STILL-OPEN-AT-HEAD judgement, a
proposed id and a severity: `{ans}`.
The agy seat (gemini, a different model family) then verified every row: `{agy}`
-- its verdict: "{agyv}".
The project owner ruled on 2026-10-02 that the working seats are GPT-6 Astra, agy and grok.  The scan is Astra's
family's own, so **YOU RULE each entry, first-hand, from the committed files**; GPT-6 Astra then confirms or overturns
your dispositions, and only then is any id assigned or any entry closed.  Nothing is fixed here; this decides what the
queue will say.

**Please do not simply ratify.**  Both texts are in front of you and both can be wrong the same way.  agy's agreement
with the scan is NOT evidence -- weigh only its FLAGS: rule FIRST the rows it marks WRONG or UNSETTLED, the rows it
says are not id-less after all (`ID-LESS: no`), and the rows whose severity it moved, and say which reading the source
supports.  The premises I am least sure of, stated so that you check them: that each quoted entry is really in the
cited lines; that "still open" was re-derived from the SOURCE and not from the queue's own words; and that a proposed
id is new.

**For EVERY row of the scan's table{partrows}, one line:**
`<line range> | ID-LESS: yes / no "<the id it carries>" | OPEN: YES / NO: <what the source shows> / UNSETTLED: <what would settle it> | DISPOSITION: NEW <HIGH/MED/LOW> <proposed-id> "<one-line title>" / FILED <existing id> / CLOSE-RECORD "<the sentence that should replace the stale one>" / KEEP-AS-IS "<why no id is owed>" / UNSETTLED | <the lines you read, path:line at {rev}>`
- **NEW** -- open, id-less, and deserving its own queue row.  The proposed id is short, lowercase, without spaces, and
  must not already exist anywhere in `OUTSTANDING_BUGS.md` at `{rev}`, in
  `{disp}` or in
  `{hdisp}` (grep all three).
- **FILED** -- the same defect already has an id elsewhere in the file (name it and its line), or the entry sits under
  an id'd heading (then `ID-LESS: no`).
- **CLOSE-RECORD** -- the prose says OPEN while the source at `{rev}` shows the defect gone: cite the lines that fix it
  and give the replacement sentence.  A closure is the costly direction: RIGHT only if you saw the fix in the source.
- **KEEP-AS-IS** -- history, a rule or a note; not a defect and not owed work.
Severity, in plain engineering terms: **HIGH** -- guest-controlled input ends the emulator process, makes one of its
memory accesses leave a buffer's bounds, or stops it making progress; **MED** -- guest-visible behaviour differs from
the hardware documentation; **LOW** -- diagnostics, latent code, records, test coverage.  Grade from the source, not
from the queue's, the scan's or agy's grade; an id graded HIGH gets a round, so be hardest there.

Then: **(a)** the entries agy lists as MISSED by the scan in this range (its section (a)){misspart} -- rule each in the
same line format, and add any you find yourself; **(b)** rows that are ONE defect recorded twice (agy's section (b)):
say which row carries it; **(c)** `DENOMINATOR: <s> rows ruled from the source, <t> from the two seats' texts alone, of {nrows}`.
"""


def main():
    os.makedirs(os.path.join(OUT, "briefs"), exist_ok=True)
    common = COMMON.format(rev=REV)
    queue = []
    total = 0
    for n in range(1, 10):
        nn = "%02d" % n
        ans = os.path.join(CP, "idless_" + nn, "codex_idless_%s.txt" % nn)
        agy = os.path.join(AP, "idlessverify_" + nn, "agy_idlessverify_%s.txt" % nn)
        first = io.open(os.path.join(CP, "briefs", "idless_%s.md" % nn), encoding="utf-8").readline()
        m = re.search(r"lines (\d+)-(\d+)", first)
        lo, hi = int(m.group(1)), int(m.group(2))
        t = io.open(ans, encoding="utf-8", errors="replace").read()
        v = re.search(r"VERDICT: (\d+)", t)
        ta = io.open(agy, encoding="utf-8", errors="replace").read()
        va = re.search(r"VERDICT: ([^\n]*)", ta)
        if not v or not va:
            sys.exit("REFUSED: slice %s lacks a VERDICT line in the scan or the verification" % nn)
        claimed = int(v.group(1))
        starts = sorted(int(ROW.match(l).group(1)) for l in t.splitlines() if ROW.match(l))
        if len(starts) != claimed:
            print("NOTE slice %s: the scan's VERDICT says %d entries, its table has %d parsed rows" % (nn, claimed, len(starts)))
        total += claimed
        parts = [("", lo, hi, "", "", claimed, "")]
        if claimed > 70 and len(starts) >= 2:
            mid = starts[len(starts) // 2 - 1]
            na = sum(1 for s in starts if s <= mid)
            parts = [("a", lo, hi, " -- PART A", " whose line range STARTS at or below %d" % mid, na,
                      " that start at or below %d" % mid),
                     ("b", lo, hi, " -- PART B", " whose line range STARTS above %d" % mid, len(starts) - na,
                      " that start above %d" % mid)]
        for suf, plo, phi, part, partrows, nrows, misspart in parts:
            tag = nn + suf
            text = BODY.format(tag=tag, TAG=tag.upper(), lo=plo, hi=phi, rev=REV, part=part, partrows=partrows, nrows=nrows,
                               claimed=claimed, ans=ans, agy=agy, agyv=va.group(1).strip().rstrip("."), common=common,
                               misspart=misspart, disp=os.path.join(ARTS, "astraretro_20260926", "DISPOSITIONS.md"),
                               hdisp=os.path.join(ARTS, "astraretro_harness_20260926", "HARNESS_DISPOSITIONS.md"))
            p = os.path.join(OUT, "briefs", "grok_ruleidless_%s.md" % tag)
            if os.path.exists(p):
                sys.exit("REFUSED: %s exists" % p)
            io.open(p, "w", encoding="utf-8", newline="\n").write(text)
            queue.append("%s\t%s" % (tag, p))
            print("%s: %d rows (%s lines %d-%d)%s" % (tag, nrows, "slice %s" % nn, plo, phi, partrows))
    io.open(os.path.join(HERE, "idless_queue.tsv"), "w", encoding="utf-8", newline="\n").write("\n".join(queue) + "\n")
    print("%d briefs; %d claimed entries in all; queue %s" % (len(queue), total, os.path.join(HERE, "idless_queue.tsv")))


if __name__ == "__main__":
    main()
