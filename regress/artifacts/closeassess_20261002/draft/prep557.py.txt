def git(*a, binary=False):
    r = subprocess.run(["git", "-C", SEC] + list(a), capture_output=True)
    if r.returncode:
        sys.exit("REFUSED: git %s: %s" % (" ".join(a), r.stderr.decode("utf-8", "replace")))
    return r.stdout if binary else r.stdout.decode("utf-8")


def md5(b):
    return hashlib.md5(b).hexdigest()[:8]


def rd(p):
    return io.open(p, encoding="utf-8", errors="replace").read().replace("\r\n", "\n")


def put(rel, data):
    p = os.path.join(ST, *rel.split("/"))
    os.makedirs(os.path.dirname(p), exist_ok=True)
    open(p, "wb").write(data)
    return p


def verdict(p):
    v = re.findall(r"VERDICT: ([^\n]*)", rd(p))
    if not v:
        sys.exit("REFUSED: no VERDICT line in %s" % p)
    return v[0].replace("`", "").strip().rstrip(".")


def census():
    """the census counts, DERIVED from the two artifacts every time the note is generated (prep reads them in the
    artifacts dir, post reads the committed copies at the same path), never transcribed; each identity is checked"""
    g = rd(os.path.join(D, "grok", "grok_closedcensus.txt"))
    by = dict((m.group(1), m.group(2)) for m in re.finditer(r"^\d+ \| ([A-Za-z0-9_]+) \| (SHIPPED|NOTHING|WORK)\b", g, re.M))
    ship = sorted(r for r, v in by.items() if v == "SHIPPED")
    noth = sorted(r for r, v in by.items() if v == "NOTHING")
    alive = sorted(r for r, v in by.items() if v == "WORK")
    a = re.search(r"VERDICT: (\d+) confirmed, (\d+) wrong, 0 not checked, of (\d+) rows", rd(os.path.join(D, "aconf_census", "codex_aconf_census.txt")))
    if not a or int(a.group(3)) != len(by):
        sys.exit("REFUSED: Astra's census verdict line is missing or counts a different population than grok's %d rows" % len(by))
    alive5 = sorted(set(MEMBERS) | set(LANDED))
    overlap = sorted(set(ship) & set(alive5))
    fill_census = [r for r in FILL if r in ship]
    cen = dict(rows=len(by), ship=len(ship), noth=len(noth), alive=len(alive), conf=int(a.group(1)), wrong=int(a.group(2)),
               alive5=alive5, overlap=overlap, fill_census=len(fill_census))
    if cen["ship"] + cen["noth"] + cen["alive"] != cen["rows"] or cen["conf"] + cen["wrong"] != cen["rows"]:
        sys.exit("REFUSED: the census verdicts do not partition the %d rows: %r" % (cen["rows"], cen))
    if set(ship) != set(fill_census) | set(NOFILL) | set(LANDED):
        sys.exit("REFUSED: grok's shipped set is not FILL-from-census + NOFILL + LANDED: %r" % (sorted(set(ship) ^ (set(fill_census) | set(NOFILL) | set(LANDED)))))
    if set(FILL) - set(ship) != {"xfamscope"} or len(alive5) != 5 or overlap != ["boilerplateprefix", "smhang"]:
        sys.exit("REFUSED: the fills outside the census, the alive five or the overlap are not what the texts say: %r" % cen)
    if cen["ship"] + cen["noth"] + len(alive5) - len(overlap) != cen["rows"]:
        sys.exit("REFUSED: shipped + nothing owed + alive - overlap != rows: %r" % cen)
    return cen

def main():
    dry = "--dry" in sys.argv
    args = [a for a in sys.argv[1:] if a != "--dry"]
    if len(args) != 1:
        sys.exit(__doc__)
    closing = args[0]
    head = git("rev-parse", "--short=8", "HEAD").strip()
    if not git("log", "-1", "--format=%s", "HEAD").startswith("#556 "):
        sys.exit("REFUSED: HEAD %s is not #556's commit" % head)
    if os.path.isdir(ST):
        for root, dirs, files in os.walk(ST, topdown=False):
            for f in files:
                os.remove(os.path.join(root, f))
            for d in dirs:
                os.rmdir(os.path.join(root, d))
    os.makedirs(ST, exist_ok=True)
    man = ["# #557 staged committed versions -- HEAD %s%s" % (head, "  (DRY)" if dry else "")]

    # 1. every sha re-checked: it exists, and it touched the file Astra named or names the row/round in its subject
    for rid, (shas, how) in FILL.items():
        for sha in shas.split():           # a row may name two commits
            subj = git("log", "-1", "--format=%s", sha).strip()
            files = git("show", "--name-only", "--format=", sha).split("\n")
            paths = re.findall(r"(?:src|regress|tools)/[A-Za-z0-9_./-]+", how)
            okf = any(p in files for p in paths)
            okn = (rid in subj) or any(("#%s" % n) in subj for n in re.findall(r"#(\d+)", how))
            if not (okf or okn):
                sys.exit("REFUSED: %s = %s: the commit neither touches %s nor names the row in its subject (%s)" % (rid, sha, paths, subj[:60]))
            man.append("sha ok  %-18s %s  %s" % (rid, sha, subj[:70]))

    # 2. the checker: the working copy, which must compile, pass on the working ledger, and differ from HEAD only as
    #    parked_check_boilerplate_v2.diff shows
    ck = os.path.join(SEC, "tools", "pipeline", "check_boilerplate.py")
    src = open(ck, "rb").read()
    if b"EXCERPT = 60" not in src or b"[:PREFIX]" in src:
        sys.exit("REFUSED: the working check_boilerplate.py is not the repaired one")
    r = subprocess.run([sys.executable, ck], capture_output=True, text=True)
    if "BOILERPLATE_PASS" not in r.stdout:
        sys.exit("REFUSED: the repaired checker does not pass on the working ledger: %s" % r.stdout[-300:])
    put("tools/pipeline/check_boilerplate.py", src)
