def block(s, first, last, edits, width=118, what=""):
    """Re-wrap the whole lines from the one holding `first' to the one holding `last', after the (old, new) edits
    applied to their whitespace-joined text.  Each marker and each old text must be unique."""
    lines = s.split("\n")
    a = [i for i, l in enumerate(lines) if first in l]
    b = [i for i, l in enumerate(lines) if last in l]
    if len(a) != 1 or len(b) != 1 or b[0] < a[0]:
        sys.exit("%s: block markers not unique/ordered (%d, %d)" % (what, len(a), len(b)))
    seg = lines[a[0]:b[0] + 1]
    ind = re.match(r"^\s*", seg[0]).group(0)
    text = " ".join(l.strip() for l in seg)
    for old, new in edits:
        n = text.count(old)
        if n != 1:
            sys.exit("%s: edit text not unique (%d): %r" % (what, n, old[:70]))
        text = text.replace(old, new)
    out = textwrap.fill(text, width=width, initial_indent=ind, subsequent_indent=ind,
                        break_on_hyphens=False, break_long_words=False)
    return "\n".join(lines[:a[0]] + out.split("\n") + lines[b[0] + 1:])


def rep(s, old, new, what):
    n = s.count(old)
    if n != 1:
        sys.exit("%s: TEXT NOT UNIQUE (%d): %r" % (what, n, old[:70]))
    return s.replace(old, new)

def ast_nodoc(src):
    t = ast.parse(src)
    if isinstance(t.body[0], ast.Expr) and isinstance(getattr(t.body[0], "value", None), ast.Constant):
