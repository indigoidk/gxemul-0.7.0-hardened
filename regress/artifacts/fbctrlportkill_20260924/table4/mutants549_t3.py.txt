def apply(base, edits, name):
    s = base
    for old, new in edits:
        n = s.count(old)
        if n != 1:
            sys.exit("REFUSED %s: %d matches for %r" % (name, n, old[:60]))
        s = s.replace(old, new)
    return s


def main():
    fixp, headp, mutp, out = sys.argv[1:5]
    os.makedirs(out, exist_ok=True)
    base = io.open(fixp, encoding="utf-8", newline="").read()
    head = io.open(headp, encoding="utf-8", newline="").read()
    assert "\r" not in base and "\r" not in head
    ns = {"__name__": "mutants549"}
    exec(io.open(mutp, encoding="utf-8").read(), ns)
    rows = [(n, c, e) for n, c, e in ns["MUTANTS"]] + TARGETS + PASS2
    names = [r[0] for r in rows]
    assert len(names) == len(set(names)), "duplicate mutant name"
    for name, cls, edits in rows:
        s = head if edits is None else apply(base, edits, name)
        if s == base:
            sys.exit("REFUSED %s: the edit changed nothing" % name)
        io.open(os.path.join(out, name + ".c"), "w", encoding="utf-8", newline="").write(s)
        print("%-15s %s" % (name, cls))
    print("MUTANTS_WRITTEN=%d" % len(rows))


if __name__ == "__main__":
    main()
