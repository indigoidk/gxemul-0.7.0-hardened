#!/usr/bin/env python3
"""#549 fbctrlportkill -- the DETECTOR for dev_fbctrl.c's port selector, on an UNMODIFIED `-E testmips`.

machine_test.c installs `fbctrl addr=0x12f00000` in testmips's own setup (and in default_test() for the other test
machines).  Every access below is a real guest sb/sh/sw/sd/lw/ld placed with `put w` and executed through real address
decode and memory_rw -- stepped one instruction at a time, except session F, which free-runs to a breakpoint (session B
also steps a `dsll` between its stores, to double its value register).  No machine
or device is added to reach the site.  t0 = 0xffffffffb2f00000, kseg1 of DEV_FBCTRL_ADDRESS; PORT is +0x00, DATA +0x10.

THE DEFECT (HEAD before #549): the PORT arm STORED the selector, as an int, with only a warning.  An int out of range --
`sw 9` -- made the next DATA access end the emulator process in the DATA arm's range check; a wide value was truncated
onto a legal port and serviced (0x100000003 selected port 3: kill table 2's M_revert).  #549 REFUSES any selector >= 8,
compared on the full 64-bit value, at the PORT arm (the previous, valid selector is kept) and announces the refusal
once per device, `(once)`.

SESSIONS (each on a fresh emulator)
  C  controls: PORT=5, DATA=0x2a, read both back (LIVENESS); X1=123 and Y1=123, COMMAND=GET_RESOLUTION, read X1 and Y1
     (DEVICE SIGNATURE: the model zeroes both without X11 -- RAM at +0x10 would read back 2, the last value written
     there, and a banked register WITHOUT command execution would read 123); no `(once)` line may appear.
  A  accept: all EIGHT selectors, written at widths 1, 2, 4 and 8, each read back through PORT; ports 1-7 round-trip
     a distinct DATA value (port 0 is COMMAND and is written only with NOP); no `(once)` line may appear.
  P  refusal, anchored to a previous selector: PORT=5 (DATA 0x2a), then eleven out-of-range writes -- 8, 9, 16 (sw),
     0x200000000, 0x100000003, all-ones (sd), all-ones (sw), 0xff (sb), 0x100 (sh), 16 (sb), 16 (sh) -- each PRECEDED
     by a fresh PORT=5 and followed by a PORT read that must still say 5; then PORT=7 and a refused 9 must leave 7 --
     twelve refusals.  The `(once)` count must be exactly 1 right after the FIRST refusal and still 1 after the twelfth,
     and the one announce must name the first refused value, 0x8.
  R  re-arm: PORT=5, refuse 9, read PORT, a DATA write to port 5, a NOP command (PORT=0, DATA=0), a DATA READ of port 5,
     a GET_RESOLUTION command (PORT=0, DATA=2), PORT=5, refuse 9: the count is still 1 and PORT keeps 5.  Its own
     session because DATA traffic inside P can end the process: when the F2 prototype put such traffic in P, after
     its refused 8, an edit that accepts 8 (M_gt) ended the emulator at P's step 4 and blanked 14 of P's rows
     (regress/artifacts/fbctrlportkill_20260924/opus549_measured/f2/proto_d4inP_summary.out).
  K  the witness pair, repaired: PORT=5, DATA=0x2a, PORT=9 refused, then a DATA READ (must survive and read port 5's
     0x2a) and a DATA WRITE of 0x77 (must land in port 5: read back through PORT=5).
  F  free-running: a refused PORT write at power-on and a guest `lw` of PORT execute INSIDE `continue`, which stops at a
     breakpoint after them; the lw must read 0 and the announce must name 0x9.  This invocation passes -V and no -v,
     so while free-running debug() is below its threshold and returns without printing, while fatal() prints: a refusal
     announced with debug() would pass every stepped row and fail here.
  V  cells: COMMAND=GET_RESOLUTION, X1=123, one refusal while COMMAND holds 2 and PORT=1 -- PORT must still read 1,
     X1 must stay 123, and that FIRST refusal must announce, naming 0x9; ports 1-7 filled with 0x11*p (the last fill
     selects 7), five refusals (9, 16, 0x100000003 by sd, 0xff by sb, 8), PORT read at once (it must still say 7),
     then all eight cells read back and the count is still 1.
  W  widths: selector 0 written by sb, sh, sw and sd and selector 7 by sb, sh and sw, each after PORT=5 (accepted);
     8 by sb, sh and sd after PORT=5 (refused -- the FIRST, by sb, must announce); 0x8000000000000000 by sd after
     PORT=5 (refused); 9 after PORT=6 (keeps 6); then `lb`, `lh` and `ld` of PORT with selector 5 (exactly 5 each).
  X2 X3 X4 X6 X7  five fresh devices, each: PORT=k, then its FIRST refusal (9 by sh after 2; 0x100000005 by sd after 3
     -- wide, and 5 as an int, never 3; all-ones by sd after 4 -- the largest value, and negative as a signed one; 9 by
     sb after 6; 0x100 by sh after 7): it announces exactly once, the announce names that value, and PORT keeps k.
  Y  COMMAND = 1: PORT=0 and DATA=1 run SET_RESOLUTION with X1 = Y1 = 0 from power-on, which dev_fb_resize() declines
     with a message ("size too small ... ignored"), so nothing is resized; PORT=5; its FIRST refusal, 9 by sw -- PORT
     keeps 5, and it announces once, naming 0x9; SET_RESOLUTION again, as traffic between refusals; PORT=5 and a second
     refusal, 0x100000003 by sd (3 as an int, never 5) -- PORT keeps 5 and the count is still 1; COMMAND reads back 1.
  Z  silence across selectors: PORT=2 and its FIRST refusal (9), then PORT=0, 1, 2, 3 and 4, each followed by a refusal
     of 9 -- the count is still 1 after all six, and PORT keeps 4.
  B  single bits: PORT=5, then EVERY single-bit value from 2^3 up that the width can carry -- 2^3..2^7 by sb,
     2^3..2^15 by sh, 2^3..2^31 by sw, 2^3..2^63 by sd, 108 refusals -- with PORT set to 5 again before each width
     and read back once after it; a register doubles between stores (dsll).  A refusal that let one of them through
     would store it as an int -- 2^k, INT_MIN or 0, never 5 -- and every later refusal of that width would keep it,
     so one read per width sees it.  The count must be exactly 1 after all 108.  Added by the verifying pass (the
     Opus 5.5 seat): the compare narrowed to ignore bits 5-7, 9-31 or 34-62 scored the full 111 rows, because no v5
     refusal had all of its bits from 3 up inside one of those ranges.

CONTRACT (clause e) -- AN ENUMERATION, NOT A LIST.
THE FLOOR.  The fixed PORT arm's body tests three things -- the access direction, the written value (`idata >=
DEV_FBCTRL_NPORTS`, on all 64 bits) and the latch (`!warned_port`) -- and it reads and writes `current_port`.  Every
cell of those is held: the value's magnitude, width and class, every bit the compare tests set alone at every width
that can carry it (B), the latch unset and set, a read and a write.  The
previous selector (current_port's VALUE), the COMMAND cell and the data cells are NOT tested by the body -- the fix
makes no partition of them -- so an edit conditioned on one of them is coverage beyond the floor (clause e): the rows
hold the values listed below and claim nothing more.  For each EFFECT, each value listed is held by the session(s)
named, and the one-axis edit of every listed cell is measured red in
regress/artifacts/fbctrlportkill_20260924/table6/TABLE.out (under an earlier edit's name where the generated text
was identical to it: table5/CELLMAP.out) -- except the read of selector 0, whose edit is EQUIVALENT (NOT CLAIMED
(4)).  This is the class closure: pass 2, the F2 pass, the closing pass and the closing F2 pass each found such an
edit on a value the rows had left empty, and the verifying pass found one on a range of bits (B).
  a refusal keeps the previous selector -- width sb sh sw sd (P, W, B); previous 0 (F), 1 (V), 2 3 (X), 4 (X4, Z), 5
      (P, W, Y, B), 6 (W, X6), 7 (P, V, X7); COMMAND 0 (P, B), 1 (Y), 2 (V); latch unset (the first refusal of P, F,
      V, W, X2, X3, X4, X6, X7, Y and B -- Z's and R's first refusals are never read back) and set (the rest)
  the first refusal announces once -- width sw (P, V, F, Y, Z) sb (W, X6) sh (X2, X7) sd (X3, X4); previous 0 (F), 1
      (V), 2 (X2, Z), 3 4 6 7 (X), 5 (P, W, Y); COMMAND 0 (P, F, W, X, Z), 1 (Y), 2 (V); the first-refused value from 8
      (P, W) to all-ones (X4)
  the later refusals are silent -- width sb (P, V, B) sh (P, W, B) sw (P, V, W, R, Z, B) sd (P, V, W, Y, B); previous
      0-4 (Z), 5 (P, W, R, Y, B), 6 (W), 7 (P, V); COMMAND 0 (P, W, Z, B), 1 (Y), 2 (V, R); across PORT reads, a DATA
      write, a DATA read, a NOP and a GET_RESOLUTION (R), a SET_RESOLUTION (Y), accepted selectors, twelve refusals
      (P) and 108 (B)
  the announce names the refused value -- 0x8 (P), 0x9 (F, V, X2, X6, Y), 0x100 (X7), the wide 0x100000005 (X3),
      all-ones (X4): width sw (P, F, V, Y) sb (X6) sh (X2, X7) sd (X3, X4); previous 0 (F), 1 (V), 2 3 4 6 7 (X), 5 (P,
      Y); COMMAND 0 (P, F, X), 1 (Y), 2 (V)
  a PORT read returns the selector -- lb lh ld (W) and lw (every session but K); every selector 0-7 (A), though a
      read of selector 0 cannot tell a skipped load from a real one (odata starts at 0: NOT CLAIMED (4)); COMMAND 0
      (A), 1 (Y), 2 (V); latch unset (A) and set (P)
  a refusal leaves the cells alone -- all eight cells after refusals of 9, 16, 0x100000003 (sd), 0xff (sb) and 8 at
      previous selector 7 with COMMAND 2 (V); X1 after a refusal of 9 (sw) at previous 1 with COMMAND 2 (V); port 5's
      cell after a refused 9 (sw) at previous 5 with COMMAND 0 (K); COMMAND itself after a refused 0x100000003 (sd) at
      previous 5 with COMMAND 1 (Y -- its second refusal only: the SET_RESOLUTION before it rewrites COMMAND, so a
      write by the first refusal would be overwritten before the read) -- and no other cell, width, selector or
      COMMAND value (NOT CLAIMED (8))
  an accepted selector is stored -- 0 and 7 at every width (W, A), every selector at one width (A); latch unset (A, W)
      and set (V's fills and Z's selections, after their first refusals)
Beyond the effects: the store of an out-of-range value, clamped, masked or reset; the compare narrowed or signed (bit
63 alone, bits 32-33, 32-bit truncation) or made to ignore any set of the bits it tests, at any width (B: every
single-bit value is refused); a boundary slip at 8; an announce on an accepted selector; an announce
demoted to debug() (F); an announce suppressed above or below any threshold, or on the sign bit (the first refusals
run from 8 to all-ones); and the pre-#549 process exit on DATA.
NOT CLAIMED (disclosed).  (0) An edit conditioned on TWO axes at once -- a joint cell such as "refuse 9 only by sb
after PORT=3", "`` sets bit 32 except at selector 5", or THE LATCH CROSSED WITH ONE BIT (while `` is clear the
compare ignores bit k) -- is caught only where a row happens to hold that pair; the enumeration above is one-axis by
design, and closing every pair is the unbounded growth the carrier's floor rules refuse.  The latch x bit joint is held
at bits 3, 8 and 32 only (the first refusals 8, 0x100 and 0x100000005; all-ones decides no bit), so 58 bits are open
at latch-clear (4-7, 9-31, 33-63): measured 117/117 at bits 4 and 40 (opus549_measured/f2v6/TABLE_f2v6.out).  What it
lets through is a guest's FIRST out-of-range selector, once per device: below bit 32 the value is stored and the next
DATA access ends the process in the kept range check; from bit 32 up it is truncated onto a LEGAL port silently --
2^40 selects port 0 and the next DATA write runs as a command (WITNESS.out) -- the pre-#549 defect, for one write.  Not
owed by the floor, whose generating set for `` is below/at/past the bound, held at both latch states; the close, one
fresh device per bit at a second anchor (61 sessions, 9.7 s on the fix: TABLE_U.out), is filed as ``.  One pair is
held and is recorded as a pair, not as a class: COMMAND non-zero with the latch set, before a PORT read (R's second
refusal; V's and Y's later ones) -- in the 94-row probe R was its only holder (the Opus closing-F2 seat,
J_ref_cmdlatch).  (1) A process-global latch instead of the per-instance field, and an announce conditioned on the
machine's serial number.  The probe's `-E testmips` session has ONE fbctrl.  A second instance IS reachable without
adding a device -- two testmips machines in one process from a config file, `gxemul @config` -- and there the
process-global latch reds (measured by the Opus F2 seat, opus549_measured/f2/TABLE_twoinst.out), but that is a new
launcher this probe does not use: filed as `fbctrltwoinst`.  The latch is per instance by construction, a struct
member, untested across two machines.  (2) Any edit of the DATA arm's kept range check -- deleting it, deleting only
its `exit(1)`, weakening `>=` to `>`, or swapping its operand -- is unreachable while the PORT arm refuses, because no
out-of-range selector is ever stored; the check is KEPT as a tripwire.  It exits before it indexes, so an index of 8
cannot reach `warned_port` while it stands, and `port[8]` would be undefined behaviour, not a defined alias of the
latch.  (3) An exception for one value no row writes (``), or for a CLASS the rows never present: the low three bits
(every refused value's residue mod 8 is 0, 1, 3, 5 or 7, never 2, 4 or 6 -- `` scores the full count, and a wide
value with such a residue is accepted and truncated onto a legal port silently, 0x100000006 to port 6); an accept-side
class (every value with exactly four bits set: 0x700000001 is accepted silently as port 1 -- not always the exit); a
latch that re-arms with a period of 108 refusals or more (107 re-announces on B's 108th refusal and reds the count
row; 108 would re-announce on a 109th no session makes -- measured); an announce suppressed or mangled only for one
value no session refuses first (``) or only for a class no first refusal belongs to (values whose bits all lie in
9..31, say); an announce demoted to debug() for some values but not F's 9 -- only a free-running session tells them
apart; and A REFUSAL THAT WRITES THE CONSTANT EQUAL TO A VALUE'S ONLY ANCHOR: the rows read "keeps the previous
selector" against the stored-int oracle, so `` reads as a keep.  Sixty refused values are observed at exactly one
previous selector -- 58 single bits at 5, 0xffffffff at 5, 0x100000005 at 3 (SIM_ANCHORS.out) -- and each admits a
refusal-time write of that selector (measured); bits 3, 4 and 8 are dual-anchored (V, X7 at 7) and red.  A second
anchor for a value closes its constant form; a replaced anchor moves it (B's sd group at 7 opens `` on bits 32 and
34-62); no finite row set closes "except value k".  (A threshold is not in this class, nor is a range of bits the
COMPARE ignores: see "Beyond the effects".)  (4) Equivalent edits, THIRTEEN, each measured at the full score
(table5/TABLE.out, table6/TABLE.out): storing `(uint32_t) idata`, `(int) idata` or `idata & 7` (only 0..7 reach that
store, and each compiles to the fix's own binary), reordering the struct's fields, masking the PORT read with 7 (the
selector is always 0..7), a PORT read that skips its load at selector 0 (odata starts at 0), a latch set to any
nonzero value in place of 1, the bound written `idata > 7`, `(idata & ~7ULL) != 0` or with the literal 8 (each equal
to `idata >= DEV_FBCTRL_NPORTS` on every uint64_t), the direction tested as `writeflag != MEM_WRITE` (the arm receives
only MEM_READ and MEM_WRITE), the latch tested as `warned_port != 1` (it holds only 0 or 1), and the announce
additionally requiring a nonzero idata (a refused value is never 0).  (5) `printf` in place of `fatal()`, or the
announce without its trailing newline: indistinguishable in THIS invocation's output.  (6) Narrowing the DATA path's
widths, and the device's OTHER two process-ending calls -- fbctrl_command()'s default (any command other than 0-2,
including the documented but unimplemented FILL/COPY) and the access function's default (any offset other than
0x00/0x10): filed as `fbctrlcmdkill` and `fbctrloffkill`.  (7) Edits to this instrument's own logic are reviewed, not
rowed.  (8) The data cells' VALUES are not an axis the rows enumerate.  An edit conditioned on a cell's value
(`port[COLOR_B] == 0`, `port[Y1] != 0`) is caught only where a row's refusal or read holds a triggering state, and the
states held are: every cell 0 (F, W, X, Z, B), port 5 at 0x2a (P, K, R), X1 at 123 with COMMAND 2 (V's first refusal),
ports 1-7 at 0x11*p with COMMAND 2 (V's later refusals and the PORT read right after them), and COMMAND 1 (Y).  The
cells-alone effect is claimed only as listed: no cell is read back after an `sh` refusal or after a refusal at
previous selector 0, 2, 3, 4 or 6, and after a refusal at previous 5 the only cells read back are port 5's under
COMMAND 0 (K) and COMMAND's under COMMAND 1 (Y, its second refusal only).
LIMITS OF A ROW.  Session P re-selects PORT=5 before each refusal, so a refused selector cannot carry over into the
next row WHEN THE LEGAL SELECTION LANDS; an edit that also refuses legal selections defeats the re-select, and later
rows then inherit its value (M_acclatch).  Every read-back is taken after its session ends, so a session that dies
part-way blanks the rows it had already passed (M_acc4: session A died at step 5 and every accept row read `-`).  And
session A's "accept selector 0 written by sw" cannot see an IGNORED store -- the selector is already 0 at power-on --
so the cell "0 by sw is accepted" is held by W's row after PORT=5, not by A's.  Session B reads PORT once per width, so
each of its rows says that SOME single-bit value of that width got through, not which: it prints the last value that
GOT THROUGH, which names the bit for k <= 31 (2^k; INT_MIN prints 0xffffffff80000000) and reads 0 for bits 32-63.  B's
"keeps 5" is verified against the stored-int oracle only: a refusal that WRITES 5 reads as a keep (NOT CLAIMED (3)).  B
depends on the emulated `` executing -- the encoding is pinned and the disassembly control checks the word, not the
doubling; table 6's width-differentiated reds (bit 9 red at sh/sw/sd and never sb, bit 16 and up never sh, bit 32 and
up never sw) are the evidence each 2^k reached the compare.  After a refusal, the DATA accesses made without a fresh
selector are K's read, K's write and R's write; R's DATA read re-selects port 5 first.  Every refusal row rests on
`fatal()` printing and returning on this path (src/core/debugmsg.c's `fatal()`: no early-out and no exit) -- twelve
refusals in P and 108 in B.

Every stepped word is compared with the emulator's own step line (the hand-assembled-encoding trap).
Output: one row per check, then FBCTRLP_CONTROL=OK|FAILED, FBCTRLP_RESULT=n/t, FBCTRLP_DONE.  The exit status is 0
only when the controls are OK and every row is green: a red row is a nonzero exit, not only a crash.
usage: fbctrl_port_probe.py <gxemul-binary>
"""
import os
import pty
import re
import select
import signal
import sys
import time

BIN = sys.argv[1]

STUBADDR = 0xffffffff80100000
CODE = 0xffffffff80100020
FBCTRL = 0xffffffffb2f00000

ONCE_RE = re.compile(r"\[ fbctrl: port number 0x[0-9a-f]+ is out of range; the selector is unchanged \(once\) \]")
#  The refused value, captured whole: "0x8" can match only 0x8, never 0x80 (the digits run to " is").
VAL_RE = re.compile(r"\[ fbctrl: port number (0x[0-9a-f]+) is out of range; the selector is unchanged \(once\) \]")

REG = dict(zr=0, v0=2, v1=3, a0=4, a1=5, a2=6, a3=7, t0=8, t1=9, t2=10, t3=11, t4=12, t5=13, t6=14, t7=15,
           s0=16, s1=17, s2=18, s3=19, s4=20, s5=21, s6=22, s7=23, t8=24, t9=25)
OPS = dict(sb=0x28, sh=0x29, sw=0x2b, sd=0x3f, lb=0x20, lh=0x21, lw=0x23, ld=0x37)
LOADS = ("lb", "lh", "lw", "ld")


def ins(op, rt, off, base="t0"):
    if op == "dsll":                                     # SPECIAL dsll rt,rt,off: rd == rt, the shift in sa
        return (REG[rt] << 16) | (REG[rt] << 11) | ((off & 31) << 6) | 0x38
    return (OPS[op] << 26) | (REG[base] << 21) | (REG[rt] << 16) | (off & 0xffff)


def text(op, rt, off, base="t0"):
    if op == "dsll":
        return "dsll %s,%s,%d" % (rt, rt, off)
    return "%s %s,%d(%s)" % (op, rt, off, base)


ANCHOR = (0x0f << 26) | (REG["t7"] << 16)                   # lui t7,0 -- the breakpoint anchor, never executed
#  Pinned independently of the helper: a wrong register field would otherwise pass silently.
assert ins("sw", "t1", 0) == 0xad090000
assert ins("lw", "t2", 0x10) == 0x8d0a0010
assert ins("sd", "t5", 0) == 0xfd0d0000
assert ins("sb", "t8", 0) == 0xa1180000
assert ins("sh", "t9", 0) == 0xa5190000
assert ins("ld", "s3", 0) == 0xdd130000
assert ins("sb", "zr", 0) == 0xa1000000
assert ins("lb", "a1", 0) == 0x81050000
assert ins("lh", "a2", 0) == 0x85060000
assert ANCHOR == 0x3c0f0000
assert ins("dsll", "t1", 1) == 0x00094878
assert ins("dsll", "t4", 1) == 0x000c6078


def stub_path():
    p = "/tmp/gx_fbctrl549p_%d.bin" % os.getpid()
    with open(p, "wb") as f:
        f.write(b"\x00\x00\x00\x00")
    return p


def session(label, prog, regs, reads, stub, free_run=False):
    """prog: list of (op, rt, off, tag).  Steps each (or free-runs to an anchor after the last), then prints reads.
    Returns dict: status, exit, badis, vals, once_after (tag -> cumulative (once) count right after that step), buf."""
    pid, fd = pty.fork()
    if pid == 0:
        os.execvp(BIN, [BIN, "-V", "-E", "testmips", "-M", "64", "0x%x:%s" % (STUBADDR, stub)])
        os._exit(127)
    buf = ""
    dead = [False]

    def rd(t=0.4):
        nonlocal buf
        r, _, _ = select.select([fd], [], [], t)
        if fd not in r:
            return True
        try:
            d = os.read(fd, 65536)
        except OSError:
            dead[0] = True
            return False
        if not d:
            dead[0] = True
            return False
        buf += d.decode("latin1", "replace")
        return True

    def wait(mark=0, echo=None, timeout=60):
        t = time.time()
        while time.time() - t < timeout:
            if not rd():
                return False
            resp = buf[mark:]
            if echo is not None and echo not in resp:
                continue
            if len(buf) > mark and resp.rstrip().endswith("GXemul>"):
                return True
        return False

    def send(x, timeout=60):
        b = (x + "\n").encode("latin1")
        _mark = len(buf)
        n = 0
        try:
            while n < len(b):
                n += os.write(fd, b[n:])
        except OSError:
            dead[0] = True
            return False
        return wait(mark=_mark, echo=x if x else None, timeout=timeout)

    words = [ins(op, rt, off) for op, rt, off, _t in prog] + [ANCHOR]
    status, badis, vals, once_after, steps = "?", [], {}, {}, 0
    if not wait(timeout=120):
        status = "NO-PROMPT"
    else:
        for i, w in enumerate(words):
            send("put w 0x%x, 0x%08x" % (CODE + 4 * i, w))
        for name, val in regs:
            send("%s=0x%x" % (name, val & 0xffffffffffffffff))
        send("pc=0x%x" % CODE)
        status = "SURVIVED"
        if free_run:
            send("breakpoint add 0x%x" % (CODE + 4 * len(prog)))
            mark = len(buf)
            b = b"continue\n"
            try:
                os.write(fd, b)
            except OSError:
                dead[0] = True
            if not wait(mark=mark, timeout=60) or dead[0]:
                status = "HOST-DIED" if dead[0] else "NO-BREAK"
            elif ("%016x" % (CODE + 4 * len(prog))) not in buf[mark:] and \
                    ("%08x" % ((CODE + 4 * len(prog)) & 0xffffffff)) not in buf[mark:]:
                status = "NO-BREAK"
            else:
                steps = len(prog)
                once_after["free"] = len(ONCE_RE.findall(buf))
        else:
            for i, (op, rt, off, tag) in enumerate(prog):
                mark = len(buf)
                if not send("step 1", timeout=30) or dead[0]:
                    status = "HOST-DIED"
                    break
                steps += 1
                seg = buf[mark:]
                w = words[i]
                m = re.search(r"%016x:?\s+%08x\s+([^\r\n<]+)" % (CODE + 4 * i, w), seg)
                got = re.sub(r"\s+", " ", m.group(1)).strip() if m else "(no step line)"
                if not got.startswith(text(op, rt, off)):
                    badis.append("+0x%02x planted 0x%08x -> %r, expected %r" % (4 * i, w, got, text(op, rt, off)))
                if tag:
                    once_after[tag] = len(ONCE_RE.findall(buf))
        if status == "SURVIVED":
            for name in reads:
                got = None
                for _ in range(2):
                    mark = len(buf)
                    if not send("print %s" % name, timeout=30) or dead[0]:
                        break
                    m = re.search(r"(?m)^0x([0-9a-fA-F]+)\s*$", buf[mark:])
                    if m:
                        got = int(m.group(1), 16)
                        break
                vals[name] = got
    try:
        os.write(fd, b"quit\n")
    except OSError:
        pass
    t1 = time.time()
    while time.time() - t1 < 20.0:
        if not rd(0.1):
            break
    try:
        os.close(fd)
    except OSError:
        pass
    rpid, st = 0, None
    t1 = time.time()
    while rpid == 0 and time.time() - t1 < 10.0:
        try:
            rpid, st = os.waitpid(pid, os.WNOHANG)
        except OSError:
            break
        if rpid == 0:
            time.sleep(0.05)
    if rpid == 0:
        try:
            os.kill(pid, signal.SIGKILL)
            _, st = os.waitpid(pid, 0)
        except OSError:
            st = None
    code = os.WEXITSTATUS(st) if st is not None and os.WIFEXITED(st) else None
    print("  session %-2s %-9s steps=%d/%d exit=%s once=%d" % (label, status, steps, len(prog), code,
                                                              len(ONCE_RE.findall(buf))))
    for b in badis:
        print("      DISASSEMBLY MISMATCH: %s" % b)
    return {"status": status, "exit": code, "badis": badis, "vals": vals, "once_after": once_after,
            "steps": steps, "buf": buf, "n": len(prog)}

def hx(v):
    return "-" if v is None else "0x%x" % v


def first_val(S):
    """the value the FIRST announce of a session names, or '-'"""
    m = VAL_RE.findall(S["buf"])
    return m[0] if m else "-"

#  read that never ran cannot pass -- selector 0 would otherwise match a register's reset value); DATA value 0x11*p

ACC = [(0, "sw"), (1, "sb"), (2, "sh"), (3, "sw"), (4, "sd"), (5, "sb"), (6, "sh"), (7, "sd")]

    PROG_A.append(("sw", VAL_REG[k], 16, None))                        # DATA = 0x11*p (NOP for port 0)

    row("Z PORT keeps 4 after its refusals", hx(Z["vals"].get("s0")), "0x4")
    print("SINGLE BITS (session B: %s %d/%d)" % (B["status"], B["steps"], B["n"]))
    for _w, _r, bits, rb in B_GROUPS:
        row("B refuse every single bit 2^3..2^%d (%s), PORT keeps 5" % (bits - 1, _w), hx(B["vals"].get(rb)),
            "0x5")
    row("B announce exactly 1 across its %d refusals" % NB, B["once_after"].get("blast"), 1)
    print("FIRST REFUSALS (sessions X2-X7: %s)" % " ".join("%s %d/%d" % (S["status"], S["steps"], S["n"])
                                                          for S in X))
    for (lab, k, w, v), S in zip(X_CELLS, X):
        row("%s refuse 0x%x (%s) after PORT=%d, PORT keeps %d" % (lab, v, w, k, k), hx(S["vals"].get("s0")), hx(k))
        row("%s that FIRST refusal announces exactly once" % lab, S["once_after"].get("x"), 1)
        row("%s the announce names the refused value 0x%x" % (lab, v), first_val(S), "0x%x" % v)
    control = all(ctl)
    n_ok, n = sum(RESULTS), len(RESULTS)
    print("FBCTRLP_CONTROL=%s" % ("OK" if control else "FAILED"))
    print("FBCTRLP_RESULT=%d/%d" % (n_ok, n))
    print("FBCTRLP_DONE")
    return 0 if control and n_ok == n else 1
