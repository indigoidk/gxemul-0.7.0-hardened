def main():
    src, dst = sys.argv[1], sys.argv[2]
    s = io.open(src, encoding="utf-8", newline="").read()
    assert "\r" not in s
    for old, new in EDITS:
        s = sub1(s, old, new)
    io.open(dst, "w", encoding="utf-8", newline="").write(s)
    print("make_v6 ->", dst, "(%d edits)" % len(EDITS))


if __name__ == "__main__":
    main()
