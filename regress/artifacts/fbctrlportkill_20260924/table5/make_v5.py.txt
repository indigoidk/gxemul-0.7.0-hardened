def sub1(s, old, new):

    if n != 1:

    return s.replace(old, new)

def make(s):
    s = sub1(s, R_OLD, R_NEW)
    s = sub1(s, V_OLD, V_NEW)
    s = sub1(s, X_OLD, X_NEW)
    i = s.index(C_HEAD)
    j = s.index(C_TAIL, i) + len(C_TAIL)
    if s.count(C_HEAD) != 1 or s.count(C_TAIL) != 1:
        raise SystemExit("REFUSED: contract anchors are not unique")
    s = s[:i] + C_NEW + s[j:]
    for old, new in CODE_EDITS:
        s = sub1(s, old, new)
    return s


def main():
    src = io.open(sys.argv[1], encoding="utf-8", newline="").read()
    assert "\r" not in src
    out = make(src)
    doc = out.split('"""')[1]
    if chr(92) in doc:
        raise SystemExit("REFUSED: a backslash in the docstring (check_score_claims reads it bare)")
    io.open(sys.argv[2], "w", encoding="utf-8", newline="").write(out)
    print("make_v5 ->", sys.argv[2])


if __name__ == "__main__":
    main()
