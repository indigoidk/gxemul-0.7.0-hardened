#!/bin/bash
# Dry run: stage FROZEN copies of the seat's scripts in /tmp/opusf2v6_scripts, then exercise the generator and the
# sd7 probe patcher against the repo files (read-only), writing only under /tmp/opusf2v6_dry.
set -e
SPW=/mnt/c/Users/User00/AppData/Local/Temp/claude/C--DocumentNoSnc-cc-GXEMUL/8e474f41-4d47-4c99-bf11-3406a84c4fcd/scratchpad/opusf2v6_seat
SEC=/mnt/c/DocumentNoSnc/cc/GXEMUL/GXEMUL-SEC
rm -rf /tmp/opusf2v6_scripts /tmp/opusf2v6_dry
mkdir -p /tmp/opusf2v6_scripts /tmp/opusf2v6_dry
cp "$SPW"/gen_f2v6.py "$SPW"/mkprobe_sd7.py "$SPW"/u_sessions.py "$SPW"/run_all.sh /tmp/opusf2v6_scripts/
sed -i 's/\r$//' /tmp/opusf2v6_scripts/*
cd /tmp/opusf2v6_dry
python3 /tmp/opusf2v6_scripts/gen_f2v6.py "$SEC/src/devices/dev_fbctrl.c" v
cat v/MANIFEST.out
echo "--- D2_eq32_w5 vs fix"; diff "$SEC/src/devices/dev_fbctrl.c" v/D2_eq32_w5.c || true
echo "--- N1_mask4 vs fix"; diff "$SEC/src/devices/dev_fbctrl.c" v/N1_mask4.c || true
echo "--- B_mod107 vs fix"; diff "$SEC/src/devices/dev_fbctrl.c" v/B_mod107.c || true
python3 /tmp/opusf2v6_scripts/mkprobe_sd7.py "$SEC/regress/fbctrl_port_probe.py" p_sd7.py
echo "--- sd7 probe vs v6"; diff "$SEC/regress/fbctrl_port_probe.py" p_sd7.py || true
python3 -c "import ast; ast.parse(open('p_sd7.py').read()); print('sd7 parses')"
bash -n /tmp/opusf2v6_scripts/run_all.sh && echo "run_all parses"
md5sum /tmp/opusf2v6_scripts/*
echo DRY_DONE
