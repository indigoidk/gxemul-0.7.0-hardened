#!/usr/bin/env python3
"""A prototype of D1's repair (NOT a proposed probe edit): one FRESH device per single-bit value, so each value's refusal
is that device's FIRST -- the latch is clear.  Each session: PORT=<anchor> (legal, leaves the latch clear), a store of
2^k (sw below bit 32, sd from 32 up), then a PORT read into a poisoned register.  Checks per session: PORT keeps the
anchor; exactly one announce right after the refusal; the announce names 2^k; and the session's own controls (it
survived, every step ran, the disassembly matched).  Uses the frozen v6 probe's own session().
usage: u_sessions.py <gxemul> <frozen v6 probe> [anchor=7] [bits: all | 58]"""
import importlib.util
import os
import sys
import time

BIN, PROBE = sys.argv[1], sys.argv[2]
ANCHOR = int(sys.argv[3]) if len(sys.argv) > 3 else 7
WHICH = sys.argv[4] if len(sys.argv) > 4 else "all"
sys.argv = [PROBE, BIN]
spec = importlib.util.spec_from_file_location("fbp", PROBE)
fbp = importlib.util.module_from_spec(spec)
spec.loader.exec_module(fbp)
BITS = list(range(3, 64)) if WHICH == "all" else [k for k in range(3, 64) if k not in (3, 8, 32)]
stub = fbp.stub_path()
t0 = time.time()
bad_ctl, bad_keep, bad_once, bad_name = [], [], [], []
try:
    for k in BITS:
        w = "sw" if k < 32 else "sd"
        S = fbp.session("U%02d" % k, [("sw", "a0", 0, None), (w, "t1", 0, "u"), ("lw", "s0", 0, None)],
                        [("t0", fbp.FBCTRL), ("a0", ANCHOR), ("t1", 1 << k), ("s0", 0xdead)], ["s0"], stub)
        got = S["vals"].get("s0")
        if S["status"] != "SURVIVED" or S["steps"] != 3 or S["badis"]:
            bad_ctl.append("%d:%s/%d/%d" % (k, S["status"], S["steps"], len(S["badis"])))
        if got != ANCHOR:
            bad_keep.append("%d:%s" % (k, fbp.hx(got)))
        if S["once_after"].get("u") != 1:
            bad_once.append("%d:%s" % (k, S["once_after"].get("u")))
        if fbp.first_val(S) != "0x%x" % (1 << k):
            bad_name.append("%d:%s" % (k, fbp.first_val(S)))
finally:
    os.unlink(stub)
print("U_ANCHOR=%d U_SESSIONS=%d U_WALL=%.1fs" % (ANCHOR, len(BITS), time.time() - t0))
print("U_CONTROL_BAD=%d %s" % (len(bad_ctl), " ".join(bad_ctl)))
print("U_KEEP_BAD=%d %s" % (len(bad_keep), " ".join(bad_keep)))
print("U_ONCE_BAD=%d %s" % (len(bad_once), " ".join(bad_once)))
print("U_NAME_BAD=%d %s" % (len(bad_name), " ".join(bad_name)))
print("U_DONE")
