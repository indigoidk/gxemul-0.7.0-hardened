#!/usr/bin/env python3
"""fable5 pass-1 experiment 2: (S8) USR-mode plain load through a page warmed by a privileged
access; (S8b) the same USR load COLD; (S10) a Thumb-state data abort's LR_abt; (S11) a Thumb
branch to an unmapped VA.  Same session machinery as exp.py; transcripts beside this file."""
import os, pty, re, select, signal, sys, time

BIN = sys.argv[1]
HERE = os.path.dirname(os.path.abspath(__file__))
STUB = os.path.join(HERE, "stub.bin")
CODE, DEST, PREF, DATA, THUMB = 0x8000, 0x9000, 0x8400, 0x8500, 0x8100
POISON = 0xDEADBEEF


def session(name, cmds, settle=2.0, spin=2.0):
    pid, fd = pty.fork()
    if pid == 0:
        os.execv(BIN, [BIN, "-V", "-A", "-E", "testarm", "-M", "64", "0x%x:%s" % (CODE, STUB)])
        os._exit(127)
    buf, eof = "", False

    def rd(t):
        nonlocal buf, eof
        end = time.time() + t
        while time.time() < end:
            try:
                r, _, _ = select.select([fd], [], [], 0.1)
            except OSError:
                eof = True
                return
            if not r:
                continue
            try:
                d = os.read(fd, 65536)
            except OSError:
                eof = True
                return
            if not d:
                eof = True
                return
            buf += d.decode("utf-8", "replace")

    rd(settle)
    for c in cmds:
        if eof:
            break
        try:
            os.write(fd, (c + "\n").encode())
        except OSError:
            eof = True
            break
        rd(0.2)
    rd(spin)
    alive = True
    deadline = time.time() + (12.0 if eof else 2.0)
    while time.time() < deadline:
        try:
            wpid, _st = os.waitpid(pid, os.WNOHANG)
        except ChildProcessError:
            alive = False
            break
        if wpid == pid:
            alive = False
            break
        time.sleep(0.05)
    if alive:
        try:
            os.kill(pid, signal.SIGKILL)
            os.waitpid(pid, 0)
        except OSError:
            pass
    try:
        os.close(fd)
    except OSError:
        pass
    t = buf.replace("\r", "")
    with open(os.path.join(HERE, "S_%s.txt" % name), "w") as f:
        f.write(t)
    return t, alive


def printed(t, reg):
    return [int(x, 16) for x in re.findall(r"print\s+%s\s*\n\s*(0x[0-9a-f]+)" % reg, t)]


def dumped(t, addr):
    label = "0x%08x" % (addr & ~0xF)
    idx = (addr & 0xF) // 4
    vals = []
    for line in t.splitlines():
        if line.startswith(label):
            parts = line[12:48].split()
            if len(parts) == 4 and len(parts[idx]) == 8:
                vals.append(int.from_bytes(bytes.fromhex(parts[idx]), "little"))
            elif "memory access failed" in line:
                vals.append("FAILED")
    return vals


def put(addr, w):
    return "put w 0x%x, 0x%08x" % (addr, w)


def plant(words, at):
    return [put(at + 4 * i, w) for i, w in enumerate(words)]


ARM = [0xE3A01901, 0xE3A02001, 0xEE021F10, 0xEE032F10, 0xEE110F10, 0xE3800001, 0xEE010F10, 0xEE080F17]
HANDLER = lambda marker: [0xE3A05A09, 0xE10F6000, 0xE5856000, 0xEE157F10, 0xE5857004, 0xEE168F10,
                          0xE5858008, 0xE585E00C, 0xE3A09000 | marker, 0xE5859010, 0xEAFFFFFE]
FALL = [0xE3A05A09, 0xE5853014, 0xEAFFFFFE]
TABLE = [
    put(0x4000, 0x00000C02), put(0x4008, 0x00300C02), put(0x4010, 0x00005001),
    put(0x4014, 0x00500C22), put(0x4018, 0x00600002), put(0x5000, 0x00600FF2),
    put(0x0C, 0xEA0020FB), put(0x10, 0xEA00213A),
    put(0x200000, 0x22222222), put(0x300000, 0x33333333), put(0x600000, 0x66666666),
    put(0x9100, 0x11111111),
] + [put(DEST + o, POISON) for o in range(0, 0x18, 4)] \
  + plant(HANDLER(0xA5), PREF) + plant(HANDLER(0xDA), DATA)
NH = 11


def run(name, stim, steps_list, extra_before=(), tail=()):
    code = ARM + stim + FALL
    cmds = TABLE + list(extra_before) + plant(code, CODE) + ["pc=0x%x" % CODE]
    for s in steps_list:
        cmds += s if isinstance(s, list) else ["step %d" % s]
    cmds += ["dump 0x%x 0x%x" % (DEST, DEST + 0x20)] + list(tail)
    t, alive = session(name, cmds)
    print("=" * 78)
    print("SESSION %s  alive=%s  code_len=%d" % (name, alive, len(code)))
    print("  DEST[0..5] =", ["0x%08x" % v if isinstance(v, int) else v
                           for v in [(dumped(t, DEST + o) or ["?"])[-1] for o in range(0, 0x18, 4)]])
    for reg in ("pc", "lr", "cpsr", "r0", "r4", "r6", "r7"):
        vals = printed(t, reg)
        if vals:
            print("  print %-4s ->" % reg, ["0x%08x" % v for v in vals])
    for ln in t.splitlines():
        if ("arm memory fault" in ln or "arm_exception" in ln or "FATAL" in ln or "WARNING" in ln
                or "unimplemented" in ln.lower() or "TODO" in ln or "not running" in ln.lower()
                or "could not" in ln or "(once)" in ln or "declin" in ln.lower()):
            print("  LOG:", ln.strip()[:150])
    return t


# S8: SVC load of the AP=0 section (priv RO -> succeeds, warms host_load), then MSR CPSR_c,#0x10
#     (USR32), then a plain USR load of the SAME VA.  Architecture (S=1,R=0, AP=0): user -> no
#     access -> permission fault 0x0d.  One step N, so no debugger invalidation in between.
run("S8_usr_warm",
    [0xE3A03606, 0xE5934000, 0xE321F010, 0xE5936000],   # mov r3,#0x600000 ; ldr r4,[r3] ; msr cpsr_c,#0x10 ; ldr r6,[r3]
    [8 + 4 + 1 + NH, ["print r4", "print r6", "print cpsr"]])

# S8b: the same USR load with NO prior privileged touch of that VA (cold): the walk is consulted.
run("S8b_usr_cold",
    [0xE3A03606, 0xE321F010, 0xE5936000],               # mov r3,#0x600000 ; msr cpsr_c,#0x10 ; ldr r6,[r3]
    [8 + 3 + 1 + NH, ["print r6", "print cpsr"]])

# S10: Thumb-state DATA abort.  ARM: mov r3,#0x100000 ; mov r5,#0x8100 ; orr r5,r5,#1 ; bx r5.
#      Thumb at 0x8100: 6818 = ldr r0,[r3,#0] ; e7fe = b . ; then whatever the model does.
#      DDI 0100I A2.6.6: R14_abt = aborted instruction + 8 = 0x8108.  The brief's reading of
#      arm_exception(): retaddr = pc + 4 + 2.
run("S10_thumb_data_abort",
    [0xE3A03601, 0xE3A05C81, 0xE3855001, 0xE12FF315],   # mov r3,#0x100000 ; mov r5,#0x8100 ; orr r5,r5,#1 ; bx r5
    [8 + 4, ["print pc", "print cpsr"], 1, ["print pc", "print lr", "print cpsr", "print r0"],
     1, ["print pc"], NH],
    extra_before=[put(THUMB, 0xE7FE6818)])

# S11: Thumb-state BRANCH to an unmapped VA (Thumb bx r3, r3 = 0x100001).  Architecture:
#      prefetch abort, R14_abt = 0x100000 + 4.  The model: PC_TO_POINTERS at the top of
#      run_instr (generic) or the Thumb interpreter's own fetch (cpu_arm.c:1273, running = 0)?
run("S11_thumb_branch_unmapped",
    [0xE3A03601, 0xE3833001, 0xE3A05C81, 0xE3855001, 0xE12FF315],  # mov r3,#0x100000 ; orr r3,r3,#1 ; mov r5,#0x8100 ; orr r5,r5,#1 ; bx r5
    [8 + 5, ["print pc", "print cpsr"], 1, ["print pc", "print lr", "print cpsr"],
     1, ["print pc", "print lr"], NH],
    extra_before=[put(THUMB, 0xE7FE4718)])             # 4718 = bx r3 ; e7fe = b .

print("EXP2_DONE")
