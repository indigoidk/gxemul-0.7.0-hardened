#!/usr/bin/env python3
"""explore0: output formats -- put w echo, unassemble, step lines, print, dump, and the reset control word."""
import sys
from gxsess import session

BIN = sys.argv[1]
cmds = [
    "put w 0x8000, 0xee113f10",      # mrc p15,0,r3,c1,c0,0
    "put w 0x8004, 0xe5843000",      # str r3,[r4]
    "put w 0x8008, 0xeafffffe",      # b .
    "r4=0x9000",
    "pc=0x8000",
    "unassemble 0x8000 0x800c",
    "print cpsr",
    "print pc",
    "step 3",
    "dump 0x9000 0x9010",
    "print r3",
    "dump 0x03fff000 0x03fff030",
    "dump 0x03ffffe0 0x04000000",
    "dump 0x0 0x20",
]
t, alive, st = session(BIN, cmds)
print(t)
print("ALIVE=%s STATUS=%s" % (alive, st))
print("EXPLORE0_DONE")
