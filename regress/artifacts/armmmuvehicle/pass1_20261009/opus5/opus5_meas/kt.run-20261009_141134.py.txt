#!/usr/bin/env python3
"""opus5 seat -- a PROTOTYPE of the armmmuvehicle rows, used ONLY as a kill-table instrument.

Two row sets, scored separately:
  B.*  the brief's PROPOSED rows (section 3), implemented as written: controls in one process
       (in TWO session shapes -- CTL_A plants and runs in ONE debugger interaction, CTL_B has an
       arming `step 1' + dump re-entry first, the arm_msrmode_probe shape), abort rows one process
       each in the CTL_B shape.
  X.*  rows this seat proposes in addition.
usage: kt.py <gxemul-binary> <label>
"""
import os
import re
import sys
import time
from gxsess import session, dumped, dumpcmd, HERE

BIN = sys.argv[1]
LABEL = sys.argv[2] if len(sys.argv) > 2 else "?"
OUT = os.path.join(HERE, "kt_out")
os.makedirs(OUT, exist_ok=True)

CODE, PH, DH, DEST, LIVE = 0x8000, 0x8400, 0x8500, 0x9000, 0x9100
POISON = 0xDEADBEEF


def b_to(src, dst):
    return 0xEA000000 | (((dst - (src + 8)) >> 2) & 0xFFFFFF)


W = dict(mcr_c2_r1=0xEE021F10, mcr_c3_r2=0xEE032F10, mrc_c1_r0=0xEE110F10, orr_r0_1=0xE3800001,
         mcr_c1_r0=0xEE010F10, mcr_c8_r0=0xEE080F17, nop=0xE1A00000, mcr_c5_r6=0xEE056F10,
         mcr_c6_r7=0xEE067F10, ldr_r3_r5=0xE5953000, str_r3_r5=0xE5853000, str_r3_r4=0xE5843000,
         mrc_c1_r3=0xEE113F10, mrc_c2_r3=0xEE123F10, mrc_c5_r3=0xEE153F10, mrc_c6_r3=0xEE163F10,
         mrs_r3=0xE10F3000, mrs_r3_s=0xE14F3000, b_self=0xEAFFFFFE,
         ldr_r3_r8=0xE5983000, ldr_r3_r9=0xE5993000, ldr_r3_sl=0xE59A3000, ldr_r3_fp=0xE59B3000,
         mcr_c2_r7=0xEE027F10,                      # mcr p15,0,r7,c2,c0,0  (TTB <- r7)
         bic_r0_s=0xE3C00C01,                       # bic r0,r0,#0x100  (S)
         orr_r0_r=0xE3800C02)                       # orr r0,r0,#0x200  (R)


def str_r3(off):
    return 0xE5843000 | off


def str_lr(off):
    return 0xE584E000 | off


def mov_r3(imm8):
    return 0xE3A03000 | imm8


def handler(marker):
    return [str_lr(4), W["mrs_r3_s"], str_r3(8), W["mrs_r3"], str_r3(12), W["mrc_c5_r3"], str_r3(16),
            W["mrc_c6_r3"], str_r3(20), mov_r3(marker), str_r3(24), W["b_self"]]


TTB, TTB2 = 0x4000, 0xC000
L1 = {TTB + 0x00: 0x00000C02, TTB + 0x08: 0x00300C02, TTB + 0x10: 0x00005001,
      TTB + 0x14: 0x00500C22, TTB + 0x18: 0x00600002,
      TTB + 0x1C: 0x000001E0,                     # X: VA 0x007 type-0 descriptor WITH domain bits (15)
      TTB + 0x20: 0x00005461,                     # X: VA 0x008 coarse @0x5400 in DOMAIN 3
      TTB2 + 0x00: 0x00000C02, TTB2 + 0x08: 0x00700C02}   # X: second table (VA 0x002 -> PA 0x007)
L2 = {0x5000: 0x00600FF2,                         # VA 0x00400000 -> PA 0x00600000 (AP 3)
      0x5008: 0x00702FF2,                         # X: VA 0x00402000 -> PA 0x00702000 (anchor, index 2)
      0x500C: 0x00703002}                         # X: VA 0x00403000 -> PA 0x00703000, small page AP 0
# the brief plants only what its rows read: identity vs remap targets and LIVE
DATA_B = {0x00200000: 0x22222222, 0x00300000: 0x33333333, 0x00400000: 0x44444444,
          0x00600000: 0x66666666, LIVE: 0x11111111}
DATA_X = {0x00700000: 0x77777777, 0x00702000: 0x77202000, 0x00402000: 0x44402000,
          0x00703000: 0x77303000}


def program(code_ops):
    w = {}
    for i, v in enumerate(code_ops):
        w[CODE + 4 * i] = v
    for i, v in enumerate(handler(0xDA)):
        w[DH + 4 * i] = v
    for i, v in enumerate(handler(0xBF)):
        w[PH + 4 * i] = v
    w[0x0C] = b_to(0x0C, PH)
    w[0x10] = b_to(0x10, DH)
    for o in range(0, 0x20, 4):
        w[DEST + o] = POISON
    w.update(L1)
    w.update(L2)
    w.update(DATA_B)
    w.update(DATA_X)
    return w


ARM = [W["mcr_c2_r1"], W["mcr_c3_r2"], W["mrc_c1_r0"], W["orr_r0_1"], W["mcr_c1_r0"], W["mcr_c8_r0"]]
FAULT_RE = re.compile(r"\{ arm memory fault: vaddr=0x([0-9a-f]+) domain=(\d+)")
rows = []
NS = [0]


def row(name, got, want):
    rows.append((name, got, want))


def run(name, code, regs, steps, shape="B", post=()):
    """shape A: plants + `step steps' in ONE interaction.  shape B: plants, `step 1', a dump
    (debugger re-entry: every dyntrans translation is cleared), then `step steps-1'."""
    NS[0] += 1
    w = program(code)
    cmds = ["put w 0x%x, 0x%08x" % (a, v) for a, v in sorted(w.items())]
    cmds += ["%s=0x%x" % (r, v) for r, v in regs] + ["pc=0x%x" % CODE]
    if shape == "A":
        cmds += ["step %d" % steps]
    else:
        cmds += ["step 1", dumpcmd(DEST, DEST + 0x1C), "step %d" % (steps - 1)]
    cmds += [dumpcmd(DEST, DEST + 0x3C), "dump 0x8000 0x8010", "dump 0x0 0x20"] + list(post) \
        + ["print 0x5eedd0ae"]
    t, alive, st = session(BIN, cmds, per_cmd=0.25, fast=0.1, spin=2.0)
    with open(os.path.join(OUT, "%s_%s.txt" % (LABEL, name)), "w") as f:
        f.write(t)
    armed = (dumped(t, CODE) == code[0] and dumped(t, 0x10) == b_to(0x10, DH)
             and "5eedd0ae" in t)
    d = {o: dumped(t, DEST + o) for o in range(0, 0x40, 4)}
    return t, alive, armed, d, FAULT_RE.findall(t)


def hx(v):
    return "none" if v is None else ("POISON" if v == POISON else "%x" % v)


def ab(name, d, alive, armed, marker=0xDA):
    if not armed:
        return "unarmed"
    if not alive:
        return "dead"
    if d[0x18] != marker:
        return "nohandler(%s)" % hx(d[0x18])
    return None


# ------------------------------------------------------------------ B: controls, two shapes
ctl = [W["ldr_r3_fp"], str_r3(0x3C)] + ARM + [
    W["mrc_c1_r3"], str_r3(0x20), W["mrc_c2_r3"], str_r3(0x24),
    W["ldr_r3_r8"], str_r3(0x28), W["ldr_r3_r9"], str_r3(0x2C),
    mov_r3(0xC1), str_r3(0x38), W["b_self"]]
for shp in ("A", "B"):
    t, alive, armed, d, fl = run("ctl" + shp, ctl, [("r1", TTB), ("r2", 1), ("r4", DEST),
                                                     ("r8", 0x00200000), ("r9", 0x00400000),
                                                     ("fp", LIVE)], len(ctl) + 1, shape=shp,
                                 post=["dump 0x00100000 0x00100010"])
    bad = None if (armed and alive and d[0x38] == 0xC1) else ("unarmed" if not armed else
                                                              ("dead" if not alive else "nomarker"))
    row("B.live" + shp, bad or hx(d[0x3C]), "11111111")
    row("B.mmubit" + shp, bad or hx((d[0x20] or 0) & 1), "1")
    row("B.ttbrd" + shp, bad or hx(d[0x24]), "4000")
    row("B.remap" + shp, bad or hx(d[0x28]), "33333333")
    row("B.pagemap" + shp, bad or hx(d[0x2C]), "66666666")
    hole = re.search(r"^0x00100000\s+\(memory access failed\)", t, re.M)
    row("B.holedump" + shp, bad or ("failed" if hole else "read"), "failed")

# ------------------------------------------------------------------ B: abort rows (shape B)
STEPS = 6 + 1 + 1 + 11
regs0 = [("r1", TTB), ("r2", 1), ("r4", DEST)]
for nm, va, wfsr in (("dtrans", 0x00100000, 0x05), ("pgtrans", 0x00401000, 0x07),
                     ("domain", 0x00500000, 0x19)):
    t, alive, armed, d, fl = run(nm, ARM + [W["ldr_r3_r5"], W["str_r3_r4"], W["b_self"]],
                                 regs0 + [("r5", va)], STEPS + 2)
    bad = ab(nm, d, alive, armed)
    row("B.%s fsr" % nm, bad or hx(d[0x10]), "%x" % wfsr)
    if nm == "dtrans":
        row("B.dtrans far", bad or hx(d[0x14]), "100000")
        row("B.dtrans mode", bad or hx((d[0x0C] or 0) & 0x1F), "17")
        row("B.dtrans lr", bad or hx(d[0x04]), "8020")
# perm: AP=0 LOAD accepted (value), then STR faults
code = ARM + [W["ldr_r3_r5"], str_r3(0x1C), mov_r3(0x77), W["str_r3_r5"], W["str_r3_r4"], W["b_self"]]
t, alive, armed, d, fl = run("perm", code, regs0 + [("r5", 0x00600000)], STEPS + 3 + 2)
bad = ab("perm", d, alive, armed)
row("B.perm fsr", bad or hx(d[0x10]), "d")
row("B.perm acceptld", ("unarmed" if not armed else hx(d[0x1C])), "66666666")
# pref trans: sentinels in FSR/FAR first, then B to the unmapped section
pcode = [W["mcr_c5_r6"], W["mcr_c6_r7"]] + ARM + [b_to(CODE + 4 * 8, 0x00100000), W["b_self"]]
t, alive, armed, d, fl = run("ptrans", pcode, regs0 + [("r6", 0xA4), ("r7", 0xFA4F0F00)],
                             2 + 6 + 1 + 1 + 11 + 2)
bad = ab("ptrans", d, alive, armed, marker=0xBF)
row("B.ptrans lr", bad or hx(d[0x04]), "100004")
row("B.ptrans mode", bad or hx((d[0x0C] or 0) & 0x1F), "17")
row("B.ptrans fsrsent", bad or hx(d[0x10]), "a4")
row("B.ptrans farsent", bad or hx(d[0x14]), "fa4f0f00")

# ------------------------------------------------------------------ X: proposed additional rows
for nm, va, dacr, wfsr, wfar in (("dtrans_odd", 0x00123458, 1, 0x05, 0x00123458),
                                 ("faultdesc", 0x00700000, 1, 0x05, 0x00700000),
                                 ("pgtrans_d3", 0x00800000, 0x41, 0x37, 0x00800000),
                                 ("pgdom_d3", 0x00800000, 0x01, 0x3B, 0x00800000)):
    t, alive, armed, d, fl = run(nm, ARM + [W["ldr_r3_r5"], W["str_r3_r4"], W["b_self"]],
                                 [("r1", TTB), ("r2", dacr), ("r4", DEST), ("r5", va)], STEPS + 2)
    bad = ab(nm, d, alive, armed)
    row("X.%s fsr" % nm, bad or hx(d[0x10]), "%x" % wfsr)
    row("X.%s far" % nm, bad or hx(d[0x14]), "%x" % wfar)
# page permission fault (small page AP=0): accept LOAD, then STR faults 0x0f
code = ARM + [W["ldr_r3_r5"], str_r3(0x1C), mov_r3(0x77), W["str_r3_r5"], W["str_r3_r4"], W["b_self"]]
t, alive, armed, d, fl = run("pageperm", code, regs0 + [("r5", 0x00403000)], STEPS + 3 + 2)
bad = ab("pageperm", d, alive, armed)
row("X.pageperm fsr", bad or hx(d[0x10]), "f")
row("X.pageperm acceptld", ("unarmed" if not armed else hx(d[0x1C])), "77303000")
# S=0,R=1 written by the guest: AP=0 LOAD accepted, STR faults 0x0d
code = [W["mcr_c2_r1"], W["mcr_c3_r2"], W["mrc_c1_r0"], W["bic_r0_s"], W["orr_r0_r"], W["orr_r0_1"],
        W["mcr_c1_r0"], W["mcr_c8_r0"], W["ldr_r3_r5"], str_r3(0x1C), mov_r3(0x77), W["str_r3_r5"],
        W["str_r3_r4"], W["b_self"]]
t, alive, armed, d, fl = run("sr", code, regs0 + [("r5", 0x00600000)], STEPS + 2 + 3 + 2)
bad = ab("sr", d, alive, armed)
row("X.sr fsr", bad or hx(d[0x10]), "d")
row("X.sr acceptld", ("unarmed" if not armed else hx(d[0x1C])), "66666666")
# non-aborting X controls: guest-warm + c8, L2 anchor at index 2, TTB switch + c8
code = [W["nop"], W["ldr_r3_r8"], str_r3(0x2C)] + ARM + [
    W["ldr_r3_r8"], str_r3(0x20),            # c8 load-bearing: must be the REMAPPED word
    W["ldr_r3_r9"], str_r3(0x24),            # L2 anchor (index 2)
    W["mcr_c2_r7"], W["mcr_c8_r0"], W["ldr_r3_r8"], str_r3(0x28),   # TTB switch
    mov_r3(0xC2), str_r3(0x38), W["b_self"]]
t, alive, armed, d, fl = run("ctl2", code, regs0 + [("r8", 0x00200000), ("r9", 0x00402000),
                                                    ("r7", TTB2)], len(code) + 1)
bad = None if (armed and alive and d[0x38] == 0xC2) else ("unarmed" if not armed else
                                                          ("dead" if not alive else "nomarker"))
row("X.guestwarm pre", bad or hx(d[0x2C]), "22222222")
row("X.guestwarm post", bad or hx(d[0x20]), "33333333")
row("X.l2anchor", bad or hx(d[0x24]), "77202000")
row("X.ttbswitch", bad or hx(d[0x28]), "77777777")
# TTB written WITH low bits: the c2 write mask is what keeps the walk on 0x4000
t, alive, armed, d, fl = run("c2low", ctl, [("r1", TTB | 0x3FF8), ("r2", 1), ("r4", DEST),
                                             ("r8", 0x00200000), ("r9", 0x00400000), ("fp", LIVE)],
                             len(ctl) + 1)
bad = None if (armed and alive and d[0x38] == 0xC1) else ("unarmed" if not armed else
                                                          ("dead" if not alive else "nomarker"))
row("X.c2low remap", bad or hx(d[0x28]), "33333333")

nb = sum(1 for n, g, w in rows if n.startswith("B.") and g == w)
tb = sum(1 for n, g, w in rows if n.startswith("B."))
nx = sum(1 for n, g, w in rows if n.startswith("X.") and g == w)
tx = sum(1 for n, g, w in rows if n.startswith("X."))
print("KT label=%s binary=%s sessions=%d" % (LABEL, BIN, NS[0]))
for n, g, w in rows:
    print("  %-22s got=%-20s want=%-12s %s" % (n, g, w, "ok" if g == w else "RED"))
print("KT_BRIEF=%d/%d KT_EXTRA=%d/%d" % (nb, tb, nx, tx))
print("KT_DONE")
