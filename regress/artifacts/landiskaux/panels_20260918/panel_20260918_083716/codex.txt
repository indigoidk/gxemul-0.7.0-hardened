## 1. SETTLED PREMISES (from pass 1, panels 20260918_080238 and _080056, adjudicated by fable5)
* The address is `devinit->addr + 0x2c`, a LITERAL byte offset: NetBSD and OpenBSD landisk `wdc_obio.c` define
  `WDC_OBIO_AUXREG_OFFSET 0x2c` and map `or_addr + WDC_OBIO_AUXREG_OFFSET` unscaled; `AUXREG_SIZE` = 2. `0x16` is
  macppc's factorisation (`0x16 << 4 = 0x160`), not landisk's. Codex's disassembly of the 7.6 ramdisk: `add #44,r5`.
* Boot census (four runs, two instruments): OpenBSD 7.6 makes exactly 12 byte writes to 0x1400002c, 0 reads; the
  first is 0x0c = SRST|4BIT. `wdc_altstatus` at 0x14000206 receives ZERO accesses.
* Whole-machine collision check: landisk registers nothing between `wdc` (0x14000000..0x1400000f) and RAM at
  0x1e000000 except the old altstatus at 0x14000206. 0x1400002c..2d is free.
* The pre-switch `alt_status_addr = devinit->addr + 0x206;` is UNTOUCHED and still serves every machine without a case.
* The switch keys on machine_type: HPCMIPS 1004, MACPPC 2004, IQ80321 5005, LANDISK 6004 — no cross-arm reach.
* The fix DELIVERS the five WDCTL_IDS (nIEN) writes; the aux handler acts on WDCTL_RST only, so nIEN is NOT implemented.

    pid, fd = pty.fork()
    if pid == 0:
        os.execv(BIN, [BIN, "-V", "-E", "landisk", "-M", "64", KERNEL])
        os._exit(127)
    buf = ""

    def rd(t):
        nonlocal buf
        end = time.time() + t
        while time.time() < end:
            r, _, _ = select.select([fd], [], [], 0.25)
            if not r:
                continue
            try:
                d = os.read(fd, 65536)
            except OSError:
                return False
            if not d:
                return False
            buf += d.decode("latin-1", "replace")
        return True

    alive = rd(settle)
    for i, c in enumerate(cmds):
        try:
            os.write(fd, (c + "\n").encode("latin-1"))
        except OSError:
            alive = False
            break
        if not rd(spin if i == len(cmds) - 1 else 0.35):
            alive = False
            break
    try:
        os.write(fd, b"quit\n")
        rd(0.6)
    except OSError:
        pass
    try:
        os.close(fd)
    except OSError:
        pass
    try:
        os.waitpid(pid, 0)
    except OSError:
        pass
    return buf, alive


def printed(t, reg):
    m = re.search(r"print\s+%s\s*\r?\n\s*(0x[0-9a-f]+)" % reg, t)
    return m.group(1) if m else "absent"


def prog(ctrl, ctrlval=0x0c, ndata=8):
    """Fill the ring, one byte store of ctrlval to ctrl, then read status and both aux addresses."""
    ops = ([MOV_W_R8_AT_R7] * ndata
           + [MOV_B_R9_AT_R10, MOV_B_AT_R1_R2, MOV_B_AT_R3_R4, MOV_B_AT_R5_R6, BRA_SELF, NOP])
    cmds = ["r7=0xb4000000", "r8=0x1234", "r10=0x%x" % ctrl, "r9=0x%x" % ctrlval,
            "r1=0xb400000e", "r3=0xb400002c", "r5=0xb4000206",
            "r2=0xaa", "r4=0xaa", "r6=0xaa"]
    cmds += ["put h 0x%x, 0x%04x" % (AT + 2 * i, op) for i, op in enumerate(ops)]
    cmds += ["pc=0x%x" % AT, "step %d" % (ndata + 4), "print r2", "print r4", "print r6"]
    return cmds

# What this protects

# truncated. Truncation is round-toward-zero, which is the mode the SH-4 resets into, so
# there was no visible defect until you look at a guest that CHANGES the mode -- and

#
