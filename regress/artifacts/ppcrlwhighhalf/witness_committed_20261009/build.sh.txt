#!/bin/bash
# Build committed HEAD (from a git archive tar, no working-tree content) in a WSL /tmp clone for the ppcrlwhighhalf witness.
set -u
TAR="$1"; W=/tmp/gxppcwit
rm -rf "$W"; mkdir -p "$W"; cd "$W" || exit 3
tar -xf "$TAR" || { echo "TAR_FAILED"; exit 3; }
./configure > configure.log 2>&1; echo "CONFIGURE_RC=$?"
make -j2 > make.log 2>&1; echo "MAKE_RC=$?"
if [ -x gxemul ] && [ gxemul -nt src/cpus/cpu_ppc_instr.c ]; then echo "BINARY_FRESH=yes"; else echo "BINARY_FRESH=no"; fi
echo "cpu_ppc_instr.c md5: $(md5sum src/cpus/cpu_ppc_instr.c | cut -c1-8)"
ls -la --time-style=+%F_%T gxemul src/cpus/cpu_ppc_instr.c 2>&1
echo "PPCWIT_BUILD_DONE"
