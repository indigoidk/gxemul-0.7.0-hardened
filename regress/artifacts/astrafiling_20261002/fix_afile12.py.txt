#!/usr/bin/env python3
"""fix_afile12.py -- GPT-6 Astra's afile_12: "0 files wrongly in the list; 0 files wrongly left out; 1 sentences
wrong" -- the CHANGELOG block's "The files." lists five of the six tracked files the commit changes and leaves out
CHANGELOG.md itself ("should also name `CHANGELOG.md` -- prepends this #554 block to the existing changelog").  The
bullet is added in its words.  A Write-tool file: no shell layer touches the text."""
import io
import os
import py_compile

P = os.path.join(os.path.dirname(os.path.abspath(__file__)), "prep554.py")
s = io.open(P, encoding="utf-8", newline="").read()
old = "**The files.**\n* `OUTSTANDING_BUGS.md` -- the 2026-10-02 `#554` section:"
new = ("**The files.**\n* `CHANGELOG.md` -- this `#554` block, prepended to the existing changelog.\n"
       "* `OUTSTANDING_BUGS.md` -- the 2026-10-02 `#554` section:")
if s.count(old) != 1:
    raise SystemExit("REFUSED: %d matches" % s.count(old))
io.open(P, "w", encoding="utf-8", newline="").write(s.replace(old, new))
py_compile.compile(P, doraise=True)
print("prep554.py: The files now names CHANGELOG.md; it compiles")
