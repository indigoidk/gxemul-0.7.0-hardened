def table():
    rows = []
    for l in io.open(os.path.join(F, "FILING.tsv"), encoding="utf-8"):
        if l.startswith("#") or not l.strip():
            continue
        w, where, key, kind, sev, ident, title, src = l.rstrip("\n").split("\t")
        rows.append(dict(wave=w, where=where, key=key, kind=kind, sev=sev, id=ident, title=title, src=src, by=""))
    ov = os.path.join(F, "FILING_OVERLAY.tsv")
    applied = []
    if os.path.isfile(ov):
        for n, l in enumerate(io.open(ov, encoding="utf-8"), 1):
            if l.startswith("#") or not l.strip():
                continue
            ident, field, value, who = (l.rstrip("\n").split("\t") + ["", "", ""])[:4]
            if field == "ADD":
                w, where, key, kind, sev, title = value.split("|", 5)
                if any(r["id"] == ident for r in rows):
                    sys.exit("REFUSED: overlay line %d ADDs %s, which the table has" % (n, ident))
                rows.append(dict(wave=w, where=where, key=key, kind=kind, sev=sev, id=ident, title=title, src="(added by the review pass)", by=who))
            else:
                hit = [r for r in rows if r["id"] == ident]
                if len(hit) != 1:
                    sys.exit("REFUSED: overlay line %d names %s, found %d times" % (n, ident, len(hit)))
                if field == "DROP":
                    rows.remove(hit[0])
                elif field == "note":
                    hit[0]["by"] = (hit[0]["by"] + "; " if hit[0]["by"] else "") + value
                elif field in ("sev", "title", "id", "kind"):
                    hit[0][field] = value
                    hit[0]["by"] = (hit[0]["by"] + "; " if hit[0]["by"] else "") + "%s corrected: %s" % (field, who)
                else:
                    sys.exit("REFUSED: overlay line %d: unknown field %r" % (n, field))
            applied.append("%s %s" % (ident, field))
    ids = [r["id"] for r in rows]
    dup = sorted(set(i for i in ids if ids.count(i) > 1))
    if dup or "?" in ids:
        sys.exit("REFUSED: duplicate or missing ids after the overlay: %s" % (dup or "?"))
    return rows, applied

def regrade_bugs(text):
    """HEAD's OUTSTANDING_BUGS.md with the re-graded ids' headings corrected in place; returns (text, report lines)."""
    lines, rep = text.split("\n"), []
    for rid, new, why in REGRADES:
        n = 0
        for i, l in enumerate(lines):
            if not re.match(r"^#{2,4} ", l):
                continue
            m = re.search(r"(`%s` \()(LOW|MED|HIGH|low|med|high)\b" % re.escape(rid), l)
            if m and m.group(2).upper() != new:
                lines[i] = l[:m.start(2)] + "%s%s; it said %s until #554 re-graded it, 2026-10-02" % (
                    new, ", the grade disputed between two seats" if rid in DISPUTED_REGRADES else "", m.group(2)) + l[m.end(2):]
                n += 1
        rep.append("%s -> %s: %d heading(s) corrected in place" % (rid, new, n))
    return "\n".join(lines), rep

def ledger():
    raw = open(P, "rb").read()
    L = json.loads(raw.decode("utf-8"))
    dump = lambda x: json.dumps(x, indent=1, ensure_ascii=False).replace("\n", "\r\n").encode("utf-8")
    if dump(L) != raw:
        sys.exit("REFUSED: the ledger does not round-trip byte-identical; its format changed")
    filed = json.load(io.open(os.path.join(REC, "filed554.json"), encoding="utf-8"))
    ids = {x["id"] for x in L["rows"]}
    clash = [r["id"] for r in filed if r["id"] in ids]
    if clash:
        sys.exit("REFUSED: %d ids already exist: %s" % (len(clash), " ".join(clash[:8])))
    for rid, new, why in REGRADES:
        r = [x for x in L["rows"] if x["id"] == rid]
        if len(r) != 1:
            sys.exit("REFUSED: re-grade target %s found %d times" % (rid, len(r)))
        old = str(r[0].get("sev"))
        r[0]["sev"] = new
        r[0]["state_note"] = ("[%s] RE-GRADED %s -> %s by #554 from the Astra review campaign (%s; the files are under regress/artifacts/"
                              "astraretro_20260926/ and astraretro_harness_20260926/).  " % (DATE, old, new, why)) + (r[0].get("state_note") or "")
    L["rows"].extend(filed)
    open(P, "wb").write(dump(L))
    print("ledger: +%d rows filed by #554 (%d rows now); %d existing rows re-graded" % (len(filed), len(L["rows"]), len(REGRADES)))


def setcommits():
    """setcommits <row> <sha> [...]: record a commit on existing rows (a previous round's pending record; this unit's
    own closed row once its sha exists)."""
    raw = open(P, "rb").read()
    L = json.loads(raw.decode("utf-8"))
    dump = lambda x: json.dumps(x, indent=1, ensure_ascii=False).replace("\n", "\r\n").encode("utf-8")
    if dump(L) != raw:
        sys.exit("REFUSED: the ledger does not round-trip byte-identical; its format changed")
    a = sys.argv[2:]
    for rid, sha in zip(a[0::2], a[1::2]):
        r = [x for x in L["rows"] if x["id"] == rid]
        if len(r) != 1:
            sys.exit("REFUSED: row %s found %d times" % (rid, len(r)))
        r[0]["commits"] = sha
    open(P, "wb").write(dump(L))
    print("ledger: commits recorded -- " + ", ".join("%s %s" % p for p in zip(a[0::2], a[1::2])))


if __name__ == "__main__":
