#!/usr/bin/env python3
"""Assemble #554's COMMITTED versions into <records>/stage554 -- no write to the working tree -- and write
spec554.json (the per-unit committer's spec, with the artifact file list) and msg554.txt.

usage: prep554.py <afile_06 verdict + adjudication> <afile_11 verdict> <grok's second HIGH re-read, one line> <grok's read of the three added rows, one line> <afile_12 verdict> [--dry]

#554 (astrafile) is a RECORDS unit: it files the rows of the Astra review campaign and commits the campaign's
artifacts.  Every number in the texts is derived here (records554.table(), the dispositions files, the file walk), never
typed.  Refuses when HEAD is not #529's commit, when the receipt or the hook edit is not what this unit expects, or
when an artifact text file carries a NUL byte (the hook's section Y would refuse the commit anyway)."""
import difflib
import hashlib
import io
import json
import os
import re
import subprocess
import sys
import time

REC = os.path.dirname(os.path.abspath(__file__))
sys.path.insert(0, REC)
import records554 as R

SEC = R.SEC
ST = os.path.join(REC, "stage554")
ARTS = os.path.join(SEC, "regress", "artifacts")
DIRS = ["astraretro_20260926", "astraretro_harness_20260926", "grokjobs_20260930", "agypack_20261001",
        "codexpack_20260928", "codexfarm_20260926", "astrafiling_20261002"]
TEXT_EXT = (".c", ".h", ".py", ".sh", ".md", ".json", ".txt", ".tsv")
SHA529 = "7440e129"


def git(*a, binary=False):
    r = subprocess.run(["git", "-C", SEC] + list(a), capture_output=True)
    if r.returncode:
        sys.exit("REFUSED: git %s: %s" % (" ".join(a), r.stderr.decode("utf-8", "replace")))
    return r.stdout if binary else r.stdout.decode("utf-8")


def md5(b):
    return hashlib.md5(b).hexdigest()[:8]


def put(rel, data):
    p = os.path.join(ST, *rel.split("/"))
    os.makedirs(os.path.dirname(p), exist_ok=True)
    open(p, "wb").write(data)
    return p


def counts(rows):
    c = {}
    for r in rows:
        c[(r["wave"], r["sev"])] = c.get((r["wave"], r["sev"]), 0) + 1
    return c


def main():
    dry = "--dry" in sys.argv
    args = [a for a in sys.argv[1:] if a != "--dry"]
    if len(args) != 5:
        sys.exit(__doc__)
    astra, closing, grokhigh2, grok3, closing2 = args
    head = git("rev-parse", "--short=8", "HEAD").strip()
    if not git("log", "-1", "--format=%s", "HEAD").startswith("#529 "):
        sys.exit("REFUSED: HEAD %s is not #529's commit" % head)
    if os.path.isdir(ST):
        for root, dirs, files in os.walk(ST, topdown=False):
            for f in files:
                os.remove(os.path.join(root, f))
            for d in dirs:
                os.rmdir(os.path.join(root, d))
    os.makedirs(ST, exist_ok=True)
    man = []
    allrows, applied = R.table()
    rows = [r for r in allrows if r["id"] not in R.CLOSED]      # the OPEN rows; the one this commit fixes is filed closed
    if len(allrows) - len(rows) != len(R.CLOSED):
        sys.exit("REFUSED: the row this commit closes is not in the table")
    c = counts(rows)
    nrow = len(rows)
    low = dict((w, sum(1 for r in rows if r["wave"] == w and r["sev"] == "LOW")) for w in ("retro", "harness"))
    nind = nrow - low["retro"] - low["harness"]
    kinds = dict((k, sum(1 for r in rows if r["kind"] == k)) for k in ("NEW", "GAP", "ID"))
    high = sum(v for (w, s), v in c.items() if s == "HIGH")
    disp = [r["id"] for r in rows if r["sev"] == "HIGH" and "HIGH IS DISPUTED" in r["by"]]
    nregr = sum(1 for x in R.REGRADES if x[2].startswith("retro"))
    nregh = sum(1 for x in R.REGRADES if x[2].startswith("harness"))
    held10 = [x for x in disp if x not in ("vgadirtyarm", "gdromcondhow")]
    if nregr + nregh != len(R.REGRADES) or "vgadirtyarm" not in disp or "gdromcondhow" not in disp or len(held10) != 10 \
       or tuple(R.DISPUTED_REGRADES) != ("ptywaitstatus",):
        sys.exit("REFUSED: re-grades %d + %d of %d, or the disputed HIGH rows are not the twelve the texts name: %s"
                 % (nregr, nregh, len(R.REGRADES), disp))

    # 1. the artifact file list: everything under the campaign's directories except the seats' stderr streams
    files, skipped, sk_bytes, nul, empty = [], 0, 0, [], []
    complete = re.compile(r"seat answered|SEAT WALLED|REFUSED|The answer is LOST|PANEL_COMPLETE|rc=")
    now = time.time()
    for d in DIRS:
        for root, _, fs in os.walk(os.path.join(ARTS, d)):
            for f in sorted(fs):
                p = os.path.join(root, f)
                if f.endswith(".err"):
                    skipped += 1
                    sk_bytes += os.path.getsize(p)
                    continue
                if os.path.getsize(p) == 0:
                    empty.append(os.path.relpath(p, SEC).replace(os.sep, "/"))     # the committer refuses an empty file
                    continue
                if now - os.path.getmtime(p) < 120 and not dry:
                    sys.exit("REFUSED: %s changed in the last two minutes -- a job in flight?" % p)
                if f.endswith(".launch.log") and not complete.search(io.open(p, encoding="utf-8", errors="replace").read()):
                    sys.exit("REFUSED: %s has no completion line -- a job in flight?" % p)
                if f.endswith(TEXT_EXT) and b"\0" in open(p, "rb").read():
                    nul.append(os.path.relpath(p, SEC))
                files.append(os.path.relpath(p, SEC).replace(os.sep, "/"))
    if nul:
        sys.exit("REFUSED: %d artifact text file(s) carry a NUL byte (the hook's section Y): %s" % (len(nul), "; ".join(nul[:6])))
    tracked = set(git("ls-files", "regress/artifacts").split("\n"))
    files = [f for f in files if f not in tracked]
    empty = [f for f in empty if f not in tracked]
    man.append("artifacts: %d files to add under %d directories; %d *.err streams (%.0f MB) and %d zero-byte files left on "
               "disk, untracked" % (len(files), len(DIRS), skipped, sk_bytes / 1048576.0, len(empty)))

    # 2. the receipt for #529 (the working file must be HEAD's plus exactly that one line) and the hook edit
    tsv_w = open(os.path.join(SEC, "regress", "pass2_ledger.tsv"), "rb").read().decode("utf-8")
    tsv_h = git("show", "HEAD:regress/pass2_ledger.tsv")
    extra = tsv_w[len(tsv_h):] if tsv_w.startswith(tsv_h) else None
    if extra is None or extra.count("\n") != 1 or not extra.startswith(SHA529 + "\t"):
        sys.exit("REFUSED: the working pass2_ledger.tsv is not HEAD's plus one line for %s" % SHA529)
    put("regress/pass2_ledger.tsv", tsv_w.encode("utf-8"))
    hk_w = open(os.path.join(SEC, "tools", "hooks", "round_reminder.sh"), "rb").read().decode("utf-8")
    hk_h = git("show", "HEAD:tools/hooks/round_reminder.sh")
    dd = [l for l in difflib.unified_diff(hk_h.split("\n"), hk_w.split("\n"), lineterm="", n=0) if l[:1] in "+-" and not l.startswith(("+++", "---"))]
    if len(dd) != 4 or "5.6-SOL" in hk_w or "./panel.sh for six" in hk_w:
        sys.exit("REFUSED: the working round_reminder.sh is not HEAD's with its two JSON lines edited (%d diff lines)" % len(dd))
    put("tools/hooks/round_reminder.sh", hk_w.encode("utf-8"))

    # 3. OUTSTANDING_BUGS.md: the re-graded headings in place, then the section
    ob, rrep = R.regrade_bugs(git("show", "HEAD:OUTSTANDING_BUGS.md"))
    nhead = sum(int(re.search(r": (\d+) heading", x).group(1)) for x in rrep)
    body = io.open(os.path.join(REC, "bugs554.body.md"), encoding="utf-8", newline="").read()
    sec = SECTION.format(nrow=nrow, nind=nind, lr=low["retro"], lh=low["harness"], high=high,
                         rh=c.get(("retro", "HIGH"), 0), rm=c.get(("retro", "MED"), 0), hh=c.get(("harness", "HIGH"), 0),
                         hm=c.get(("harness", "MED"), 0), nnew=kinds["NEW"], ngap=kinds["GAP"], nid=kinds["ID"],
                         nreg=len(R.REGRADES), nhead=nhead, ndisp=len(disp), nregr=nregr, nregh=nregh,
                         hdisputed=", ".join("`%s`" % x for x in held10),
                         regr="; ".join("`%s` -> %s (%s)" % x for x in R.REGRADES),
                         peer="; ".join("`%s` -> %s (%s)" % x for x in R.PEER_REGRADES)) + body
    put("OUTSTANDING_BUGS.md", (ob.rstrip("\n") + "\n" + sec).encode("utf-8"))
    man += ["bugs: " + x for x in rrep if not x.endswith(": 0 heading(s) corrected in place")]

    # 4. ledger.json and matrix.html (HEAD's ledger + #529's sha + this unit's rows and re-grades)
    lp = put("tools/pipeline/ledger.json", git("show", "HEAD:tools/pipeline/ledger.json", binary=True))
    env = dict(os.environ, LEDGER_PATH=lp)
    for a in (["setcommits", "crimewidthkill", SHA529], ["ledger"]):
        r = subprocess.run([sys.executable, os.path.join(REC, "records554.py")] + a, env=env, capture_output=True, text=True)
        if r.returncode:
            sys.exit("REFUSED: records554 %s: %s%s" % (a[0], r.stdout, r.stderr))
        man.append(r.stdout.strip())
    # every re-grade is APPLIED in what gets committed -- asserted, because the closing check of the repairs (afile_06)
    # read the committed tree, found `ptywaitstatus` still MED there, and said so
    staged = dict((x["id"], x) for x in json.loads(open(lp, "rb").read().decode("utf-8"))["rows"])
    obs = open(os.path.join(ST, "OUTSTANDING_BUGS.md"), "rb").read().decode("utf-8")
    for rid, new, why in R.REGRADES:
        if str(staged[rid].get("sev")) != new:
            sys.exit("REFUSED: the staged ledger does not carry %s -> %s" % (rid, new))
        # EVERY heading that prints a grade for the id must print the new one: the first form of this check passed as
        # soon as ONE heading did, so a correct heading could mask a stale sibling (GPT-6 Astra, afile_09)
        stale = [m.group(0)[-70:] for m in re.finditer(r"(?m)^#{2,4} [^\n]*?`%s` \((LOW|MED|HIGH|low|med|high)\b" % re.escape(rid), obs)
                 if m.group(1).upper() != new]
        if stale:
            sys.exit("REFUSED: %d staged heading(s) of %s still print the old grade: ...%s" % (len(stale), rid, stale[0]))
    man.append("re-grades: all %d applied in the staged ledger; every heading that prints a grade for one of them prints the new one" % len(R.REGRADES))
    mp = os.path.join(ST, "tools", "pipeline", "matrix.html")
    r = subprocess.run([sys.executable, os.path.join(SEC, "tools", "pipeline", "gen_dashboard.py"), "--out", mp],
                       env=dict(os.environ, GXLEDGER=lp), capture_output=True, text=True)
    if r.returncode or not os.path.isfile(mp):
        sys.exit("REFUSED: gen_dashboard: %s%s" % (r.stdout, r.stderr))

    # 5. CHANGELOG, the message
    retro_n = sum(int(x) for x in re.findall(r"(?m)^## Batch \S+ -- [^\n]*?\((\d+) findings\)", io.open(os.path.join(ARTS, DIRS[0], "DISPOSITIONS.md"), encoding="utf-8").read()))
    ht = io.open(os.path.join(ARTS, DIRS[1], "HARNESS_DISPOSITIONS.md"), encoding="utf-8").read()
    harn_n = sum(int(x) for x in re.findall(r"(?m)^## Group \d\d -- (\d+) findings", ht))
    hk = re.search(r"Governing kinds over \d+ findings: ([^\n]*)\.", ht).group(1)
    fmt = dict(ndisp=len(disp), nrow=nrow, nind=nind, lr=low["retro"], lh=low["harness"], high=high, retro_n=retro_n, harn_n=harn_n, hk=hk,
               nfiles=len(files), nerr=skipped, nempty=len(empty), errmb="%.0f" % (sk_bytes / 1048576.0), nreg=len(R.REGRADES), nhead=nhead,
               nov=len(applied), astra=astra, closing=closing, grokhigh2=grokhigh2, grok3=grok3, closing2=closing2, nregr=nregr, nregh=nregh, nnew=kinds["NEW"], ngap=kinds["GAP"], nid=kinds["ID"],
               nledger=nind + 2 + len(R.CLOSED))
    blk = CHANGELOG.format(**fmt)
    put("CHANGELOG.md", (blk.rstrip("\n") + "\n\n" + git("show", "HEAD:CHANGELOG.md")).encode("utf-8"))
    io.open(os.path.join(REC, "msg554.txt"), "w", encoding="utf-8", newline="\n").write(MESSAGE.format(**fmt))
    # absolute paths and a list of argv lists, as commit_unit.py reads them (spec529.json is the working precedent)
    spec = {"unit": "#554", "head_subject_re": "^#529 ", "stage_dir": ST,
            "files": ["CHANGELOG.md", "OUTSTANDING_BUGS.md", "regress/pass2_ledger.tsv", "tools/hooks/round_reminder.sh",
                      "tools/pipeline/ledger.json", "tools/pipeline/matrix.html"],
            "neutral": ["tools/pipeline/check_probe_wiring.py"], "add_dirs": [], "add_files": files, "exclude_dirs": [],
            "message": os.path.join(REC, "msg554.txt"), "sidecars": [], "pre": [],
            "post": [["python", os.path.join(REC, "post554.py")]],
            "no_merge": ["tools/pipeline/ledger.json", "tools/pipeline/matrix.html"]}
    io.open(os.path.join(REC, "spec554.json"), "w", encoding="utf-8", newline="\n").write(json.dumps(spec, indent=1) + "\n")
    for t in (sec, blk, MESSAGE.format(**fmt)):
        if "<OWED" in t or "{" in t and re.search(r"\{[a-z_]+\}", t):
            sys.exit("REFUSED: a placeholder survives in a generated text")

    # 6. manifest
    for root, _, fs in os.walk(ST):
        for f in sorted(fs):
            p = os.path.join(root, f)
            rel = os.path.relpath(p, ST).replace(os.sep, "/")
            if rel == "MANIFEST.txt":
                continue
            new = open(p, "rb").read()
            r0 = subprocess.run(["git", "-C", SEC, "show", "HEAD:" + rel], capture_output=True)
            old = r0.stdout if r0.returncode == 0 else b""
            d2 = list(difflib.unified_diff(old.decode("utf-8", "replace").splitlines(), new.decode("utf-8", "replace").splitlines(), lineterm="", n=0))
            man.append("%s  %-44s +%d -%d  (HEAD %s)" % (md5(new), rel, sum(1 for l in d2 if l.startswith("+") and not l.startswith("+++")),
                                                        sum(1 for l in d2 if l.startswith("-") and not l.startswith("---")), md5(old)))
    man.insert(0, "# #554 staged committed versions -- HEAD %s%s" % (head, "  (DRY)" if dry else ""))
    io.open(os.path.join(ST, "MANIFEST.txt"), "w", encoding="utf-8", newline="\n").write("\n".join(man) + "\n")
    print("\n".join(man))
