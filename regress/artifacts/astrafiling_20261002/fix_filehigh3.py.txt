import io

import py_compile

def patch(path, edits):
    s = io.open(path, encoding="utf-8", newline="").read()
    for old, new in edits:
        if s.count(old) != 1:
            raise SystemExit("REFUSED: %s: %d matches for %r" % (os.path.basename(path), s.count(old), old[:70]))
        s = s.replace(old, new)
    io.open(path, "w", encoding="utf-8", newline="").write(s)
    py_compile.compile(path, doraise=True)
    print("%s: %d edits; it compiles" % (os.path.basename(path), len(edits)))
