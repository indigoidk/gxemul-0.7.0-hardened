def agy_verdicts(name, table):
    """(line number, 'Q1: .. | Q2: .. | Q3: ..'): agy's OWN three verdict words for a row, cut out of its line"""
    lines = io.open(os.path.join(os.path.dirname(OUT), "agy", table), encoding="utf-8").read().split("\n")
    hit = [(n, l) for n, l in enumerate(lines, 1) if l.startswith(name + " |")]
    if len(hit) != 1:
        raise SystemExit("REFUSED: %d lines of %s for %s" % (len(hit), table, name))
    heads = [re.match(r"\s*(Q[123]: [A-Z/ ]+?)\s*(?:--|$)", p) for p in hit[0][1].split("|")[1:4]]
    if not all(heads):
        raise SystemExit("REFUSED: agy's line for %s does not carry three verdicts" % name)
    return hit[0][0], " | ".join(h.group(1) for h in heads)
