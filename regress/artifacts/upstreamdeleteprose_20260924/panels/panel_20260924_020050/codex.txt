 256 def bad_commits(rows):
 257     """Rows whose `commits` is a placeholder rather than hashes or a stated non-repo note."""
 258     out = []
 259     for rid, row in sorted(rows.items()):
 260         c = (row.get("commits") or "").strip()
 261         if not c or NONREPO.search(c):
 262             continue
 263         for tok in c.split():
 264             if PLACEHOLDER.match(tok) or not re.fullmatch(r"[0-9a-f]{7,40}", tok):
 265                 out.append((rid, row.get("state", "?"), c))
 266                 break
 267     return out
