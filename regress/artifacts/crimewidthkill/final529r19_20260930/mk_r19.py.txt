def md5(s):
    return hashlib.md5(s.encode("utf-8")).hexdigest()[:8]


def strip_doc(t):
    if t.body and isinstance(t.body[0], ast.Expr) and isinstance(getattr(t.body[0], "value", None), ast.Constant) \
            and isinstance(t.body[0].value.value, str):
        t.body = t.body[1:]
    return t

    sys.exit("r18 is not 81f8894e")

       "     arm's diagnostic, bypassing CONTROL (this said \"does nothing\" until 2026-09-28).  A round that fixes that lane\n")

    sys.exit("REFUSED: the r18 sentence is not unique")

    sys.exit("REFUSED: r19's AST minus its docstring differs from r18's")
