{
	char buf[DEBUG_BUFSIZE + 1];
	buf[0] = buf[DEBUG_BUFSIZE] = 0;

	while (*s) {

			for (size_t i = 0; i < debug_indent * DEBUG_INDENTATION; i++)
				printf(" ");
		}

		printf("%c", *s);

		debug_old_currently_at_start_of_line = false;
		if (*s == '\n' || *s == '\r')
			debug_old_currently_at_start_of_line = true;
		s++;
	}
