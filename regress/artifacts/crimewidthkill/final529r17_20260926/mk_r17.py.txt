def skeleton(src):
    t = ast.parse(src)
    if isinstance(t.body[0], ast.Expr) and isinstance(t.body[0].value, ast.Constant):
        t.body = t.body[1:]
    for n in ast.walk(t):
        if isinstance(n, ast.Call) and getattr(n.func, "id", None) == "row" and len(n.args) >= 4:
            a = n.args[3]
            if isinstance(a, ast.BinOp) and isinstance(a.left, ast.Constant):
                a.left = ast.Constant("NOTE")
            elif isinstance(a, ast.Constant):
                n.args[3] = ast.Constant("NOTE")
    return ast.dump(t)
