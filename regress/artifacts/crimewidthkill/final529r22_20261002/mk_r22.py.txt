def md5(s):
    return hashlib.md5(s.encode("utf-8")).hexdigest()[:8]


def strip_doc(t):
    if t.body and isinstance(t.body[0], ast.Expr) and isinstance(getattr(t.body[0], "value", None), ast.Constant) \
            and isinstance(t.body[0].value.value, str):
        t.body = t.body[1:]
    return t


s = io.open(R21, encoding="utf-8", newline="").read()
if md5(s) != "8da85812":
    sys.exit("r21 is not 8da85812")
PAIRS = [
    ("`sw +0x04': F3, 157/157 -- guest-equivalent;",
     "`sw +0x04': F3, 157/157 on the 157-row test -- guest-equivalent;"),
]
r22 = s
for old, new in PAIRS:
    if r22.count(old) != 1:
        sys.exit("REFUSED: find-text not unique (%d): %r" % (r22.count(old), old[:60]))
    r22 = r22.replace(old, new)
if ast.dump(strip_doc(ast.parse(s))) != ast.dump(strip_doc(ast.parse(r22))):
    sys.exit("REFUSED: r22's AST minus its docstring differs from r21's")
# the edited line must sit inside the module docstring (the checker reads the DOCSTRING), not in a comment or a string

    sys.exit("REFUSED: the edit did not land in the module docstring")

print("r22 %s  (r21 %s)  %s" % (md5(r22), md5(s), OUT))
