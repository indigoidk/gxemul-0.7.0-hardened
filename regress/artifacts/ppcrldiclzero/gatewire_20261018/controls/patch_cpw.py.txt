#!/usr/bin/env python3
"""patch_cpw.py <in> <out> -- #565's three edits to tools/pipeline/check_probe_wiring.py: the MANIFEST comment on
ppc_rldicl_probe.py gains a dated WIRED note; its EXEMPT tuple and the comment block above it are deleted; and a
DISCHARGED note joins the earlier ones.  Every anchor must match exactly once; applied alike to HEAD's copy (the commit)
and to the working-tree copy (which carries other uncommitted lines)."""
import io, sys
src, dst = sys.argv[1], sys.argv[2]
t = io.open(src, encoding="utf-8", newline="").read()
def once(a, b, what):
    global t
    assert t.count(a) == 1, (what, t.count(a))
    t = t.replace(a, b)
once("    #  regress/artifacts/astraretro_harness_20260926/HARNESS_DISPOSITIONS.md)\n    \"ppc_rldicl_probe.py\":\n",
     "    #  regress/artifacts/astraretro_harness_20260926/HARNESS_DISPOSITIONS.md)\n"
     "    #  [#565, 2026-10-18] WIRED: gate_ppc.sh (gate 13) now runs it and scores every row by name, with its run, green,\n"
     "    #  red and SCORE counts pinned; its EXEMPT entry is DISCHARGED below.  \"EXEMPT below rather than wired\" above is\n"
     "    #  historical.\n"
     "    \"ppc_rldicl_probe.py\":\n", "MANIFEST note")
a = t.index("    #  #522's detector.  Single-leg, 0.2 s, green on the fix.  It lacks a gate that RUNS a\n")
end_tok = "approach, not a prerequisite; filed to wiringcliff\"),\n"
b = t.index(end_tok, a) + len(end_tok)
assert t.count("    #  #522's detector.  Single-leg, 0.2 s, green on the fix.") == 1
assert t[a:b].count('"ppc_rldicl_probe.py":') == 1 and b - a < 3000, (b - a)
t = t[:a] + t[b:]
once("    #  fbpending_drain_probe.py's exemption is DISCHARGED, 2026-09-24 (#548) -- wired into\n"
     "    #  gate_arm.sh beside footbridge_sites_probe.py, all four timers, --expect-cap 762.\n"
     "    #  Deleted in the SAME edit as the wiring, for the same reason.\n",
     "    #  fbpending_drain_probe.py's exemption is DISCHARGED, 2026-09-24 (#548) -- wired into\n"
     "    #  gate_arm.sh beside footbridge_sites_probe.py, all four timers, --expect-cap 762.\n"
     "    #  Deleted in the SAME edit as the wiring, for the same reason.\n"
     "    #  ppc_rldicl_probe.py's exemption is DISCHARGED, 2026-10-18 (#565) -- wired into gate_ppc.sh beside the\n"
     "    #  rlwimi/rldimi blocks on the same -E bareppc -M 64, every row scored by name.  It came due 2026-10-22 after\n"
     "    #  two re-dates by owner ruling.  Deleted in the SAME edit as the wiring, for the same reason.\n",
     "DISCHARGED note")
assert '"ppc_rldicl_probe.py":         ("2026-10-22",' not in t
io.open(dst, "w", encoding="utf-8", newline="").write(t)
print("patched", dst)
