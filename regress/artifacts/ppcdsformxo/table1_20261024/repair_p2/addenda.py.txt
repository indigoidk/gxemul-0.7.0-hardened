def append(path, block):
    t = io.open(path, encoding="utf-8", newline="").read()
    nl = "\r\n" if "\r\n" in t else "\n"
    if not t.endswith(nl):
        t += nl
    t += nl + block.replace("\n", nl)
    io.open(path, "w", encoding="utf-8", newline="").write(t)
    print("%s: +%d lines (%s)" % (os.path.basename(path), block.count("\n"), "CRLF" if nl == "\r\n" else "LF"))

def fix(path, pairs):
    t = io.open(path, encoding="utf-8", newline="").read()
    for old, new in pairs:
        n = t.count(old); assert n == 1, "%s: %r occurs %d times" % (path, old[:60], n)
        t = t.replace(old, new)
    io.open(path, "w", encoding="utf-8", newline="").write(t); print("%s: %d replacements" % (os.path.basename(path), len(pairs)))
