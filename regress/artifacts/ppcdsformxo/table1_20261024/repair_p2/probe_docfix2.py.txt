"""#571 F2 (Astra AMEND F19/F26 + Grok AMEND F19): the SECOND docstring-only revision of regress/ppc_dsform_probe.py -- the
sign-joint bullet's warrant replaced (the bodies COPY the DS sign into imm and never branch on it, and they do not read RA;
Grok's wording) and the two remaining unmentioned non-floor classes disclosed (stdu with RA = RS at DS = 0; the reserved
encodings at a positive DS; Astra's F26).  Asserts the input is revision 1 (59e83491...), prints the diff, verifies
code-identity, writes probe_docfix2.diff beside the script."""
import difflib, hashlib, io, os, re, shutil, sys

SEC = "C:/DocumentNoSnc/cc/GXEMUL/GXEMUL-SEC"
P = SEC + "/regress/ppc_dsform_probe.py"
HERE = os.path.dirname(os.path.abspath(__file__))
old = io.open(P, encoding="utf-8", newline="").read()
assert hashlib.md5(old.encode("utf-8")).hexdigest() == "59e83491a5819a599c610b1aa8e8b847", "not revision 1"
shutil.copy2(P, os.path.join(HERE, "ppc_dsform_probe.py.rev1_59e83491"))
a = ("    * the sign of DS crossed with RA -- a negative DS at RA = 0 and a positive DS at RA != 0\n"
     "      for ld, ldu, std and stdu, and any negative DS for lwa -- and the reserved encodings at\n"
     "      DS = 0 or RA = 0: the case bodies read neither the sign of DS nor RA, so these are not\n"
     "      floor cells; not rowed;\n")
b = ("    * the sign of DS crossed with RA -- a negative DS at RA = 0 and a positive DS at RA != 0\n"
     "      for ld, ldu, std and stdu, and any negative DS for lwa -- the reserved encodings at\n"
     "      DS = 0, at RA = 0 or at a positive DS, and stdu with RA = RS at DS = 0: the case bodies\n"
     "      copy the sign of DS into imm and never branch on it, and they do not read RA (the\n"
     "      common tail tests only imm == 0 and ra == 0 && update), so these are not floor cells;\n"
     "      not rowed -- remaining non-floor coverage debt;\n")
assert old.count(a) == 1
new = old.replace(a, b)
io.open(P, "w", encoding="utf-8", newline="").write(new)
diff = "".join(difflib.unified_diff(old.splitlines(True), new.splitlines(True),
                                    "a/regress/ppc_dsform_probe.py (revision 1, md5 59e83491a5819a599c610b1aa8e8b847)",
                                    "b/regress/ppc_dsform_probe.py (revision 2)", n=2))
io.open(os.path.join(HERE, "probe_docfix2.diff"), "w", encoding="utf-8", newline="").write(diff)
sys.stdout.write(diff)
strip = lambda t: "\n".join(l for l in re.sub(r'(?s)^#!.*?"""(.*?)"""', '"""DOC"""', t, count=1).splitlines() if not l.strip().startswith("#"))
assert strip(old) == strip(new), "a non-docstring, non-comment line changed"
print("NEW md5 %s  lines %d  bytes %d" % (hashlib.md5(new.encode("utf-8")).hexdigest(), new.count("\n"), len(new.encode("utf-8"))))
print("CODE UNCHANGED: docstring-only edit verified")
