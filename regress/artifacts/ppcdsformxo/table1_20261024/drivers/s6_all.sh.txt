#!/bin/bash
#  T2 (#571) spec section 6: every kill-table leg in order, each through s6_leg.sh (frozen copies), then L5b.
#  Foreground, one leg at a time.  Console goes to table1/s6_all.out via the caller.
set -u
SP=/mnt/c/Users/User00/AppData/Local/Temp/claude/C--DocumentNoSnc-cc-GXEMUL/8e474f41-4d47-4c99-bf11-3406a84c4fcd/scratchpad/t2/build
SEC=/mnt/c/DocumentNoSnc/cc/GXEMUL/GXEMUL-SEC
T=$SEC/regress/artifacts/ppcdsformxo/table1_20261024
STAMP=$(date +%H%M%S)
cp "$SP/s6_leg.sh" "$SP/s6_leg.run-$STAMP.sh"; cp "$SP/legs571.py" "$SP/legs571.run-$STAMP.py"
echo "=== $(date '+%F %T') S6 table: frozen driver copies s6_leg.run-$STAMP.sh / legs571.run-$STAMP.py"
md5sum "$SP/legs571.run-$STAMP.py" "$SEC/regress/ppc_dsform_probe.py"
for pair in "3 armHEAD" "4b case58HEAD" "4c case62HEAD" "5 identity" "6a noupd58" "6b noupd62" "7 rawimm" "8a lwaxo0flags" "8b lwazero1" "8c lwasize3zero0" "9a xo0asxo1" "9b xo0plus1" "10 updall" "11 sharedmask" "12a xo623open" "12b xo583open"; do
  set -- $pair
  sed "s#\$SP/legs571.py#$SP/legs571.run-$STAMP.py#" "$SP/s6_leg.run-$STAMP.sh" > "$SP/s6_leg.cur.sh"
  bash "$SP/s6_leg.cur.sh" "$1" "$2"
  echo "--------------------------------------------------------------------------------"
done
echo "=== $(date '+%F %T') L5b: the binary absent"
python3 -u "$SEC/regress/ppc_dsform_probe.py" /tmp/gxt2/NO_SUCH_BINARY/gxemul L5b_absent "$T" > "$T/L5b_absent.out" 2>&1
echo "L5b_RC=$?"; tail -n 3 "$T/L5b_absent.out"; grep -c 'PPC_DSFORM_PROBE_DONE' "$T/L5b_absent.out" | sed 's/^/L5b sentinel count: /'
echo "=== $(date '+%F %T') legs dir after the table"
ls -la /tmp/gxt2/legs 2>/dev/null
echo "TABLE_DONE"
