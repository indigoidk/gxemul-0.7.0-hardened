#!/bin/bash
# fable5 REGRESS seat (8n devices batch): build the five trees SEQUENTIALLY from git-archive tarballs
# in a private WSL scratch dir.  Touches NO project tree.
set -u
S=/mnt/c/Users/User00/AppData/Local/Temp/claude/C--DocumentNoSnc-cc-GXEMUL/8e474f41-4d47-4c99-bf11-3406a84c4fcd/scratchpad/fable_rgd
D=/tmp/gxfable_rgd
rm -rf "$D"; mkdir -p "$D"
build() {   # label sha devfile
  local L=$1 SHA=$2 DEV=$3
  mkdir -p "$D/$L"
  echo "=== extract $SHA -> $D/$L $(date +%T)"
  tar -xf "$S/work/$SHA.tar" -C "$D/$L"; echo "TAR_RC=$?"
  echo "=== configure $L $(date +%T)"
  (cd "$D/$L" && ./configure > configure.log 2>&1; echo "CONFIGURE_RC=$?")
  echo "=== make $L $(date +%T)"
  (cd "$D/$L" && make -j4 > make.log 2>&1; echo "MAKE_RC=$?")
  ls -la --time-style=full-iso "$D/$L/gxemul" "$D/$L/$DEV" 2>&1
  if [ "$D/$L/gxemul" -nt "$D/$L/$DEV" ]; then echo "$L: binary NEWER than $DEV  OK"; else echo "$L: binary NOT newer than $DEV  FAIL"; fi
  echo "$L devfile md5: $(md5sum "$D/$L/$DEV")"
  echo "$L make.log lines matching 'error': $(grep -ci 'error' "$D/$L/make.log")"
}
build sgi_commit  7a41b3f  src/devices/dev_sgi_re.c
build sgi_parent  a264a34d src/devices/dev_sgi_re.c
build land_commit e16a8b7  src/devices/dev_wdc.c
build land_parent 686d179c src/devices/dev_wdc.c
build pcc2_commit bb8efbf  src/devices/dev_pcc2.c
echo "BUILD_DONE $(date +%T)"
