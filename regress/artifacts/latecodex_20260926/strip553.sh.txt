#!/bin/bash
#  #551i: the comment-stripped preprocessor output of #551f/g/h, raw and whitespace-normalized; #551g's record gives
#  c10d453c for f, g and i (HEAD).  #553's MINE_j.c must give the same.
cd /mnt/c/Users/User00/AppData/Local/Temp/claude/C--DocumentNoSnc-cc-GXEMUL/8e474f41-4d47-4c99-bf11-3406a84c4fcd/scratchpad/latecodex553 || exit 9
for f in MINE_i.c MINE_j.c; do
    a=$(gcc -fpreprocessed -dD -E -P "$f" 2>/dev/null | md5sum | cut -c1-8)
    b=$(gcc -fpreprocessed -dD -E -P "$f" 2>/dev/null | tr -s ' \t\n' ' ' | md5sum | cut -c1-8)
    echo "$f raw=$a normalized=$b"
done
echo STRIP553_DONE
