#!/usr/bin/env python3
"""mk_553.py -- #553 latecodex's two comment-only file edits, built from HEAD (never from the working tree, which
carries #529's lines in the checker): the flagship's replacement texts (latecodex_20260926/fable5_latecodex.md, L1+L2
into regress/diff_wdc_identify.c's contract (B); L6 and L7 into tools/pipeline/check_probe_wiring.py's #550 block).
Writes MINE_j.c and cpw553.py beside this script; refuses unless every find-text is unique and every changed line is a
comment line.  One wording change from the ruling's text: "three tallies" -> "three tally targets" (two tallies and
"both" -- the seat's own words)."""
import difflib, hashlib, io, os, subprocess, sys, textwrap

HERE = os.path.dirname(os.path.abspath(__file__))
SEC = r"C:\DocumentNoSnc\cc\GXEMUL\GXEMUL-SEC"


def head(rel):
    return subprocess.run(["git", "-C", SEC, "show", "HEAD:" + rel], capture_output=True, check=True).stdout.decode("utf-8")


def md5(s):
    return hashlib.md5(s.encode("utf-8")).hexdigest()[:8]


def rep(s, old, new, what):
    if s.count(old) != 1:
        sys.exit("%s: find-text found %d times" % (what, s.count(old)))
    return s.replace(old, new)


def cwrap(text, width=110):
    return "".join(" *  " + l + "\n" for l in textwrap.wrap(text, width=width - 4, break_on_hyphens=False,
                                                          break_long_words=False))


# ---- diff_wdc_identify.c ------------------------------------------------------------------------------------------
c = head("regress/diff_wdc_identify.c")
if md5(c) != "3580dd0a":
    sys.exit("HEAD's diff_wdc_identify.c is not #551i (3580dd0a)")
OLD_C = (" *  reading `0 failures'.  Hence no\n"
         " *  edit that leaves the covered rows running and the comparators' compiled behaviour intact hides a covered\n"
         " *  failure: not the trailer's count, token or exit rewritten singly or together, not a reset of either\n"
         " *  tally or both at any placement, not rows moved or deferred past the verdict with their counts forged\n"
         " *  (measured on #551f's pass: regress/artifacts/wdcidselfcheck_20260925/measured/run551f.out, scored by an\n"
         " *  EMULATION of gate 2's checks and lane; opus551_measured_f/analyze9.out, its 41 rows re-run with gate 2's\n"
         " *  own lib.sh; gate 2's own selfmutant.py on five of those cells -- this said \"gate 2's own lib.sh and\n"
         " *  selfmutant.py\" of all of it: GPT-6 Astra, confirming #551).  Not caught -- the (A) residual\n")
NEW_TEXT = ("reading `0 failures'.  Hence no edit that leaves the covered rows running, their INPUTS and the comparators' "
            "compiled behaviour intact hides a covered failure: not the trailer's count, token or exit rewritten singly or "
            "together, not a reset of either tally or both at any placement, not rows moved or deferred past the verdict "
            "with their counts forged.  The reset claim holds BY READING at every placement and either conditionality -- "
            "the report line is printed inside the comparator, before any statement outside it runs, and the check refuses "
            "a second captured line -- and is MEASURED for nine of the 24 reset shapes (three tally targets x "
            "conditional/unconditional x four placements: pass551f/meas551f.py.txt builds the unconditional single-tally "

            "#551f's pass: regress/artifacts/wdcidselfcheck_20260925/measured/run551f.out, scored by an EMULATION of gate "
            "2's checks and lane; opus551_measured_f/analyze9.out, its 41 rows re-run with gate 2's own lib.sh; gate 2's "
            "own selfmutant.py on five of those cells -- this said \"gate 2's own lib.sh and selfmutant.py\" of all of "
            "it: GPT-6 Astra, confirming #551).  Not caught -- the (A) residual")
c2 = rep(c, OLD_C, cwrap(NEW_TEXT), "contract (B)")

# ---- check_probe_wiring.py ------------------------------------------------------------------------------------------
w = head("tools/pipeline/check_probe_wiring.py")
w2 = rep(w, "#  These are wiring jobs, not design ones.  NOT every probe here prints a verdict token or exits\n",
         "#  Most of these are wiring jobs; two are not -- wdc_altoffset_sig_probe.py owes a redesign (its oracle cannot\n"
         "#  tell the outcomes apart) and wdc_altoffset_4bit_probe.py an oracle choice, each said at its own entry (this\n"
         "#  line said all of them were wiring jobs: GPT-6 Astra, late on the F2b pass).  NOT every probe here prints a\n"
         "#  verdict token or exits\n", "L6")
w2 = rep(w2,
         "    #  What remains is the instrument: this probe prints no RESULT/CONTROL lines.  (This sentence went\n"
         "    #  on \"so a gate could only score its exit status\" until #550: false -- it prints ok/FAIL rows,\n"
         "    #  `SCORE n/m' and a DONE line, which a gate can score by rows, as gate_arm.sh scores mk48's.)\n",
         "    #  What remains is a gate block: this probe prints ok/FAIL rows, `SCORE n/m' and a DONE line, which a\n"
         "    #  gate can score by rows, as gate_arm.sh scores mk48's; it prints no RESULT/CONTROL lines, and adding\n"

i = w2.index('    "ppc_rldicl_probe.py":         ("2026-10-08",\n')
j = w2.index("),\n", i) + 3
old_tuple = w2[i:j]
if "RESULT/CONTROL protocol a gate parses" not in old_tuple:
    sys.exit("L7 tuple: not the expected text: %r" % old_tuple[:200])
new_tuple = ('    "ppc_rldicl_probe.py":         ("2026-10-08",\n'

for name, old, new, ok in (("MINE_j.c", c, c2, lambda l: l.lstrip().startswith(("/*", "*", "*/")) or not l.strip()),

                            or "ppc_rldicl_probe.py" in l or l.strip().startswith('"'))):
    changed = [l[1:] for l in difflib.unified_diff(old.split("\n"), new.split("\n"), lineterm="", n=0)
               if l[:1] in "+-" and not l.startswith(("+++", "---"))]
    bad = [l for l in changed if not ok(l)]
    if bad:
        sys.exit("%s: a non-comment line changed: %r" % (name, bad[0]))
    io.open(os.path.join(HERE, name), "w", encoding="utf-8", newline="").write(new)
    print("wrote %s %s (%d changed lines)" % (name, md5(new), len(changed)))
print("old tuple was:\n" + old_tuple)
