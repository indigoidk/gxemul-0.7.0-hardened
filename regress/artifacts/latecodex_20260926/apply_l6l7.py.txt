#!/usr/bin/env python3
"""apply_l6l7.py <checker.py> -- apply #553's two comment corrections (L6, L7) to a copy of check_probe_wiring.py that
carries other units' lines (the working tree's), exactly as mk_553.py applied them to HEAD's; verify the result's L6/L7
region equals cpw553.py's.  Used after #553's commit to keep the working checker = committed + #529's pending lines."""
import io, os, sys

HERE = os.path.dirname(os.path.abspath(__file__))
p = sys.argv[1]
s = io.open(p, encoding="utf-8", newline="").read()
ref = io.open(os.path.join(HERE, "cpw553.py"), encoding="utf-8", newline="").read()
head_l6 = "#  These are wiring jobs, not design ones.  NOT every probe here prints a verdict token or exits\n"
new_l6 = ref[ref.index("#  Most of these are wiring jobs"):ref.index("#  verdict token or exits\n") + len("#  verdict token or exits\n")]
if s.count(head_l6) != 1:
    sys.exit("REFUSED: L6 anchor found %d times" % s.count(head_l6))
s = s.replace(head_l6, new_l6)
old_p = ("    #  What remains is the instrument: this probe prints no RESULT/CONTROL lines.  (This sentence went\n"
         "    #  on \"so a gate could only score its exit status\" until #550: false -- it prints ok/FAIL rows,\n"
         "    #  `SCORE n/m' and a DONE line, which a gate can score by rows, as gate_arm.sh scores mk48's.)\n")
a = ref.index("    #  What remains is a gate block")
new_p = ref[a:ref.index("GPT-6 Astra, late on the F2b pass.)\n", a) + len("GPT-6 Astra, late on the F2b pass.)\n")]
if s.count(old_p) != 1:
    sys.exit("REFUSED: L7 paragraph found %d times" % s.count(old_p))
s = s.replace(old_p, new_p)
key = '    "ppc_rldicl_probe.py":         ("2026-10-08",\n'
i = s.index(key); j = s.index("),\n", i) + 3
ri = ref.index(key); rj = ref.index("),\n", ri) + 3
s = s[:i] + ref[ri:rj] + s[j:]
io.open(p, "w", encoding="utf-8", newline="").write(s)
print("applied L6/L7 to", p)
