#!/usr/bin/env python3
"""#550: the one sibling comment in precommit_check.sh that names the exemption block's date."""
import io, sys
src, dst = sys.argv[1], sys.argv[2]
raw = io.open(src, encoding="utf-8", newline="").read()
eol = "\r\n" if "\r\n" in raw else "\n"
s = raw.replace("\r\n", "\n")
old = "#  block to 2026-09-27 by owner ruling.)\n"
new = ("#  block to 2026-09-27 by owner ruling; #550 dispersed it to ten individual dates, 2026-10-08\n"
       "#  to 2026-11-17, by owner ruling 2026-09-24.)\n")
if s.count(old) != 1:
    sys.exit("REFUSED: %d matches" % s.count(old))
s = s.replace(old, new)
io.open(dst, "w", encoding="utf-8", newline="").write(s.replace("\n", eol))
print("edit550_pcc ->", dst)
