def sub1(s, old, new):
    n = s.count(old)
    if n != 1:
        sys.exit("REFUSED: %d matches for %r" % (n, old[:80]))
    return s.replace(old, new)


def entry(key_line_old, date, why):
    """key_line_old: the exact `    "name":   ("2026-09-27",` line(s) prefix; returns (old, new)."""
    return (key_line_old, "%s%s.\n%s" % (TAG, why, key_line_old.replace('"2026-09-27"', '"%s"' % date)))
