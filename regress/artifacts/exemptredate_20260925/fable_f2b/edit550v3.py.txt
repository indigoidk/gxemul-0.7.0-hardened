    c = s.count(old)

    return s.replace(old, new)

def main():
    which, src, dst = sys.argv[1], sys.argv[2], sys.argv[3]
    raw = io.open(src, encoding="utf-8", newline="").read()
    eol = "\r\n" if "\r\n" in raw else "\n"
    s = raw.replace("\r\n", "\n")
    for old, new, n in (CPW if which == "cpw" else GOFF):
        s = subn(s, old, new, n)
    io.open(dst, "w", encoding="utf-8", newline="").write(s.replace("\n", eol))
    print("edit550v3", which, "->", dst)


if __name__ == "__main__":
    main()
