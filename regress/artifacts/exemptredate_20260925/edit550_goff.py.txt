#!/usr/bin/env python3
"""#550: gate 2's SC_EXEMPT -- ieee_store re-dated with its own reason (owner ruling 2026-09-25); wdc_identify NOT
re-dated, because the owner ruled it FIXED (#551).  usage: python edit550_goff.py <in gate_offline.sh> <out>"""
import io
import sys

src, dst = sys.argv[1], sys.argv[2]
raw = io.open(src, encoding="utf-8", newline="").read()
eol = "\r\n" if "\r\n" in raw else "\n"
s = raw.replace("\r\n", "\n")
EDITS = [
    ('#  and RED from 2026-09-28.  Owner row: `scexempt` -- discharge, re-decide or delete before then.\n'
     'SC_EXEMPT="ieee_store:2026-09-27 wdc_identify:2026-09-27"\n',
     '#  and RED from 2026-09-28.  Owner row: `scexempt` -- discharge, re-decide or delete before then.\n'
     '#  #550 (2026-09-25), OWNER RULING 2026-09-25 -- the two are decided SEPARATELY, not as a pair.\n'
     '#    ieee_store    RE-DATED to 2026-11-30, its reason above unchanged: it has no comparator at all,\n'
     '#                  so a sentinel needs its six inline counters moved onto one first -- the larger\n'
     '#                  refactor, dated after the wdc work so it falls due as its own decision.\n'
     '#    wdc_identify  NOT re-dated: the owner ruled it FIXED -- #551 moves its rows onto a shared\n'
     '#                  comparator with a sentinel and moves the stem to SC_COVERED before 2026-09-28.\n'
     '#                  Until #551 lands, its 2026-09-27 date stands and reds from the 28th.\n'
     'SC_EXEMPT="ieee_store:2026-11-30 wdc_identify:2026-09-27"\n'),
]
for old, new in EDITS:
    if s.count(old) != 1:
        sys.exit("REFUSED: %d matches for %r" % (s.count(old), old[:70]))
    s = s.replace(old, new)
io.open(dst, "w", encoding="utf-8", newline="").write(s.replace("\n", eol))
print("edit550_goff ->", dst)
