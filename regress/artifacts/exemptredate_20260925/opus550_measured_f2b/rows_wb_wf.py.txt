#!/usr/bin/env python3
"""opus550 f2b: evaluate wdc_altoffset_probe.py's ROWS table with the probe's own constants (only the code before
ROWS is executed -- definitions and constants, no session) and count rows whose committed-leg want (wb) differs from
the fix-leg want (wf).  A row with wb == wf is satisfied by the fix binary on both legs."""
import re, sys
src = open(sys.argv[1], encoding="utf-8").read()
m = re.search(r"^ROWS = \[", src, re.M)
depth, end = 0, None
for i in range(m.end() - 1, len(src)):
    if src[i] == "[": depth += 1
    elif src[i] == "]":
        depth -= 1
        if depth == 0:
            end = i + 1; break
ns = {"__name__": "rowscount"}
sys.argv = [sys.argv[0]]
exec(compile(src[:m.start()], "pre_rows", "exec"), ns)
rows = eval(src[m.start():end].split("=", 1)[1], ns)
same = [r for r in rows if r[2] == r[3]]
diff = [r for r in rows if r[2] != r[3]]
print("ROWS entries: %d   wb == wf: %d   wb != wf: %d" % (len(rows), len(same), len(diff)))
for r in diff:
    print("  wb != wf: %-58s wb=0x%02x wf=0x%02x" % (r[0][:58], r[2], r[3]))
print("ROWS_WB_WF_DONE")
