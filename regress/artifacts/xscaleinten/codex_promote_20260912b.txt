    def wait_prompt(self, timeout=30, mark=None):
        if mark is None:
            mark = len(self.buf)
        self.ready = False
        deadline = time.monotonic() + timeout
        while not self.dead:
            remaining = deadline - time.monotonic()
            if remaining <= 0:
