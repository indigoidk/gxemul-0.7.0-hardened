#!/usr/bin/env python3
# Make the pre-#473 parent into "re-arm fix ONLY": the re-arm's `= 0` becomes `= 32768`,
# nothing else -- no clamp move, no refill.  Exactly one site, or exit 3.
import io, sys
path = sys.argv[1]
TAB = chr(9)
old = "if (d->atapi_len > 32768)\n" + TAB * 7 + "d->atapi_len = 0;"
new = "if (d->atapi_len > 32768)\n" + TAB * 7 + "d->atapi_len = 32768;"
with io.open(path, "r", encoding="latin-1", newline="") as f:
    src = f.read()
n = src.count(old)
if n != 1:
    sys.stderr.write("rearm_only: count=%d (want 1)\n" % n)
    sys.exit(3)
with io.open(path, "w", encoding="latin-1", newline="") as f:
    f.write(src.replace(old, new))
print("REARM_ONLY applied")
