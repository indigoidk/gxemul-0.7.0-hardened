#!/usr/bin/env python3
# HEAD with #495's write-arm guard nulled (`if (flag && phase != CMDOUT)` -> `if (0)`),
# i.e. the verbatim reinstatement of the defect #495 closed.  Exactly one site, or exit 3.
import io, sys
path = sys.argv[1]
T = chr(9)
old = T*3 + "if (d->atapi_cmd_in_progress &&\n" + T*3 + "    d->atapi_phase != PHASE_CMDOUT) {\n"
new = T*3 + "if (0) {\n"
with io.open(path, "r", encoding="latin-1", newline="") as f:
    src = f.read()
n = src.count(old)
if n != 1:
    sys.stderr.write("no495: count=%d (want 1)\n" % n)
    sys.exit(3)
with io.open(path, "w", encoding="latin-1", newline="") as f:
    f.write(src.replace(old, new))
print("NO495 applied")
