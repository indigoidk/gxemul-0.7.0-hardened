#!/usr/bin/env python3
# Replace EXACTLY ONE occurrence of OLD with NEW in FILE.  Any other count is a FAIL
# (exit 3), so a mutant whose find-text is absent or ambiguous can never be scored
# as "survived" -- it turns every find-text into a presence pin.
import io, sys
path, old, new = sys.argv[1], sys.argv[2], sys.argv[3]
with io.open(path, "r", encoding="latin-1", newline="") as f:
    src = f.read()
n = src.count(old)
if n != 1:
    sys.stderr.write("MUTANT-TEXT count=%d (want 1) for %r\n" % (n, old))
    sys.exit(3)
with io.open(path, "w", encoding="latin-1", newline="") as f:
    f.write(src.replace(old, new))
print("MUTATED 1 site")
