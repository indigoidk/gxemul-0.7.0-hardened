   ```c
   {
   	uint32_t iw = ic->arg[0];
   	int rs = (iw >> 21) & 31, ra = (iw >> 16) & 31;
   	int sh = ((iw >> 11) & 31) | ((iw & 2) << 4);
   	int mb = ((iw >> 6) & 31) | (iw & 0x20);
   	int rc = ic->arg[0] & 1;
   	int m, n = sh;
   	uint64_t tmp, s = cpu->cd.ppc.gpr[rs];

   		s = (s << 1) | b;
   	}
   	m = mb; tmp = 0;
   	do {
   		tmp |= ((uint64_t)1 << (63-m));
   		m = (m + 1) & 63;
   	} while (m != ((64 - sh) & 63));

   	cpu->cd.ppc.gpr[ra] |= (tmp & s);

   ```
