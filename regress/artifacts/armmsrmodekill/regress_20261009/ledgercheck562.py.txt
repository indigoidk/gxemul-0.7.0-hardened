import sys, json, re
diff = open(sys.argv[1], encoding="utf-8", errors="replace").read().splitlines()
# reconstruct added JSON objects crudely: group '+' lines between '+    {' and '+    },' / '+    }'
objs=[]; cur=None
for l in diff:
    if not l.startswith('+'): continue
    s=l[1:]
    if re.match(r'^\s{4}\{\s*$', s): cur=[]
    elif cur is not None and re.match(r'^\s{4}\},?\s*$', s):
        objs.append('\n'.join(cur)); cur=None
    elif cur is not None: cur.append(s)
seats={}; nodir=[]
for o in objs:
    m=re.search(r'"seat":\s*"([^"]+)"', o)
    if not m: continue
    seat=m.group(1); seats[seat]=seats.get(seat,0)+1
    if seat=="fable5": continue
    if not re.search(r'panel_(pass1|pass2|pass2r|pass2r3|pass2r4|pass2r5|f2)_2026100[789]|opus_pass2|witness_20261007', o):
        nodir.append(o[:160].replace('\n',' '))
print("added seat entries:", sum(seats.values()), seats)
print("non-fable5 entries naming no panel/opus dir:", len(nodir))
for n in nodir: print("  ", n)
# per panel dir: which seats cite it
for d in ("panel_pass1_20261007","panel_f2_20261007","panel_pass2_20261007","panel_pass2r_20261007","panel_pass2r3_20261008","panel_pass2r4_20261008","panel_pass2r5_20261009"):
    ss=sorted(set(re.search(r'"seat":\s*"([^"]+)"',o).group(1) for o in objs if d in o and re.search(r'"seat":',o)))
    print("  %-26s %d seats: %s" % (d, len(ss), ",".join(ss)))
