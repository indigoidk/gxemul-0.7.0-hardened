#!/bin/bash
# gate14.sh -- #562: the two-part clearance of every build tree, then gate 14 ALONE on the working tree.
SEC=/mnt/c/DocumentNoSnc/cc/GXEMUL/GXEMUL-SEC
FILES="src/cpus/cpu_arm.c src/cpus/cpu_arm_instr.c src/cpus/cpu_arm_instr_dpi.c src/include/cpu_arm.h"
echo "== two-part clearance $(date '+%F %T')"
for T in /mnt/c/DocumentNoSnc/cc/GXEMUL/build /tmp/gxsec-build /mnt/c/DocumentNoSnc/cc/GXEMUL/build-asan; do
    [ -e "$T/.MUTANT" ] && echo "  $T: SENTINEL PRESENT"
    for f in $FILES; do
        cmp -s "$SEC/$f" "$T/$f" && p1=ok || p1=FAIL
        [ "$T/gxemul" -nt "$T/$f" ] && p2=ok || p2=FAIL
        echo "  $T $f part1=$p1 part2=$p2"
    done
done
for f in $FILES; do cmp -s "$SEC/$f" "/mnt/c/DocumentNoSnc/cc/GXEMUL/est/$f" && echo "  est $f part1=ok part2=N/A (source tree)" || echo "  est $f part1=FAIL"; done
echo "== gate 14 $(date '+%F %T')  probe md5 $(md5sum $SEC/regress/arm_msrmode_probe.py | cut -c1-8)  gate md5 $(md5sum $SEC/regress/gate_arm.sh | cut -c1-8)"
export LOGDIR=/tmp/gx562gate14
rm -rf "$LOGDIR"; mkdir -p "$LOGDIR"
bash "$SEC/regress/gate_arm.sh"
echo "GATE14_RC=$?"
echo "GATE14_DONE $(date '+%F %T')"
