def git(*a, binary=False):
    r = subprocess.run(["git", "-C", SEC] + list(a), capture_output=True)
    if r.returncode:
        sys.exit("REFUSED: git %s: %s" % (" ".join(a), r.stderr.decode("utf-8", "replace")))
    return r.stdout if binary else r.stdout.decode("utf-8")


def md5(b):
    return hashlib.md5(b).hexdigest()[:8]


def rd(p):
    return io.open(p, encoding="utf-8", errors="replace").read().replace("\r\n", "\n")


def put(rel, data):
    p = os.path.join(ST, *rel.split("/"))
    os.makedirs(os.path.dirname(p), exist_ok=True)
    open(p, "wb").write(data)
    return p


def verdict(p):
    v = re.findall(r"VERDICT: ([^\n]*)", rd(p))
    if not v:
        sys.exit("REFUSED: no VERDICT line in %s" % p)
    return v[0].replace("`", "").strip().rstrip(".")


def answers():
    """one dict per agy brief: verdict, flags (PARTLY/OWED item numbers as the answer gives them), size, walled"""
    out = {}
    for a in ANSWERS:
        txt = os.path.join(D, a, "agy_%s.txt" % a)
        err = os.path.join(D, a, "agy_%s.err" % a)
        if a in WALLED:
            e = rd(err) if os.path.isfile(err) else ""
            if "quota" not in e:
                sys.exit("REFUSED: %s is recorded as WALLED but its .err carries no quota text" % a)
            out[a] = dict(verdict="SEAT WALLED (agy quota: %s)" % re.search(r"Resets in [0-9hms]+", e).group(0), flags=[], size=0, walled=True,
                          err_line=e.strip().split("\n")[0][:160])
            continue
        t = rd(txt)
        v = verdict(txt)
        n = int(re.match(r"(\d+) of", v).group(1))
        if a.startswith("ifiled"):
            flags = [int(m.group(1)) for m in re.finditer(r"^(\d+) \| .*?\| COVERS: PARTLY", t, re.M)]
        elif a.startswith("ikeep"):
            flags = [int(m.group(1)) for m in re.finditer(r"^(\d+) \| OWED:", t, re.M)]
        else:
            flags = [int(m.group(1)) for m in re.finditer(r"^(\d+) \| .*?\b(WRONG|FLAG|MISCITED|NOT FOUND)\b", t, re.M)]
        if len(flags) != n:
            sys.exit("REFUSED: %s: verdict says %d flagged, %d flag lines found: %r" % (a, n, len(flags), flags))
        out[a] = dict(verdict=v, flags=flags, size=len(t.encode("utf-8")), walled=False)
    if {k: v["flags"] for k, v in out.items() if v["flags"]} != {**PARTLY, **OWED}:
        sys.exit("REFUSED: the flags in the answers are not the ones the disposition file records: %r" % {k: v["flags"] for k, v in out.items()})
    return out

def main():
    dry = "--dry" in sys.argv
    args = [a for a in sys.argv[1:] if a != "--dry"]
    if len(args) != 1:
        sys.exit(__doc__)
    astra = args[0]
    head = git("rev-parse", "--short=8", "HEAD").strip()
    if not git("log", "-1", "--format=%s", "HEAD").startswith("#557 "):
        sys.exit("REFUSED: HEAD %s is not #557's commit" % head)
    if os.path.isdir(ST):
        for root, dirs, files in os.walk(ST, topdown=False):
            for f in files:
                os.remove(os.path.join(root, f))
            for d in dirs:
                os.rmdir(os.path.join(root, d))
    os.makedirs(ST, exist_ok=True)
    man = ["# #558 staged committed versions -- HEAD %s%s" % (head, "  (DRY)" if dry else "")]
    ans = answers()
    for a in ANSWERS:
        man.append("agy %-10s %s%s" % (a, ans[a]["verdict"][:70], ("  flags " + str(ans[a]["flags"])) if ans[a]["flags"] else ""))
    ap = os.path.join(D, "aconf_agyflags", "codex_aconf_agyflags.txt")
    if not dry and verdict(ap) != astra:
        sys.exit("REFUSED: the verdict given (%r) is not the one in %s (%r)" % (astra, ap, verdict(ap)))

    # the artifacts (counted before the ledger so the main-loop entry can name the count)
    afiles, skipped, empty = [], 0, 0
    complete = re.compile(r"seat answered|SEAT WALLED|REFUSED|The answer is LOST|PANEL_COMPLETE|rc=")
    tracked = set(git("ls-files", "regress/artifacts").split("\n"))
    now = time.time()
    for d in DIRS:
        for root, _, fs in os.walk(os.path.join(ARTS, d)):
            for f in sorted(fs):
                p = os.path.join(root, f)
                rel = os.path.relpath(p, SEC).replace(os.sep, "/")
                keep_err = f == "agy_icite2_2.err"        # the wall is the whole answer; every other .err stays on disk
                if rel in tracked or (f.endswith(".err") and not keep_err):
                    skipped += f.endswith(".err")
                    continue
                if os.path.getsize(p) == 0:
                    empty += 1
                    continue
                if now - os.path.getmtime(p) < 120 and not dry:
                    sys.exit("REFUSED: %s changed in the last two minutes -- a job in flight?" % p)
                if f.endswith(".launch.log") and not complete.search(rd(p)):
                    if not dry:
                        sys.exit("REFUSED: %s has no completion line -- a job in flight?" % p)
                    man.append("(dry) IN FLIGHT, would refuse: %s" % rel)
                if f.endswith(TEXT_EXT) and b"\0" in open(p, "rb").read():
                    sys.exit("REFUSED: %s carries a NUL byte" % rel)
                afiles.append(rel)
    man.append("artifacts: %d files to add under %s; %d *.err and %d zero-byte files left on disk" % (len(afiles), ", ".join(DIRS), skipped, empty))

    # the ledger
    raw = git("show", "HEAD:tools/pipeline/ledger.json", binary=True)
    L = json.loads(raw.decode("utf-8"))
    dump = lambda x: json.dumps(x, indent=1, ensure_ascii=False).replace("\n", "\r\n").encode("utf-8")
    if dump(L) != raw:
        sys.exit("REFUSED: the ledger does not round-trip byte-identical")
    nent = edit_ledger(L, head, astra, ans, len(afiles))
    lp = put("tools/pipeline/ledger.json", dump(L))
    mp = os.path.join(ST, "tools", "pipeline", "matrix.html")
    r = subprocess.run([sys.executable, os.path.join(SEC, "tools", "pipeline", "gen_dashboard.py"), "--out", mp],
                       env=dict(os.environ, GXLEDGER=lp), capture_output=True, text=True)
    if r.returncode or not os.path.isfile(mp):
        sys.exit("REFUSED: gen_dashboard: %s%s" % (r.stdout, r.stderr))
    man.append("ledger: %d entries on %s; %d rows filed (%s); boilerplateprefix.commits = %s" % (nent, ROW, len(NEWROWS), ", ".join(r_[0] for r_ in NEWROWS), head))

    # the texts
    np_ = sum(len(v) for v in PARTLY.values())
    no_ = sum(len(v) for v in OWED.values())
    mbv = verdict(os.path.join(SEC, *ASTRA_MB.split("/")))
    fmt = dict(unit=UNIT, head557=head, astra=astra, nfiles=len(afiles), npartly=np_, nowed=no_, nflags=np_ + no_, mbv=mbv,
               newids=", ".join("`%s`" % r_[0] for r_ in NEWROWS),
               newrows="\n".join("#### `%s` (%s, filed by %s) -- %s\n%s" % (rid, sev, UNIT, title, note) for rid, sev, title, note in NEWROWS),
               carriers=", ".join("`%s`" % c for c in sorted(c for c in CARRIERS if c != "qhm88kstops")), live=", ".join("`%s`" % c for c in sorted(LIVE)),
               wall=ans["icite2_2"]["err_line"],
