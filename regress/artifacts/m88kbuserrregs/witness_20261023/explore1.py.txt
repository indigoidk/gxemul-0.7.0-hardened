#!/usr/bin/env python3
"""T36 exploration (not a deliverable): what do the CMMU PFSR/PFAR registers hold after a refused device
access on the committed build?  Stepped sessions, debugger fed on stdin (vme_default_probe.py's shape)."""
import os, re, subprocess, sys, tempfile

BIN = sys.argv[1]
IMAGES = sys.argv[2] if len(sys.argv) > 2 else "/mnt/c/DocumentNoSnc/CC/GXEMUL/_images"
TMP = tempfile.mkdtemp(prefix="t36x_")
DUMMY = os.path.join(TMP, "dummy.bin")
open(DUMMY, "wb").write(b"\0\0\0\0")
CODE, H = 0x10000, 0x10100


def enc(op, rd, rs1, imm):
    return (op << 26) | (rd << 21) | (rs1 << 16) | imm


def LD(rd, rs, off): return enc(0x05, rd, rs, off)
def ST(rd, rs, off): return enc(0x09, rd, rs, off)
def OR(rd, rs, imm): return enc(0x16, rd, rs, imm)
def BR(frm, to): return 0xc0000000 | (((to - frm) >> 2) & 0x3ffffff)
JMP = lambda rs2: 0xf400c000 | rs2


def run(machine_args, regs, blocks, nsteps, extra=()):
    cmds = ["r%d=0x%x" % (r, v) for r, v in regs]
    for at, words in blocks:
        for i, w in enumerate(words):
            cmds.append("put w 0x%x, 0x%08x" % (at + 4 * i, w))
        cmds.append("unassemble 0x%x 0x%x" % (at, at + 4 * len(words)))
    cmds += ["pc=0x%x" % CODE, "step %d" % nsteps, "reg", "reg 0,0"] + list(extra) + ["quit"]
    argv = [BIN, "-V", "-x"] + machine_args
    p = subprocess.run(argv, input=("\n".join(cmds) + "\n").encode(), stdout=subprocess.PIPE,
                       stderr=subprocess.STDOUT, timeout=120, cwd=IMAGES)
    return p.returncode, p.stdout.decode("latin-1", "replace").replace("\r", "")


def program(cmmu_d, cmmu_i, access):
    vec = [(0x10, [OR(7, 0, 2), BR(0x14, H)]), (0x18, [OR(7, 0, 3), BR(0x1c, H)])]
    code = [LD(10, 3, 0x0),            # IDR (device signature)
            ST(20, 3, 0x108), ST(21, 3, 0x10c), ST(22, 2, 0x108), ST(23, 2, 0x10c),
            LD(11, 3, 0x108), LD(12, 3, 0x10c),
            access,
            OR(6, 0, 0x1234), BR(CODE + 4 * 9, H)]
    hnd = [LD(13, 3, 0x108), LD(14, 3, 0x10c), LD(15, 2, 0x108), LD(16, 2, 0x10c), OR(31, 0, 0x5678),
           0xc0000000]
    regs = [(3, cmmu_d), (2, cmmu_i), (20, 0x5a5a5a5a), (21, 0x13572468), (22, 0x2468ace0), (23, 0x0badf00d),
            (4, 0xdeadbeef), (6, 0xdeadbeef), (7, 0xdeadbeef), (10, 0xdeadbeef), (11, 0xdeadbeef),
            (12, 0xdeadbeef), (13, 0xdeadbeef), (14, 0xdeadbeef), (15, 0xdeadbeef), (16, 0xdeadbeef),
            (31, 0xdeadbeef)]
    return vec + [(CODE, code), (H, hnd)], regs


MV = ["-E", "mvme88k", "-e", "mvme187", "-M", "64", "0x%x:%s" % (CODE, DUMMY)]
LU_DUMMY = ["-E", "luna88k", "-e", "luna-88k", "0x%x:%s" % (CODE, DUMMY)]
LU_RIG = ["-e", "luna-88k", "-d", "R:" + IMAGES + "/liveimage-luna88k-raw-20250518.img", "boot"]

cases = [
    ("mvme187 unreadable ld 0xfff46010", MV, 0xfff7f000, 0xfff77000, [(5, 0xfff46010)], LD(4, 5, 0)),
    ("mvme187 unreadable st 0xfff46010", MV, 0xfff7f000, 0xfff77000, [(5, 0xfff46010), (8, 0xa5a5a5a5)], ST(8, 5, 0)),
    ("mvme187 vme pop1 ld 0xfff40090", MV, 0xfff7f000, 0xfff77000, [(5, 0xfff40090)], LD(4, 5, 0)),
    ("mvme187 ifetch jmp 0xfff46010", MV, 0xfff7f000, 0xfff77000, [(5, 0xfff46010)], JMP(5)),
    ("luna dummy 0x3ffffff0 ld", LU_DUMMY, 0xfff06000, 0xfff07000, [(5, 0x3ffffff0)], LD(4, 5, 0)),
    ("luna rig 0x3ffffff0 ld", LU_RIG, 0xfff06000, 0xfff07000, [(5, 0x3ffffff0)], LD(4, 5, 0)),
]
for name, margs, d, i, xregs, acc in cases:
    blocks, regs = program(d, i, acc)
    rc, out = run(margs, regs + xregs, blocks, 24)
    print("=" * 30, name, "rc=%d" % rc)
    print(out[-6000:])
print("EXPLORE1_DONE")
