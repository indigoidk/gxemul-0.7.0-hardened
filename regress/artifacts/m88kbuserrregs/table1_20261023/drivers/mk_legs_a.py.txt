#!/usr/bin/env python3
"""T36 / #570 kill table -- variant maker, group A (reinstatements, selector, outer guard): L3 L4 L6 L6b L7 L19.
Each variant is written to variants/<leg>/<basename>; every replacement must match exactly once (else refuse)."""
import io
import os
import sys

SEC = "C:/DocumentNoSnc/cc/GXEMUL/GXEMUL-SEC"
HERE = os.path.dirname(os.path.abspath(__file__))
OUT = os.path.join(HERE, "variants")
T = "\t"
MRW, LUNA = "src/cpus/memory_rw.c", "src/devices/dev_luna88k.c"


def rd(p):
    with io.open(p, "r", encoding="latin-1", newline="") as f:
        return f.read()


def put(leg, rel, text):
    d = os.path.join(OUT, leg)
    os.makedirs(d, exist_ok=True)
    with io.open(os.path.join(d, os.path.basename(rel)), "w", encoding="latin-1", newline="") as f:
        f.write(text)
    print("WROTE %-4s %s" % (leg, rel))


def one(leg, rel, old, new):
    src = rd(os.path.join(SEC, rel))
    n = src.count(old)
    if n != 1:
        print("REFUSE %s: anchor found %d times" % (leg, n))
        sys.exit(1)
    put(leg, rel, src.replace(old, new))


put("L3", MRW, rd(os.path.join(HERE, "head4", "memory_rw.c")))      # the arm, verbatim from HEAD
put("L4", LUNA, rd(os.path.join(HERE, "head4", "dev_luna88k.c")))   # the LUNA case, verbatim from HEAD
SEL = T * 6 + "    CACHE_INSTRUCTION ? 0 : 1];"
one("L6", MRW, SEL, T * 6 + "    CACHE_INSTRUCTION ? 1 : 1];")
one("L6b", MRW, SEL, T * 6 + "    CACHE_INSTRUCTION ? 0 : 0];")
GUARD = T * 4 + "if (res <= 0 && !no_exceptions) {"
one("L7", MRW, GUARD, T * 4 + "if (res <= 0) {")
one("L19", MRW, GUARD, T * 4 + "if (res <= 1 && !no_exceptions) {")
print("MK_LEGS_A_DONE")
