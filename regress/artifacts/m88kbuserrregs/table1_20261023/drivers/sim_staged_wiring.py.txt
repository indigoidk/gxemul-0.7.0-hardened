#!/usr/bin/env python3
"""T36 / #570: simulate 'the two .py files are staged' for check_probe_wiring.py WITHOUT touching any git index:
import the checker (the given copy) and patch domain() to return git ls-files' list plus the two new paths.
Read-only.  usage: sim_staged_wiring.py <checker copy.py> <GXEMUL-SEC/regress>"""
import importlib.util
import sys

spec = importlib.util.spec_from_file_location("cpw", sys.argv[1])
cpw = importlib.util.module_from_spec(spec)
spec.loader.exec_module(cpw)
orig = cpw.domain
NEW = ["m88k_buserr_probe.py", "artifacts/m88kbuserrregs/witness_20261023/m88k_buserr_probe.py"]


def domain(regress):
    files, mode = orig(regress)
    return sorted(set(files) | set(NEW)), mode + " + the two #570 files (SIMULATED staging)"


cpw.domain = domain
rc = cpw.main(["-v", "--regress", sys.argv[2]])
print("SIM_RC=%d" % rc)
