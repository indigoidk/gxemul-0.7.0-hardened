#!/bin/bash
# T36 / #570 builder -- md5 of every object and the binary in a build tree (read-only), to an output file.
# usage: obj_md5.sh <tree> <out file>
set -u
T=$1; OUT=$2
( cd "$T" && find src -name '*.o' -print0 | sort -z | xargs -0 md5sum; md5sum gxemul ) > "$OUT"
echo "objects+binary hashed: $(wc -l < "$OUT") -> $OUT"
grep -E 'dev_m8820x\.o|dev_luna88k\.o|cpu_m88k\.o| gxemul$' "$OUT"
