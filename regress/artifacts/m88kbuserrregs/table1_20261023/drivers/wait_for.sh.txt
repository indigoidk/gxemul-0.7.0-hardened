#!/bin/bash
# T36 builder: block until <pattern> appears in <file> or <max> seconds pass; then print progress context.
# usage: wait_for.sh <file> <pattern> <max seconds>
set -u
F=$1; P=$2; MAX=$3
t0=$(date +%s)
while [ $(( $(date +%s) - t0 )) -lt "$MAX" ]; do
    if grep -q -- "$P" "$F" 2>/dev/null; then echo "FOUND after $(( $(date +%s) - t0 ))s"; break; fi
    sleep 20
done
grep -q -- "$P" "$F" 2>/dev/null || echo "NOT YET after ${MAX}s"
echo "$(date '+%T')  lines=$(wc -l < "$F")  last: $(tail -1 "$F" | cut -c1-120)"
echo "  newest gate log: $(ls -t /tmp/gxt36/gatelogs14/ 2>/dev/null | head -1)"
