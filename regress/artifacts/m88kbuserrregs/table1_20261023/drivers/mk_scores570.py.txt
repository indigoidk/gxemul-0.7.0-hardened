ROW_RE = re.compile(r"^(\S+(?: \S+)?)\s{2,}.*\s(ok|FAIL)$")


def parse(path):
    with io.open(path, encoding="latin-1") as f:
        text = f.read()
    rows = {}
    for line in text.splitlines():
        if line.startswith(("INFO ", "row ", "  unasm mismatch")):
            continue
        m = re.match(r"^(.{16}) ", line)
        if m and (line.endswith(" ok") or line.endswith(" FAIL")):
            rows[m.group(1).strip()] = line.endswith(" ok")
    kv = dict(re.findall(r"^M88K_BUSERR_([A-Z]+)=(.*)$", text, re.M))
    return rows, kv, "M88K_BUSERR_PROBE_DONE" in text
