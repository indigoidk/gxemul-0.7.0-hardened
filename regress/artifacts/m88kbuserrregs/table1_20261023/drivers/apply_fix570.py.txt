for rel, edits in EDITS.items():
    p = os.path.join(ROOT, rel)
    with io.open(p, "r", encoding="latin-1", newline="") as f:
        src = f.read()
    if "\r" in src:
        print("REFUSE %s: CR characters present" % rel)
        rc = 1
        continue
    new = src
    ok = True
    for old, rep in edits:
        n = new.count(old)
        if n != 1:
            print("REFUSE %s: old text found %d times (want 1): %r" % (rel, n, old[:70]))
            ok = False
            break
        new = new.replace(old, rep)
    if ok and rel.endswith("dev_m8820x.c") and new.count("\n") != src.count("\n"):
        print("REFUSE %s: line count would change %d -> %d" % (rel, src.count("\n"), new.count("\n")))
        ok = False
    if not ok:
        rc = 1
        continue
    with io.open(p, "w", encoding="latin-1", newline="") as f:
        f.write(new)
    print("EDITED %-28s lines %d -> %d, %d edit(s)" % (rel, src.count("\n"), new.count("\n"), len(edits)))
print("APPLY_FIX570_DONE rc=%d" % rc)
sys.exit(rc)
