#!/bin/bash
# T36 / #570 builder -- A11: ONE luna rig boot to `login:' in gate_ab.sh's luna_run shape (gate_ab.sh:214-221):
#   run_emu_progress LUNA_BACKSTOP LUNA_BUDGET LUNA_STALL 'login:' <log> <binary> -N -e luna-88k -d R:<image> boot
# with gate_ab.sh's constants (:110-112: 12e9 / 120 / 1800).  lib.sh's run_emu_progress runs from $IMAGES and uses
# an R: (read-only, throwaway overlay) image, so the rig image is never written.  Foreground.
# usage: boot_ab_one.sh <gxemul> <tag> <out dir>
set -u
BIN=$1; TAG=$2; OUT=$3
REG=/mnt/c/DocumentNoSnc/CC/GXEMUL/GXEMUL-SEC/regress
. "$REG/lib.sh"
LUNA_IMG="R:$IMAGES/liveimage-luna88k-raw-20250518.img"
LUNA_BUDGET=12000000000
LUNA_STALL=120
LUNA_BACKSTOP=1800
mkdir -p "$OUT"
log=$OUT/ab_luna_$TAG.log
echo "BOOT_AB_START $TAG $(date '+%F %T %z')"
echo "binary: $(md5sum "$BIN")  mtime $(stat -c '%y' "$BIN")"
echo "image : $(ls -la "$IMAGES/liveimage-luna88k-raw-20250518.img" | awk '{print $5, $6, $7, $8, $9}')"
echo "shape : run_emu_progress $LUNA_BACKSTOP $LUNA_BUDGET $LUNA_STALL 'login:' $log $BIN -N -e luna-88k -d $LUNA_IMG boot"
out=$(run_emu_progress "$LUNA_BACKSTOP" "$LUNA_BUDGET" "$LUNA_STALL" 'login:' "$log" "$BIN" -N -e luna-88k -d "$LUNA_IMG" boot)
echo "result: $out"
echo "markers 'LUNA-88K BOOT':'M88100':'login:' = $(count "$log" 'LUNA-88K BOOT'):$(count "$log" 'M88100'):$(count "$log" 'login:')"
echo "log: $(wc -c < "$log") bytes, $(wc -l < "$log") lines; last non-blank line: $(grep -a . "$log" | tail -1 | cut -c1-120)"
echo "BOOT_AB_DONE $TAG $(date '+%F %T %z')"
