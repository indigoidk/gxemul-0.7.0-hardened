def rd(p):
    return io.open(p, encoding="utf-8", newline="").read()


def main():
    dry = "--dry" in sys.argv
    texts = {}
    bad = []
    for key, old, new in FIXES:
        for p in COPIES[key]:
            t = texts.setdefault(p, rd(p))
            crlf = "\r\n" in t
            o, n = (old.replace("\n", "\r\n"), new.replace("\n", "\r\n")) if crlf else (old, new)
            c = t.count(o)
            if c != 1:
                bad.append("%s: %r occurs %d times" % (os.path.basename(p), old[:60], c))
                continue
            texts[p] = t.replace(o, n, 1)
    if bad:
        for b in bad:
            sys.stderr.write("REFUSED: %s\n" % b)
        sys.stderr.write("NOTHING WRITTEN\n")
        return 5
    if dry:
        print("DRY RUN: %d fixes apply cleanly to %d files" % (len(FIXES), len(texts)))
        return 0
    for p, t in texts.items():
        bk = os.path.join(C, os.path.basename(os.path.dirname(p)) + "__" + os.path.basename(p) + ".prefix568")
        if not os.path.exists(bk):
            shutil.copy2(p, bk)
        io.open(p, "w", encoding="utf-8", newline="").write(t)
    print("APPLIED %d fixes to %d files" % (len(FIXES), len(texts)))
    return 0


if __name__ == "__main__":
    sys.exit(main())
