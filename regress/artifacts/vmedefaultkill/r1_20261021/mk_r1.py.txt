def rd(p):
    with io.open(p, "r", encoding="utf-8", newline="") as f:
        return f.read().replace("\r\n", "\n")


def section(text, start, stop):
    """text from the line starting with `start' up to (not including) the line starting with `stop' (or EOF)."""
    a = text.index(start)
    b = text.find(stop, a + len(start)) if stop else -1
    return text[a:b if b >= 0 else len(text)].rstrip() + "\n"


def shorten(diff, old, new):
    lines = diff.split("\n")
    assert lines[0].startswith("--- ") and lines[1].startswith("+++ "), "not a unified diff header"
    lines[0], lines[1] = "--- " + old, "+++ " + new
    return "\n".join(lines).rstrip() + "\n"
