#!/usr/bin/env python3
"""explore3.py <gxemul> -- raw announce text, for the record (not a row source).  Free-running, through a pty, the
same way the probe drives it: (A) the probe's six non-word FIRST accesses; (B) two registers read-then-written and
two written-then-read, then the same four pairs again.  Prints every `continue' segment verbatim."""
import os, pty, select, sys, tempfile, time

BIN = sys.argv[1]
TD = tempfile.mkdtemp(prefix="vmeexp3_")
DUMMY = os.path.join(TD, "dummy.bin")
open(DUMMY, "wb").write(b"\x00\x00\x00\x00")
ARGV = [BIN, "-V", "-x", "-E", "mvme88k", "-e", "mvme187", "-M", "64", "0x10000:" + DUMMY]
P = "GXemul> "


def enc(op26, rd, rs1, imm):
    return (op26 << 26) | (rd << 21) | (rs1 << 16) | imm


def ptyrun(cmds):
    pid, fd = pty.fork()
    if pid == 0:
        os.execv(BIN, ARGV)
        os._exit(127)
    buf = [""]

    def pump(t, pred):
        end = time.time() + t
        while time.time() < end:
            if pred():
                return True
            r, _, _ = select.select([fd], [], [], 0.05)
            if r:
                try:
                    d = os.read(fd, 65536)
                except OSError:
                    return False
                if not d:
                    return False
                buf[0] += d.decode("latin-1", "replace")
        return pred()

    pump(20, lambda: P in buf[0])
    for c in cmds:
        s = len(buf[0])
        os.write(fd, (c + "\n").encode())
        if c == "continue":
            pump(30, lambda s=s: buf[0].find("BREAKPOINT", s) >= 0 and
                 buf[0].find(P, buf[0].find("BREAKPOINT", s)) >= 0)
        else:
            pump(10, lambda s=s: buf[0].find(P, s) >= 0)
    os.write(fd, b"quit\n")
    pump(3, lambda: False)
    os.close(fd)
    os.waitpid(pid, 0)
    return buf[0].replace("\r", "")


def conts(t):
    return [p.partition("\n")[2] for p in t.split(P)[1:] if p.startswith("continue")]


def prog(at, words):
    return ["put w 0x%x, 0x%08x" % (at + 4 * i, w) for i, w in enumerate(words)]


SEED = ["r3=0xfff40000", "r2=0xa5a5a5a5", "r12=0xa1b2c3d4", "r10=0x11223344", "r11=0x55667788"]
print("=== (A) non-word FIRST accesses: ld.bu 0x04, ld.hu 0x0c, ld.d 0x10, st.b 0x14, st.h 0x1c, st.d 0x20")
w = [enc(0x03, 4, 3, 0x04), enc(0x02, 5, 3, 0x0c), enc(0x04, 6, 3, 0x10),
     enc(0x0b, 12, 3, 0x14), enc(0x0a, 12, 3, 0x1c), enc(0x08, 10, 3, 0x20), 0xc0000000]
t = ptyrun(SEED + prog(0x10000, w) + ["breakpoint add 0x%x" % (0x10000 + 4 * 6), "pc=0x10000", "continue"])
for s in conts(t):
    print(s, end="")
print("=== (B) pass 1: 0x04 read-then-write, 0x08 read-then-write, 0x0c write-then-read, 0x10 write-then-read")
p1 = [enc(0x05, 4, 3, 0x04), enc(0x09, 2, 3, 0x04), enc(0x05, 4, 3, 0x08), enc(0x09, 2, 3, 0x08),
      enc(0x09, 2, 3, 0x0c), enc(0x05, 4, 3, 0x0c), enc(0x09, 2, 3, 0x10), enc(0x05, 4, 3, 0x10), 0xc0000000]
t = ptyrun(SEED + prog(0x10000, p1) + prog(0x10200, p1[:-1] + [0xc0000000])
           + ["breakpoint add 0x%x" % (0x10000 + 4 * 8), "breakpoint add 0x%x" % (0x10200 + 4 * 8),
              "pc=0x10000", "continue", "pc=0x10200", "continue"])
for i, s in enumerate(conts(t)):
    print("--- pass %d segment:" % (i + 1))
    print(s, end="")
print("EXPLORE3_DONE")
