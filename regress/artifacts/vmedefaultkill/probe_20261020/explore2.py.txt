#!/usr/bin/env python3
"""explore2.py <gxemul> -- one-off measurements behind three statements in the probe header / SUMMARY.
Not a row source.  (a) a ZERO word at the vector: what stepping into it does; (b) how much host text a
population-1 FAULT prints while FREE-RUNNING versus stepped; (c) whether a population-2 WRITE announce
is visible free-running (no scored row pins the write direction)."""
import os, pty, select, subprocess, sys, tempfile, time

BIN = sys.argv[1]
TD = tempfile.mkdtemp(prefix="vmeexp2_")
DUMMY = os.path.join(TD, "dummy.bin")
open(DUMMY, "wb").write(b"\x00\x00\x00\x00")
ARGV = [BIN, "-V", "-x", "-E", "mvme88k", "-e", "mvme187", "-M", "64", "0x10000:" + DUMMY]
P = "GXemul> "


def enc(op26, rd, rs1, imm):
    return (op26 << 26) | (rd << 21) | (rs1 << 16) | (imm & 0xffff)


def piped(cmds):
    inp = ("\n".join(cmds + ["quit"]) + "\n").encode("latin-1")
    p = subprocess.run(ARGV, input=inp, stdout=subprocess.PIPE, stderr=subprocess.STDOUT, timeout=60)
    return p.returncode, p.stdout.decode("latin-1", "replace").replace("\r", "")


def ptyrun(cmds):
    pid, fd = pty.fork()
    if pid == 0:
        os.execv(BIN, ARGV)
        os._exit(127)
    buf = [""]

    def pump(t, pred):
        end = time.time() + t
        while time.time() < end:
            if pred():
                return True
            r, _, _ = select.select([fd], [], [], 0.05)
            if r:
                try:
                    d = os.read(fd, 65536)
                except OSError:
                    return False
                if not d:
                    return False
                buf[0] += d.decode("latin-1", "replace")
        return pred()

    pump(20, lambda: P in buf[0])
    for c in cmds:
        s = len(buf[0])
        os.write(fd, (c + "\n").encode())
        if c == "continue":
            pump(30, lambda s=s: buf[0].find("BREAKPOINT", s) >= 0 and
                 buf[0].find(P, buf[0].find("BREAKPOINT", s)) >= 0)
        else:
            pump(10, lambda s=s: buf[0].find(P, s) >= 0)
    os.write(fd, b"quit\n")
    pump(3, lambda: False)
    os.close(fd)
    os.waitpid(pid, 0)
    return buf[0].replace("\r", "")


def seg_after(t, cmd):
    """text between `GXemul> <cmd>' and the next prompt"""
    i = t.find(P + cmd + "\n")
    if i < 0:
        return ""
    j = t.find(P, i + len(P))
    return t[i:j if j >= 0 else len(t)]


print("=== (a) ZERO word at the vector 0x18, ld at 0x90, step 2 (the 2026-10-09 shape)")
rc, t = piped(["r3=0xfff40000", "r4=0xdeadbeef",
               "put w 0x10000, 0x%08x" % enc(0x05, 4, 3, 0x90),
               "put w 0x10004, 0x%08x" % enc(0x16, 5, 0, 0x1234),
               "pc=0x10000", "step 2", "reg"])
s = seg_after(t, "step 2")
print("rc=%d  'UNIMPLEMENTED instruction at 0x18' in step output: %s" % (rc, "UNIMPLEMENTED instruction at 0x18" in s))
print("pc line: %s" % [ln for ln in t.splitlines() if " pc  = " in ln])
print("--- step 2 segment:")
print(s)

print("=== (b) population-1 faults FREE-RUNNING: 8 loads 0x90..0xac, handler addu r1,r1,1 ; rte at 0x18")
cmds = ["r1=0x0", "r3=0xfff40000", "r4=0xdeadbeef",
        "put w 0x18, 0x%08x" % enc(0x18, 1, 1, 1), "put w 0x1c, 0xf400fc00"]
for i in range(8):
    cmds.append("put w 0x%x, 0x%08x" % (0x10000 + 4 * i, enc(0x05, 4, 3, 0x90 + 4 * i)))
cmds += ["put w 0x10020, 0xc0000000", "unassemble 0x18 0x20", "breakpoint add 0x10020",
         "pc=0x10000", "continue", "reg"]
t = ptyrun(cmds)
s = seg_after(t, "continue")
print("free-running segment: default-arm debug lines=%d  memory_rw 'failed' lines=%d  'exception DATA_ACCESS' lines=%d  any '[ vme:'=%d"
      % (s.count("[ vme: unimplemented READ from offset"), s.count("[ reading from device 'vme'"),
         s.count("exception DATA_ACCESS"), s.count("[ vme:")))
print("r1 (faults counted): %s" % [ln for ln in seg_after(t, "reg").splitlines() if "r1 " in ln])
print("--- continue segment:")
print(s)
print("=== (b') the same 8 loads STEPPED (step 24: 8 x (load + addu + rte))")
cmds2 = [c for c in cmds if c not in ("breakpoint add 0x10020", "continue", "reg")] + ["step 24", "reg"]
rc, t = piped(cmds2)
s = seg_after(t, "step 24")
print("rc=%d stepped segment: default-arm debug lines=%d  memory_rw 'failed' lines=%d  'exception DATA_ACCESS' lines=%d"
      % (rc, s.count("[ vme: unimplemented READ from offset"), s.count("[ reading from device 'vme'"),
         s.count("exception DATA_ACCESS")))
print("r1: %s" % [ln for ln in seg_after(t, "reg").splitlines() if "r1 " in ln])

print("=== (c) population-2 WRITE announce FREE-RUNNING: st 0xa5a5a5a5 to MASTER3 0x1c twice, then ld 0x1c")
cmds = ["r2=0xa5a5a5a5", "r3=0xfff40000", "r4=0xdeadbeef",
        "put w 0x10000, 0x%08x" % enc(0x09, 2, 3, 0x1c),
        "put w 0x10004, 0x%08x" % enc(0x09, 2, 3, 0x1c),
        "put w 0x10008, 0x%08x" % enc(0x05, 4, 3, 0x1c),
        "put w 0x1000c, 0xc0000000", "breakpoint add 0x1000c", "pc=0x10000", "continue", "reg"]
t = ptyrun(cmds)
s = seg_after(t, "continue")
print("free-running segment '(once)' lines: %d" % s.count("(once)"))
print(s)
print("r4: %s" % [ln for ln in seg_after(t, "reg").splitlines() if "r4 " in ln])
print("EXPLORE2_DONE")
