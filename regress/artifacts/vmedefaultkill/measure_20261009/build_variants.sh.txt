#!/bin/bash
# Build two SKETCH variants of committed HEAD's dev_vme.c default: arm for the 7n-4 "measure first" probe.
# FAULT  : the arm returns 0 (memory_rw then raises the m88k data-access exception).
# DECLINE: the arm sets odata = 0 and breaks to the common tail (reads give 0, writes dropped).
# Neither is a fix; both exist only to measure what the guest would see.  Scratch trees in /tmp only.
set -u
TAR="$1"
for V in fault decline; do
  W=/tmp/gxvme_$V
  rm -rf "$W"; mkdir -p "$W"; cd "$W" || exit 3
  tar -xf "$TAR" || { echo "TAR_FAILED $V"; exit 3; }
  python3 - "$V" <<'PY'
import sys
v = sys.argv[1]
p = "src/devices/dev_vme.c"
s = open(p).read()
old = "\t\texit(1);\n\t}\n\n\tif (writeflag == MEM_READ)\n\t\tmemory_writemax64(cpu, data, len, odata);\n"
assert s.count(old) == 1, "anchor count %d" % s.count(old)
new = {"fault": "\t\treturn 0;\n\t}\n\n\tif (writeflag == MEM_READ)\n\t\tmemory_writemax64(cpu, data, len, odata);\n",
       "decline": "\t\todata = 0;\n\t\tbreak;\n\t}\n\n\tif (writeflag == MEM_READ)\n\t\tmemory_writemax64(cpu, data, len, odata);\n"}[v]
open(p, "w").write(s.replace(old, new))
print("EDIT_APPLIED", v)
PY
  ./configure > configure.log 2>&1; echo "CONFIGURE_RC_$V=$?"
  make -j2 > make.log 2>&1; echo "MAKE_RC_$V=$?"
  if [ -f gxemul ] && [ gxemul -nt src/devices/dev_vme.c ]; then echo "BINARY_FRESH_$V=yes"; else echo "BINARY_FRESH_$V=no"; fi
  echo "dev_vme.c md5 $V: $(md5sum src/devices/dev_vme.c | cut -c1-8)"
  diff <(tar -xOf "$TAR" src/devices/dev_vme.c) src/devices/dev_vme.c
done
echo "VME_VARIANTS_DONE"
