#!/usr/bin/env python3
"""timer_explore.py <gxemul> <outdir> -- EXPLORATORY measurement for T19 question 5 (not a detector row): what the 8253
model on unmodified malta does to IRQ 0 across a specific EOI.  Programs counter 0 (mode 2, count 0x1000, in_use becomes 1
on the first access), waits a long counted loop, then reads IRR, applies EOI 0x60, reads IRR at once, waits again, reads
IRR again, plus ISR and the poll byte.  Everything printed is raw; nothing here is a want.  Imports the probe module for
its encoders, Emu driver and session runner."""
import importlib.util, os, sys, time

HERE = os.path.dirname(os.path.abspath(__file__))
spec = importlib.util.spec_from_file_location("p", os.path.join(HERE, "i8259_eoi_probe.py"))
p = importlib.util.module_from_spec(spec)
ARGS = list(sys.argv)
BIN = ARGS[1] if len(ARGS) > 1 else "/tmp/gxt19/base/gxemul"
OUT = ARGS[2] if len(ARGS) > 2 else "/tmp/gxt19/timer"
sys.argv = ["x", BIN, "-", OUT]          # the probe module parses argv at import: <base> <fix|-> <outdir>
spec.loader.exec_module(p)
PIT_C0, PIT_CTL = 0x40, 0x43
LONG = 60000     # 3 instructions per iteration: ~180k instructions per wait


def long_delay():
    p.e_li(p.t4, LONG)
    p.emit(p.addiu(p.t4, p.t4, -1), "addiu t4,t4,-1")
    p.emit(p.bne(p.t4, p.zero, -2), "bne t4,zr,0x")
    p.emit(p.NOP, "nop")


def build_timer():
    p.begin()
    p.e_ocw3_read(p.v1, p.OCW3_IRR)          # IRR before touching the PIT
    p.e_out(PIT_CTL, 0x34)                   # counter 0, LSB/MSB, mode 2, binary
    p.e_out(PIT_C0, 0x00)
    p.e_out(PIT_C0, 0x10)                    # count 0x1000
    long_delay(); long_delay(); long_delay()
    p.e_ocw3_read(p.a0, p.OCW3_IRR)          # IRR after ~540k instructions
    p.e_ocw3_read(p.a1, p.OCW3_ISR)
    p.e_li(p.t5, p.OCW3_IRR)
    p.e_li(p.t6, 0x60)                       # specific EOI, line 0
    p.emit(p.sb(p.t6, p.PIC1_CMD, p.t0), "sb t6,%d(t0)" % p.PIC1_CMD)
    p.emit(p.sb(p.t5, p.PIC1_CMD, p.t0), "sb t5,%d(t0)" % p.PIC1_CMD)
    p.e_in(p.a2, p.PIC1_CMD)                 # IRR two instructions after the EOI
    long_delay(); long_delay(); long_delay()
    p.e_ocw3_read(p.a3, p.OCW3_IRR)          # IRR after another ~540k instructions
    p.e_ocw3_read(p.s0, p.OCW3_POLL)
    p.e_in(p.s1, p.PIT_C0 if hasattr(p, "PIT_C0") else PIT_C0)   # the counter's low byte (informational)
    prog, halt = p.end()
    return prog, halt, {"ram": "v0", "irr_before": "v1", "irr_after_wait": "a0", "isr": "a1", "irr_after_eoi": "a2",
                        "irr_after_wait2": "a3", "poll": "s0", "pit_lsb": "s1"}


def main():
    os.makedirs(OUT, exist_ok=True)
    p.RAWDIR = os.path.join(OUT, "raw"); os.makedirs(p.RAWDIR, exist_ok=True)
    lines = []

    def out(s=""):
        lines.append(s); print(s); sys.stdout.flush()
    out("timer_explore -- EXPLORATORY: the 8253 model's IRQ 0 across a specific EOI on unmodified evbmips -e malta")
    out("binary %s   %s" % (BIN, time.strftime("%Y-%m-%d %H:%M:%S")))
    for rep in range(2):
        prog, halt, regmap = build_timer()
        got, enc_ok, hit = p.run_session(BIN, prog, halt, os.path.join(p.RAWDIR, "timer%d.bin" % rep), regmap, "timer%d" % rep, out)
        out("  run %d: encodings_ok=%s parked=%s" % (rep, enc_ok, hit))
        out("  raw: " + "  ".join("%s=%s" % (k, p.show(v)) for k, v in got.items()))
        out("  bit0: before=%s after_wait=%s after_eoi=%s after_wait2=%s" % tuple(
            None if got.get(k) is None else got[k] & 1
            for k in ("irr_before", "irr_after_wait", "irr_after_eoi", "irr_after_wait2")))
    out("TIMER_EXPLORE_DONE")
    with open(os.path.join(OUT, "RESULTS_timer.txt"), "w") as f:
        f.write("\n".join(lines) + "\n")


if __name__ == "__main__":
    main()
